# ===== Heuristic Code =====
#{RankAdaptive_Softmax_SelectiveDualDecay combines vectorized logistic-utility evaluation and median thresholding with rank-adaptive exponential weighting for penalization, coupled with a two-tiered neighbor decay prioritizing mutual k-nearest unvisited edges over general unvisited edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.max(utilities)

    p_indices = np.where(penalize_mask)[0]
    p_u = u_idx[p_indices]
    p_v = v_idx[p_indices]
    p_costs = costs[p_indices]
    p_used = used[p_indices]
    p_utils = utilities[p_indices]
    max_util = np.max(utilities) + 1e-8

    m = len(p_indices)
    if m > 1:
        ranks = np.argsort(np.argsort(p_utils)).astype(np.float64)
        norm_ranks = ranks / (m - 1.0)
    else:
        norm_ranks = np.ones(1, dtype=np.float64)

    exp_weights = np.exp(norm_ranks)
    soft_factors = exp_weights / np.mean(exp_weights)

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * soft_factors
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    k = min(8, n - 1)
    k_nn_indices = np.argpartition(edge_distance, kth=k, axis=1)[:, :k + 1]
    is_knn = np.zeros((n, n), dtype=bool)
    row_idx = np.repeat(np.arange(n), k + 1)
    col_idx = k_nn_indices.ravel()
    is_knn[row_idx, col_idx] = True
    np.fill_diagonal(is_knn, False)
    mutual_knn = is_knn & is_knn.T

    strong_decay = 0.03
    light_decay = 0.015

    for u in penalized_nodes:
        unused = (edge_n_used[u] == 0)
        unused[u] = False

        knn_unvisited = unused & mutual_knn[u]
        non_knn_unvisited = unused & (~mutual_knn[u])

        if np.any(knn_unvisited):
            current_vals = updated_edge_distance[u, knn_unvisited]
            dec_vals = np.maximum(1e-6, current_vals * (1.0 - strong_decay))
            updated_edge_distance[u, knn_unvisited] = dec_vals
            updated_edge_distance[knn_unvisited, u] = dec_vals

        if np.any(non_knn_unvisited):
            current_vals = updated_edge_distance[u, non_knn_unvisited]
            dec_vals = np.maximum(1e-6, current_vals * (1.0 - light_decay))
            updated_edge_distance[u, non_knn_unvisited] = dec_vals
            updated_edge_distance[non_knn_unvisited, u] = dec_vals

    return updated_edge_distance
