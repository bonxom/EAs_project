# ===== Heuristic Code =====
#{This algorithm introduces an adaptive global perturbation decay that smoothly relaxes previously accumulated matrix distortions towards an estimate of the base metric while applying standardized utility-driven penalties to high-cost tour edges and selective incentive discounts to candidate unexplored alternatives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Extract active tour edges
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Adaptive global evaporation / relaxation:
    # Estimate base cost metric by discounting cumulative penalties based on historical usage
    # This prevents metric distortion from growing unboundedly across successive iterations.
    decay_factor = 0.96
    effective_mask = edge_n_used > 0
    np.fill_diagonal(effective_mask, False)

    if np.any(effective_mask):
        # Relax the matrix towards the unperturbed distance baseline
        approx_base = updated_edge_distance / (1.0 + 0.05 * np.log1p(edge_n_used.astype(np.float64)))
        updated_edge_distance[effective_mask] = (
            approx_base[effective_mask] + decay_factor * (updated_edge_distance[effective_mask] - approx_base[effective_mask])
        )

    # Re-sample edge costs after relaxation
    costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)

    # Dual-force utility calculation using sigmoid weighting on standardized edge costs
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Compute scaled penalties for top utility tour edges
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * 0.5
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Reward/discount promising unused or low-frequency incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.03
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            discounts = current_vals * decay_ratio
            updated_edge_distance[u, unused_mask] -= discounts
            updated_edge_distance[unused_mask, u] -= discounts

    # Enforce symmetric zero-diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)
    return updated_edge_distance
