# ===== Heuristic Code =====
#{A Guided Local Search matrix update that computes endpoint structural motif rigidity via incident usage Shannon entropy to augment edge utility scores and guide penalties toward entrapped subpaths, while selectively discounting unexplored alternative incident edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Compute node-level usage entropy for structural motif rigidity
    unique_nodes = np.unique(u_idx)
    sub_used = edge_n_used[unique_nodes, :].astype(np.float64)
    row_sums = np.sum(sub_used, axis=1, keepdims=True)

    # Normalized distribution and Shannon entropy per node
    probs = np.divide(sub_used, row_sums, out=np.zeros_like(sub_used), where=row_sums > 0)
    with np.errstate(divide='ignore', invalid='ignore'):
        log_probs = np.where(probs > 1e-12, np.log2(probs), 0.0)
    entropies = -np.sum(probs * log_probs, axis=1)

    # Max possible entropy is log2(N - 1)
    max_possible_entropy = np.log2(max(edge_distance.shape[0] - 1, 1))
    norm_entropy = entropies / (max_possible_entropy + 1e-8)

    # Rigidity: lower entropy and higher exploration indicates frozen/rigid search basin
    explored_factor = np.tanh(row_sums.squeeze(-1) / 10.0)
    rigidity_arr = (1.0 - norm_entropy) * explored_factor

    rigidity_map = np.zeros(edge_distance.shape[0], dtype=np.float64)
    rigidity_map[unique_nodes] = rigidity_arr

    # Structural motif rigidity for each edge in the tour
    edge_rigidity = 0.5 * (rigidity_map[u_idx] + rigidity_map[v_idx])

    # Augment utility calculation with structural motif rigidity
    utilities = (costs * (1.0 + 2.0 * edge_rigidity)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]
    p_rigidity = edge_rigidity[penalize_mask]

    # Calculate directed penalties targeting persistently rigid subpaths
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * (1.0 + p_rigidity)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Reward discounting on unexplored/under-explored alternatives for penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
