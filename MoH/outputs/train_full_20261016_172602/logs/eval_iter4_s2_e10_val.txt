# ===== Heuristic Code =====
#{Modulate guided local search edge penalties using spatial density and incident distance dispersion to heavily penalize inefficient edges in dense clusters while protecting structural boundary bridges, coupled with geometry-aware utility prioritization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decayed frequency tracking
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric triplet evaluation: detour ratio and acute turn deficit
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Spatial density and centrality estimation:
    # Nodes in dense clusters have small mean incident distances to k-nearest neighbors
    # whereas isolated / boundary nodes have large mean distances and low dispersion
    k_nn = min(8, num_nodes - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    knn_dists = sorted_dists[:, 1:k_nn + 1]
    local_scale = np.mean(knn_dists, axis=1) + 1e-8

    # Density factor: higher when endpoints are located in dense regions
    density_u = 1.0 / local_scale[u_idx]
    density_v = 1.0 / local_scale[v_idx]
    mean_density = np.mean(1.0 / local_scale)
    relative_density = np.sqrt((density_u * density_v)) / (mean_density + 1e-8)

    # Relative bridge dampening: long edges relative to local scale are penalized heavily
    # if in dense clusters, but protected if endpoints naturally have large local scales
    edge_isolation = 0.5 * (local_scale[u_idx] + local_scale[v_idx])
    density_modulation = np.clip(relative_density * (d_uv / (edge_isolation + 1e-8)), 0.5, 2.5)

    # Utilities combine base distance, frequency damping, geometry factor, and cluster density
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor) * (0.8 + 0.2 * density_modulation)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    base_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        mod_lambda = base_lambda * density_modulation[idx]

        penalty = mod_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Incentivize promising shortcuts that cut high-detour obtuse geometries
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
