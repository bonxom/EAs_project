# ===== Heuristic Code =====
# {Applies a cooling decay across previously penalized edges to evaporate obsolete penalties while injecting directed, asymmetric burst penalties on chronically overused edges to break cyclic stagnation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Cooling mechanism: identify previously penalized edges and apply a memory-guided discount factor
    decay_factor = 0.985
    penalized_mask = edge_n_used > 0
    updated_edge_distance[penalized_mask] = (
        updated_edge_distance[penalized_mask] * decay_factor
    )

    # Base penalty for edges meeting the utility threshold
    updated_edge_distance[sel_u, sel_v] += base_penalty
    updated_edge_distance[sel_v, sel_u] += base_penalty

    # Asymmetric burst perturbation targeting chronic cycle stagnation on high-frequency edges
    max_usage = np.max(tour_counts)
    if max_usage > 1:
        stagnant_threshold = 0.75 * max_usage
        stagnant_mask = (tour_counts >= stagnant_threshold) & (tour_counts > 0)

        stag_u = u_idx[stagnant_mask]
        stag_v = v_idx[stagnant_mask]

        burst_ratio = tour_counts[stagnant_mask] / max_usage
        burst_penalty = base_penalty * (1.0 + burst_ratio)

        # Asymmetric directed injection to break cyclic symmetry in search trajectories
        updated_edge_distance[stag_u, stag_v] += 1.25 * burst_penalty
        updated_edge_distance[stag_v, stag_u] += 0.75 * burst_penalty

    return updated_edge_distance
