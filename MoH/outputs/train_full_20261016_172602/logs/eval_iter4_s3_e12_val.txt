# ===== Heuristic Code =====
#{Replaces full distance sorting with fast k-NN partitioning, substitutes the arithmetic density mean with harmonic and bottleneck density coupling to safeguard bridge edges, and applies smooth continuous metric-scaled penalization modulated by edge usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    # Efficient local density estimation using argpartition/partition instead of full sort
    k = min(6, n - 1)
    k_part = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(k_part, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Bottleneck and harmonic density coupling to protect inter-cluster bridge edges
    dens_u = node_density[u_idx]
    dens_v = node_density[v_idx]
    min_density = np.minimum(dens_u, dens_v)
    harmonic_density = (2.0 * dens_u * dens_v) / (dens_u + dens_v + 1e-6)

    # Bridge edges connect distinct sparse/dense regions or cross sparse voids; prioritize intra-cluster swaps
    edge_criticality = np.sqrt(harmonic_density * min_density)

    # Inverse frequency damping to avoid cyclic over-penalization
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.35 * usage)

    # Smooth continuous length-based activation without abrupt threshold cliffs
    smooth_activation = 1.0 / (1.0 + np.exp(-3.5 * (relative_length - 1.1)))

    # Adaptive penalty scaled proportionally to the tour's natural edge length scale
    base_step = 0.15 * avg_tour_length
    penalty = base_step * relative_length * edge_criticality * usage_factor * (0.4 + 1.2 * smooth_activation)

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
