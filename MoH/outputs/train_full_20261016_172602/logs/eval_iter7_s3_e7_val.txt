# ===== Heuristic Code =====
#{This algorithm replaces static thresholding with a bi-objective Pareto-rank dominance scoring over edge length and recency followed by adaptive Boltzmann softmax amplification to selectively penalize sub-optimal edges while protecting necessary topological bridges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    # Calculate k-nearest neighbor density for structural context
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Tour edge attributes
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_counts)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_counts)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Bi-objective Pareto dominance ranking: Objective 1 = tour_edge_lengths (maximize), Objective 2 = -edge_counts (maximize)
    # Edge A dominates Edge B if (len_A >= len_B and -count_A >= -count_B) with at least one strict inequality.
    len_diff = tour_edge_lengths[:, None] - tour_edge_lengths[None, :]
    count_diff = -edge_counts[:, None] - (-edge_counts[None, :])

    dominates = (len_diff >= 0) & (count_diff >= 0) & ((len_diff > 0) | (count_diff > 0))
    # Number of candidates dominating each edge determines its Pareto non-optimality rank (lower count = more dominant)
    pareto_dom_count = np.sum(dominates, axis=0).astype(float)

    # Softmax Boltzmann weighting with adaptive temperature to focus penalty pressure
    # Higher utility for edges that are heavily dominant (long length and low prior penalty history)
    pareto_utility = -pareto_dom_count
    norm_utility = (pareto_utility - np.mean(pareto_utility)) / (np.std(pareto_utility) + 1e-6)

    temperature = max(0.5, 1.5 / (1.0 + 0.05 * freq_mean))
    exp_weights = np.exp(norm_utility / temperature)
    boltzmann_multiplier = 1.0 + (exp_weights / (np.max(exp_weights) + 1e-6)) * 0.8

    # Apply pressure selectively
    final_penalty = combined_penalty * boltzmann_multiplier

    np.add.at(updated_edge_distance, (u_idx, v_idx), final_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), final_penalty)

    return updated_edge_distance
