# ===== Heuristic Code =====
#{An adaptive rank-based ruin operator dynamically selects and penalizes the top-k tour edges by computing an intensity quantile scaled by tour edge repetition frequency to diversify stagnant local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    if n_edges == 0:
        return updated_edge_distance

    # Measure stagnation via edge repetition in the current tour
    mean_count = np.mean(tour_counts)
    max_count = np.max(tour_counts)
    stagnation_factor = np.clip((mean_count + 0.5 * max_count) / 10.0, 0.0, 1.0)

    # Scale ruin size k dynamically between 1 edge and up to 15% of the tour
    min_k = 1
    max_k = max(1, int(np.ceil(0.15 * n_edges)))
    k = int(min_k + np.round(stagnation_factor * (max_k - min_k)))
    k = min(n_edges, max(1, k))

    # Efficiently extract top-k edges using partition without sorting full array
    top_k_indices = np.argpartition(utilities, -k)[-k:]

    # Proportional penalty scaled with search intensity and edge length
    avg_edge = np.mean(tour_lengths)
    penalty_base = 0.3 * avg_edge * (1.0 + 0.5 * stagnation_factor)

    sel_u = u_idx[top_k_indices]
    sel_v = v_idx[top_k_indices]

    updated_edge_distance[sel_u, sel_v] += penalty_base
    updated_edge_distance[sel_v, sel_u] += penalty_base

    return updated_edge_distance
