# ===== Heuristic Code =====
#{A deterministic guided local search update that uses np.argpartition to penalize strictly the top bottleneck edge having the maximal utility with an adaptive fallback to the second-highest edge only if it shares the near-exact maximum utility, ensuring controlled and minimal structural disruption.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(utilities)
    if n_edges <= 2:
        top_indices = np.argsort(utilities)[-1:]
    else:
        # Partition to find the top 2 candidate bottleneck edges efficiently
        top2_part = np.argpartition(utilities, -2)[-2:]
        if utilities[top2_part[0]] < utilities[top2_part[1]]:
            idx_max, idx_sub = top2_part[1], top2_part[0]
        else:
            idx_max, idx_sub = top2_part[0], top2_part[1]

        # Penalize strictly the single maximum utility edge unless the runner-up is an exact numerical tie
        if np.isclose(utilities[idx_max], utilities[idx_sub], rtol=1e-9, atol=1e-12):
            top_indices = np.array([idx_max, idx_sub])
        else:
            top_indices = np.array([idx_max])

    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    sel_u = u_idx[top_indices]
    sel_v = v_idx[top_indices]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
