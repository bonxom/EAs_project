# ===== Heuristic Code =====
#{Replaces the static sigmoid ranking with a Pareto-utility regret metric measuring the ratio of each tour edge's cost against the best available non-tour incident alternative edges, combined with local density and frequency regularization to effectively penalize inefficient edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Local k-NN density estimation
    k = min(6, n - 1)
    sorted_dists = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Tour edge endpoints
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Predecessor and successor in tour to exclude incident tour edges
    prev_u = np.roll(local_opt_tour, 1)
    next_v = np.roll(local_opt_tour, -2)

    # Create temporary distance copies for u and v to find best non-tour alternatives
    # For node u_i, tour neighbors are prev_u and v_i; for v_i, tour neighbors are u_i and next_v
    # Partition small k candidates to quickly extract minimum non-tour distance
    c_u = np.copy(edge_distance[u_idx])
    rows = np.arange(n)
    c_u[rows, u_idx] = np.inf
    c_u[rows, prev_u] = np.inf
    c_u[rows, v_idx] = np.inf
    min_alt_u = np.min(c_u, axis=1)

    c_v = np.copy(edge_distance[v_idx])
    c_v[rows, v_idx] = np.inf
    c_v[rows, u_idx] = np.inf
    c_v[rows, next_v] = np.inf
    min_alt_v = np.min(c_v, axis=1)

    # Regret / Pareto alternative baseline (harmonic or minimum best alternative)
    best_alt = 0.5 * (min_alt_u + min_alt_v)
    regret_ratio = tour_edge_lengths / (best_alt + 1e-6)
    pareto_utility = np.clip(regret_ratio - 1.0, 0.0, None)
    pareto_norm = pareto_utility / (np.mean(pareto_utility) + 1e-6)

    # Edge usage and density adjustments
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Adaptive penalty calculation
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    penalty = pareto_norm * relative_length * usage_factor * density_factor * (avg_tour_length * 0.15)

    # Top regret threshold amplifier
    threshold = np.percentile(regret_ratio, 75)
    high_regret_mask = regret_ratio >= threshold
    penalty[high_regret_mask] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
