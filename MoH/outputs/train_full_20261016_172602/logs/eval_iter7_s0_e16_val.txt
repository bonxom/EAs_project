# ===== Heuristic Code =====
#{This algorithm identifies high-utility tour edges and applies a neighborhood-directed spatial penalty by penalizing the selected edges while diffusing a geometrically decayed penalty across their incident tour and k-nearest neighbor subgraphs to effectively ruin and escape the local spatial basin.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]
    if n < 4:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    prev_u = np.roll(local_opt_tour, 1)
    next_v = np.roll(local_opt_tour, -2)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.35 * avg_edge

    selected_mask = utilities >= (0.95 * max_utility)
    sel_indices = np.flatnonzero(selected_mask)

    if sel_indices.size == 0:
        sel_indices = np.array([np.argmax(utilities)])

    sel_u = u_idx[sel_indices]
    sel_v = v_idx[sel_indices]

    # Primary edge ruin
    np.add.at(updated_edge_distance, (sel_u, sel_v), base_penalty)
    np.add.at(updated_edge_distance, (sel_v, sel_u), base_penalty)

    # Incident edge diffusion (adjacent links along the tour)
    incident_penalty = 0.5 * base_penalty
    p_u = prev_u[sel_indices]
    n_v = next_v[sel_indices]

    np.add.at(updated_edge_distance, (p_u, sel_u), incident_penalty)
    np.add.at(updated_edge_distance, (sel_u, p_u), incident_penalty)
    np.add.at(updated_edge_distance, (sel_v, n_v), incident_penalty)
    np.add.at(updated_edge_distance, (n_v, sel_v), incident_penalty)

    # Spatial k-nearest neighbor ruin around endpoint nodes
    k = min(5, n - 1)
    seed_nodes = np.unique(np.concatenate([sel_u, sel_v]))

    # Vectorized k-nearest neighbor lookup for all seed nodes
    sub_dists = edge_distance[seed_nodes]
    knn_indices = np.argpartition(sub_dists, kth=k, axis=1)[:, 1:k+1]

    knn_penalty = 0.25 * base_penalty
    row_seeds = np.repeat(seed_nodes, k)
    col_neighbors = knn_indices.ravel()

    np.add.at(updated_edge_distance, (row_seeds, col_neighbors), knn_penalty)
    np.add.at(updated_edge_distance, (col_neighbors, row_seeds), knn_penalty)

    np.fill_diagonal(updated_edge_distance, 0.0)
    return updated_edge_distance
