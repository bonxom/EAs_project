#{Dynamic recency-weighted exponential penalty assignment combined with memory-based utility normalization and dissipation across long-unvisited adjacent edges to systematically escape local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Recency-weighted frequency surrogate via sub-linear saturating exponential scale
    effective_frequency = 1.0 - np.exp(-0.25 * used)

    # Cost-to-frequency utility metric balancing high edge cost against historical entrapment
    utilities = costs / (1.0 + 3.0 * effective_frequency)

    # Dynamic selection of high-utility edges to disrupt local optimality
    util_median = np.median(utilities)
    util_max = np.max(utilities) + 1e-12
    penalize_mask = utilities >= util_median

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_eff_freq = effective_frequency[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Penalties scaled by normalized utility and inversely modulated by effective saturation
    penalties = (0.35 * p_costs) * (1.0 + (p_utils / util_max)) / (1.0 + p_eff_freq)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Memory-based penalty dissipation on long-unvisited/underutilized incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    sub_matrix_used = edge_n_used[penalized_nodes]
    sub_matrix_dist = updated_edge_distance[penalized_nodes]

    # Dissipate distance for edges that have experienced negligible recent visits
    dissipation_rate = 0.05
    unvisited_mask = (sub_matrix_used == 0)
    # Exclude self-loops
    unvisited_mask[np.arange(len(penalized_nodes)), penalized_nodes] = False

    dissipation_amounts = sub_matrix_dist * dissipation_rate * unvisited_mask
    updated_edge_distance[penalized_nodes] -= dissipation_amounts

    # Enforce exact matrix symmetry after dissipation
    updated_edge_distance[:, penalized_nodes] = updated_edge_distance[penalized_nodes, :].T

    # Enforce non-negativity constraint
    np.maximum(updated_edge_distance, 0.0, out=updated_edge_distance)

    return updated_edge_distance
