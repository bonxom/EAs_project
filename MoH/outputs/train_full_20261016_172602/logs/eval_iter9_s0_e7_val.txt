# ===== Heuristic Code =====
#{A non-uniform utility-proportional ruin strategy that applies graded penalties based on normalized excess utility and softly penalizes the incident edges of the most critical bottleneck vertices to escape deep local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    min_utility = np.min(utilities)
    avg_edge = np.mean(tour_lengths)

    # Define soft threshold to capture a band of high-utility edges
    threshold = min_utility + 0.85 * (max_utility - min_utility)
    eps = 1e-12

    selected_mask = utilities >= threshold
    if not np.any(selected_mask):
        selected_mask = utilities >= (max_utility - eps)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_utilities = utilities[selected_mask]

    # Graded penalty proportional to relative excess utility
    scaling = (sel_utilities - threshold) / (max_utility - threshold + eps)
    penalties = (0.2 + 0.4 * scaling) * avg_edge

    # Direct symmetric penalty application on selected tour edges
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    # Neighborhood ruin: identify high-utility bottleneck vertices and apply soft incident penalties
    worst_vertices = np.unique(np.concatenate([sel_u, sel_v]))
    incident_penalty = 0.05 * avg_edge

    updated_edge_distance[worst_vertices, :] += incident_penalty
    updated_edge_distance[:, worst_vertices] += incident_penalty
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
