# ===== Heuristic Code =====
#{We introduce a non-maximum suppression (NMS) penalty dispersion mechanism that iteratively selects top geometric-utility tour edges while dampening adjacent edge scores to distribute perturbations uniformly around the cycle, paired with corresponding shortcut discounts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Smooth decay of historical usage memory
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Evaluate geometric inefficiency and turn deficit across triplets
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Non-maximum suppression (NMS) penalty dispersion
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    selected_indices = []
    active_utilities = np.copy(utilities)

    for _ in range(k):
        best_idx = int(np.argmax(active_utilities))
        if active_utilities[best_idx] <= -1e8:
            break
        selected_indices.append(best_idx)
        # Suppress selected edge
        active_utilities[best_idx] = -1e9
        # Dampen immediate cyclic neighbors to prevent localized over-penalization
        prev_idx = (best_idx - 1) % num_nodes
        next_idx = (best_idx + 1) % num_nodes
        active_utilities[prev_idx] *= 0.35
        active_utilities[next_idx] *= 0.35

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply distributed penalties
    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Shortcut discount for highest detour segments associated with selected perturbations
    for idx in selected_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
