# ===== Heuristic Code =====
#{This algorithm refines Guided Local Search edge penalization by dynamically amplifying penalties on above-median utility edges using rank-weighted soft distribution and excess-cost scaling relative to tour median cost, coupled with symmetric incident unused-edge discounting.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    med_cost = np.median(costs) + 1e-8
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    num_penalized = len(p_utils)
    if num_penalized > 0:
        # Rank-weighted soft distribution among penalized edges
        sort_ranks = np.argsort(np.argsort(p_utils)).astype(np.float64)
        rank_weights = (sort_ranks + 1.0) / float(num_penalized)

        # Excess-cost amplification factor
        excess_cost_ratio = np.maximum(1.0, p_costs / med_cost)

        # Differential penalty synthesis
        base_penalty = p_costs / (1.0 + np.sqrt(p_used + 1.0))
        penalties = base_penalty * (1.0 + rank_weights) * excess_cost_ratio

        updated_edge_distance[p_u, p_v] += penalties
        updated_edge_distance[p_v, p_u] += penalties

        penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
        decay_ratio = 0.02
        for u in penalized_nodes:
            unused_mask = (edge_n_used[u] == 0)
            unused_mask[u] = False
            if np.any(unused_mask):
                current_vals = updated_edge_distance[u, unused_mask]
                discounted = current_vals * (1.0 - decay_ratio)
                updated_edge_distance[u, unused_mask] = discounted
                updated_edge_distance[unused_mask, u] = discounted

    return updated_edge_distance
