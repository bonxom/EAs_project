# ===== Heuristic Code =====
#{Adaptive Guided Local Search edge penalty mechanism selecting high-utility edges via dynamic standard deviation thresholding and scaling penalties inversely by tour coefficient of variation to preserve metric structure.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    n_nodes = len(local_opt_tour)
    if n_nodes < 2:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    mean_util = np.mean(utilities)
    std_util = np.std(utilities)

    # Adaptive threshold: select edges with utility significantly higher than average
    threshold = mean_util + 0.5 * std_util
    selected_mask = utilities >= threshold

    # Ensure at least the single maximum utility edge is selected if distribution is flat
    if not np.any(selected_mask):
        selected_mask = utilities == np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)

    # Dynamically scale penalty based on dispersion of tour edge lengths
    cv = std_edge / (avg_edge + 1e-12)
    penalty_factor = np.clip(0.1 + 0.2 * cv, 0.05, 0.4)
    penalty = penalty_factor * avg_edge

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
