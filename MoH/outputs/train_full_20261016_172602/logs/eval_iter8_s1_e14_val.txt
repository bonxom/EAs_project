# ===== Heuristic Code =====
#{This strategy evaluates tour edge utilities via cost-frequency balancing to penalize high-utility edges, then applies targeted surrogate discounts to the k-nearest low-frequency incident alternatives weighted inversely by baseline metric distance and visitation history.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Compute edge penalties calibrated by historical usage and utility scale
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k_candidates = min(5, n - 2)
    if k_candidates <= 0:
        return updated_edge_distance

    # Surrogate-guided directed attraction on incident alternative edges
    for u in penalized_nodes:
        # Exclude self-loop
        cand_mask = np.ones(n, dtype=bool)
        cand_mask[u] = False
        cand_indices = np.where(cand_mask)[0]

        base_dists = edge_distance[u, cand_indices].astype(np.float64)
        cand_used = edge_n_used[u, cand_indices].astype(np.float64)

        # Prioritize edges by low baseline distance and low usage frequency
        surrogate_metric = base_dists * (1.0 + np.log1p(cand_used))

        if len(cand_indices) > k_candidates:
            top_k_rel = np.argpartition(surrogate_metric, k_candidates)[:k_candidates]
            chosen_cands = cand_indices[top_k_rel]
            chosen_dists = base_dists[top_k_rel]
            chosen_used = cand_used[top_k_rel]
        else:
            chosen_cands = cand_indices
            chosen_dists = base_dists
            chosen_used = cand_used

        # Magnitude inversely proportional to baseline metric distance and usage count
        mean_base = np.mean(chosen_dists) + 1e-8
        rel_closeness = mean_base / (chosen_dists + 1e-8)
        norm_closeness = rel_closeness / np.sum(rel_closeness)

        # Scale discount proportionally to the local penalty context
        discount_strength = 0.08 / (1.0 + chosen_used)
        discounts = updated_edge_distance[u, chosen_cands] * discount_strength * norm_closeness

        updated_edge_distance[u, chosen_cands] -= discounts
        updated_edge_distance[chosen_cands, u] = updated_edge_distance[u, chosen_cands]

    # Ensure strictly positive metric distances
    np.maximum(updated_edge_distance, 1e-6, out=updated_edge_distance)
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
