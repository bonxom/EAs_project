# ===== Heuristic Code =====
#{Dual-space guided ruin-and-recreate that evaluates edge utilities via fast 1-tree/MST alpha-nearness slack and usage-entropy regularization, applying asymmetric-free backbone penalties and multi-fragment ruin incentives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy distance matrix and initialize problem parameters
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)

    # Evaporate prior edge usage frequency memory
    edge_n_used *= 0.95

    # Step 2: Compute approximate 1-tree / MST backbone proxy via Prim-like greedy metric
    # Efficiently compute minimum outgoing neighbor slack (alpha-nearness proxy)
    # alpha(u, v) = c(u, v) - max(min_cost(u), min_cost(v))
    dist_no_diag = np.copy(edge_distance)
    np.fill_diagonal(dist_no_diag, np.inf)
    min_out = np.min(dist_no_diag, axis=1)

    # Step 3: Compute dual-space utility along current tour edges
    d_uv = edge_distance[u_idx, v_idx]
    alpha_slack = d_uv - 0.5 * (min_out[u_idx] + min_out[v_idx])
    alpha_slack = np.maximum(0.0, alpha_slack)

    usage_uv = edge_n_used[u_idx, v_idx]

    # Tour edge utility: higher slack relative to MST backbone and higher cost relative to usage
    # Regularized utility captures information-theoretic divergence from tree backbone
    regularized_utility = (d_uv + 1.2 * alpha_slack) / (1.0 + np.sqrt(usage_uv))

    # Step 4: Determine multi-fragment ruin size and select top penalization candidates
    ruin_budget = max(2, int(np.ceil(0.12 * num_nodes)))
    sorted_edge_order = np.argsort(regularized_utility)
    penalized_edges = sorted_edge_order[-ruin_budget:]

    global_mean_cost = np.mean(edge_distance[edge_distance < np.inf])
    base_penalty = 0.28 * global_mean_cost

    # Step 5: Apply guided backbone penalties to break stagnant basin
    for idx in penalized_edges:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        penalty_scale = 1.0 + np.log1p(edge_n_used[u, v])
        penalty = base_penalty * penalty_scale * (1.0 + alpha_slack[idx] / (global_mean_cost + 1e-8))

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Step 6: Targeted ruin-recreate incentive: reward high-potential backbone shortcuts
    # Select skip-1 and skip-2 alternatives around penalized fragments
    recreate_budget = min(ruin_budget, num_nodes)
    for idx in penalized_edges[:recreate_budget]:
        u = u_idx[idx]
        target_v = tour[(idx + 2) % num_nodes]
        alt_cost = edge_distance[u, target_v]
        slack_alt = alt_cost - 0.5 * (min_out[u] + min_out[target_v])

        # Only discount promising connections that respect backbone bounds
        if edge_n_used[u, target_v] < 1.5 and slack_alt < 0.5 * global_mean_cost:
            discount = 0.08 * global_mean_cost / (1.0 + edge_n_used[u, target_v])
            new_val = max(1e-6, updated_edge_distance[u, target_v] - discount)
            updated_edge_distance[u, target_v] = new_val
            updated_edge_distance[target_v, u] = new_val

    return updated_edge_distance
