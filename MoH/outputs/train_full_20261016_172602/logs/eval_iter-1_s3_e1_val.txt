#{Guided local search edge penalty incorporating normalized tour utility, 4-neighborhood curvature context, topological cluster bridging, and frequency-scaled adaptive penalization.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)
    p = np.roll(local_opt_tour, 1)
    w = np.roll(local_opt_tour, -2)

    tour_lens = edge_distance[u, v]
    usage = edge_n_used[u, v]

    # 1. Classical Guided Local Search (GLS) utility
    # Utility = c_{ij} / (1 + count_{ij})
    gls_utility = tour_lens / (1.0 + usage)

    # 2. Multi-scale topological density (k-nearest neighbors)
    k = min(8, n - 1)
    k_nn_dists = np.partition(edge_distance, k, axis=1)[:, 1:k]
    node_scale = np.mean(k_nn_dists, axis=1) + 1e-6
    cluster_scale = np.sqrt(node_scale[u] * node_scale[v])
    bridging_ratio = tour_lens / cluster_scale

    # 3. Extended local context (2-hop neighborhood slack & curvature)
    local_adj_mean = 0.5 * (edge_distance[p, u] + edge_distance[v, w]) + 1e-6
    local_slack = tour_lens / local_adj_mean

    # 4. Global statistics
    mean_l = np.mean(tour_lens) + 1e-6
    std_l = np.std(tour_lens)
    cv = std_l / mean_l

    # Normalized composite utility
    norm_gls = gls_utility / (np.max(gls_utility) + 1e-6)
    norm_bridge = np.tanh(bridging_ratio / (np.median(bridging_ratio) + 1e-6))
    norm_slack = np.clip(local_slack / 2.0, 0.0, 2.0)

    composite_score = 0.5 * norm_gls + 0.3 * norm_bridge + 0.2 * norm_slack

    # Dynamic percentile selection for targeted perturbations
    cutoff_pct = np.clip(70.0 - 20.0 * cv, 40.0, 80.0)
    threshold = np.percentile(composite_score, cutoff_pct)

    # Adaptive penalty scaling proportional to tour edge scale
    base_penalty = 0.3 * mean_l / (1.0 + 0.5 * usage)
    multiplier = np.where(composite_score >= threshold, 1.5 + 0.5 * np.clip(cv, 0.0, 1.0), 0.5)

    penalties = base_penalty * multiplier * composite_score

    np.add.at(updated_edge_distance, (u, v), penalties)
    np.add.at(updated_edge_distance, (v, u), penalties)

    return updated_edge_distance
