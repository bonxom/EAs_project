# ===== Heuristic Code =====
#{A purely functional Guided Local Search penalty heuristic that evaluates tour edge utilities using vectorized roll indexing, selects edges matching the maximum utility within a tight relative tolerance, and symmetrically penalizes them proportional to the mean tour edge length without mutating input state.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy of edge_distance to ensure pure functional execution without side effects
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge endpoints along the local optimal tour using vectorized indexing
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Evaluate tour edge lengths and their historical usage frequencies
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate utility according to the standard Voudouris and Tsang formulation
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Determine the additive penalty scaled by the average edge length of the current tour
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Identify edges that achieve the maximum utility within a narrow tolerance
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically apply penalties to both directed representations of the selected edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
