#{Replace the raw utility with a topological-bottleneck-aware utility that scales edge costs by the endpoint distances relative to the average incident edge lengths in the local tour, applying an edge-adaptive dynamic penalty to break sub-tour symmetries.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Extract edge lengths and usage counts
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Structural context: measure relative incident edge lengths at vertices u and v
    prev_u_idx = np.roll(local_opt_tour, 1)
    prev_lengths = edge_distance[prev_u_idx, u_idx]
    next_lengths = np.roll(tour_lengths, -1)

    # Vertex-local baseline: bottleneck cross-edges stand out prominently against adjacent segments
    local_baseline = 0.5 * (prev_lengths + next_lengths) + 1e-8
    relative_prominence = tour_lengths / local_baseline

    # Topological bottleneck-aware utility
    utilities = (tour_lengths * (1.0 + 0.5 * relative_prominence)) / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    threshold = 0.95 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_lengths = tour_lengths[selected_mask]
    sel_counts = tour_counts[selected_mask]

    # Adaptive dynamic penalty proportional to individual edge weight and inverse of usage
    penalties = sel_lengths * (0.25 + 0.15 / (1.0 + sel_counts))

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
