# ===== Heuristic Code =====
#{Update edge penalties along the tour by evaluating geometric bridging ratios against endpoint k-NN neighborhood scales combined with adjacent predecessor-successor slack ratios to aggressively penalize bottleneck bridge edges while discounting frequently used connections.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Compute local neighborhood scale using k-NN average distance per node
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_scale = np.mean(sorted_dists[:, 1:k + 1], axis=1) + 1e-8

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Predecessor and successor edge lengths along the tour
    pred_lengths = np.roll(tour_edge_lengths, 1)
    succ_lengths = np.roll(tour_edge_lengths, -1)
    adj_baseline = 0.5 * (pred_lengths + succ_lengths) + 1e-8

    # Local adjacent edge slack ratio: detects spikes relative to adjacent tour edges
    slack_ratio = tour_edge_lengths / adj_baseline

    # Geometric bridging ratio: edge length normalized by geometric mean of endpoint scales
    geom_scale = np.sqrt(k_nn_scale[u_idx] * k_nn_scale[v_idx])
    bridging_ratio = tour_edge_lengths / geom_scale

    # Global relative length to tour average
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-8
    global_relative_length = tour_edge_lengths / avg_tour_length

    # Combined bridge bottleneck score
    bottleneck_score = 0.5 * bridging_ratio + 0.5 * slack_ratio

    # Frequency dampening (usage factor)
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.3 * usage)

    # Sigmoid-based rank penalty balanced with structural bottleneck penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (global_relative_length - 1.0)))
    bridge_penalty = bottleneck_score * usage_factor

    freq_mean = np.mean(usage)
    w_global = 0.4 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = (w_global * rank_penalty + w_local * bridge_penalty) * geom_scale

    # Amplify penalties on top quartile bottleneck edges
    threshold = np.percentile(combined_penalty, 75)
    combined_penalty[combined_penalty >= threshold] *= 1.5

    # Apply symmetric additive penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
