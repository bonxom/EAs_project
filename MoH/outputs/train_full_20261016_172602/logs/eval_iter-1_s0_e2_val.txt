# ===== Heuristic Code =====
#{This heuristic identifies central bottleneck nodes based on cumulative historical usage across adjacent edges and penalizes all incident tour edges proportional to their joint node congestion and length, effectively dispersing traffic away from recurring structural focal points.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    # Identify tour edges
    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    avg_tour_edge = np.mean(tour_lengths) if tour_lengths else 1.0

    # Compute node central congestion based on cumulative incident edge usage
    node_usage = np.sum(edge_n_used, axis=1)
    mean_node_usage = np.mean(node_usage)
    std_node_usage = np.std(node_usage) + 1e-6
    node_centrality = (node_usage - mean_node_usage) / std_node_usage

    # Calculate utilities based on edge length and joint bottleneck centrality of endpoint nodes
    utilities = []
    for u, v in tour_edges:
        centrality_factor = 1.0 + max(0.0, float(node_centrality[u] + node_centrality[v]))
        util = (edge_distance[u, v] * centrality_factor) / (1.0 + edge_n_used[u, v])
        utilities.append(util)

    max_utility = max(utilities) if utilities else 0.0
    penalty_scale = 0.3 * avg_tour_edge

    # Penalize the most critical bottleneck edges in the local optimum
    for (u, v), util in zip(tour_edges, utilities):
        if util >= 0.75 * max_utility:
            add_penalty = penalty_scale * (util / (max_utility + 1e-8))
            updated_edge_distance[u, v] += add_penalty
            updated_edge_distance[v, u] += add_penalty

    return updated_edge_distance
