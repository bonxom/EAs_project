# ===== Heuristic Code =====
# {This algorithm adaptively modulates the penalty cardinality k and scales edge penalties by combining stagnation overlap, node entropy-based rigidity, and consecutive trapped subpath detection.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # Apply historical decay
    edge_n_used *= 0.95

    # Calculate edge utilities along the tour
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    # Calculate stagnation indicators
    overlap_ratio = np.mean(tour_used > 1.0)
    avg_tour_usage = np.mean(tour_used)
    max_used = np.max(edge_n_used) + 1e-12
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (avg_tour_usage / max_used)

    # Calculate node entropy and structural rigidity
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)
    rigidity_variance = np.var(entropy_rigidity)

    # Dynamically scale k based on stagnation and rigidity variance
    k_ratio = np.clip(0.10 + 0.15 * overlap_ratio + 0.10 * rigidity_variance, 0.05, 0.35)
    k = max(1, int(np.ceil(k_ratio * num_nodes)))

    top_k_indices = np.argsort(utilities)[-k:]
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Subpath trappedness: mark top-k membership and compute consecutive trappedness
    is_top_k = np.zeros(num_nodes, dtype=bool)
    is_top_k[top_k_indices] = True
    consecutive_trapped = (
        is_top_k.astype(float)
        + 0.5 * np.roll(is_top_k, 1).astype(float)
        + 0.5 * np.roll(is_top_k, -1).astype(float)
    )

    # Node centrality
    node_centrality = np.sum(edge_n_used, axis=1)
    norm_centrality = node_centrality / (np.max(node_centrality) + 1e-12)

    # Adaptive penalty scaling
    avg_dist = np.mean(edge_distance)
    adaptive_lambda = (0.15 + 0.20 * np.tanh(stagnation_factor)) * avg_dist

    # Apply targeted penalties to selected trapped edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        c_mod = 0.3 * (norm_centrality[u] + norm_centrality[v])
        r_mod = 0.3 * (entropy_rigidity[u] + entropy_rigidity[v])
        subpath_mod = 0.4 * consecutive_trapped[idx]
        modulation = 1.0 + c_mod + r_mod + subpath_mod

        penalty = adaptive_lambda * edge_n_used[u, v] * modulation
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
