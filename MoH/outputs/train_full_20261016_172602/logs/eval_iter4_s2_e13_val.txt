# ===== Heuristic Code =====
#{Multi-hop skip-connection geometric evaluation across horizons k=2, 3, and 4 to compute composite hull-slack and path detour metrics, guiding logarithmic frequency-scaled penalties on structural bottlenecks and multi-horizon shortcut discounts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Decay frequency memory
    edge_n_used *= 0.96

    u_idx = tour
    v_idx = np.roll(tour, -1)
    d_uv = edge_distance[u_idx, v_idx]

    # Multi-hop skip evaluation (k = 2, 3, 4 hops ahead)
    # k=2: u -> v -> w
    w2_idx = np.roll(tour, -2)
    d_vw2 = edge_distance[v_idx, w2_idx]
    d_uw2 = edge_distance[u_idx, w2_idx]
    detour_k2 = (d_uv + d_vw2) / (d_uw2 + 1e-8)

    denom_k2 = 2.0 * d_uv * d_vw2 + 1e-8
    cos_turn_k2 = np.clip((d_uv**2 + d_vw2**2 - d_uw2**2) / denom_k2, -1.0, 1.0)
    acute_deficit_k2 = 1.0 - cos_turn_k2

    # k=3: u -> ... -> w3 (accumulating path slack)
    w3_idx = np.roll(tour, -3)
    d_w2w3 = edge_distance[w2_idx, w3_idx]
    path_len_k3 = d_uv + d_vw2 + d_w2w3
    d_uw3 = edge_distance[u_idx, w3_idx]
    detour_k3 = path_len_k3 / (d_uw3 + 1e-8)

    # k=4: u -> ... -> w4
    w4_idx = np.roll(tour, -4)
    d_w3w4 = edge_distance[w3_idx, w4_idx]
    path_len_k4 = path_len_k3 + d_w3w4
    d_uw4 = edge_distance[u_idx, w4_idx]
    detour_k4 = path_len_k4 / (d_uw4 + 1e-8)

    # Composite multi-hop hull slack: weighted combination capturing local turns and wide loop concavities
    multi_hop_detour = 0.50 * detour_k2 + 0.30 * detour_k3 + 0.20 * detour_k4
    geom_inefficiency = multi_hop_detour * (1.0 + 0.40 * acute_deficit_k2)

    # Smooth spatial metric across edge incident vertices
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k_penalty = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k_penalty:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize inefficient, heavily utilized tour edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Reward shortcuts across multi-hop skip connections that exhibit high detour ratios
    # 2-hop shortcuts
    top_k2_shortcuts = np.argsort(detour_k2)[-k_penalty:]
    for idx in top_k2_shortcuts:
        u, w = u_idx[idx], w2_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    # 3-hop macro shortcuts
    k_macro = max(1, int(np.ceil(0.07 * num_nodes)))
    top_k3_shortcuts = np.argsort(detour_k3)[-k_macro:]
    for idx in top_k3_shortcuts:
        u, w = u_idx[idx], w3_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 0.5:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
