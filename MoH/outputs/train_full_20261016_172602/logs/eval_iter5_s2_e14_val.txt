# ===== Heuristic Code =====
#{Guided Local Search with multi-scale geometric penalization and variable-depth ruin-and-recreate chords that identify and discount high-gain skip-1 and skip-2 uncrossing shortcuts across entangled sub-tours.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Node sequences along the current local optimum tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Decay history of penalized edges
    edge_n_used *= 0.96

    # 1-step and 2-step tour segment lengths
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wx = edge_distance[w_idx, x_idx]

    # Chord lengths for variable-depth shortcutting
    d_uw = edge_distance[u_idx, w_idx]  # skip-1 chord
    d_ux = edge_distance[u_idx, x_idx]  # skip-2 chord

    # Geometric sharpness and detour inefficiencies for 2-edge paths (u -> v -> w)
    detour_2step = (d_uv + d_vw) / (d_uw + 1e-8)
    denom_2step = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom_2step, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_2step * (1.0 + 0.5 * acute_deficit)

    # Detour inefficiency for 3-edge paths (u -> v -> w -> x)
    detour_3step = (d_uv + d_vw + d_wx) / (d_ux + 1e-8)

    # GLS Edge utilities: penalize long, geometrically inefficient, low-penalty edges
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Variable-depth shortcut incentives: evaluate skip-1 and skip-2 candidate chords
    gain_skip1 = (d_uv + d_vw) - d_uw
    gain_skip2 = (d_uv + d_vw + d_wx) - d_ux

    # Ruin-and-recreate chord selection
    k_ruin = max(1, int(np.ceil(0.08 * num_nodes)))
    top_skip1 = np.argsort(gain_skip1)[-k_ruin:]
    top_skip2 = np.argsort(gain_skip2)[-k_ruin:]

    # Incentivize promising skip-1 bridging chords
    for idx in top_skip1:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    # Incentivize promising skip-2 bridging chords to break multi-edge serpentine detours
    for idx in top_skip2:
        u, x = u_idx[idx], x_idx[idx]
        freq = edge_n_used[u, x]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, x] = max(1e-6, updated_edge_distance[u, x] - discount)
            updated_edge_distance[x, u] = updated_edge_distance[u, x]

    return updated_edge_distance
