# ===== Heuristic Code =====
#{This algorithm constructs a dual-scale topological tension metric combining Delaunay-proxy 2-hop slack ratios with hyperbolic cluster-bridge potential, dynamically scaled by tour length dispersion and super-linear usage damping to systematically penalize stubborn sub-optimal bridge edges while protecting compact local connections.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Edge indices in the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # 2. Local topological scale estimation (k-NN proxy for local cluster scale)
    k = min(6, n)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_scale = np.mean(sorted_dists[:, 1:k], axis=1)
    cluster_scale = 0.5 * (k_nn_scale[u_idx] + k_nn_scale[v_idx]) + 1e-8

    # 3. Dual-scale tension: Cluster bridge potential + Delaunay-proxy 2-hop slack
    # Bridge potential measures if the edge traverses across sparse cluster boundaries
    bridge_potential = np.tanh(np.maximum(0.0, tour_edge_lengths / cluster_scale - 1.0))

    # 2-hop neighborhood proxy tension
    prev_u = np.roll(local_opt_tour, 1)
    next_v = np.roll(local_opt_tour, -2)
    two_hop_proxy = 0.5 * (edge_distance[prev_u, u_idx] + edge_distance[v_idx, next_v]) + 1e-8
    slack_ratio = tour_edge_lengths / two_hop_proxy
    tension_metric = np.maximum(slack_ratio - 1.0, 0.0) / (slack_ratio + 1.0)

    # 4. Search phase state from tour dispersion (coefficient of variation of tour edges)
    mean_len = np.mean(tour_edge_lengths) + 1e-8
    std_len = np.std(tour_edge_lengths)
    dispersion_state = np.clip(std_len / mean_len, 0.1, 1.5)

    # 5. Super-linear usage damping to severely penalize recurrent pseudo-optimal cycles
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.4 * usage + 0.08 * (usage ** 1.5))

    # Standard normalized GLS utility baseline modulated by topological metrics
    rel_length = tour_edge_lengths / mean_len
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (rel_length - 1.0)))

    # Synthesis of dual-context penalty
    dual_context_penalty = (0.5 * rank_penalty + 0.3 * bridge_potential + 0.2 * tension_metric)
    scaled_penalty = dispersion_state * mean_len * 0.15 * dual_context_penalty * usage_factor

    # Upper-quartile amplification for dominant non-local bridges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_mask = tour_edge_lengths >= threshold
    scaled_penalty[high_mask] *= 1.4

    # Symmetrically accumulate penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), scaled_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), scaled_penalty)

    return updated_edge_distance
