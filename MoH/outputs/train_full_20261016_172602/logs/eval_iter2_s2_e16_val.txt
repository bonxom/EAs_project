# ===== Heuristic Code =====
#{Guided local search penalizes geometrically inefficient, high-utility tour edges while selectively discounting 2-hop shortcut chords only when relocating the bypassed intermediate node yields a positive net tour length improvement estimate.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate shortcut promotion with regret-guided re-insertion check
    removal_savings = (d_uv + d_vw) - d_uw
    candidate_order = np.argsort(removal_savings)[::-1]

    discount_budget = k
    for idx in candidate_order:
        if discount_budget <= 0:
            break
        u, v, w = u_idx[idx], v_idx[idx], w_idx[idx]
        save = removal_savings[idx]
        if save <= 0:
            continue

        freq_uw = edge_n_used[u, w]
        if freq_uw >= 1.0:
            continue

        # Evaluate best alternative insertion cost for v into any existing tour edge (a, b) not adjacent to v
        valid_mask = (tour != u) & (tour != v) & (tour != w)
        if not np.any(valid_mask):
            continue

        a_nodes = tour[valid_mask]
        b_nodes = np.roll(tour, -1)[valid_mask]

        insert_costs = edge_distance[a_nodes, v] + edge_distance[v, b_nodes] - edge_distance[a_nodes, b_nodes]
        min_insert_cost = np.min(insert_costs)

        net_gain = save - min_insert_cost
        if net_gain > 0:
            gain_factor = min(1.0, net_gain / (global_mean_dist + 1e-8))
            discount = 0.12 * global_mean_dist * (1.0 / (1.0 + freq_uw)) * (0.5 + 0.5 * gain_factor)
            new_dist = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_dist
            updated_edge_distance[w, u] = new_dist
            discount_budget -= 1

    return updated_edge_distance
