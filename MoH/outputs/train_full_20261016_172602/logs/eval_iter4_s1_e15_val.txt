# ===== Heuristic Code =====
#{A dual-force GLS heuristic that applies a bounded leaky-accumulator memory with geometric global decay to historic perturbations, selectively penalizing critical high-utility tour edges while discounting viable unvisited incident alternatives.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to preserve original data
    updated_edge_distance = np.copy(edge_distance).astype(np.float64)
    n = edge_distance.shape[0]

    # Global geometric dissipation (leaky accumulator) to revert towards base metric and prevent metric explosion
    dissipation_rate = 0.985
    base_costs = edge_distance.astype(np.float64)
    updated_edge_distance = base_costs + dissipation_rate * (updated_edge_distance - base_costs)

    # Extract current tour edges
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = updated_edge_distance[u_idx, v_idx]
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Compute tour edge utility combining normalized cost and sublinear visit frequency
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-np.clip(z_scores, -10.0, 10.0)))
    utilities = (costs * (1.0 + weights)) / (1.0 + np.sqrt(used + 1.0))

    # Identify candidate edges exceeding the median utility threshold
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Calculate bounded penalty increments to avoid runaway metric distortion
    max_util = np.max(utilities) + 1e-8
    penalty_scale = 0.25 * (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + (p_utils / max_util))
    max_penalty_cap = 0.5 * p_costs
    bounded_penalties = np.minimum(penalty_scale, max_penalty_cap)

    # Apply penalties symmetrically
    updated_edge_distance[p_u, p_v] += bounded_penalties
    updated_edge_distance[p_v, p_u] += bounded_penalties

    # Dual-force: Discount promising unvisited/under-explored incident edges of penalized vertices
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_ratio = 0.02

    for u in penalized_nodes:
        unvisited_mask = (edge_n_used[u] == 0)
        unvisited_mask[u] = False
        if np.any(unvisited_mask):
            current_vals = updated_edge_distance[u, unvisited_mask]
            # Ensure discount cannot push edges below 70% of their base cost
            base_vals = base_costs[u, unvisited_mask]
            discounted = np.maximum(current_vals * (1.0 - discount_ratio), 0.70 * base_vals)
            updated_edge_distance[u, unvisited_mask] = discounted
            updated_edge_distance[unvisited_mask, u] = discounted

    # Maintain strict zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
