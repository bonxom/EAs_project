# ===== Heuristic Code =====
#{This strategy replaces arithmetic density averaging with an asymmetric bottleneck density metric that penalizes bridge edges between dense and sparse regions while shielding mutual k-nearest neighbor edges from excessive penalization.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Compute k-nearest neighbor scale and node densities
    k = min(6, n - 1) if n > 1 else 1
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_dists = sorted_dists[:, 1:k+1]
    k_nn_avg = np.mean(k_nn_dists, axis=1) + 1e-6
    node_density = 1.0 / k_nn_avg
    norm_density = node_density / (np.mean(node_density) + 1e-6)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # 2. Asymmetric bottleneck density scaling
    # Penalizes bridge connections between sparse and dense clusters where local moves struggle
    d_u = norm_density[u_idx]
    d_v = norm_density[v_idx]
    density_ratio = np.maximum(d_u / (d_v + 1e-6), d_v / (d_u + 1e-6))
    max_density = np.maximum(d_u, d_v)
    bottleneck_density = max_density * (1.0 + 0.5 * np.log1p(density_ratio))

    # 3. Mutual k-nearest neighborhood shielding flag
    # True nearest-neighbor edges should be protected from excessive penalties
    k_threshold_u = sorted_dists[u_idx, k]
    k_threshold_v = sorted_dists[v_idx, k]
    is_knn_u = tour_edge_lengths <= k_threshold_u
    is_knn_v = tour_edge_lengths <= k_threshold_v
    knn_shield = np.where(is_knn_u & is_knn_v, 0.4, np.where(is_knn_u | is_knn_v, 0.75, 1.25))

    # 4. Length and usage factors
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])

    # 5. Composite local penalty with bottleneck and neighborhood shielding
    local_penalty = relative_length * usage_factor * bottleneck_density * knn_shield

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Scale penalties relative to baseline edge lengths to ensure consistent impact
    penalty_magnitude = combined_penalty * avg_tour_length * 0.1

    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    penalty_magnitude[high_penalty_mask] *= 1.4

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty_magnitude)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty_magnitude)

    return updated_edge_distance
