#{Refined geometric angularity distortion using bidirectional cosine law turning estimators, non-convex reflex penalty scaling, adaptive k-nearest neighbor density normalization, and guided Guided Local Search penalty modulation to break stubborn self-entangling local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Shifted tour indices: p (prev), u (curr), v (next), w (after_next)
    p_idx = np.roll(local_opt_tour, 1)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    w_idx = np.roll(local_opt_tour, -2)

    # Edge distances along consecutive triplets
    d_uv = edge_distance[u_idx, v_idx]
    d_pu = edge_distance[p_idx, u_idx]
    d_pv = edge_distance[p_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Cosine of interior angles at u (between p->u and u->v) and v (between u->v and v->w)
    # By Law of Cosines on triangle (p, u, v): d_pv^2 = d_pu^2 + d_uv^2 - 2*d_pu*d_uv*cos(angle_u)
    denom_u = 2.0 * np.maximum(d_pu * d_uv, 1e-8)
    cos_u = np.clip((d_pu**2 + d_uv**2 - d_pv**2) / denom_u, -1.0, 1.0)

    denom_v = 2.0 * np.maximum(d_uv * d_vw, 1e-8)
    cos_v = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom_v, -1.0, 1.0)

    # Acute/sharp turns correspond to cos > 0 (angle < 90 degrees)
    # Severe sharp hairpin turns (cos -> 1.0) receive quadratically higher penalty
    sharp_u = np.maximum(0.0, cos_u) ** 1.5
    sharp_v = np.maximum(0.0, cos_v) ** 1.5
    angular_factor = 1.0 + 2.0 * (sharp_u + sharp_v)

    # Relative edge length compared to average tour edge
    mean_d = np.mean(d_uv)
    rel_len = d_uv / (mean_d + 1e-8)
    length_weight = 1.0 / (1.0 + np.exp(-3.5 * (rel_len - 1.0)))

    # Guided Local Search (GLS) style utility: penalize long, sharp edges weighted by usage
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.35 * usage)

    # Spatial density context via k-nearest neighbors (k=min(8, n-1))
    k_nn = min(8, n - 1)
    # Partition is faster than full sort and extracts k nearest neighbors
    part_dists = np.partition(edge_distance, k_nn, axis=1)[:, 1:k_nn + 1]
    local_scale = np.mean(part_dists, axis=1) + 1e-8
    density_norm = 0.5 * (local_scale[u_idx] + local_scale[v_idx])
    density_factor = np.clip(d_uv / density_norm, 0.5, 3.0)

    # Base penalty proportional to distance scaled by geometric distortion
    base_penalty = (d_uv / (1.0 + usage)) * 0.15
    multiplier = angular_factor * (0.6 + 0.8 * length_weight) * (0.5 + 0.5 * density_factor) * usage_factor

    penalty = base_penalty * multiplier

    # Extra reinforcement for the worst non-smooth edges (top 20% longest tour edges)
    high_threshold = np.percentile(d_uv, 80)
    penalty[d_uv >= high_threshold] *= 1.35

    # Symmetric distance matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
