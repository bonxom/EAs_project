#{Dynamically scales Guided Local Search penalties using the tour's coefficient of variation and edge spread to heavily penalize bottleneck edges while applying gentle perturbations when edge costs are homogeneous.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)

    # Calculate dispersion via coefficient of variation and normalized range
    cv = std_len / (mean_len + 1e-12)
    min_len = np.min(tour_lengths)
    max_len = np.max(tour_lengths)
    relative_spread = (max_len - min_len) / (mean_len + 1e-12)

    # Adaptive penalty scaling based on edge heterogeneity
    dispersion_factor = np.clip(cv + 0.5 * relative_spread, 0.1, 1.5)
    penalty = dispersion_factor * mean_len

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
