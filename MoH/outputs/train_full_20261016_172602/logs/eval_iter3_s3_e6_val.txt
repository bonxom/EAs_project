#{Guided local search edge penalization scaling distance adjustments via 2-hop detour triangle inequality slack and localized density context, modulated by usage frequency.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u = tour
    v = np.roll(tour, -1)
    prev_u = np.roll(tour, 1)
    next_v = np.roll(tour, -2)

    d_uv = edge_distance[u, v]
    avg_tour_edge = np.mean(d_uv) + 1e-8

    # 2-hop detour slack: compare d(u, v) against shortcuts/detours with predecessor and successor
    # High detour ratio indicates an acute zig-zag edge ripe for 2-opt or 3-opt reconnection
    chord_prev = edge_distance[prev_u, v]
    chord_next = edge_distance[u, next_v]
    base_prev = edge_distance[prev_u, u]
    base_next = edge_distance[v, next_v]

    detour_slack_prev = (base_prev + d_uv) - chord_prev
    detour_slack_next = (d_uv + base_next) - chord_next
    relative_slack = np.maximum(0.0, np.maximum(detour_slack_prev, detour_slack_next)) / avg_tour_edge

    # Relative edge cost normalized to tour average
    rel_cost = d_uv / avg_tour_edge

    # Frequency damping factor following GLS principles
    freq = edge_n_used[u, v]
    freq_damping = 1.0 / (1.0 + 0.35 * freq)

    # Local context: k-NN approximation via top-k partition instead of full sort
    k = min(5, n - 1)
    if k > 0:
        k_smallest = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
        local_scale = np.mean(k_smallest, axis=1) + 1e-8
        context_weight = 0.5 * (d_uv / local_scale[u] + d_uv / local_scale[v])
    else:
        context_weight = rel_cost

    # Combined penalty prioritizing redundant zig-zag edges while preserving necessary bridges
    penalty_magnitude = 0.5 * avg_tour_edge * (0.6 * rel_cost + 0.4 * relative_slack) * np.tanh(context_weight)
    penalty = penalty_magnitude * freq_damping

    # Symmetric distance matrix update
    np.add.at(updated_edge_distance, (u, v), penalty)
    np.add.at(updated_edge_distance, (v, u), penalty)

    return updated_edge_distance
