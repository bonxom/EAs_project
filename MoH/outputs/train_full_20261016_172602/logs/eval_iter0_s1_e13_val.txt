# ===== Heuristic Code =====
#{This algorithm penalizes high-utility tour edges and proactively boosts cross-reconnecting 2-opt and 3-opt shortcut chords created by severing penalized edges via quadratic/geometric distance discounts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    costs = np.array([edge_distance[u, v] for u, v in tour_edges], dtype=np.float64)
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8

    utilities = np.zeros(n, dtype=np.float64)
    for i in range(n):
        u, v = tour_edges[i]
        used = edge_n_used[u, v]
        z_score = (costs[i] - mean_cost) / std_cost
        weight = 1.0 / (1.0 + np.exp(-z_score))
        utilities[i] = (costs[i] * (1.0 + weight)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalized_indices = []

    for idx in range(n):
        if utilities[idx] >= threshold:
            u, v = tour_edges[idx]
            used = edge_n_used[u, v]
            penalty = (costs[idx] / (1.0 + np.sqrt(used + 1.0))) * (1.0 + utilities[idx] / max_util)
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty
            penalized_indices.append(idx)

    # Proactively identify 2-opt reconnecting chords formed by severing pairs of penalized edges
    m = len(penalized_indices)
    for i in range(m):
        idx_a = penalized_indices[i]
        u1, v1 = tour_edges[idx_a]
        for j in range(i + 1, m):
            idx_b = penalized_indices[j]
            # Ensure edges are non-adjacent in the tour
            diff = abs(idx_a - idx_b)
            if diff <= 1 or diff >= n - 1:
                continue
            u2, v2 = tour_edges[idx_b]

            # The two alternative 2-opt reconnecting chord pairs are (u1, u2) and (v1, v2)
            for x, y in ((u1, u2), (v1, v2)):
                used = edge_n_used[x, y]
                if used <= 2:
                    # Apply a geometric reward proportional to combined utilities and inversely to used count
                    relative_saving = (costs[idx_a] + costs[idx_b]) / (2.0 * edge_distance[x, y] + 1e-8)
                    boost = 0.045 * (1.0 / (1.0 + used)) * np.clip(relative_saving, 0.5, 2.0)
                    discount = updated_edge_distance[x, y] * boost
                    updated_edge_distance[x, y] -= discount
                    updated_edge_distance[y, x] -= discount

    return updated_edge_distance
