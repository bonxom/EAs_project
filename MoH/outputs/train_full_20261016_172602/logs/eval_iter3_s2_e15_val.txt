#{Adaptive guided local search with cycle-stagnation memory that dynamically scales the ruin fraction and modulates edge utility penalties using rank-weighted deterministic dispersion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour, dtype=int)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Apply standard memory decay to avoid saturation over long horizons
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour calculation and turn-angle inefficiency
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base frequencies on current tour edges
    base_freq = edge_n_used[u_idx, v_idx]

    # Assess stagnation using frequency distribution variance and max concentration
    mean_freq = np.mean(base_freq)
    var_freq = np.var(base_freq)
    stagnation_ratio = float(np.clip(mean_freq / (1.0 + np.sqrt(var_freq)), 0.0, 1.0))

    # Adaptively scale ruin budget k between 14% and 30% based on detected stagnation
    min_ratio = 0.14
    max_ratio = 0.30
    ruin_fraction = min_ratio + (max_ratio - min_ratio) * stagnation_ratio
    k = max(1, min(num_nodes, int(np.ceil(ruin_fraction * num_nodes))))

    # Deterministic rank-proportional utility evaluation
    base_dist = d_uv
    raw_utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Deterministic pseudo-dispersion perturbation based on node index phase
    deterministic_phase = 0.15 * np.sin(u_idx * 1.61803398875 + v_idx * 0.61803398875)
    perturbed_utilities = raw_utilities * (1.0 + stagnation_ratio * deterministic_phase)

    # Select edges to penalize
    ranked_indices = np.argsort(perturbed_utilities)
    top_k_indices = ranked_indices[-k:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = (0.30 + 0.20 * stagnation_ratio) * global_mean_dist

    # Apply rank-modulated and stagnation-aware penalty
    for rank_pos, idx in enumerate(top_k_indices):
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        normalized_rank = (rank_pos + 1.0) / float(k)
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v]) + (0.5 * stagnation_ratio * normalized_rank)
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]

        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Detour-based shortcut promotion
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    discount_base = (0.10 + 0.05 * stagnation_ratio) * global_mean_dist
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = discount_base / (1.0 + freq)
            new_cost = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_cost
            updated_edge_distance[w, u] = new_cost

    return updated_edge_distance
