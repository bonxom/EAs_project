# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    n_edges = len(utilities)

    mean_util = np.mean(utilities)
    std_util = np.std(utilities)
    cv = std_util / (mean_util + 1e-12)

    # Adaptive quantile: high CV focuses on top outliers; low CV expands selection
    quantile_level = np.clip(100.0 * (1.0 - np.clip(0.05 + 0.15 * (1.0 / (1.0 + cv)), 0.05, 0.25)), 75.0, 95.0)
    cutoff = np.percentile(utilities, quantile_level)

    selected_mask = utilities >= cutoff
    if not np.any(selected_mask):
        selected_mask = utilities >= np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Continuous weighting based on utility excess over cutoff
    max_util = np.max(utilities)
    util_range = np.maximum(max_util - cutoff, 1e-12)
    scaled_penalties = base_penalty * (1.0 + (utilities[selected_mask] - cutoff) / util_range)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += scaled_penalties
    updated_edge_distance[sel_v, sel_u] += scaled_penalties

    return updated_edge_distance
