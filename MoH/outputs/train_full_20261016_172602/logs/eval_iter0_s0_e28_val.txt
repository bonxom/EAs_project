# ===== Heuristic Code =====
#{Guided Local Search with Incident Neighborhood Decay and Clustering Penalty: Identify bottleneck edges with high utility, apply primary penalties along with geometrically decayed incident edge penalties around bottleneck endpoints, and scale penalties by local tour clustering to prevent short-distance detour oscillations.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to avoid modifying original array
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge endpoints along the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Compute tour edge lengths and usage counts
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Standard GLS utility computation
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Filter bottleneck edges near maximum utility
    threshold = 0.95 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Mean tour length and edge clustering metric (ratio of adjacent edge lengths)
    avg_tour_edge = np.mean(tour_lengths)
    prev_lengths = np.roll(tour_lengths, 1)
    next_lengths = np.roll(tour_lengths, -1)
    local_density = (prev_lengths + tour_lengths + next_lengths) / (3.0 * (avg_tour_edge + 1e-9))
    density_scale = 1.0 + 0.5 * np.clip(local_density[selected_mask], 0.5, 2.0)

    # Base penalty scaled by local edge density
    base_penalty = 0.35 * avg_tour_edge * density_scale
    direct_penalty = 0.85 * base_penalty

    # Direct penalty applied to selected bottleneck edges
    for u, v, pen in zip(sel_u, sel_v, direct_penalty):
        updated_edge_distance[u, v] += pen
        updated_edge_distance[v, u] += pen

    # Target incident nodes connected to the bottlenecks
    incident_nodes = np.unique(np.concatenate([sel_u, sel_v]))

    # Distribute an incident penalty with a localized decay factor based on average bottleneck penalty
    avg_incident_pen = 0.15 * np.mean(base_penalty)
    updated_edge_distance[incident_nodes, :] += avg_incident_pen
    updated_edge_distance[:, incident_nodes] += avg_incident_pen

    # Tour-adjacent edges connecting to bottleneck nodes get an extra dampening penalty
    prev_u = np.roll(local_opt_tour, 1)
    tour_incident_mask = np.isin(u_idx, incident_nodes) | np.isin(prev_u, incident_nodes)
    tour_inc_u = prev_u[tour_incident_mask]
    tour_inc_v = u_idx[tour_incident_mask]
    tour_adjacent_pen = 0.25 * avg_incident_pen
    updated_edge_distance[tour_inc_u, tour_inc_v] += tour_adjacent_pen
    updated_edge_distance[tour_inc_v, tour_inc_u] += tour_adjacent_pen

    # Reset diagonal distances to 0.0
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
