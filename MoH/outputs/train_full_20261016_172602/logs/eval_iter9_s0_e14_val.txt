#{Adaptively modulate penalty magnitude using tour edge dispersion metrics and usage saturation, dynamically balancing exploration and matrix distortion through standard deviation and relative edge extremities.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    max_edge = np.max(tour_lengths)

    # Instance-aware adaptive scaling based on dispersion and peak-to-mean ratio
    dispersion_factor = std_edge / (avg_edge + 1e-9)
    extremity_factor = max_edge / (avg_edge + 1e-9)
    mean_usage = np.mean(tour_counts)
    depth_decay = 1.0 / np.sqrt(1.0 + mean_usage)

    # Dynamically scale lambda to prevent premature distortion on high-variance or heavily explored tours
    adaptive_lambda = (0.15 + 0.15 * np.clip(dispersion_factor, 0.2, 1.5) + 0.1 * np.tanh(extremity_factor - 1.0)) * depth_decay
    penalty = adaptive_lambda * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
