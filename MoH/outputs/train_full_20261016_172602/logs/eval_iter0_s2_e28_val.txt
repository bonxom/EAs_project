# ===== Heuristic Code =====
#{An advanced dual-frequency memory strategy that precisely separates fast-decaying recency from persistent frequency, penalizes high-recency cyclic bottlenecks with stagnation-amplified penalties, and symmetrically applies directed exploration discounts to the most promising under-visited k-nearest incident edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # Decouple dual-frequency memory:
    # Long-term frequency is tracked in integer scale, fast recency in fractional scale
    recency = edge_n_used - np.floor(edge_n_used)
    recency *= 0.60  # Fast-decay recency tabu component

    edge_n_used *= 0.985  # Slow-decay cumulative frequency

    tour_dist = edge_distance[u_idx, v_idx]
    tour_freq = edge_n_used[u_idx, v_idx]

    # GLS-style utility prioritizing long, infrequently penalized edges
    utilities = tour_dist / (1.0 + np.floor(tour_freq))

    # Adaptive penalty budget: scale dynamically between 8% and 15% of nodes
    k = max(2, int(np.round(0.10 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        recency[u, v] = min(1.0, recency[u, v] + 0.65)
        recency[v, u] = recency[u, v]

    # Re-embed recency memory into the fractional part
    edge_n_used += 0.08 * recency

    # Tour-level cyclic trap detection
    tour_recency = recency[u_idx, v_idx]
    cyclic_stagnation = np.mean(tour_recency[top_k_indices])

    mean_dist = np.mean(tour_dist)
    adaptive_lambda = (0.25 + 0.20 * np.tanh(2.5 * cyclic_stagnation)) * mean_dist

    # Apply adaptive penalties to bottleneck edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        recency_mult = 1.0 + 2.0 * recency[u, v]
        penalty = adaptive_lambda * (1.0 + np.log1p(tour_freq[idx])) * recency_mult
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Directed exploration: reward / discount top under-visited incident edges
    discount_magnitude = 0.08 * adaptive_lambda
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        for node in (u, v):
            dists = edge_distance[node]
            used = edge_n_used[node]
            # Find eligible incident edges: strictly positive distance and low cumulative usage
            eligible = (dists > 0) & (used < 0.40)
            if np.any(eligible):
                cand_idx = np.where(eligible)[0]
                # Pick the lowest original distance candidate among under-visited
                best_neighbor = cand_idx[np.argmin(dists[cand_idx])]
                disc = discount_magnitude / (1.0 + used[best_neighbor])
                new_d = max(1e-6, updated_edge_distance[node, best_neighbor] - disc)
                updated_edge_distance[node, best_neighbor] = new_d
                updated_edge_distance[best_neighbor, node] = new_d

    return updated_edge_distance
