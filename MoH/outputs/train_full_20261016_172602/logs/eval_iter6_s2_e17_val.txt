# ===== Heuristic Code =====
#{This algorithm implements precision Guided Local Search with exact maximum-utility edge penalization scaled by the current local optimum tour cost, eliminating disruptive edge discounting and preserving metric integrity.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)

    d_edges = edge_distance[u_idx, v_idx]
    penalties = edge_n_used[u_idx, v_idx]

    # Tour length and adaptive penalty parameter lambda (classic GLS formulation)
    tour_cost = np.sum(d_edges)
    lambda_param = 0.25 * (tour_cost / float(num_nodes))

    # Calculate standard GLS utilities
    utilities = d_edges / (1.0 + penalties)
    max_util = np.max(utilities)

    # Identify edges with maximum utility (allowing for slight numerical tolerance)
    target_mask = utilities >= (max_util - 1e-9)

    # In case of ties or very few candidates, penalize all edges achieving maximum utility
    selected_indices = np.where(target_mask)[0]

    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        updated_edge_distance[u, v] += lambda_param
        updated_edge_distance[v, u] += lambda_param

    return updated_edge_distance
