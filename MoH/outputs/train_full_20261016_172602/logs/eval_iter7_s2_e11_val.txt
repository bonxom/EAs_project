# ===== Heuristic Code =====
#{This heuristic detects local search stagnation by monitoring tour repetition frequencies stored within edge memory, adaptively adjusting decay rates and applying tabu-like masks on over-penalized edges to force exploration into orthogonal sub-tours.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Measure tour overlap with memory to detect stagnation
    current_edge_counts = edge_n_used[u_idx, v_idx]
    mean_tour_freq = np.mean(current_edge_counts)
    stagnation_ratio = np.clip(mean_tour_freq / 5.0, 0.0, 1.0)

    # Adaptive decay: slows down to 0.99 under stagnation to maintain memory pressure
    adaptive_decay = 0.95 + 0.04 * stagnation_ratio
    edge_n_used *= adaptive_decay

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Compute base GLS utilities
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Tabu mask: under stagnation, prevent repeatedly penalizing hyper-frequent edges to force orthogonal moves
    if stagnation_ratio > 0.4:
        freq_threshold = np.percentile(base_freq, 75.0)
        tabu_mask = base_freq >= freq_threshold
        utilities[tabu_mask] *= 0.1

    k = max(1, int(np.ceil((0.14 + 0.06 * stagnation_ratio) * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.30 + 0.15 * stagnation_ratio) * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = (0.10 + 0.05 * stagnation_ratio) * global_mean_dist * (1.0 / (1.0 + freq))
            val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = val
            updated_edge_distance[w, u] = val

    return updated_edge_distance
