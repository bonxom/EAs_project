# ===== Heuristic Code =====
# {Vectorized guided local search penalization targeting maximum utility tour edges while propagating fractional context penalties to adjacent predecessor and successor incident edges to disrupt cycling near the local optimum.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Extract directed edge indices along the tour: (u, v)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Compute base edge lengths, usage frequencies, and guided utilities
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Identify high-utility bottleneck edges
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    # Primary edges to penalize
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Predecessor edges incident to u: (pred_u, u)
    pred_u = np.roll(local_opt_tour, 1)[selected_mask]

    # Successor edges incident to v: (v, succ_v)
    succ_v = np.roll(local_opt_tour, -2)[selected_mask]

    # Apply primary symmetric penalties to selected edges
    np.add.at(updated_edge_distance, (sel_u, sel_v), penalty)
    np.add.at(updated_edge_distance, (sel_v, sel_u), penalty)

    # Apply fractional context penalties to adjacent incident edges
    adjacent_penalty = 0.25 * penalty
    np.add.at(updated_edge_distance, (pred_u, sel_u), adjacent_penalty)
    np.add.at(updated_edge_distance, (sel_u, pred_u), adjacent_penalty)
    np.add.at(updated_edge_distance, (sel_v, succ_v), adjacent_penalty)
    np.add.at(updated_edge_distance, (succ_v, sel_v), adjacent_penalty)

    return updated_edge_distance
