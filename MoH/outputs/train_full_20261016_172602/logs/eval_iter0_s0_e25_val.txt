# ===== Heuristic Code =====
# {Dynamic top-k rank-based perturbation dynamically selects candidate bottleneck edges using variance-adapted k and applies penalties inversely scaled by cumulative frequency to preserve structurally necessary edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility calculation
    utilities = tour_lengths / (1.0 + tour_counts)

    n = len(local_opt_tour)
    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    cv = std_len / (mean_len + 1e-8)

    # Dynamically determine k based on the coefficient of variation of edge lengths
    # Higher variance indicates a few prominent bottleneck edges, requiring smaller k; uniform lengths need broader perturbation
    k = int(np.clip(np.round(n * (0.05 + 0.15 * (1.0 / (1.0 + cv)))), 1, max(1, n // 3)))

    # Identify indices of top-k utility edges
    top_k_indices = np.argpartition(utilities, -k)[-k:]

    sel_u = u_idx[top_k_indices]
    sel_v = v_idx[top_k_indices]
    sel_counts = tour_counts[top_k_indices]

    # Scale penalty inversely with logarithmic visit history to avoid over-penalizing structurally essential edges
    base_penalty = 0.3 * mean_len
    penalty_scale = 1.0 / np.log2(2.0 + sel_counts)
    penalties = base_penalty * penalty_scale

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
