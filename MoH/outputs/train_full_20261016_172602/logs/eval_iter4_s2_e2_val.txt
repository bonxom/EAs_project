# ===== Heuristic Code =====
# {Enhance guided edge penalization by synthesizing geometric detour and acute angle deficits with normalized spatial closeness centrality and historical node congestion fields to disproportionately penalize congested central bottlenecks while discounting shortcut chords.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Track triplet vertices along the current tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay historical edge usage to maintain recency
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Invariable Core: geometric detour and acute turn angle deficit
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Spatial closeness centrality calculation: nodes closer to all other nodes have higher centrality
    mean_node_dist = np.mean(edge_distance, axis=1)
    centrality = 1.0 / (mean_node_dist + 1e-8)
    centrality_norm = centrality / (np.mean(centrality) + 1e-8)

    # Congestion field per node derived from cumulative edge usage
    node_usage = np.sum(edge_n_used, axis=1)
    mean_node_usage = np.mean(node_usage) + 1e-8
    congestion_norm = node_usage / mean_node_usage

    # Synthesized node-level centrality & congestion factor for consecutive tour edges
    node_factor = 0.5 * (centrality_norm[u_idx] * congestion_norm[u_idx] + 
                         centrality_norm[v_idx] * congestion_norm[v_idx])

    local_geom = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Combine distance, recency frequency, geometric defect, and topological centrality
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.25 * local_geom) * (1.0 + 0.25 * node_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize inefficient edges with centrality-compounded weights
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.35 * local_geom[idx]
        topo_weight = 1.0 + 0.35 * node_factor[idx]

        penalty = adaptive_lambda * freq_penalty * geom_weight * topo_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Discount promising shortcut chords across high-detour turns
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
