# ===== Heuristic Code =====
#{Replaces monotonic usage damping with an adaptive tabu-decay memory kernel that imposes a sharp short-term recency barrier before decaying into an asymptotic regional-dispersion penalty as edge usage saturates.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Analyze local neighborhood topology
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Sigmoidal edge length penalty relative to average tour edge length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Adaptive Tabu-Decay Memory Kernel:
    # High recency barrier for freshly re-entered edges decaying to an asymptotic threshold
    # that converts from edge-specific tabooing to regional dispersion under stagnation.
    usage = edge_n_used[u_idx, v_idx]
    tabu_barrier = 1.8 * np.exp(-0.75 * usage)
    saturation_dispersion = 0.4 / (1.0 + 0.15 * usage)
    adaptive_memory_factor = tabu_barrier + saturation_dispersion

    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * adaptive_memory_factor * density_factor

    # Balance global rank-based perturbation against local adaptive-memory perturbation
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.08 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Amplified perturbation on the longest edges in the tour
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Symmetrically apply penalties to distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
