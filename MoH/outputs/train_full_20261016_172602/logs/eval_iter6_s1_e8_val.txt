# ===== Heuristic Code =====
#{Adaptively penalize high-utility tour edges above a dynamic percentile threshold based on search stagnation while selectively discounting the nearest geometric neighbors of penalized nodes to focus diversification and avoid metric distortion.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    max_used = np.max(edge_n_used)
    stagnation_ratio = float(np.mean(used)) / (float(max_used) + 1.0) if max_used > 0 else 0.0

    percentile = 70.0 - (stagnation_ratio * 25.0)
    percentile = float(np.clip(percentile, 45.0, 80.0))
    threshold = np.percentile(utilities, percentile)
    penalize_mask = utilities >= threshold

    if not np.any(penalize_mask):
        penalize_mask = utilities == np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]
    max_util = np.max(utilities) + 1e-8

    boost_factor = 1.0 + 0.3 * np.minimum(p_used / (np.mean(p_used) + 1.0), 2.5)
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * boost_factor

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.015 + 0.01 * (1.0 - stagnation_ratio)
    k_candidates = min(5, n - 1)

    for u in penalized_nodes:
        orig_row = edge_distance[u]
        sorted_indices = np.argsort(orig_row)
        valid_targets = []
        for v in sorted_indices:
            if v != u and edge_n_used[u, v] == 0:
                valid_targets.append(v)
                if len(valid_targets) >= k_candidates:
                    break

        if valid_targets:
            tgt = np.array(valid_targets, dtype=np.intp)
            discounts = updated_edge_distance[u, tgt] * decay_ratio
            updated_edge_distance[u, tgt] -= discounts
            updated_edge_distance[tgt, u] = updated_edge_distance[u, tgt]

    return updated_edge_distance
