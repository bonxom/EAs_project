#{A Guided Local Search perturbation strategy that selects maximum utility tour edges, scales penalty additions boundedly, and applies a global exponential evaporation discount over all previously accumulated edge modifications to retain metric geometry.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Create a copy of the edge distance matrix to avoid modifying the input
    updated_edge_distance = np.copy(edge_distance)

    # Step 2: Global memory evaporation to restore original geometry and avoid permanent distortion
    evaporation_rate = 0.985
    updated_edge_distance = updated_edge_distance * evaporation_rate

    # Step 3: Extract tour edges and edge properties
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Step 4: Compute normalized utility balancing edge cost and usage frequency
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    cost_weights = 1.0 / (1.0 + np.exp(-z_scores))

    utilities = (costs * (1.0 + cost_weights)) / (1.0 + used)

    # Step 5: Identify high-utility edges exceeding median utility
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]
    max_util = np.max(utilities) + 1e-8

    # Step 6: Bounded perturbation calculation based on relative utility and usage damping
    bounded_penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Step 7: Apply symmetric penalty updates to the edge distance matrix
    updated_edge_distance[p_u, p_v] += bounded_penalties
    updated_edge_distance[p_v, p_u] += bounded_penalties

    # Ensure zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
