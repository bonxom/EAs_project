# ===== Heuristic Code =====
#{Normalize edge costs by the mean distance of local nearest neighbors of incident vertices to compute utility, selectively penalizing only the most inefficient edges relative to their local sparseness.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    # Calculate the average incident edge weight for each node among its k nearest neighbors
    k = min(max(3, n // 5), n - 1)
    # Sort distances row-wise to get the k nearest non-zero neighbors (excluding self at index 0)
    sorted_dists = np.sort(edge_distance, axis=1)
    local_mean = np.mean(sorted_dists[:, 1:k + 1], axis=1)

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    avg_tour_edge = np.mean(tour_lengths) if len(tour_lengths) > 0 else 1.0

    # Normalize edge costs by the local neighborhood scales of u and v
    utilities = []
    for u, v in tour_edges:
        norm_cost = edge_distance[u, v] / (0.5 * (local_mean[u] + local_mean[v]) + 1e-8)
        util = norm_cost / (1.0 + edge_n_used[u, v])
        utilities.append(util)

    max_utility = max(utilities) if utilities else 0.0
    penalty_scale = 0.3 * avg_tour_edge

    # Apply penalty to edges with utility close to the maximum
    for (u, v), util in zip(tour_edges, utilities):
        if util >= 0.85 * max_utility:
            add_penalty = penalty_scale / (1.0 + edge_n_used[u, v])
            updated_edge_distance[u, v] += add_penalty
            updated_edge_distance[v, u] += add_penalty

    return updated_edge_distance
