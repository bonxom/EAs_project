# ===== Heuristic Code =====
#{Vectorized guided local search update penalizing tour edges that simultaneously exceed the mean tour cost and rank in the top utility tier scaled by frequency-discounted edge penalties.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    avg_tour_cost = np.mean(tour_lengths)
    max_utility = np.max(utilities)

    # Combined selection: edges with above-average cost and top-tier utility
    cost_mask = tour_lengths > avg_tour_cost
    utility_threshold = 0.85 * max_utility
    selected_mask = cost_mask & (utilities >= utility_threshold)

    # Fallback to absolute maximum utility if no edges satisfy both conditions
    if not np.any(selected_mask):
        selected_mask = utilities >= (0.95 * max_utility)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Proportional utility penalty modulated by average edge cost
    penalties = 0.3 * avg_tour_cost * (utilities[selected_mask] / (max_utility + 1e-12))

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
