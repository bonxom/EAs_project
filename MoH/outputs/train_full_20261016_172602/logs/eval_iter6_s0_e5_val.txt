#{A radical structural reformulation that reconstructs base geometric distances by stripping accumulated penalties, calculates normalized multi-attribute utility incorporating local neighborhood centrality and historical frequency, and applies an adaptive, rank-discounted penalty to the highest-utility edges to diversify escape trajectories.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_lengths = updated_edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Reconstruct estimated baseline length to eliminate penalty distortion in utility evaluation
    # GLS standard penalty model: distance = d_0 + lambda * count * avg_d
    mean_curr = np.mean(current_lengths)
    n = len(local_opt_tour)
    scale_factor = mean_curr / (1.0 + 0.1 * np.mean(counts))
    base_lengths = np.maximum(current_lengths - 0.25 * scale_factor * counts, 0.05 * current_lengths)

    # Structural centrality: relative cost against average incident edges of the endpoints
    u_mean = np.mean(updated_edge_distance[u_idx], axis=1)
    v_mean = np.mean(updated_edge_distance[v_idx], axis=1)
    incident_mean = 0.5 * (u_mean + v_mean)
    centrality_factor = base_lengths / (incident_mean + 1e-9)

    # Multi-attribute utility: base cost balanced by usage count and relative centrality
    utilities = (base_lengths * centrality_factor) / (1.0 + counts)

    # Dynamic adaptive penalty scaling based on tour length variance
    tour_std = np.std(base_lengths)
    adaptive_lambda = (0.25 + 0.15 * (tour_std / (mean_curr + 1e-9))) * mean_curr

    # Select top edges to penalize using rank thresholding
    max_util = np.max(utilities)
    threshold = 0.90 * max_util
    selected_mask = utilities >= threshold

    # If no edge meets threshold or degenerate, penalize the single absolute max
    if not np.any(selected_mask):
        selected_mask = utilities == max_util

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Rank-proportional penalty application
    util_norm = utilities[selected_mask] / (max_util + 1e-9)
    penalties = adaptive_lambda * util_norm

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
