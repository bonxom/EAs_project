# ===== Heuristic Code =====
#{This algorithm replaces static k-NN density with an edge-bypass regret metric, penalizing long tour edges that possess cheap 2-opt alternative reconnections scaled inversely by historical edge usage.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_edge_lens = edge_distance[u_idx, v_idx]
    mean_len = np.mean(current_edge_lens)
    std_len = np.std(current_edge_lens) + 1e-6
    norm_len = (current_edge_lens - mean_len) / std_len

    # Identify bypass regret: the relative ease with which an edge (u, v) can be replaced
    # Sample candidate alternative edges to reconstruct a shortcut
    # For each edge e_i = (u_i, v_i), find minimum cost to bridge u_i or v_i to nearby tour positions
    u_dist = edge_distance[u_idx]
    # Small neighborhood bypass cost: second closest tour node for u_idx not equal to v_idx or u_idx predecessor
    pred_idx = np.roll(local_opt_tour, 1)

    # Vectorized computation of alternative bridge costs for u_idx
    # Sort closest nodes to u_idx along tour
    sub_tour_dists = edge_distance[np.ix_(u_idx, local_opt_tour)]
    # Set current neighbors to infinity to find nearest alternative tour connection
    diag_indices = np.arange(n)
    succ_indices = (diag_indices + 1) % n
    pred_indices = (diag_indices - 1) % n

    sub_tour_dists[diag_indices, diag_indices] = np.inf
    sub_tour_dists[diag_indices, succ_indices] = np.inf
    sub_tour_dists[diag_indices, pred_indices] = np.inf

    min_alt_u = np.min(sub_tour_dists, axis=1)

    # Bypass feasibility metric: lower alt cost relative to current length means high substitution regret / high bypass potential
    bypass_ratio = min_alt_u / (current_edge_lens + 1e-6)
    bypass_factor = 1.0 / (1.0 + np.exp(2.0 * (bypass_ratio - 1.0)))

    # Historical edge frequency damping
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.25 * usage)

    # Base guided local search penalty formulation enhanced by regret
    length_rank_penalty = 1.0 / (1.0 + np.exp(-3.0 * norm_len))
    penalty = mean_len * 0.15 * length_rank_penalty * (1.0 + bypass_factor) * usage_factor

    # Top quartile reinforcement
    high_threshold = np.percentile(current_edge_lens, 75)
    penalty[current_edge_lens >= high_threshold] *= 1.35

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
