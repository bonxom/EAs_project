# ===== Heuristic Code =====
#{This algorithm adaptively penalizes maximum-utility tour edges by estimating base metric costs through penalty inversion, scaling the penalty by local variance and search depth to intensely break suboptimal basins.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_lengths = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Estimate original base distance by correcting for prior accumulated penalties
    # This prevents utility distortion over progressive GLS iterations
    mean_length = np.mean(current_lengths)
    std_length = np.std(current_lengths)
    n = len(local_opt_tour)

    # Adaptive penalty scaling parameter lambda based on edge cost distribution
    # Higher dispersion intensifies penalties on outlier long edges
    lambda_factor = 0.3 * (1.0 + (std_length / (mean_length + 1e-12)))
    base_penalty = lambda_factor * (mean_length / float(n))

    estimated_base_lengths = np.maximum(0.0, current_lengths - counts * base_penalty)
    utilities = estimated_base_lengths / (1.0 + counts)

    max_utility = np.max(utilities)
    threshold = 0.999 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Fine-grained intensification: apply scaled penalties tailored to individual edge over-lengths
    edge_penalties = base_penalty * (1.0 + (current_lengths[selected_mask] / (mean_length + 1e-12)))

    updated_edge_distance[sel_u, sel_v] += edge_penalties
    updated_edge_distance[sel_v, sel_u] += edge_penalties

    return updated_edge_distance
