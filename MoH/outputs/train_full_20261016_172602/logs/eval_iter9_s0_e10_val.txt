# ===== Heuristic Code =====
#{A quantile-based selection scheme identifies the top candidate edges according to utility and applies smooth, distribution-scaled penalties to prevent stagnation on utility plateaus without abrupt thresholds.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    # Dynamically select the top quantile of high-utility edges (at least 1 edge, up to top 15%)
    k = max(1, int(np.ceil(0.15 * n_edges)))

    # Vectorized top-k selection via partition to avoid full sort overhead
    partition_idx = np.argpartition(utilities, -k)[-k:]
    top_utilities = utilities[partition_idx]

    min_top_util = np.min(top_utilities)
    max_top_util = np.max(top_utilities)
    util_range = max_top_util - min_top_util

    # Scale penalty smoothly between 0.5 and 1.0 based on relative rank/magnitude in top-k
    weights = np.ones_like(top_utilities) if util_range == 0.0 else 0.5 + 0.5 * ((top_utilities - min_top_util) / util_range)

    avg_edge = np.mean(tour_lengths)
    penalties = 0.3 * avg_edge * weights

    sel_u = u_idx[partition_idx]
    sel_v = v_idx[partition_idx]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
