#{Dual-force guided local search with an evaporative memory framework that adaptively penalizes bottleneck tour edges based on frequency-adjusted cost utility while applying frequency-regulated discounts to under-explored incident alternatives that decay towards metric fidelity as visitation accumulates.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    utilities = (costs * (1.0 + weights)) / (1.0 + used)
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Adaptive penalty scaling inversely proportional to visit count to prevent over-penalization
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Evaporative memory model for incident edges: discount under-explored edges with evaporation
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    base_discount_rate = 0.03

    for u in penalized_nodes:
        deg_used = edge_n_used[u].astype(np.float64)
        valid_mask = np.ones(n, dtype=bool)
        valid_mask[u] = False
        valid_mask[p_v[p_u == u]] = False
        valid_mask[p_u[p_v == u]] = False

        if np.any(valid_mask):
            # As visit frequency grows, bonus evaporates asymptotically: exp(-lambda * used)
            decay_factor = np.exp(-0.25 * deg_used[valid_mask])
            discount = edge_distance[u, valid_mask] * base_discount_rate * decay_factor

            updated_edge_distance[u, valid_mask] -= discount
            updated_edge_distance[valid_mask, u] -= discount

    # Guarantee metric non-negativity
    np.maximum(updated_edge_distance, 0.0, out=updated_edge_distance)

    return updated_edge_distance
