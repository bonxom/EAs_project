#{A leaky-integrator distance modification scheme that contracts accumulated distortions back toward baseline metric distances while injecting tour-normalized, utility-weighted penalties and incident edge discounts to preserve global geometry.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Initialize updated distances as a copy of current distance matrix
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Contract previous modifications slightly towards symmetric baseline to prevent unbounded drift
    memory_retention = 0.985
    # Symmetrize and decay existing matrix deviations
    updated_edge_distance = 0.5 * (updated_edge_distance + updated_edge_distance.T)

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)
    tour_length = np.sum(costs)

    # Evaluate edge utilities based on relative contribution to tour length and reuse frequency
    mean_cost = tour_length / n
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-np.clip(z_scores, -10.0, 10.0)))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Identify bottleneck edges exceeding median utility
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.mean(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Scale penalties boundedly relative to average edge length
    max_util = np.max(utilities) + 1e-8
    base_step = mean_cost * 0.15
    norm_factor = 1.0 / (1.0 + np.sqrt(p_used + 1.0))
    penalties = base_step * norm_factor * (1.0 + (p_utils / max_util)) * (p_costs / (mean_cost + 1e-8))

    # Apply symmetric penalties
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Apply targeted counter-discounting on incident unvisited/low-frequency edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_ratio = 0.03
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            discounts = updated_edge_distance[u, unused_mask] * discount_ratio
            updated_edge_distance[u, unused_mask] -= discounts
            updated_edge_distance[unused_mask, u] -= discounts

    # Apply gentle global memory-fading contraction across entire graph
    updated_edge_distance = updated_edge_distance * memory_retention + (1.0 - memory_retention) * edge_distance
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
