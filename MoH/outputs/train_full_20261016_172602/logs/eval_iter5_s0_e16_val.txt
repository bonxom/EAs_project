# ===== Heuristic Code =====
#{Recreate the edge penalty mechanism by scaling the perturbation inversely with endpoint local nearest-neighbor scales, safeguarding mandatory topological bottleneck edges from excessive over-penalization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Compute guided local search utilities
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Context-aware penalization: assess local neighborhood scale for endpoints
    n = edge_distance.shape[0]
    k = min(3, max(1, n - 1))

    # Partition to find k-nearest neighbor distances for each vertex (ignoring self-loop with infinity)
    masked_dists = np.copy(edge_distance)
    np.fill_diagonal(masked_dists, np.inf)
    knn_dists = np.partition(masked_dists, k - 1, axis=1)[:, :k]
    local_scale = np.mean(knn_dists, axis=1)

    # Incident edge context: protect topological bottleneck edges where local density is low
    edge_scale = 0.5 * (local_scale[sel_u] + local_scale[sel_v])
    global_avg = np.mean(tour_lengths)

    # Modulate penalty inversely with bottleneck ratio (edge_length / local_scale)
    bottleneck_factor = np.clip(edge_scale / (tour_lengths[selected_mask] + 1e-8), 0.2, 1.5)
    penalty = 0.3 * global_avg * bottleneck_factor

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
