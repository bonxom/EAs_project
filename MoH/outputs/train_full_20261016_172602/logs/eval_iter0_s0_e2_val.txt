# ===== Heuristic Code =====
#{Vectorize tour edge extraction to compute edge utilities, determine an adaptive threshold at 85% of maximum utility, and apply simultaneous symmetric penalties scaled by the average tour edge length and inverse visitation counts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n == 0:
        return updated_edge_distance

    u_indices = np.asarray(local_opt_tour)
    v_indices = np.roll(u_indices, -1)

    tour_lengths = edge_distance[u_indices, v_indices]
    tour_counts = edge_n_used[u_indices, v_indices]

    inv_visitation = 1.0 / (1.0 + tour_counts)
    utilities = tour_lengths * inv_visitation
    max_utility = np.max(utilities)

    if max_utility > 0:
        avg_tour_edge = np.mean(tour_lengths)
        threshold = 0.85 * max_utility
        qualifying = utilities >= threshold

        sel_u = u_indices[qualifying]
        sel_v = v_indices[qualifying]
        penalties = (0.25 * avg_tour_edge) * inv_visitation[qualifying]

        np.add.at(updated_edge_distance, (sel_u, sel_v), penalties)
        np.add.at(updated_edge_distance, (sel_v, sel_u), penalties)

    return updated_edge_distance
