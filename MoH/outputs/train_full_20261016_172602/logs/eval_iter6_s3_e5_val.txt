#{Scale-invariant utility-guided penalty targeted strictly at the highest-cost and least-diversified bottleneck edges via partition-based neighborhood density and dynamic thresholding.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    mean_len = np.mean(tour_lengths)
    if mean_len < 1e-12:
        return updated_edge_distance

    # Fast local density estimation using argpartition instead of full sorting
    k = min(5, n - 1)
    small_part = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    node_scale = np.mean(small_part, axis=1) + 1e-8
    edge_scale = 0.5 * (node_scale[u_idx] + node_scale[v_idx])

    # Scale-invariant excess length ratio relative to local context
    relative_strain = tour_lengths / edge_scale

    # Classical GLS utility formulation modulated by spatial strain
    usage = edge_n_used[u_idx, v_idx]
    gls_utility = (tour_lengths / (1.0 + usage)) * (relative_strain ** 0.5)

    # Selective penalization: strictly target the top quartile of bottleneck edges
    util_threshold = np.percentile(gls_utility, 75)
    target_mask = gls_utility >= util_threshold

    if np.any(target_mask):
        # Scale-adaptive penalty step calibrated against the average tour edge length
        target_u = u_idx[target_mask]
        target_v = v_idx[target_mask]

        # Penalties scaled to distance matrix units to ensure invariance across problem scales
        penalty_magnitudes = 0.15 * mean_len * (1.0 + (gls_utility[target_mask] / (np.max(gls_utility) + 1e-8)))

        np.add.at(updated_edge_distance, (target_u, target_v), penalty_magnitudes)
        np.add.at(updated_edge_distance, (target_v, target_u), penalty_magnitudes)

    return updated_edge_distance
