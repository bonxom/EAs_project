# ===== Heuristic Code =====
# {Dynamically scale the adaptive penalty and shortcut discount using a search-stagnation metric derived from tour edge re-use and node centrality, while preserving geometric turn-detour inefficiency evaluations.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Trace edge history and decay
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric turn deficit and detour ratio calculation
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Measure stagnation through frequency of current tour edges
    mean_tour_freq = np.mean(base_freq)
    max_tour_freq = np.max(base_freq)
    stagnation_factor = mean_tour_freq + 0.3 * max_tour_freq

    # Spatial centrality: ratio of global mean distance to node-mean distance
    global_mean_dist = np.mean(edge_distance)
    node_mean_dist = np.mean(edge_distance, axis=1)
    centrality = global_mean_dist / (node_mean_dist + 1e-8)
    u_centrality = 0.5 * (centrality[u_idx] + centrality[v_idx])

    # Dynamic penalty scaling via hyperbolic tangent modulation
    adaptive_lambda = (0.18 + 0.22 * np.tanh(stagnation_factor)) * global_mean_dist

    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor) * (0.8 + 0.4 * u_centrality)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight * (0.8 + 0.4 * u_centrality[idx])
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Dual-action shortcut incentives modulated by stagnation
    shortcut_incentive = (0.08 + 0.08 * np.tanh(stagnation_factor)) * global_mean_dist
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = shortcut_incentive * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
