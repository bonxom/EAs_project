# ===== Heuristic Code =====
#{Adaptive stagnation-aware percentile and Pareto-dominance thresholding dynamically modulates the set of penalized tour edges, escalating from sharp bottleneck disruptions to deep multi-edge basin-escape perturbations as edge reuse intensifies.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Calculate stagnation metric based on tour edge reuse frequency and tour cost variance
    mean_used = np.mean(used)
    max_used = np.max(used)
    stagnation_ratio = mean_used / (max_used + 1.0)

    # Cost-frequency utility calculation with non-linear edge weight scaling
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + np.sqrt(used + 1.0))

    # Adaptive percentile cutoff: narrow (top ~10-15%) under low stagnation; wide (up to ~60%) under high stagnation
    target_percentile = np.clip(85.0 - 50.0 * stagnation_ratio, 35.0, 92.0)
    percentile_threshold = np.percentile(utilities, target_percentile)

    # Approximate Pareto dominance: edges that have both high relative cost and high reuse
    cost_rank = np.argsort(np.argsort(costs)) / float(n)
    used_rank = np.argsort(np.argsort(used)) / float(n)
    pareto_score = 0.6 * cost_rank + 0.4 * used_rank
    pareto_threshold = np.percentile(pareto_score, target_percentile)

    # Edge must meet either the dynamic utility percentile or the Pareto dominance criterion
    penalize_mask = (utilities >= percentile_threshold) | (pareto_score >= pareto_threshold)

    # Fallback: guarantee at least the single worst bottleneck edge is penalized
    if not np.any(penalize_mask):
        penalize_mask[np.argmax(utilities)] = True

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    max_util = np.max(utilities) + 1e-8
    stagnation_multiplier = 1.0 + 0.75 * stagnation_ratio
    penalties = (p_costs / (1.0 + np.log1p(p_used))) * (1.0 + p_utils / max_util) * stagnation_multiplier

    # Apply symmetric edge penalties
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Discount unused incident edges to guide reconstruction toward novel search spaces
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02 + 0.03 * stagnation_ratio

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
