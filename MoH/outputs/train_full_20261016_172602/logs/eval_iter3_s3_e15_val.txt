#{We implement a dual-horizon frequency memory mechanism that decomposes edge usage into short-term deviations and long-term accumulation, dynamically modulating penalties through tour-length variance to balance exploratory anti-clustering and targeted exploitation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Baseline k-nearest-neighbor density representation
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Trajectory variance metric to balance exploration vs exploitation
    tour_length_std = np.std(tour_edge_lengths)
    coeff_variation = tour_length_std / (avg_tour_length + 1e-6)

    # Dual-horizon usage memory extraction
    edge_counts = edge_n_used[u_idx, v_idx]
    long_term_global = edge_counts
    # Short-term recency proxy based on local usage relative to current tour usage mean
    mean_tour_usage = np.mean(edge_counts)
    short_term_local = np.maximum(0.0, edge_counts - mean_tour_usage)

    # Non-linear dual-scale frequency dampener
    # As variation increases, shift weight towards anti-clustering short-term penalties; otherwise prioritize long-term dispersion
    explore_weight = 1.0 / (1.0 + np.exp(-5.0 * (coeff_variation - 0.5)))
    dampener_long = 1.0 / (1.0 + 0.15 * np.log1p(long_term_global))
    dampener_short = np.exp(-0.4 * short_term_local)

    adaptive_usage_factor = (1.0 - explore_weight) * dampener_long + explore_weight * dampener_short

    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    local_penalty = relative_length * adaptive_usage_factor * density_factor

    w_global = 0.6 / (1.0 + 0.1 * mean_tour_usage)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Outlier thresholding
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
