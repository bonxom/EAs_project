# ===== Heuristic Code =====
#{Filter edges exceeding the 75th length percentile and penalize them using multi-scale bridging normalized by geometric cluster density with sub-linear usage decay to prevent penalty dilution.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)

    tour_lens = edge_distance[u, v]
    usage = edge_n_used[u, v]

    # Strict elite rank-based thresholding (75th percentile of tour edge length)
    length_threshold = np.percentile(tour_lens, 75.0)
    eligible_mask = tour_lens >= length_threshold

    if np.any(eligible_mask):
        k = min(6, n - 1)
        k_nn_dists = np.partition(edge_distance, k, axis=1)[:, 1:k]
        node_scale = np.mean(k_nn_dists, axis=1) + 1e-6
        cluster_scale = np.sqrt(node_scale[u] * node_scale[v])

        # Bridging metric scaled relative to geometric mean cluster density
        bridging_ratio = tour_lens / cluster_scale

        # Sub-linear usage decay modulation
        usage_factor = 1.0 / (1.0 + 0.35 * np.sqrt(usage))

        # Penalties computed strictly on eligible long edges
        penalties = np.zeros(n, dtype=edge_distance.dtype)
        p_u = u[eligible_mask]
        p_v = v[eligible_mask]

        # Combine normalized bridging metric with usage factor
        base_penalties = (bridging_ratio[eligible_mask] * cluster_scale[eligible_mask]) * usage_factor[eligible_mask]
        penalties[eligible_mask] = base_penalties

        np.add.at(updated_edge_distance, (p_u, p_v), base_penalties)
        np.add.at(updated_edge_distance, (p_v, p_u), base_penalties)

    return updated_edge_distance
