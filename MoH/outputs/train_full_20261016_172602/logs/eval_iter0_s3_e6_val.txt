# ===== Heuristic Code =====
# {Dual-stage utilization-damped penalty strategy that combines O(n*k) partition-based local density estimation, dynamic search-history adaptation, and sparsity-aware basin-breaking boosts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)
    p = np.roll(local_opt_tour, 1)
    w = np.roll(local_opt_tour, -2)

    tour_lens = edge_distance[u, v]
    usage = edge_n_used[u, v]

    # Efficient O(n*k) local neighborhood scale via partition
    k = min(6, n - 1)
    k_nn_dists = np.partition(edge_distance, k, axis=1)[:, 1:k]
    node_scale = np.mean(k_nn_dists, axis=1) + 1e-6
    pair_scale = 0.5 * (node_scale[u] + node_scale[v])

    # Normalized length metrics
    mean_len = np.mean(tour_lens) + 1e-6
    std_len = np.std(tour_lens)
    cv = std_len / mean_len
    rel_len = tour_lens / mean_len

    # Topological context: slack ratio and relative density ratio
    adj_mean = 0.5 * (edge_distance[p, u] + edge_distance[v, w]) + 1e-6
    slack_ratio = tour_lens / adj_mean
    sparsity_ratio = tour_lens / pair_scale

    # Global sigmoid rank penalty and local context penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (rel_len - 1.0)))
    local_penalty = 0.6 * rel_len + 0.25 * slack_ratio + 0.15 * np.tanh(sparsity_ratio - 1.0)

    # Dynamic exploration/exploitation weighting adapted to search history
    freq_mean = np.mean(usage)
    w_global = 0.65 / (1.0 + 0.08 * freq_mean)
    w_local = 1.0 - w_global
    base_penalty = w_global * rank_penalty + w_local * local_penalty

    # Damping by edge usage to promote exploration of unvisited alternatives
    usage_damping = 1.0 / (1.0 + 0.3 * usage)
    penalties = base_penalty * usage_damping

    # Adaptive sparsity-aware thresholding to break stubborn basin attractors
    cutoff_pct = np.clip(75.0 - 15.0 * cv, 50.0, 85.0)
    threshold = np.percentile(base_penalty, cutoff_pct)
    boost_mask = base_penalty >= threshold

    # Boost factor scales with edge sparsity and cumulative visit frequency
    basin_boost = 1.25 + 0.2 * np.clip(sparsity_ratio[boost_mask] - 1.0, 0.0, 2.0) + 0.05 * np.log1p(usage[boost_mask])
    penalties[boost_mask] *= basin_boost

    np.add.at(updated_edge_distance, (u, v), penalties)
    np.add.at(updated_edge_distance, (v, u), penalties)

    return updated_edge_distance
