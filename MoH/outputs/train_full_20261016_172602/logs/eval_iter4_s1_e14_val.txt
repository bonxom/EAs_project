# ===== Heuristic Code =====
#{A dual-force guided local search that penalizes high-utility tour edges while applying rank-biased topological discounting to incident k-nearest neighbor edges scaled by the length advantage over the penalized edge.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # 1. Edge cost & frequency evaluation along current local optimal tour
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # 2. Penalize high-utility edges symmetrically
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # 3. Rank-biased topological discounting on candidate neighbors of penalized endpoints
    k = min(10, n - 1)
    if k > 0:
        # Map each penalized node to the maximum penalized cost of its incident penalized edges
        node_pen_cost = {}
        for u, v, cost in zip(p_u, p_v, p_costs):
            if u not in node_pen_cost or cost > node_pen_cost[u]:
                node_pen_cost[u] = cost
            if v not in node_pen_cost or cost > node_pen_cost[v]:
                node_pen_cost[v] = cost

        rank_weights = 1.0 / np.arange(1, k + 1, dtype=np.float64)
        base_rate = 0.05

        for u, base_cost in node_pen_cost.items():
            row_dist = edge_distance[u].copy()
            row_dist[u] = np.inf
            # Exclude currently connected tour neighbors from bonus discounting
            # (Tour indices of u in local_opt_tour)
            # Find candidate k-nearest neighbors
            candidate_indices = np.argpartition(row_dist, k)[:k]
            sorted_candidates = candidate_indices[np.argsort(row_dist[candidate_indices])]

            c_dists = edge_distance[u, sorted_candidates].astype(np.float64)
            c_used = edge_n_used[u, sorted_candidates].astype(np.float64)

            # Length ratio reward: shorter candidate edges relative to penalized cost receive higher boost
            length_ratios = np.clip(base_cost / (c_dists + 1e-8), 0.5, 3.0)
            usage_damping = 1.0 / (1.0 + np.sqrt(c_used))

            discounts = base_rate * rank_weights * length_ratios * usage_damping * c_dists

            # Symmetrically apply topological discount without letting distance drop to zero
            new_vals = np.maximum(updated_edge_distance[u, sorted_candidates] - discounts, 1e-6)
            updated_edge_distance[u, sorted_candidates] = new_vals
            updated_edge_distance[sorted_candidates, u] = new_vals

    return updated_edge_distance
