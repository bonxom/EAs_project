# ===== Heuristic Code =====
#{Scale Guided Local Search penalties to physical distance units using a scale-invariant formulation proportional to average tour edge length, normalized cost, and inverse edge usage frequency.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_edge = np.mean(tour_edge_lengths)
    max_tour_edge = np.max(tour_edge_lengths)

    # Scale-invariant normalized edge cost factor
    normalized_cost = tour_edge_lengths / (max_tour_edge + 1e-8)

    # Inverse visitation frequency factor to penalize heavily-used edges less over time
    usage_factor = 1.0 / (1.0 + edge_n_used[u_idx, v_idx])

    # Guided Local Search parameter lambda ensuring penalties match dimensional units of distance
    gls_lambda = 0.3
    edge_penalties = gls_lambda * avg_tour_edge * normalized_cost * usage_factor

    # Symmetrically update distances for edges in the local optimum tour
    np.add.at(updated_edge_distance, (u_idx, v_idx), edge_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), edge_penalties)

    return updated_edge_distance
