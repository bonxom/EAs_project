# ===== Heuristic Code =====
#{A vectorized utility-guided edge penalization scheme that cools existing accumulated penalties through geometric discounting across the entire matrix while selectively applying fresh adaptive penalties to maximal utility edges of the current local optimum.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to ensure input is not directly modified
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive node pairs along the tour via vectorized shift
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utilities relative to their usage frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Estimate and cool/evaporate accumulated past penalties using edge_n_used guidance
    # Geometric decay factor for past penalties
    decay_rate = 0.95
    penalty_scale = 0.3 * avg_edge

    # Estimated current penalty matrix based on accumulated edge usage
    # Apply geometric cooling across non-zero visited edges to prevent permanent metric distortion
    penalty_mask = edge_n_used > 0
    cooling_delta = (1.0 - decay_rate) * penalty_scale * (edge_n_used / (1.0 + edge_n_used))
    updated_edge_distance[penalty_mask] = np.maximum(
        edge_distance[penalty_mask] - cooling_delta[penalty_mask],
        0.05 * edge_distance[penalty_mask]
    )

    # Identify and symmetrically penalize edges meeting the maximum utility threshold
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply discrete penalty increments symmetrically to high-utility edges
    updated_edge_distance[sel_u, sel_v] += penalty_scale
    updated_edge_distance[sel_v, sel_u] += penalty_scale

    # Maintain zero diagonal for valid TSP distance matrix structure
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
