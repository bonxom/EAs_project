#{Refine the Guided Local Search penalty matrix by replacing global average distance scaling with edge-specific baseline distance scaling modulated by node centrality and entropy gating, preserving fine-grained geometric structure.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    """
    Update the edge distance matrix using edge-specific baseline distance scaling
    combined with dynamic node centrality and entropy-gated usage penalties.
    """
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    # Identify edges in the local optimal tour
    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % num_nodes]) for i in range(num_nodes)]

    # Calculate utility values using edge-specific baseline distance and usage
    utilities = np.array([edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in tour_edges])
    max_util = np.max(utilities)

    # Increment usage count for edges achieving maximum utility
    for idx, (u, v) in enumerate(tour_edges):
        if np.isclose(utilities[idx], max_util):
            edge_n_used[u, v] += 1
            edge_n_used[v, u] += 1

    # Compute normalized node centrality based on edge usage
    node_centrality = np.sum(edge_n_used, axis=1)
    max_centrality = np.max(node_centrality) + 1e-12
    norm_centrality = node_centrality / max_centrality
    centrality_field = norm_centrality[:, None] + norm_centrality[None, :]

    # Compute entropy gate based on the dispersion of edge usage per node
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_gate = 1.0 - (node_entropy / max_entropy)
    gate_matrix = entropy_gate[:, None] + entropy_gate[None, :]

    # Edge-specific baseline scaling factor (lambda = 0.25 * edge_distance)
    # Scales penalty relative to individual edge costs rather than global average
    edge_lambda = 0.25 * edge_distance
    penalty_matrix = edge_lambda * edge_n_used * (1.0 + 0.5 * (centrality_field + gate_matrix))

    updated_edge_distance += penalty_matrix
    return updated_edge_distance
