# ===== Heuristic Code =====
# {Penalize high-utility tour edges and apply directed potential discounts exclusively to the lowest-cost non-tour k-nearest neighbor edges incident to penalized vertices to facilitate bridging broken tour segments.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    tour_prev = np.roll(u_idx, 1)

    k_cand = min(5, n - 3)
    if k_cand > 0:
        base_discount = 0.04
        for u in penalized_nodes:
            # Exclude self and direct tour neighbors from shortcut candidates
            orig_row = edge_distance[u].copy()
            orig_row[u] = np.inf

            # Find tour neighbors of u
            u_pos = np.where(u_idx == u)[0][0]
            nxt = v_idx[u_pos]
            prv = tour_prev[u_pos]
            orig_row[nxt] = np.inf
            orig_row[prv] = np.inf

            # Select k lowest original-cost non-tour candidate shortcut edges
            best_candidates = np.argpartition(orig_row, k_cand)[:k_cand]

            # Apply targeted discount weighted by usage frequency and cost efficiency
            cand_used = edge_n_used[u, best_candidates].astype(np.float64)
            cand_discount = base_discount / (1.0 + np.sqrt(cand_used))

            reduction = updated_edge_distance[u, best_candidates] * cand_discount
            updated_edge_distance[u, best_candidates] -= reduction
            updated_edge_distance[best_candidates, u] = updated_edge_distance[u, best_candidates]

    return updated_edge_distance
