# ===== Heuristic Code =====
# {An adaptive multi-timescale memory strategy that penalizes high-utility tour edges using decoupled recency fatigue and cumulative frequency, while vectorially rewarding promising unvisited incident alternatives based on search stagnation depth.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    freq = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Estimate search progression / stagnation depth from overall visitation concentration
    total_calls = np.max(edge_n_used)
    stagnation_ratio = float(np.mean(freq)) / (float(total_calls) + 1.0)
    stagnation_factor = 1.0 / (1.0 + np.exp(-6.0 * (stagnation_ratio - 0.4)))

    # Dual-timescale utility: short-term recency fatigue versus long-term frequency
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_costs = (costs - mean_cost) / std_cost
    cost_weights = 1.0 / (1.0 + np.exp(-z_costs))

    # Recency proxy: edges appearing in consecutive cycles accumulate higher relative local counts
    recency_fatigue = freq / (np.median(freq) + 1.0)
    utilities = costs * (1.0 + cost_weights) * (1.0 + 0.5 * recency_fatigue) / (1.0 + np.log1p(freq))

    # Adaptive cutoff: shift from conservative top-quartile to broader half as stagnation grows
    quantile_cutoff = 0.75 - 0.25 * stagnation_factor
    cutoff_val = np.quantile(utilities, quantile_cutoff)
    penalize_mask = utilities >= cutoff_val

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_freq = freq[penalize_mask]
    p_recency = recency_fatigue[penalize_mask]

    # Asymmetric penalty scaling: combine short-term fatigue diversion with long-term consolidation
    penalty_scale = 0.25 + 0.35 * stagnation_factor
    edge_penalties = p_costs * penalty_scale * (1.0 + 0.2 * p_recency) / (1.0 + np.sqrt(p_freq))

    # Apply symmetric penalties on the selected subset
    updated_edge_distance[p_u, p_v] += edge_penalties
    updated_edge_distance[p_v, p_u] += edge_penalties

    # Vectorized compensatory discount on unexplored / low-frequency incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    sub_used = edge_n_used[penalized_nodes, :]
    sub_dist = updated_edge_distance[penalized_nodes, :]

    # Identify least-visited incident candidate alternatives excluding self-loops
    min_incident_freq = np.min(sub_used + np.eye(sub_dist.shape[1])[None, :, :].repeat(len(penalized_nodes), axis=0) * 1e9, axis=1, keepdims=True)
    discount_mask = (sub_used <= min_incident_freq) & (~np.eye(sub_dist.shape[1], dtype=bool)[penalized_nodes, :])

    discount_rate = 0.015 * (1.0 + stagnation_factor)
    discounts = sub_dist * discount_rate * discount_mask

    # Vectorized subtraction ensuring symmetry
    updated_edge_distance[penalized_nodes, :] -= discounts
    updated_edge_distance[:, penalized_nodes] -= discounts.T

    # Guarantee diagonal remains zero and distances non-negative
    np.fill_diagonal(updated_edge_distance, 0.0)
    np.maximum(updated_edge_distance, 0.0, out=updated_edge_distance)

    return updated_edge_distance
