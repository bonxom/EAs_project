# ===== Heuristic Code =====
#{This algorithm deepens exploitation via Guided Local Search by penalizing high-utility and acute-detour tour edges with logarithmic frequency scaling and local curvature factors while replacing unconstrained shortcut discounting with structurally bounded 2-opt cross-exchange incentives clamped within rigorous distance invariants.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Decay historical penalty memory
    edge_n_used *= 0.95

    # Extract consecutive nodes along the current tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Evaluate geometric inefficiency and turn sharpness
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Compute GLS utility with geometric inefficiency weighting
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.40 * local_geom_factor)

    # Select top-k inefficient / high-utility edges to penalize
    k = max(1, int(np.ceil(0.12 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.35 * global_mean_dist

    # Apply symmetric guided penalties with robust boundary-clamping guards
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Structurally bounded 2-opt cross-exchange incentives:
    # Identify promising non-adjacent pairs that yield high 2-opt gain candidates
    # without unconstrained discounting.
    step = max(2, num_nodes // 8)
    k_pairs = min(k, max(1, num_nodes // 6))

    # Evaluate 2-opt exchange gains: (u -> u_next) and (x -> x_next) replaced by (u -> x) and (u_next -> x_next)
    x_idx = np.roll(tour, -step)
    x_next_idx = np.roll(x_idx, -1)

    d_current = edge_distance[u_idx, v_idx] + edge_distance[x_idx, x_next_idx]
    d_candidate = edge_distance[u_idx, x_idx] + edge_distance[v_idx, x_next_idx]
    exchange_gain = d_current - d_candidate

    top_exchange_indices = np.argsort(exchange_gain)[-k_pairs:]
    for idx in top_exchange_indices:
        if exchange_gain[idx] > 0.0:
            u, x = u_idx[idx], x_idx[idx]
            v, x_next = v_idx[idx], x_next_idx[idx]

            # Bound discount to a fraction of the original edge distance and ensure lower clamp
            discount_ux = min(0.08 * edge_distance[u, x], 0.05 * global_mean_dist)
            discount_vx = min(0.08 * edge_distance[v, x_next], 0.05 * global_mean_dist)

            updated_edge_distance[u, x] = max(1e-6, updated_edge_distance[u, x] - discount_ux)
            updated_edge_distance[x, u] = updated_edge_distance[u, x]

            updated_edge_distance[v, x_next] = max(1e-6, updated_edge_distance[v, x_next] - discount_vx)
            updated_edge_distance[x_next, v] = updated_edge_distance[v, x_next]

    return updated_edge_distance
