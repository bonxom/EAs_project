#{Key idea: Adaptively penalize tour edges using GLS utility scaled by relative length above local neighborhood medians, while rewarding mutually reciprocal k-nearest low-use non-tour edges with rank-discounted bonuses to provide targeted structural escape routes.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    costs = np.array([edge_distance[u, v] for u, v in tour_edges], dtype=np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8

    # Calculate guided local search utility: edge length prioritized by frequency of usage and z-score
    utilities = np.zeros(n, dtype=np.float64)
    for i in range(n):
        u, v = tour_edges[i]
        used = edge_n_used[u, v]
        z_score = (costs[i] - mean_cost) / std_cost
        sigmoid_weight = 1.0 / (1.0 + np.exp(-z_score))
        utilities[i] = (costs[i] * (1.0 + sigmoid_weight)) / (1.0 + used)

    threshold = np.percentile(utilities, 60.0)
    max_util = np.max(utilities) + 1e-8
    penalized_nodes = set()

    # Apply penalties to top utility edges
    for idx in range(n):
        if utilities[idx] >= threshold:
            u, v = tour_edges[idx]
            used = edge_n_used[u, v]
            penalty = (costs[idx] / (1.0 + np.sqrt(used + 1.0))) * (0.8 + 0.4 * (utilities[idx] / max_util))
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty
            penalized_nodes.add(u)
            penalized_nodes.add(v)

    # Form a set of edges currently in tour for fast lookup
    tour_edge_set = set()
    for u, v in tour_edges:
        tour_edge_set.add((min(u, v), max(u, v)))

    # Identify reciprocal k-nearest shortcut candidates for penalized nodes
    k_neighbors = min(6, max(3, n // 20))
    base_bonus = 0.04
    candidate_neighbors = {}

    for u in penalized_nodes:
        dists = edge_distance[u].copy()
        dists[u] = np.inf
        sorted_indices = np.argsort(dists)
        valid_candidates = []
        for v in sorted_indices:
            edge_key = (min(u, v), max(u, v))
            # Focus on low-frequency, non-tour edges
            if edge_key not in tour_edge_set and edge_n_used[u, v] <= 2:
                valid_candidates.append(int(v))
                if len(valid_candidates) >= k_neighbors:
                    break
        candidate_neighbors[u] = valid_candidates

    # Apply reciprocal shortcut bonuses
    visited_pairs = set()
    for u in penalized_nodes:
        u_cand = candidate_neighbors[u]
        for rank_u, v in enumerate(u_cand):
            pair = (min(u, v), max(u, v))
            if pair in visited_pairs:
                continue
            if v in candidate_neighbors and u in candidate_neighbors[v]:
                rank_v = candidate_neighbors[v].index(u)
                harmonic_rank = 2.0 / ((rank_u + 1) + (rank_v + 1))
                freq_damping = 1.0 / (1.0 + edge_n_used[u, v])
                decay = edge_distance[u, v] * base_bonus * harmonic_rank * freq_damping
                updated_edge_distance[u, v] = max(1e-6, updated_edge_distance[u, v] - decay)
                updated_edge_distance[v, u] = updated_edge_distance[u, v]
                visited_pairs.add(pair)

    return updated_edge_distance
