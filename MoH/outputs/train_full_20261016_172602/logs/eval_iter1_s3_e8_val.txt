#{Scale-invariant guided penalization normalizing relative edge costs by tour length dispersion combined with soft-decay memory attenuation on unvisited edges to balance guided escape and search flexibility.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Soft decay on historical penalties across the entire distance matrix
    # Attenuates previous penalties slightly toward a base state to prevent over-constraining the search space
    decay_factor = 0.995
    updated_edge_distance = updated_edge_distance * decay_factor

    # Compute local spatial node density via k-NN
    k = min(6, n - 1)
    if k > 1:
        sorted_dists = np.sort(edge_distance, axis=1)
        k_nn_avg = np.mean(sorted_dists[:, 1:k], axis=1)
    else:
        k_nn_avg = np.ones(n)
    node_density = 1.0 / (k_nn_avg + 1e-8)
    node_density = node_density / (np.mean(node_density) + 1e-8)

    # Current tour edge extraction
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Scale-invariant normalization metrics
    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths)
    scale = std_tour_length if std_tour_length > 1e-6 else (avg_tour_length + 1e-8)

    # Relative length and frequency damping
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-8)
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Non-linear rank penalty and density-modulated local penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_factor = w_global * rank_penalty + w_local * local_penalty

    # Amplify top quartile long edges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_factor[high_penalty_mask] *= 1.5

    # Scale-invariant additive penalty injection
    injected_penalty = combined_factor * scale * 0.25

    np.add.at(updated_edge_distance, (u_idx, v_idx), injected_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), injected_penalty)

    # Ensure zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
