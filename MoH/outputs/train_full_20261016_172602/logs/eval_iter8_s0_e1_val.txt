#{Penalize contiguous two-edge segments with maximum combined utilities to disrupt stubborn subpaths using vectorized symmetric edge updates scaled by average tour length.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    edge_utilities = tour_lengths / (1.0 + tour_counts)
    segment_utilities = edge_utilities + np.roll(edge_utilities, -1)

    max_segment_utility = np.max(segment_utilities)
    threshold = 0.99 * max_segment_utility
    selected_mask = segment_utilities >= threshold

    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Penalize both constituent edges of each selected segment
    u_first = u_idx[selected_mask]
    v_first = v_idx[selected_mask]

    next_mask = np.roll(selected_mask, 1)
    u_second = u_idx[next_mask]
    v_second = v_idx[next_mask]

    all_u = np.concatenate([u_first, u_second])
    all_v = np.concatenate([v_first, v_second])

    updated_edge_distance[all_u, all_v] += penalty
    updated_edge_distance[all_v, all_u] += penalty

    return updated_edge_distance
