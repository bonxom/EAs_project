# ===== Heuristic Code =====
#{An adaptive stagnation-regulated guided local search heuristic that measures tour edge overlap against the historical memory to dynamically adjust the memory decay rate, penalty multiplier, and shortcut discount strength, escaping deep basins while preserving promising topological moves.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Measure tour overlap with existing memory to regulate stagnation
    current_edge_freqs = edge_n_used[u_idx, v_idx]
    # Edge retention ratio: high mean frequency implies repeated visits to the same tour edges
    mean_overlap = np.mean(current_edge_freqs)
    stagnation_factor = np.clip(mean_overlap / 3.0, 0.0, 3.0)

    # Adaptive decay: retain memory when exploring new territory, accelerate forgetting when stagnating
    adaptive_decay = 0.98 - 0.08 * (stagnation_factor / 3.0)
    edge_n_used *= adaptive_decay

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Dynamic selection count based on stagnation level
    k = max(1, int(np.ceil((0.12 + 0.08 * (stagnation_factor / 3.0)) * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)

    # Dynamically scale penalty multiplier up during stagnation
    penalty_multiplier = 0.25 * (1.0 + 1.25 * stagnation_factor)
    adaptive_lambda = penalty_multiplier * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Dynamically expand shortcut incentives during stagnation
    shortcut_k = max(1, int(np.ceil((0.10 + 0.06 * (stagnation_factor / 3.0)) * num_nodes)))
    high_detour_indices = np.argsort(detour_ratio)[-shortcut_k:]
    discount_scale = 0.08 * (1.0 + 1.0 * stagnation_factor) * global_mean_dist

    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = discount_scale * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
