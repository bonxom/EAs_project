# ===== Heuristic Code =====
# {Applies sub-linear frequency damping to edge utilities and selects penalization candidates using an adaptive quantile-mean threshold combined with utility-proportional penalties.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Sub-linear frequency scaling to avoid over-suppressing moderately explored edges
    utilities = tour_lengths / (1.0 + np.sqrt(tour_counts))

    # Dynamic distribution-sensitive threshold based on median and interquartile dispersion
    q75 = np.percentile(utilities, 75)
    max_utility = np.max(utilities)
    threshold = max(0.95 * max_utility, q75)

    selected_mask = utilities >= threshold
    if not np.any(selected_mask):
        selected_mask = utilities >= max_utility

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Scale-normalized edge penalty calibrated against average local tour cost
    avg_edge = np.mean(tour_lengths)
    penalty_weights = utilities[selected_mask] / max_utility
    penalty = 0.35 * avg_edge * penalty_weights

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
