#{Dynamically select top-utility edges using rank-based percentile thresholding weighted by tour-level utility dispersion to apply adaptive, non-cycling penalties.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    u_std = np.std(utilities)
    u_mean = np.mean(utilities)

    # Adaptive quantile based on utility dispersion to select a dynamic k-percentile
    dispersion = u_std / (u_mean + 1e-12)
    percentile = np.clip(100.0 * (1.0 - (0.05 + 0.15 / (1.0 + dispersion))), 80.0, 98.0)

    rank_threshold = np.percentile(utilities, percentile)
    selected_mask = utilities >= rank_threshold

    # Ensure at least the single maximum utility edge is selected
    if not np.any(selected_mask):
        selected_mask = utilities == np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    penalty_base = 0.3 * avg_edge

    # Scale penalty smoothly with edge utility relative to the selected threshold
    u_excess = (utilities[selected_mask] - rank_threshold) / (np.max(utilities) - rank_threshold + 1e-12)
    penalties = penalty_base * (1.0 + 0.5 * u_excess)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
