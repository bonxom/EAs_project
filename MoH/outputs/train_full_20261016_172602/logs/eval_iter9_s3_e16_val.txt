# ===== Heuristic Code =====
#{Computes multi-faceted edge penalties via relative length, local density, and usage attenuation, then applies a scale-invariant perturbation normalized to a controlled fraction of the total tour length to prevent distortion explosion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Calculate local density around endpoints based on k-nearest neighbor distances
    k = min(6, n)
    sorted_dists = np.partition(edge_distance, kth=range(k), axis=1)[:, 1:k]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-8)
    node_density = node_density / (np.mean(node_density) + 1e-8)

    # Extract edges from current local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    total_tour_cost = np.sum(tour_edge_lengths)
    avg_tour_length = total_tour_cost / n

    # Relative edge length and sigmoid rank penalty
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-8)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Attenuate based on edge usage frequency and modulate by endpoint density
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Adaptive weighting between global and local penalties
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Amplify top quartile penalties
    threshold = np.percentile(tour_edge_lengths, 75)
    combined_penalty[tour_edge_lengths >= threshold] *= 1.5

    # Scale-invariant normalization: total injected penalty is calibrated to a fixed budget of total tour cost
    penalty_sum = np.sum(combined_penalty)
    if penalty_sum > 1e-8:
        target_budget = 0.05 * total_tour_cost
        normalized_penalty = combined_penalty * (target_budget / penalty_sum)
    else:
        normalized_penalty = combined_penalty

    # Apply symmetric perturbation
    np.add.at(updated_edge_distance, (u_idx, v_idx), normalized_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), normalized_penalty)

    return updated_edge_distance
