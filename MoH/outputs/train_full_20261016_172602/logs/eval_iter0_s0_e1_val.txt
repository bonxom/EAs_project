#{Fully vectorizes guided local search penalty updates, computing edge utilities on the current tour in vectorized NumPy operations and scaling cumulative penalties relative to base edge distances with exponential memory decay.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]
    if n == 0:
        return updated_edge_distance

    # Vectorized extraction of tour directed edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Base tour edge properties
    tour_lengths = edge_distance[u_idx, v_idx]
    avg_tour_edge = np.mean(tour_lengths) if n > 0 else 1.0

    # Guided Local Search utility calculation: cost / (1 + count)
    counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + counts)
    max_utility = np.max(utilities) if n > 0 else 0.0

    # Exponential decay applied to active penalties in edge_n_used
    # Penalties are applied on top of the original base edge distances
    penalty_scale = 0.25 * avg_tour_edge
    decay_rate = 0.95

    # Identify edges exceeding utility threshold for guided penalization
    high_util_mask = utilities >= (0.85 * max_utility)
    penalties = np.zeros(n, dtype=edge_distance.dtype)
    penalties[high_util_mask] = penalty_scale / (1.0 + counts[high_util_mask])

    # Apply symmetric penalties vectorially
    selected_u = u_idx[high_util_mask]
    selected_v = v_idx[high_util_mask]
    selected_penalties = penalties[high_util_mask]

    updated_edge_distance[selected_u, selected_v] += selected_penalties
    updated_edge_distance[selected_v, selected_u] += selected_penalties

    return updated_edge_distance
