# ===== Heuristic Code =====
#{Adaptive soft-evaporation of past penalties across the global distance matrix combined with targeted guided penalties on high-utility bottleneck tour edges to enable fine-grained exploitation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Calculate baseline proxy: minimal distances observed across the graph or unpenalized baseline approximation
    # Apply soft evaporation: decay penalties on all edges where edge_n_used indicates accumulated bias
    decay_rate = 0.96
    has_history = edge_n_used > 0
    # Evaporate excess penalty by contracting distances that have been visited/penalized
    updated_edge_distance[has_history] = (
        updated_edge_distance[has_history] * decay_rate +
        (updated_edge_distance[has_history] / (1.0 + 0.05 * edge_n_used[has_history])) * (1.0 - decay_rate)
    )

    # Extract current tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = updated_edge_distance[u_idx, v_idx]
    mean_length = np.mean(tour_lengths)
    std_length = np.std(tour_lengths) + 1e-8

    # Standardized relative cost indicating edge inefficiency
    rel_cost = (tour_lengths - mean_length) / std_length

    # Usage counter discount to prevent penalizing already heavily avoided edges repeatedly
    counts = edge_n_used[u_idx, v_idx]
    count_weight = 1.0 / (1.0 + 0.25 * counts)

    # Local context: k-nearest neighbor scaling for fine-grained exploitation
    k = min(5, n - 1)
    if k > 0:
        nn_dists = np.partition(updated_edge_distance, k, axis=1)[:, :k]
        local_scale = 0.5 * (np.mean(nn_dists[u_idx], axis=1) + np.mean(nn_dists[v_idx], axis=1)) + 1e-8
    else:
        local_scale = np.ones(n)

    # GLS utility metric: penalize long edges that have not been over-penalized yet
    utility = (tour_lengths / local_scale) * count_weight

    # Target upper-quartile bottlenecks with fine-grained progressive penalty
    crit_threshold = np.percentile(utility, 65)
    active_bottlenecks = utility >= crit_threshold

    penalty_magnitude = 0.15 * local_scale * (1.0 + np.maximum(0.0, rel_cost)) * count_weight
    applied_penalty = np.where(active_bottlenecks, penalty_magnitude * 1.5, penalty_magnitude * 0.4)

    # Symmetrically update distances for tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), applied_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), applied_penalty)

    # Maintain zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
