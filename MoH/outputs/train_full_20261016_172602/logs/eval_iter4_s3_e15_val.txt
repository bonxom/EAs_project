# ===== Heuristic Code =====
#{This algorithm penalizes local optimum edges using an adaptive utility model scaled by inverse square-root frequency dampening and local cluster density, while applying soft-tabu memory decay to prevent saturated distance distortions.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy distance matrix to ensure purity and no side-effects
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Step 2: Global gentle evaporation/decay on previously modified edges to prevent permanent landscape distortion
    # Evaporate excess distance where edges have seen repeated usage
    excess_mask = edge_n_used > 0
    decay_rate = 0.02
    updated_edge_distance[excess_mask] = (
        updated_edge_distance[excess_mask] * (1.0 - decay_rate) +
        edge_distance[excess_mask] * decay_rate
    )

    # Step 3: Compute k-nearest-neighbor local density metric
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn = min(6, n)
    k_nn_avg = np.mean(sorted_dists[:, 1:k_nn], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Step 4: Extract tour edge indices and metrics
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Step 5: Diminishing marginal returns via inverse square-root frequency and soft tabu scaling
    freq = edge_n_used[u_idx, v_idx]
    usage_dampening = 1.0 / np.sqrt(1.0 + freq)

    # Sigmoidal priority for longer relative lengths
    length_utility = 1.0 / (1.0 + np.exp(-3.5 * (relative_length - 1.0)))

    # Context-aware spatial density scaling (penalize clustered edges more to promote structural leaps)
    density_scaling = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Combined adaptive feature utility penalty
    adaptive_penalty = (
        0.5 * avg_tour_length * length_utility * usage_dampening * np.sqrt(density_scaling)
    )

    # Boost penalty for the longest quartile of edges to force alternative connections
    p75 = np.percentile(tour_edge_lengths, 75)
    high_impact_mask = tour_edge_lengths >= p75
    adaptive_penalty[high_impact_mask] *= 1.35

    # Step 6: Symmetric application of calculated penalties to the tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), adaptive_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), adaptive_penalty)

    return updated_edge_distance
