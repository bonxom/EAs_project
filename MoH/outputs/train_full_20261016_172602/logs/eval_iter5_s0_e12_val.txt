# ===== Heuristic Code =====
#{Decay existing penalized edge distances toward their baseline state while adaptively penalizing the highest-utility edges of the local optimum tour using vectorized operations to sustain dynamic exploration without matrix saturation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Identify edges forming the current local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utility based on length and usage frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Evaporate accumulated distance perturbations across active edges
    decay_factor = 0.95
    active_mask = edge_n_used > 0
    updated_edge_distance[active_mask] = (
        updated_edge_distance[active_mask] * decay_factor
        + edge_distance[active_mask] * (1.0 - decay_factor)
    )

    # Calculate dynamic penalty and apply to near-maximal utility edges
    penalty = 0.3 * avg_edge
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
