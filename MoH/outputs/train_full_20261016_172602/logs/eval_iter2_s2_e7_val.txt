#{Adaptively modulate memory decay, geometric weighting, and penalization intensity based on tour-level edge reuse and search concentration to escape stagnation while preserving structural features.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    current_freqs = edge_n_used[u_idx, v_idx]
    mean_freq = np.mean(current_freqs)
    stagnation_ratio = float(np.clip(mean_freq / (mean_freq + 2.0), 0.0, 1.0))

    decay_rate = 0.97 - 0.06 * stagnation_ratio
    edge_n_used *= decay_rate

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn

    geom_inefficiency = detour_ratio * (1.0 + (0.45 + 0.20 * stagnation_ratio) * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    geom_weight_factor = 0.30 + 0.25 * stagnation_ratio
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + geom_weight_factor * local_geom_factor)

    k = max(1, int(np.ceil((0.12 + 0.06 * stagnation_ratio) * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = (0.28 + 0.15 * stagnation_ratio) * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + (0.40 + 0.20 * stagnation_ratio) * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    k_shortcut = max(1, int(np.ceil(0.12 * num_nodes)))
    high_detour_indices = np.argsort(detour_ratio)[-k_shortcut:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = (0.08 + 0.05 * stagnation_ratio) * global_mean_dist * (1.0 / (1.0 + freq))
            new_cost = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_cost
            updated_edge_distance[w, u] = new_cost

    return updated_edge_distance
