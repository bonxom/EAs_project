# ===== Heuristic Code =====
#{Replaces global z-score tour edge weighting with a mutual k-nearest neighbor topological indicator to heavily penalize unnatural non-mutual bridge edges while preserving natural cluster linkages and applying localized unvisited edge discounts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Recombine k-NN mutual neighborhood topology modifier
    # Adaptive k scaled logarithmically with problem size
    k = max(3, min(int(np.ceil(np.log2(n) * 2.0)), n - 1))

    # Compute k-th nearest neighbor distance threshold for each node
    # Using np.partition to efficiently identify the k-th smallest distance per row
    k_th_costs = np.partition(edge_distance, k, axis=1)[:, k]

    # An edge (u, v) is in mutual k-NN if cost(u, v) <= threshold(u) and cost(u, v) <= threshold(v)
    is_knn_u = costs <= k_th_costs[u_idx]
    is_knn_v = costs <= k_th_costs[v_idx]
    mutual_knn = (is_knn_u & is_knn_v).astype(np.float64)

    # Penalize non-mutual bridge edges heavily
    knn_multiplier = 1.0 + 0.5 * (1.0 - mutual_knn)

    # Base GLS utility scaled by mutual neighborhood topology
    utilities = (costs * knn_multiplier) / (1.0 + used)
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Compute edge penalties
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Localized discount/decay to incident unvisited alternatives
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
