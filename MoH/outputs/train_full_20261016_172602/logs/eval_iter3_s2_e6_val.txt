# ===== Heuristic Code =====
#{This algorithm penalizes long, geometrically inefficient tour edges using turn-slack guided penalties with search-state dependent sizing, while safely subsidizing prospective 3-opt/4-opt shortcut reconnections prioritized by peripheral node centrality and unvisited edge potential.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Memory decay
    edge_n_used *= 0.95

    # Tour edge lengths
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wx = edge_distance[w_idx, x_idx]
    d_uw = edge_distance[u_idx, w_idx]
    d_vx = edge_distance[v_idx, x_idx]

    # Geometric triangle-slack and angular turn deficit
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Local smoothing of geometric factor along adjacent edges
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Stagnation indicator via edge usage saturation
    tour_freqs = edge_n_used[u_idx, v_idx]
    mean_tour_freq = np.mean(tour_freqs)
    stagnation_factor = np.clip(1.0 + 0.25 * np.log1p(mean_tour_freq), 1.0, 2.0)

    # Guided utility evaluation
    utilities = (d_uv / (1.0 + tour_freqs)) * (1.0 + 0.35 * local_geom_factor)

    # Dynamic perturbation sizing
    k = max(2, int(np.clip(np.ceil(0.12 * stagnation_factor * num_nodes), 2, num_nodes // 2)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.28 * global_mean_dist * stagnation_factor

    # Apply guided penalties to inefficient and overused tour edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.4 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Prospective multi-step reconnection gain evaluation (3-opt/4-opt candidate slacks)
    gain_uw = (d_uv + d_vw) - d_uw
    gain_vx = (d_vw + d_wx) - d_vx

    # Peripheral centrality proxy: lower density / isolated nodes benefit more from structural bridges
    # Approximate peripheral score using average distance to immediate tour neighbors
    peri_u = (d_uv + np.roll(d_uv, 1)) / (global_mean_dist + 1e-8)
    peri_w = (d_vw + d_wx) / (global_mean_dist + 1e-8)
    reconnect_attractiveness = np.maximum(gain_uw, gain_vx) * (0.5 * (peri_u + peri_w))

    subsidize_k = max(1, k // 2)
    top_reconnect_indices = np.argsort(reconnect_attractiveness)[-subsidize_k:]

    # Metric-preserving targeted shortcut subsidies
    for idx in top_reconnect_indices:
        if gain_uw[idx] >= gain_vx[idx]:
            a, b = u_idx[idx], w_idx[idx]
            net_gain = gain_uw[idx]
        else:
            a, b = v_idx[idx], x_idx[idx]
            net_gain = gain_vx[idx]

        freq = edge_n_used[a, b]
        if net_gain > 0 and freq < 1.0:
            # Scale subsidy proportionally to verified prospective gain without breaching positivity
            max_allowed_discount = 0.5 * edge_distance[a, b]
            nominal_discount = 0.12 * net_gain / (1.0 + freq)
            discount = min(nominal_discount, max_allowed_discount)
            updated_edge_distance[a, b] = max(1e-6, updated_edge_distance[a, b] - discount)
            updated_edge_distance[b, a] = updated_edge_distance[a, b]

    return updated_edge_distance
