#{A fast, deterministic Guided Local Search penalty strategy that computes edge penalties using vectorized min-max relative gap scaling against tour dispersion and frequency-dampened utilization, avoiding costly full k-NN recalculations and non-linear functions.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge pairs in the tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Edge lengths and usage counts along the tour
    tour_lens = edge_distance[u_idx, v_idx]
    tour_uses = edge_n_used[u_idx, v_idx]

    min_len = np.min(tour_lens)
    max_len = np.max(tour_lens)
    span = max_len - min_len

    # Vectorized min-max relative gap scaling across the tour
    if span > 1e-8:
        rel_gap = (tour_lens - min_len) / span
    else:
        rel_gap = np.zeros_like(tour_lens)

    mean_len = np.mean(tour_lens)

    # Fast local node context approximation: ratio against mean connected edge lengths for endpoints
    # Approximates incident scale without full matrix sorting
    local_baseline = 0.5 * (np.mean(edge_distance[u_idx, :], axis=1) + np.mean(edge_distance[v_idx, :], axis=1))
    local_excess = np.maximum(0.0, (tour_lens - local_baseline) / (local_baseline + 1e-6))

    # Classical GLS utility-based scaling calibrated for TSP instances
    # Long edges penalized proportionally, mitigated by accumulated usage
    utilization_discount = 1.0 / (1.0 + tour_uses)

    # Combined penalty calibrated with linear gap boosting and mean-length normalization
    base_penalty = (0.6 * rel_gap + 0.4 * local_excess) * utilization_discount

    # Scale penalty relative to average tour edge length to guarantee effective perturbation
    penalty_magnitude = 0.15 * mean_len * base_penalty

    # Ensure symmetric distance updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty_magnitude)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty_magnitude)

    return updated_edge_distance
