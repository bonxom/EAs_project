# ===== Heuristic Code =====
#{Incorporate local density via mutual k-nearest neighbor baseline distances to penalize disproportionately long bridge edges relative to incident cluster scales while maintaining vectorized symmetric updates.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Tour edge endpoints
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    n_nodes = edge_distance.shape[0]
    # Dynamically select k based on graph size (at least 2, up to 5)
    k = min(max(2, n_nodes // 5), 5)

    # Compute the average k-nearest neighbor distance for each node to represent local density
    # Exclude self-distance (diagonal is 0) by taking partition over k+1 elements
    k_smallest = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    avg_k_dist = np.mean(k_smallest, axis=1)

    # Effective baseline distance for each tour edge based on incident vertices' local densities
    baseline_dist = 0.5 * (avg_k_dist[u_idx] + avg_k_dist[v_idx])
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Preferentially target edges that exceed their local cluster density (bridges)
    excess_lengths = np.maximum(0.0, tour_lengths - baseline_dist)
    utilities = excess_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    if max_utility > 1e-12:
        threshold = 0.90 * max_utility
        selected_mask = utilities >= threshold
    else:
        # Fallback to standard utility if no edges exceed baseline
        raw_utilities = tour_lengths / (1.0 + tour_counts)
        threshold = 0.90 * np.max(raw_utilities)
        selected_mask = raw_utilities >= threshold

    # Adaptive penalty scaled by the local incident edges
    avg_edge = np.mean(tour_lengths)
    penalty = 0.35 * avg_edge

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
