# ===== Heuristic Code =====
#{Weight edge penalties using k-nearest neighbor relative closeness thresholds to aggressively penalize bridging edges between clusters while preserving dense intra-cluster manifolds.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # Standard Guided Local Search edge utility calculation
    tour_distances = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_distances / (1.0 + tour_counts)
    max_util = np.max(utilities)
    max_mask = np.isclose(utilities, max_util)

    # Increment usage counts for maximum utility edges in the local optimum
    for i in range(num_nodes):
        if max_mask[i]:
            u, v = u_idx[i], v_idx[i]
            edge_n_used[u, v] += 1
            edge_n_used[v, u] += 1

    # Compute k-nearest neighbor distance threshold for each node (k=10)
    k = min(10, num_nodes - 1)
    sorted_dists = np.partition(edge_distance, kth=k, axis=1)
    knn_threshold = sorted_dists[:, k]

    # Relative closeness: ratio of edge distance to local k-NN scale
    sym_knn = 0.5 * (knn_threshold[:, None] + knn_threshold[None, :])
    rel_closeness = edge_distance / (sym_knn + 1e-12)
    knn_bridge_weight = np.clip(rel_closeness, 0.5, 3.0)

    # Tour overlap and stagnation tracking
    tour_usage_updated = edge_n_used[u_idx, v_idx]
    overlap_ratio = np.mean(tour_usage_updated > 1.0)

    # Adaptive base penalty parameter lambda
    mean_knn_dist = np.mean(knn_threshold)
    adaptive_lambda = (0.22 + 0.12 * overlap_ratio) * mean_knn_dist

    # Compute final penalty matrix with k-NN bridging weights
    penalty_matrix = adaptive_lambda * edge_n_used * knn_bridge_weight
    updated_edge_distance += penalty_matrix

    return updated_edge_distance
