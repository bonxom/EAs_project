# ===== Heuristic Code =====
#{Scale penalty additions directly by the actual edge length d(u,v) modulated by relative edge length, usage frequency, and local density to achieve scale-invariant guided local search.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Estimate local node density using k-nearest neighbors
    k = min(6, n - 1)
    sorted_dists = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Extract current tour edges and their attributes
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Compute relative length and utility components
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Emphasize long edges above the 75th percentile
    p75 = np.percentile(tour_edge_lengths, 75)
    max_len = np.max(tour_edge_lengths)
    scale_factor = 1.0 + 0.5 * np.maximum(0.0, (tour_edge_lengths - p75) / (max_len - p75 + 1e-6))

    # Decouple from static step sizes: multiply by actual edge length d(u,v)
    base_rate = 0.15
    final_penalty = base_rate * tour_edge_lengths * combined_penalty * scale_factor

    np.add.at(updated_edge_distance, (u_idx, v_idx), final_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), final_penalty)

    return updated_edge_distance
