# ===== Heuristic Code =====
#{Guided Local Search edge perturbation that penalizes high-utility tour edges while selectively discounting only the top-k nearest geometric neighbors of penalized vertices to guide search toward promising alternative connections without distorting distant non-viable edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Extract current tour edge indices and corresponding costs/usage
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Calculate edge utilities via normalized cost weighting and historical frequency
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Symmetric penalty augmentation for high-utility tour edges
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Targeted attraction: discount only top-k nearest geometric neighbors of penalized vertices
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k = min(5, n - 2) if n > 2 else 0
    decay_ratio = 0.02

    if k > 0:
        # Create a boolean tour-edge adjacency matrix for fast lookup of current tour edges
        tour_adj = np.zeros((n, n), dtype=bool)
        tour_adj[u_idx, v_idx] = True
        tour_adj[v_idx, u_idx] = True

        for u in penalized_nodes:
            # Candidate neighbors are distinct from u and not part of the current tour with u
            dists = edge_distance[u].astype(np.float64)
            candidate_mask = ~tour_adj[u]
            candidate_mask[u] = False

            cand_indices = np.where(candidate_mask)[0]
            if len(cand_indices) > 0:
                cand_dists = dists[cand_indices]
                actual_k = min(k, len(cand_indices))
                # Identify the top-k nearest geometric candidate neighbors
                nearest_k_idx = cand_indices[np.argpartition(cand_dists, actual_k - 1)[:actual_k]]

                # Apply targeted attraction discount symmetrically
                discounts = updated_edge_distance[u, nearest_k_idx] * decay_ratio
                updated_edge_distance[u, nearest_k_idx] -= discounts
                updated_edge_distance[nearest_k_idx, u] -= discounts

    return updated_edge_distance
