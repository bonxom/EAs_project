# {Dynamic non-linear penalty scheduling with adaptive decay driven by tour overlap and entropy-guided sub-linear scaling to prevent metric distortion and escape chronic stagnation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)
    w_idx = np.roll(u_idx, -2)

    # Calculate tour usage metrics before evaporation
    current_tour_usage = edge_n_used[u_idx, v_idx]
    overlap_ratio = np.mean(current_tour_usage > 1.0)
    avg_tour_usage = np.mean(current_tour_usage)
    max_used = np.max(edge_n_used) + 1e-12
    stagnation_score = 0.5 * overlap_ratio + 0.5 * (avg_tour_usage / max_used)

    # Adaptive decay: retain memory during exploration, accelerate decay during chronic stagnation
    decay_rate = 0.98 - 0.10 / (1.0 + np.exp(-6.0 * (stagnation_score - 0.4)))
    edge_n_used *= decay_rate

    # Utility-based edge penalization
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    # Dynamically scale selection count based on stagnation
    k_fraction = 0.12 + 0.08 * float(np.clip(stagnation_score, 0.0, 1.0))
    k = max(1, int(np.ceil(k_fraction * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Usage probability and Shannon entropy per node
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    # Node centrality in usage graph
    node_centrality = np.sum(edge_n_used, axis=1)
    norm_centrality = node_centrality / (np.max(node_centrality) + 1e-12)

    # Spatial centrality in metric space
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial_centrality = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Dynamic base penalty multiplier
    adaptive_lambda = (0.12 + 0.18 * np.tanh(2.0 * stagnation_score)) * global_mean_dist

    # Apply sub-linear power-law penalty scaled by structural rigidity
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        c_mod = 0.3 * (norm_centrality[u] + norm_centrality[v])
        sc_mod = 0.2 * (norm_spatial_centrality[u] + norm_spatial_centrality[v])
        r_mod = 0.3 * (entropy_rigidity[u] + entropy_rigidity[v])
        modulation = 1.0 + c_mod + sc_mod + r_mod

        # Sub-linear response (power 0.8) prevents extreme metric distortion on heavily visited edges
        sublinear_usage = np.power(edge_n_used[u, v], 0.8)
        penalty = adaptive_lambda * sublinear_usage * modulation

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
