#{The algorithm updates edge penalization using sub-linear count saturation and smooth memory decay to prevent late-stage over-penalization while preserving guided exploration around congested and rigid nodes.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    tour_distances = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_distances / (1.0 + tour_counts)
    max_util = np.max(utilities)
    max_mask = np.isclose(utilities, max_util)

    for i in range(num_nodes):
        if max_mask[i]:
            u, v = u_idx[i], v_idx[i]
            edge_n_used[u, v] += 1
            edge_n_used[v, u] += 1

    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    node_usage_centrality = np.sum(edge_n_used, axis=1)
    norm_usage_centrality = node_usage_centrality / (np.max(node_usage_centrality) + 1e-12)

    tour_usage_updated = edge_n_used[u_idx, v_idx]
    overlap_ratio = np.mean(tour_usage_updated > 1.0)
    mean_rigidity = np.mean(entropy_rigidity)
    stagnation_factor = np.clip(overlap_ratio + mean_rigidity, 0.0, 2.0)

    adaptive_lambda = (0.20 + 0.10 * stagnation_factor) * global_mean_dist

    spatial_field = norm_spatial[:, None] + norm_spatial[None, :]
    rigidity_field = entropy_rigidity[:, None] + entropy_rigidity[None, :]
    cluster_dispersion = norm_usage_centrality[:, None] * norm_usage_centrality[None, :]
    congestion_field = 1.0 + 0.35 * spatial_field + 0.35 * rigidity_field + 0.30 * cluster_dispersion

    # Sub-linear count saturation to dampen late-iteration over-penalization
    effective_counts = np.power(edge_n_used.astype(np.float64), 0.8)
    penalty_matrix = adaptive_lambda * effective_counts * congestion_field
    updated_edge_distance += penalty_matrix

    return updated_edge_distance
