# ===== Heuristic Code =====
#{Synthesizes acute-deficit curvature geometric utilities with dual vertex-centrality modulation to penalize structurally inefficient edges while dampening distortion on peripheral nodes and discounting promising shortcut transitions.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Trace consecutive triplets along the cyclic tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay edge historical frequencies
    edge_n_used *= 0.96

    # Distances between consecutive vertices
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Calculate geometric curvature and detour inefficiency
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Vertex centrality modulation (Parent C spatial rigidity)
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(edge_distance)
    centrality = global_mean_dist / (node_mean_dist + 1e-8)
    centrality_edge = 0.5 * (centrality[u_idx] + centrality[v_idx])
    # Clip centrality factor to prevent extreme over/under penalization
    centrality_mod = np.clip(centrality_edge, 0.6, 1.5)

    # Synthesized utility: acute-deficit geometric factor tempered by spatial centrality
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor) * centrality_mod

    # Select top-k inefficient edges to penalize
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight * centrality_mod[idx]
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Discount unvisited shortcut chords at high-detour regions
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            shortcut_centrality = 0.5 * (centrality[u] + centrality[w])
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq)) * np.clip(shortcut_centrality, 0.7, 1.3)
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
