# ===== Heuristic Code =====
#{We replace rigid top-k selection with a deterministic pseudo-stochastic softmax tournament schedule driven by tour hash stagnation and rank-modulated Boltzmann weighting to achieve non-monotonic edge perturbation and shortcut reward.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour, dtype=np.intp)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decayed edge usage frequency tracking
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Utility evaluation incorporating geometry and usage history
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # State-dependent deterministic temperature parameter derived from tour signature
    total_tour_dist = np.sum(d_uv)
    tour_signature = float(np.sum(tour * np.arange(1, num_nodes + 1))) % 1000.0
    stagnation_factor = 0.5 + 0.5 * np.sin(tour_signature / 100.0)
    temp = np.std(utilities) * (0.8 + 0.6 * stagnation_factor) + 1e-8

    # Softmax probabilities over utilities
    scaled_util = (utilities - np.max(utilities)) / temp
    exp_util = np.exp(np.clip(scaled_util, -30.0, 0.0))
    softmax_probs = exp_util / np.sum(exp_util)

    # Deterministic low-discrepancy systematic sampling for non-monotonic exploration
    target_k = max(1, int(np.round(num_nodes * (0.10 + 0.08 * stagnation_factor))))
    cum_probs = np.cumsum(softmax_probs)
    offset = (float(np.sum(tour[:min(5, num_nodes)])) % target_k) / float(target_k)
    sample_points = (np.arange(target_k) + offset) / float(target_k)
    sample_points = np.clip(sample_points, 0.0, 1.0 - 1e-9)
    selected_edges = np.searchsorted(cum_probs, sample_points)
    selected_edges = np.unique(selected_edges)

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.25 + 0.15 * stagnation_factor) * global_mean_dist

    for idx in selected_edges:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        soft_weight = 1.0 + softmax_probs[idx] * target_k
        penalty = adaptive_lambda * freq_penalty * geom_weight * soft_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Shortcut rewarding via soft-ranking
    shortcut_temp = np.std(detour_ratio) * 0.75 + 1e-8
    scaled_detour = (detour_ratio - np.max(detour_ratio)) / shortcut_temp
    shortcut_probs = np.exp(np.clip(scaled_detour, -30.0, 0.0))
    shortcut_probs /= np.sum(shortcut_probs)

    cum_detour = np.cumsum(shortcut_probs)
    shortcut_points = (np.arange(target_k) + ((offset + 0.5) % 1.0)) / float(target_k)
    shortcut_points = np.clip(shortcut_points, 0.0, 1.0 - 1e-9)
    selected_shortcuts = np.unique(np.searchsorted(cum_detour, shortcut_points))

    for idx in selected_shortcuts:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.12 * global_mean_dist * (1.0 / (1.0 + freq)) * (1.0 + shortcut_probs[idx] * target_k)
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
