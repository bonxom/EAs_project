# ===== Heuristic Code =====
#{Applies an exponential decay to historical penalty perturbations relative to the base distance matrix while injecting a structured, density-scaled penalty on the current local optimum's longest edges to prevent penalty saturation and preserve search gradients.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Estimate base distance by discounting existing edge perturbations proportional to edge_n_used
    decay_rate = 0.95
    # Apply global decay on historical cumulative penalties where edges have been penalized
    penalized_mask = edge_n_used > 0
    if np.any(penalized_mask):
        excess_penalty = updated_edge_distance[penalized_mask] - (updated_edge_distance[penalized_mask] / (1.0 + 0.05 * edge_n_used[penalized_mask]))
        updated_edge_distance[penalized_mask] -= (1.0 - decay_rate) * excess_penalty

    # Compute topological density context using 5-nearest neighbors
    sorted_dists = np.sort(updated_edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Tour edge metrics
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = updated_edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)

    # Relative length and continuous rank sigmoid penalty
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Frequency dampening and topological density weighting
    usage_factor = 1.0 / (1.0 + 0.25 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Dynamic blending of global edge disparity and local topology
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.65 / (1.0 + 0.08 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * (avg_tour_length * 0.15)

    # Sharp amplification on the worst 25% edges of the tour
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.4

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
