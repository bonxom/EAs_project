#{Adaptive guided local search penalty with scale-invariant calibration, k-NN partition-based density approximation, and selective hard-evasion targeting of high-utility bottleneck edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    # Extract tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Scale-invariance: calibrate penalty scale with tour average length
    avg_tour_len = np.mean(tour_lengths) + 1e-12
    rel_lengths = tour_lengths / avg_tour_len

    # Fast k-NN spatial density using partition instead of full sort
    k = min(5, n - 1)
    if k > 0:
        part_dists = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
        k_nn_avg = np.mean(part_dists, axis=1) + 1e-12
        node_density = 1.0 / k_nn_avg
        node_density /= np.mean(node_density)
        density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    else:
        density_factor = np.ones(n)

    # Standard GLS utility: edge length / (1 + count)
    gls_utility = tour_lengths / (1.0 + usage)

    # Reactive adaptation based on stagnation/repetition level
    mean_usage = np.mean(usage)
    evasion_factor = 1.0 + 0.15 * np.log1p(mean_usage)

    # Base penalty proportional to tour scale and local density
    base_penalty = (0.25 * avg_tour_len) * evasion_factor * (1.0 / (1.0 + 0.5 * usage)) * density_factor

    # Selective targeting: strongly penalize top utility / bottleneck edges
    util_thresh_75 = np.percentile(gls_utility, 75)
    util_thresh_90 = np.percentile(gls_utility, 90)

    penalty = np.zeros(n, dtype=np.float64)
    mask_75 = gls_utility >= util_thresh_75
    mask_90 = gls_utility >= util_thresh_90

    # Sharp evasion targeting
    penalty[mask_75] = base_penalty[mask_75] * (1.0 + rel_lengths[mask_75])
    penalty[mask_90] *= 1.8

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
