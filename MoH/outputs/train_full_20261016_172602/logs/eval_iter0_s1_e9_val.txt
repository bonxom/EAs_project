# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create a copy of the edge distance matrix to avoid modifying input
    updated_edge_distance = np.copy(edge_distance)

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Calculate utility according to the standard GLS principle: cost / (1 + penalty_count)
    utilities = costs / (1.0 + used)
    max_util = np.max(utilities)

    # Select edges with maximum utility (or very close to maximum) to penalize
    penalize_mask = utilities >= (max_util - 1e-8)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_used = used[penalize_mask]

    # Global penalty factor scaling with average tour edge cost: lambda = alpha * (L / n)
    # where L / n is the mean cost of edges in the current local optimum tour
    mean_tour_cost = np.mean(costs)
    alpha = 0.3
    base_penalty = alpha * mean_tour_cost

    # Sublinear penalty dampening to avoid over-penalizing necessary topological bridges
    dampened_penalty = base_penalty / (1.0 + 0.5 * np.log1p(p_used))

    updated_edge_distance[p_u, p_v] += dampened_penalty
    updated_edge_distance[p_v, p_u] += dampened_penalty

    return updated_edge_distance
