# ===== Heuristic Code =====
#{Integrate a 2-hop local tour neighborhood context by evaluating each tour edge against its immediate adjacent predecessors and successors alongside k-NN density and usage discounting to selectively penalize locally inefficient edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Baseline k-NN node density estimation
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Tour indices: predecessor p, current edge (u, v), successor w
    p_idx = np.roll(local_opt_tour, 1)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    w_idx = np.roll(local_opt_tour, -2)

    # Tour edge lengths
    e_curr = edge_distance[u_idx, v_idx]
    e_prev = edge_distance[p_idx, u_idx]
    e_next = edge_distance[v_idx, w_idx]

    # 2-hop local tour neighborhood context (slack ratio)
    local_adj_mean = 0.5 * (e_prev + e_next) + 1e-6
    slack_ratio = e_curr / local_adj_mean

    # Global tour normalization
    avg_tour_length = np.mean(e_curr) + 1e-6
    relative_length = e_curr / avg_tour_length

    # Usage factor and endpoint density factor
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Combined context-aware local penalty
    local_context_penalty = slack_ratio * usage_factor * density_factor
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Adaptive weighting between global rank and local structural slack
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.5 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = w_global * rank_penalty + w_local * local_context_penalty

    # Selective boost for long outliers
    threshold = np.percentile(e_curr, 75)
    high_penalty_mask = e_curr >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Symmetric in-place edge penalty accumulation
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
