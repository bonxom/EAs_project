# ===== Heuristic Code =====
#{Selectively penalize the top-tier utility edges using logarithmically saturated penalty scaling to prevent metric distortion, while selectively incentivizing unvisited k-nearest neighbor candidate edges instead of applying unguided uniform decay.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Guided Local Search utility calculation: cost / (1 + count)
    utilities = costs / (1.0 + used)

    # Selective top-tier penalization (focus on top 25-30% edges to preserve tour structure)
    threshold = np.quantile(utilities, 0.75)
    penalize_mask = utilities >= threshold

    if not np.any(penalize_mask):
        penalize_mask = utilities == np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]

    # Logarithmically dampened penalty to prevent explosive distortion and preserve triangle inequality
    # Scale proportional to mean tour edge cost, dampened by log1p(usage)
    tour_mean_cost = np.mean(costs)
    saturation_factor = 1.0 / (1.0 + np.log1p(p_used))
    penalties = 0.3 * tour_mean_cost * saturation_factor * (p_costs / (np.max(p_costs) + 1e-8))

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Targeted incentive: reward unvisited k-nearest neighbor edges for penalized nodes
    k_neighbors = min(5, n - 2)
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    for u in penalized_nodes:
        # Find candidate k-nearest neighbors based on base distance
        dists = edge_distance[u].copy()
        dists[u] = np.inf
        # Focus on unvisited k-closest candidates
        candidate_indices = np.argpartition(dists, k_neighbors)[:k_neighbors]
        unvisited_candidates = candidate_indices[edge_n_used[u, candidate_indices] == 0]

        if len(unvisited_candidates) > 0:
            incentive = 0.015 * updated_edge_distance[u, unvisited_candidates]
            updated_edge_distance[u, unvisited_candidates] -= incentive
            updated_edge_distance[unvisited_candidates, u] -= incentive

    return updated_edge_distance
