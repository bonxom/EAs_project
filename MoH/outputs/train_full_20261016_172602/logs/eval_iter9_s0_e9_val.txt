#{An adaptive Guided Local Search penalty function that dynamically scales edge penalties using the tour coefficient of variation and edge usage concentration entropy to modulate perturbation strength between fine-grained local escape and aggressive stagnation breakout.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utilities inversely weighted by search history
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Instance statistics: mean and dispersion (coefficient of variation)
    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    cv = std_edge / (avg_edge + 1e-9)

    # Entropy-based stagnation detection on edge reuse within the tour
    counts_sum = np.sum(tour_counts)
    if counts_sum > 0:
        p = (tour_counts + 1e-9) / (counts_sum + 1e-9 * len(tour_counts))
        entropy = -np.sum(p * np.log(p))
        max_entropy = np.log(len(tour_counts))
        normalized_entropy = entropy / (max_entropy + 1e-9)
    else:
        normalized_entropy = 1.0

    # High entropy indicates uniform usage (early stage/progress); low entropy indicates repeated traps (stagnation)
    stagnation_factor = 1.0 + (1.0 - normalized_entropy) * 2.0
    dynamic_scale = (0.15 + 0.35 * cv) * stagnation_factor
    penalty = dynamic_scale * avg_edge

    # Select edges meeting top utility threshold
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Vectorized symmetric update
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
