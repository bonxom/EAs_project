# ===== Heuristic Code =====
# {Penalize high-utility edges along the local optimum tour via sigmoid-standardized cost-usage utility while applying rank-weighted metric discounts exclusively to the k-nearest unused spatial neighbors of perturbed vertices to carve viable, bounded escape trajectories.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    utilities = (costs * (1.0 + weights)) / (1.0 + used)
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Symmetrically apply additive penalties
    np.add.at(updated_edge_distance, (p_u, p_v), penalties)
    np.add.at(updated_edge_distance, (p_v, p_u), penalties)

    # Identify perturbed vertices
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # Restrict discounting to top-k nearest neighbors in original metric
    k_neighbors = min(10, n - 1)
    base_discount_rate = 0.03

    for u in penalized_nodes:
        # Candidate neighbors must be distinct from u and relatively unused
        orig_dists = edge_distance[u]
        # Get sorted neighbor indices by base distance excluding self
        candidate_indices = np.argsort(orig_dists)
        candidate_indices = candidate_indices[candidate_indices != u]

        # Filter for unused or low-frequency candidate edges
        valid_candidates = candidate_indices[edge_n_used[u, candidate_indices] == 0]
        selected_candidates = valid_candidates[:k_neighbors]

        if len(selected_candidates) > 0:
            # Rank-inverse discount: rank 1 gets full base_discount, higher ranks get progressively less
            ranks = np.arange(1, len(selected_candidates) + 1, dtype=np.float64)
            rank_factors = 1.0 / ranks

            discounts = updated_edge_distance[u, selected_candidates] * (base_discount_rate * rank_factors)
            updated_edge_distance[u, selected_candidates] -= discounts
            updated_edge_distance[selected_candidates, u] -= discounts

    return updated_edge_distance
