# ===== Heuristic Code =====
# {Refine Guided Local Search penalties by dynamically scaling edge penalties according to their excess length over the k-nearest neighbor baseline of their endpoints, aggressively targeting outlier bottleneck edges while softly perturbing essential structural connections.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    tour_length = sum(tour_lengths)
    avg_edge = tour_length / float(n) if n > 0 else 1.0

    # Guided Local Search utilities
    utilities = [edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in tour_edges]
    max_utility = max(utilities) if utilities else 0.0

    if max_utility <= 0.0:
        return updated_edge_distance

    mean_used = np.mean([edge_n_used[u, v] for u, v in tour_edges])
    threshold_ratio = max(0.92, 0.99 - 0.02 * min(mean_used, 3.0))

    # Compute k-nearest neighbor baseline distances for each vertex (ignoring self-loops)
    # Using 3-nearest neighbors (or n-1 if n is small)
    k = min(3, max(1, n - 1))
    # Sort distances per row, index 1 to k gives nearest neighbors excluding self (index 0)
    sorted_dists = np.partition(edge_distance, k, axis=1)
    knn_baseline = np.mean(sorted_dists[:, 1:k+1], axis=1)

    base_lambda = 0.30 * avg_edge

    for (u, v), util in zip(tour_edges, utilities):
        if util >= threshold_ratio * max_utility:
            d_uv = edge_distance[u, v]
            # Baseline is the average of endpoint local neighborhood distances
            local_baseline = 0.5 * (knn_baseline[u] + knn_baseline[v])
            # Relative excess ratio: heavily scales outlier bottleneck edges, softens essential connections
            excess_factor = max(0.5, min(3.0, d_uv / (local_baseline + 1e-8)))

            penalty = base_lambda * excess_factor
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty

    return updated_edge_distance
