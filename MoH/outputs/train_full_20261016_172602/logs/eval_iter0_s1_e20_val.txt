# ===== Heuristic Code =====
#{Recombines Parent 1's vectorized sigmoid z-score utility and sqrt-frequency penalization with Parent 2's mutual k-NN metric to modulate utility against premature penalization of geometric neighbors, followed by efficient argpartition-based selective edge decay on unused incident k-NN edges of penalized nodes.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Efficient k-NN mutual metric computation
    k = min(8, n - 1)
    k_nn_indices = np.argpartition(edge_distance, kth=k, axis=1)[:, :k + 1]
    u_in_knn = np.any(k_nn_indices[v_idx] == u_idx[:, None], axis=1)
    v_in_knn = np.any(k_nn_indices[u_idx] == v_idx[:, None], axis=1)
    knn_mutual = (u_in_knn & v_in_knn).astype(np.float64)

    # Sigmoid z-score weighting modulated by mutual k-NN affinity
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Mutual k-NN edges receive slight protection from penalization
    centrality_mod = 1.0 + 0.25 * (1.0 - knn_mutual)
    utilities = ((costs * (1.0 + weights)) / (1.0 + used)) * centrality_mod

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Frequency-damped penalization
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Selective discount on unused incident k-NN edges for penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.025
    for u in penalized_nodes:
        knn_u = k_nn_indices[u]
        knn_u = knn_u[knn_u != u]
        unused = knn_u[edge_n_used[u, knn_u] == 0]
        if len(unused) > 0:
            current_vals = updated_edge_distance[u, unused]
            discounted = np.maximum(1e-6, current_vals * (1.0 - decay_ratio))
            updated_edge_distance[u, unused] = discounted
            updated_edge_distance[unused, u] = discounted

    return updated_edge_distance
