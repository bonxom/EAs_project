# ===== Heuristic Code =====
#{A dynamic candidate-focused guided local search penalty mechanism that modifies edge costs via targeted delta-updates on high-utility tour edges scaled by edge length variance and utilization frequency.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Ensure original matrix is not mutated directly while adhering to signature contract
    updated_edge_distance = np.copy(edge_distance)

    n_nodes = len(local_opt_tour)
    if n_nodes < 2:
        return updated_edge_distance

    # Vectorized extraction of consecutive tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    edge_costs = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utility based on Guided Local Search (GLS) formulation
    utilities = edge_costs / (1.0 + edge_counts)
    max_utility = np.max(utilities)

    if max_utility <= 0:
        return updated_edge_distance

    # Select edges within adaptive threshold of the maximum utility
    threshold = 0.95 * max_utility
    candidate_mask = utilities >= threshold

    # Adaptive penalty scaling factoring mean cost and variance
    mean_cost = np.mean(edge_costs)
    std_cost = np.std(edge_costs)
    lambda_factor = 0.3 * (mean_cost + 0.5 * std_cost) / float(n_nodes)

    # Apply delta penalties strictly to the identified candidate edge subset
    sel_u = u_idx[candidate_mask]
    sel_v = v_idx[candidate_mask]

    penalties = lambda_factor * (1.0 / (1.0 + edge_counts[candidate_mask]))

    # In-place symmetric update on candidate entries
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
