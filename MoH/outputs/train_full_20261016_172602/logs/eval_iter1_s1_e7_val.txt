# ===== Heuristic Code =====
#{Replaces tour-isolated z-score utilities with a dual-space regret formulation that scales edge penalization by substitution difficulty relative to each node's second-best alternative while discounting promising substitutes.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Dual-space information: identify alternative candidate costs for each vertex
    # Mask self-loops with infinity to find true smallest and 2nd smallest incident edges
    dist_copy = np.copy(edge_distance)
    np.fill_diagonal(dist_copy, np.inf)

    # Efficiently find top 2 closest incident distances per node via partial sort
    k_smallest = np.partition(dist_copy, 2, axis=1)[:, :2]
    k_smallest.sort(axis=1)
    d1 = k_smallest[:, 0]
    d2 = k_smallest[:, 1]

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Substitution regret: how much worse is the next best alternative from each endpoint
    regret_u = np.maximum(0.0, d2[u_idx] - costs)
    regret_v = np.maximum(0.0, d2[v_idx] - costs)
    alt_slack = np.maximum(d2[u_idx] - d1[u_idx], d2[v_idx] - d1[v_idx])

    # Replace tour-isolated utility with dual-space substitution-based utility
    # Edges with high cost, high re-usage, and low regret (easily substituted) are prioritized
    ease_of_substitution = 1.0 / (1.0 + (regret_u + regret_v) / (costs + 1e-8))
    utilities = (costs * (1.0 + ease_of_substitution)) / (1.0 + used)

    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_sub = ease_of_substitution[penalize_mask]

    # Scale penalty severity by substitution ease and frequency of use
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_sub)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Counter-discounting of incident alternative edges to steer the search landscape
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
