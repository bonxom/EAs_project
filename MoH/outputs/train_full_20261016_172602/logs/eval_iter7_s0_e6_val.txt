# ===== Heuristic Code =====
#{Dynamically scale penalty magnitude and selection selectivity based on statistical dispersion of utilities and edge usage saturation across the tour to adaptively perturb local minima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    mean_utility = np.mean(utilities)
    std_utility = np.std(utilities)

    # Adaptive threshold using standard score dispersion (coefficient of variation of utilities)
    # When utilities are dispersed, be more selective; when homogeneous, broaden selection.
    cv = std_utility / (mean_utility + 1e-12)
    threshold_factor = np.clip(1.0 - 0.15 * (1.0 / (1.0 + cv)), 0.85, 0.995)
    threshold = threshold_factor * max_utility

    selected_mask = utilities >= threshold
    if not np.any(selected_mask):
        selected_mask = utilities == max_utility

    # Scale penalty dynamically with tour edge dispersion and edge count saturation
    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    mean_used = np.mean(tour_counts)

    # Increase perturbation intensity as search stagnates in frequently traversed edges
    stagnation_factor = 1.0 + np.tanh(mean_used / 5.0)
    adaptive_weight = 0.2 + 0.3 * (std_edge / (avg_edge + 1e-12))
    penalty = adaptive_weight * stagnation_factor * avg_edge

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
