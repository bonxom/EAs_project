# ===== Heuristic Code =====
#{This strategy preserves Guided Local Search penalties based on tour edge length, visit frequency, and geometric triangle detour inefficiency, while replacing 2-step shortcuts with a deterministic multi-hop k-NN ruin-and-recreate incentive that applies distance discounts to spatial neighbors outside the local tour neighborhood to promote non-local 3-opt and 4-opt reconnections.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Tour indexing
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    prev_u = np.roll(tour, 1)
    next_w = np.roll(tour, -3)

    # Inverted position lookup for fast neighborhood checks
    pos = np.empty(num_nodes, dtype=int)
    pos[tour] = np.arange(num_nodes)

    # Memory decay
    edge_n_used *= 0.96

    # Geometric properties of consecutive triplets (u, v, w)
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Guided Local Search edge utility calculation
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Penalize worst edges
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Multi-hop k-NN ruin-and-recreate reconnection incentives around ruined subpaths
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    num_knn_candidates = min(8, num_nodes - 1)

    for idx in high_detour_indices:
        v = v_idx[idx]
        u = u_idx[idx]
        w = w_idx[idx]
        p_u = prev_u[idx]
        n_w = next_w[idx]

        # 2-neighborhood of v along the tour to exclude trivial moves
        excluded_nodes = {v, u, w, p_u, n_w}

        # Deterministic spatial nearest neighbors of central ruined node v
        nearest_neighbors = np.argsort(edge_distance[v])
        selected_count = 0

        for candidate in nearest_neighbors:
            if candidate in excluded_nodes:
                continue

            c_pos = pos[candidate]
            c_prev = tour[(c_pos - 1) % num_nodes]
            c_next = tour[(c_pos + 1) % num_nodes]

            # Discount candidate reconnection bridges (u to candidate, candidate to w, or candidate boundaries)
            reconnect_pairs = [(u, candidate), (candidate, w), (u, c_next), (c_prev, w)]
            for n1, n2 in reconnect_pairs:
                freq = edge_n_used[n1, n2]
                if freq < 1.0:
                    discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
                    new_val = max(1e-6, updated_edge_distance[n1, n2] - discount)
                    updated_edge_distance[n1, n2] = new_val
                    updated_edge_distance[n2, n1] = new_val

            selected_count += 1
            if selected_count >= 2:
                break

    return updated_edge_distance
