# ===== Heuristic Code =====
#{This strategy combines acute detour angle inefficiency with node spatial centrality and incident usage congestion to penalize geometrically inefficient edges in structurally congested clusters while selectively discounting promising shortcut chords.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Cyclic shift indices along the tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay usage frequencies to balance exploration and exploitation
    edge_n_used *= 0.96

    # Extract consecutive edge lengths and chord lengths
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Parent A: Geometric turning angle and detour ratio
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Parent B: Spatial centrality and node congestion
    # Spatial centrality: nodes closer to the rest of the network have higher centrality
    mean_dist_per_node = np.mean(edge_distance, axis=1)
    avg_tour_dist = np.mean(mean_dist_per_node) + 1e-8
    node_centrality = avg_tour_dist / (mean_dist_per_node + 1e-8)

    # Incident edge usage congestion per node
    node_usage_congestion = np.sum(edge_n_used, axis=1) / (num_nodes + 1e-8)
    mean_usage = np.mean(node_usage_congestion) + 1e-8
    norm_node_congestion = node_usage_congestion / mean_usage

    # Combined congestion factor at the shared vertex v
    structural_weight = 1.0 + 0.3 * (node_centrality[v_idx] - 1.0) + 0.2 * norm_node_congestion[v_idx]
    structural_weight = np.clip(structural_weight, 0.5, 2.5)

    # Symmetrized local geometric factor
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Modulated utilities incorporating structural centrality and frequency
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor) * structural_weight

    # Candidate edge selection
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Problem scale and adaptive penalty scaling
    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_penalty_weight = 1.0 + 0.4 * local_geom_factor[idx] + 0.2 * structural_weight[idx]
        penalty = adaptive_lambda * freq_penalty * geom_penalty_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate incentive: selectively discount promising unvisited shortcut chords
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
