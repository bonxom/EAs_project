# ===== Heuristic Code =====
#{This algorithm applies scale-invariant, targeted Guided Local Search penalties by calibrating penalty magnitudes to raw tour edge scales, computing fast partition-based local density, and selectively amplifying penalties on the highest utility-to-usage bottleneck edges to intensify structural disruption of local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Scale reference: mean and maximum tour edge length
    avg_tour_length = np.mean(tour_edge_lengths)
    if avg_tour_length < 1e-12:
        return updated_edge_distance

    # Scale-invariant standard GLS utility: length / (1 + usage)
    gls_utility = tour_edge_lengths / (1.0 + usage)

    # Fast local spatial density estimation via argpartition on the subset of tour nodes
    k = min(6, n - 1)
    if k > 1:
        sampled_dists = edge_distance[u_idx]
        part_idx = np.argpartition(sampled_dists, kth=k, axis=1)[:, 1:k+1]
        k_dists = np.take_along_axis(sampled_dists, part_idx, axis=1)
        k_nn_avg = np.mean(k_dists, axis=1)
        node_density = 1.0 / (k_nn_avg + 1e-6)
        mean_density = np.mean(node_density)
        node_density = node_density / (mean_density + 1e-6)
        density_factor = 0.5 * (node_density + np.roll(node_density, -1))
    else:
        density_factor = np.ones(n, dtype=np.float64)

    # Relative length factor with smooth sigmoid weighting
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_weight = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Base penalty proportional to the edge distance scale (scale-invariance)
    penalty_base = 0.25 * avg_tour_length * (tour_edge_lengths / (1.0 + 0.5 * usage)) / (avg_tour_length + 1e-6)
    combined_penalty = penalty_base * (0.6 * rank_weight + 0.4 * density_factor)

    # Highly targeted intensification: selectively amplify top 25% utility edges (bottlenecks)
    utility_threshold = np.percentile(gls_utility, 75)
    target_mask = gls_utility >= utility_threshold
    combined_penalty[target_mask] *= 1.8

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
