# ===== Heuristic Code =====
#{Tear down rigid deterministic top-k selection and replace it with a greedy independent edge set tournament weighted by utility, tour cost contribution, and topological sparsity to prevent contiguous over-penalization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base utilities enriched with relative tour-cost contribution
    total_tour_cost = np.sum(d_uv) + 1e-8
    rel_cost_contrib = d_uv / total_tour_cost
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor) * (1.0 + rel_cost_contrib * num_nodes)

    # Deterministic non-adjacent independent edge set selection via score tournament
    sorted_candidate_order = np.argsort(-utilities)
    target_count = max(1, int(np.ceil(0.12 * num_nodes)))

    selected_indices = []
    masked_nodes = np.zeros(num_nodes, dtype=bool)

    for idx in sorted_candidate_order:
        prev_idx = (idx - 1) % num_nodes
        next_idx = (idx + 1) % num_nodes

        # Enforce independent edge selection to avoid over-penalizing contiguous edges
        if not masked_nodes[idx] and not masked_nodes[prev_idx] and not masked_nodes[next_idx]:
            selected_indices.append(idx)
            masked_nodes[idx] = True
            if len(selected_indices) >= target_count:
                break

    # Fallback to highest utilities if strict independent set is saturated
    if len(selected_indices) < target_count:
        for idx in sorted_candidate_order:
            if idx not in selected_indices:
                selected_indices.append(idx)
                if len(selected_indices) >= target_count:
                    break

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Reward shortcuts on high detour locations
    high_detour_indices = np.argsort(detour_ratio)[-target_count:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
