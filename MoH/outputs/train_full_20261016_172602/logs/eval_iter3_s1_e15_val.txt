# ===== Heuristic Code =====
#{This algorithm implements dynamic penalty evaporation combined with an exponential counter-fatigue booster on persistent tour edges alongside targeted discount rewards on under-explored alternatives to aggressively break cyclic stagnation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy initial edge distance matrix to preserve inputs
    updated_edge_distance = np.copy(edge_distance)

    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Step 2: Global soft memory evaporation: slight decay of existing penalties on all modified edges
    max_used = np.max(edge_n_used)
    if max_used > 0:
        evap_factor = 0.995
        # Modulate evaporation softly towards original distances or scale globally
        updated_edge_distance = updated_edge_distance * evap_factor + edge_distance * (1.0 - evap_factor)
        costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)

    # Step 3: Compute edge utility with z-score cost weighting and usage penalty
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Step 4: Counter-fatigue exponential scaling for chronically reappearing stubborn edges
    # Base penalty proportional to cost and relative utility, boosted exponentially by recurring appearances
    fatigue_exponent = np.minimum(p_used / (np.mean(p_used) + 1.0), 3.0)
    boost_factor = 1.0 + 0.35 * (np.exp(fatigue_exponent) - 1.0)
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * boost_factor

    # Symmetrically apply penalties
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Step 5: Reward promising unexplored or rarely used alternative incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.025
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
