# ===== Heuristic Code =====
#{Rank-order selection of the top-k highest-utility non-adjacent edges with adaptive rank-decayed penalties to prevent disruptive simultaneous neighbor breaks while consistently escaping local minima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Sort tour edge indices by utility descending
    sorted_indices = np.argsort(-utilities)

    # Deterministically select up to k highest-utility non-adjacent edges
    target_k = max(1, min(int(np.ceil(np.sqrt(n))), n // 4))

    # Track selected edge positions along the cycle to avoid adjacent edge disruption
    taken_mask = np.zeros(n, dtype=bool)
    selected_indices = []

    for idx in sorted_indices:
        prev_idx = (idx - 1) % n
        next_idx = (idx + 1) % n
        if not taken_mask[idx] and not taken_mask[prev_idx] and not taken_mask[next_idx]:
            selected_indices.append(idx)
            taken_mask[idx] = True
            if len(selected_indices) >= target_k:
                break

    sel = np.array(selected_indices, dtype=np.intp)
    sel_u = u_idx[sel]
    sel_v = v_idx[sel]

    # Rank-proportional decay on penalty to differentiate structural priorities
    rank_weights = 1.0 / np.sqrt(np.arange(1, len(sel) + 1))
    penalties = base_penalty * rank_weights

    # Symmetrically apply additive penalization without Python-level loops
    np.add.at(updated_edge_distance, (sel_u, sel_v), penalties)
    np.add.at(updated_edge_distance, (sel_v, sel_u), penalties)

    return updated_edge_distance
