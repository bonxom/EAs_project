#{Adaptive guided local search penalty function that normalizes edge lengths against k-NN local densities and tour scale, discounts repeated edge usage, and boosts high-impact edges via a dynamic percentile cutoff modulated by the tour length coefficient of variation applied to the composite topology score.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # K-nearest neighbor baseline for node density estimation
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Active tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Tour length statistics and coefficient of variation (CV)
    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths)
    cv = std_tour_length / (avg_tour_length + 1e-6)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Continuous rank and local penalty components
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Adaptive weighting between global and local penalties
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    composite_score = w_global * rank_penalty + w_local * local_penalty

    # CV-modulated adaptive percentile cutoff on composite utility/topology score
    cutoff_pct = np.clip(75.0 - 15.0 * cv, 50.0, 85.0)
    adaptive_threshold = np.percentile(composite_score, cutoff_pct)

    # Smooth continuous amplification above adaptive threshold
    excess_ratio = np.maximum(0.0, (composite_score - adaptive_threshold) / (np.max(composite_score) - adaptive_threshold + 1e-6))
    boost_factor = 1.0 + 0.5 * (excess_ratio ** 0.5)
    combined_penalty = composite_score * boost_factor

    # Symmetric in-place penalty accumulation
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
