# ===== Heuristic Code =====
#{Multi-hop lookahead shortcut discounting over 3- and 4-hop spans combined with Delaunay-inspired proximity/intersection hazard weighting to guide edge penalization and perturbation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 5:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)

    # Asymptotic frequency decay
    edge_n_used *= 0.96

    # Tour edge lengths
    d_uv = edge_distance[u_idx, v_idx]
    global_mean_dist = float(np.mean(edge_distance))

    # Multi-hop node positions: 2-hop, 3-hop, and 4-hop forward
    w2_idx = np.roll(tour, -2)
    w3_idx = np.roll(tour, -3)
    w4_idx = np.roll(tour, -4)

    # Path distances along the current tour for 2, 3, and 4 hops
    d_v_w2 = edge_distance[v_idx, w2_idx]
    d_w2_w3 = edge_distance[w2_idx, w3_idx]
    d_w3_w4 = edge_distance[w3_idx, w4_idx]

    path_2hop = d_uv + d_v_w2
    path_3hop = path_2hop + d_w2_w3
    path_4hop = path_3hop + d_w3_w4

    # Direct shortcut distances across hops
    d_u_w2 = edge_distance[u_idx, w2_idx]
    d_u_w3 = edge_distance[u_idx, w3_idx]
    d_u_w4 = edge_distance[u_idx, w4_idx]

    # Lookahead detour ratios reflecting inefficiency of longer loops
    detour_2 = path_2hop / (d_u_w2 + 1e-8)
    detour_3 = path_3hop / (d_u_w3 + 1e-8)
    detour_4 = path_4hop / (d_u_w4 + 1e-8)

    # Proximity hazard heuristic: edges that violate Delaunay empty-circle/Gabriel properties
    # An edge (u, v) has a high crossing hazard if there exists another node k where d(u,k)^2 + d(v,k)^2 < d(u,v)^2
    # We sample k from the immediate 2-hop and 3-hop neighborhood
    gabriel_viol_2 = np.maximum(0.0, (d_uv**2) - (d_u_w2**2 + d_v_w2**2)) / (d_uv**2 + 1e-8)
    gabriel_viol_3 = np.maximum(0.0, (d_uv**2) - (d_u_w3**2 + edge_distance[v_idx, w3_idx]**2)) / (d_uv**2 + 1e-8)
    crossing_hazard = gabriel_viol_2 + 0.5 * gabriel_viol_3

    # Multi-hop compound detour factor
    detour_factor = 0.5 * detour_2 + 0.3 * detour_3 + 0.2 * detour_4
    geom_inefficiency = detour_factor * (1.0 + 0.6 * crossing_hazard)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Guided Local Search utilities
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply guided penalties to top-k inefficient / high-utility edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Adaptive multi-hop shortcut incentive: incentivize best shortcuts skipping 2, 3, or 4 nodes
    # Collect candidate shortcuts across 2-hop, 3-hop, and 4-hop paths
    gains_2 = path_2hop - d_u_w2
    gains_3 = path_3hop - d_u_w3
    gains_4 = path_4hop - d_u_w4

    best_hop = np.zeros(num_nodes, dtype=np.intp)
    best_gains = gains_2.copy()

    mask_3 = gains_3 > best_gains
    best_gains[mask_3] = gains_3[mask_3]
    best_hop[mask_3] = 1

    mask_4 = gains_4 > best_gains
    best_gains[mask_4] = gains_4[mask_4]
    best_hop[mask_4] = 2

    top_shortcut_indices = np.argsort(best_gains)[-k:]
    for idx in top_shortcut_indices:
        h = best_hop[idx]
        if h == 0:
            target = w2_idx[idx]
        elif h == 1:
            target = w3_idx[idx]
        else:
            target = w4_idx[idx]

        u = u_idx[idx]
        freq = edge_n_used[u, target]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            new_cost = max(1e-6, updated_edge_distance[u, target] - discount)
            updated_edge_distance[u, target] = new_cost
            updated_edge_distance[target, u] = new_cost

    return updated_edge_distance
