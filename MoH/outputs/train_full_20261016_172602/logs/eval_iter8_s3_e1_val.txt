#{This strategy blends adaptive global-to-local penalty weighting with efficient np.partition density estimation, specifically amplifying penalties on long edges in high-density regions that exceed the percentile threshold.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Fast k-NN computation using np.partition
    k = min(6, n - 1)
    partitioned_dists = np.partition(edge_distance, kth=k, axis=1)
    k_nn_avg = np.mean(partitioned_dists[:, 1:k+1], axis=1)

    # Node density estimation
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Edge metrics along the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Global logistic rank penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Frequency and edge usage dampening
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    edge_density = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * edge_density

    # Adaptive global-to-local schedule based on usage history
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Gating by length percentile combined with targeted high-density long-edge amplification
    p75_threshold = np.percentile(tour_edge_lengths, 75)
    percentile_mask = tour_edge_lengths >= p75_threshold
    targeted_mask = percentile_mask & (edge_density > 1.0) & (relative_length > 1.0)

    # Apply baseline upper percentile boost and additional targeted boost
    combined_penalty[percentile_mask] *= 1.5
    combined_penalty[targeted_mask] *= 1.5

    # Symmetric distance matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
