# ===== Heuristic Code =====
#{This strategy replaces static uniform frequency decay with a deterministic stagnation-triggered re-initialization ruin operator that resets stagnant low-frequency penalties, disperses top penalty concentrations via deterministic chaotic phase perturbation, and re-centers baseline edge distances when tour overlap exceeds threshold.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    global_mean_dist = np.mean(edge_distance)

    # Calculate stagnation metric based on tour edge frequency concentration
    tour_freqs = edge_n_used[u_idx, v_idx]
    mean_tour_freq = np.mean(tour_freqs)
    max_freq = np.max(edge_n_used)

    # Stagnation ratio measures the depth of repeated visits along the current tour
    stagnation_ratio = mean_tour_freq / (max_freq + 1e-8) if max_freq > 0.0 else 0.0

    if stagnation_ratio > 0.80 and max_freq > 5.0:
        # Re-initialization ruin operator: escape deep basins deterministically
        # Zero out weak/stagnant non-tour frequencies
        tour_mask = np.zeros_like(edge_n_used, dtype=bool)
        tour_mask[u_idx, v_idx] = True
        tour_mask[v_idx, u_idx] = True

        low_freq_non_tour = (~tour_mask) & (edge_n_used < 0.5 * mean_tour_freq)
        edge_n_used[low_freq_non_tour] = 0.0

        # Deterministic high-entropy perturbation on heavily penalized edges using golden ratio dispersion
        heavy_penalty_mask = edge_n_used > (0.75 * max_freq)
        coords_u, coords_v = np.nonzero(heavy_penalty_mask)
        if len(coords_u) > 0:
            phi = 0.618033988749895
            pseudo_entropy = (np.sin((coords_u + 1) * phi + (coords_v + 1) * (1.0 - phi)) + 1.0) * 0.5
            edge_n_used[coords_u, coords_v] *= (0.35 + 0.30 * pseudo_entropy)
            edge_n_used[coords_v, coords_u] = edge_n_used[coords_u, coords_v]

        # Re-center decay
        edge_n_used *= 0.85
    else:
        # Standard dynamic frequency decay
        decay_factor = 0.95 - 0.05 * stagnation_ratio
        edge_n_used *= decay_factor

    # Core geometric calculations
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate shortcuts by discounting promising chords
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
