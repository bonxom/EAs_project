# ===== Heuristic Code =====
#{An adaptive feedback-driven guided local search heuristic that dynamically adjusts edge penalty magnitude and selection threshold based on search stagnation, usage skewness, and edge utility dispersion.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    # Calculate state statistics to dynamically guide penalty strength and selection
    avg_count = np.mean(tour_counts)
    max_count = np.max(tour_counts)
    std_count = np.std(tour_counts)

    # Stagnation indicator derived from count dispersion and maximum edge reuse
    stagnation_factor = np.tanh(max_count / (avg_count + 1.0) - 1.0)

    # Base penalty scales with tour edge distribution and intensifies with stagnation
    mean_length = np.mean(tour_lengths)
    std_length = np.std(tour_lengths)
    cv_length = std_length / (mean_length + 1e-12)

    # Adaptive penalty coefficient bounded dynamically between 0.15 and 0.65
    penalty_coeff = 0.25 + 0.20 * stagnation_factor + 0.10 * np.clip(cv_length, 0.0, 1.0)
    penalty_val = penalty_coeff * (mean_length + std_length)

    # Dynamic utility threshold based on utility dispersion (stagnation broadens selection)
    util_mean = np.mean(utilities)
    util_std = np.std(utilities)
    util_max = np.max(utilities)

    # Higher stagnation lowers the threshold to penalize a broader cluster of bottleneck edges
    k = max(0.5, 1.5 - stagnation_factor)
    threshold = max(util_mean + k * util_std, 0.90 * util_max)

    selected_mask = utilities >= threshold

    # Guarantee at least the single maximum utility edge is selected
    if not np.any(selected_mask):
        selected_mask = utilities >= util_max

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Proportional edge-specific penalty weighting to penalize persistent bottleneck edges more severely
    relative_weights = 1.0 + (tour_counts[selected_mask] / (max_count + 1.0))
    penalties = penalty_val * relative_weights

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
