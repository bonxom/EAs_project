# ===== Heuristic Code =====
#{Replace 3-node detour metrics with a multi-scale chordal curvature operator across triplets and quadruplets that measures perimeter-to-chord expansion ratios and crossing tensions to penalize tangled sub-paths and incentivize direct shortcuts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Pre-calculate shifted node indices for multi-scale windows
    t_idx = np.roll(tour, 1)    # i - 1
    u_idx = tour                # i
    v_idx = np.roll(tour, -1)   # i + 1
    w_idx = np.roll(tour, -2)   # i + 2

    # Decay edge historical counts
    edge_n_used *= 0.96

    # Scale 1: Triplet (u, v, w) perimeter vs chord (u, w)
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]
    triplet_detour = (d_uv + d_vw) / (d_uw + 1e-8)

    # Acute angle / turning deficit at v
    denom_triplet = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom_triplet, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    triplet_curvature = triplet_detour * (1.0 + 0.5 * acute_deficit)

    # Scale 2: Quadruplet (t, u, v, w) perimeter vs chord (t, w) and crossing potential
    d_tu = edge_distance[t_idx, u_idx]
    d_tw = edge_distance[t_idx, w_idx]
    quad_detour = (d_tu + d_uv + d_vw) / (d_tw + 1e-8)

    # Crossing potential: diagonals vs perimeters of the 4-cycle (t, u, v, w)
    # If diagonals d_tv + d_uw are shorter than d_tu + d_vw, local sub-path is likely crossed/tangled
    d_tv = edge_distance[t_idx, v_idx]
    crossing_tension = (d_tu + d_vw) / (d_tv + d_uw + 1e-8)
    crossing_tension = np.clip(crossing_tension, 0.5, 3.0)
    quad_curvature = quad_detour * crossing_tension

    # Multi-scale curvature assigned to edge (u, v)
    # Edge (u, v) participates in triplet at index i (u->v->w) and index i-1 (t->u->v)
    triplet_factor = 0.5 * (triplet_curvature + np.roll(triplet_curvature, 1))
    quad_factor = 0.5 * (quad_curvature + np.roll(quad_curvature, 1))
    multi_scale_curvature = 0.6 * triplet_factor + 0.4 * quad_factor

    # Guided Local Search utilities incorporating multi-scale geometry
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * multi_scale_curvature)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize top-k inefficient edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * multi_scale_curvature[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate / incentivize prominent shortcut chords from high detour triplets & quadruplets
    high_triplet_indices = np.argsort(triplet_detour)[-k:]
    for idx in high_triplet_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    high_quad_indices = np.argsort(quad_detour)[-max(1, k // 2):]
    for idx in high_quad_indices:
        t, w = t_idx[idx], w_idx[idx]
        freq = edge_n_used[t, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[t, w] = max(1e-6, updated_edge_distance[t, w] - discount)
            updated_edge_distance[w, t] = updated_edge_distance[t, w]

    return updated_edge_distance
