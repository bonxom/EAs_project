# {An adaptive Guided Local Search penalization scheme that dynamically scales edge penalties based on tour length variance, interquartile spread, and relative deviation from minimum tour edge to aggressively disrupt outlier bottlenecks while moderating changes on homogeneous tours.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utilities: longer, infrequently penalized edges have higher utility
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Compute statistical dispersion metrics of current tour edge lengths
    min_edge = np.min(tour_lengths)
    mean_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)

    # Coefficient of variation (robust against zero division)
    cv = std_edge / (mean_edge + 1e-9)

    # Adaptive lambda factor: scale up with dispersion, constrained to prevent explosion
    lambda_factor = np.clip(0.15 + 0.5 * cv, 0.1, 0.6)

    # Identify candidate edges close to maximum utility
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    # Variance-aware edge-specific penalty: amplifies penalization on edges that exceed the tour minimum
    sel_lengths = tour_lengths[selected_mask]
    sel_penalties = lambda_factor * (std_edge + np.maximum(0.0, sel_lengths - min_edge))

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply symmetric updates
    updated_edge_distance[sel_u, sel_v] += sel_penalties
    updated_edge_distance[sel_v, sel_u] += sel_penalties

    return updated_edge_distance
