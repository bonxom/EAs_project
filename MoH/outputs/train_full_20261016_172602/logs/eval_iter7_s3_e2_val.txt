#{Differential synthesis combining partition-based k-NN local scaling, adjacent-edge slack ratios, and a CV-adaptive threshold for targeted GLS edge penalization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy initial edge distance matrix to preserve original data
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Step 2: Efficient k-NN scale estimation using partition rather than full sort
    k = min(6, n - 1)
    k_partition = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    node_scale = np.mean(k_partition, axis=1) + 1e-6
    node_density = 1.0 / node_scale
    norm_density = node_density / np.mean(node_density)

    # Step 3: Extract cyclic tour indices and adjacent context
    p_idx = np.roll(local_opt_tour, 1)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    w_idx = np.roll(local_opt_tour, -2)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Step 4: Contextual geometric ratio using local adjacent tour edge lengths
    adj_u_lens = edge_distance[p_idx, u_idx]
    adj_v_lens = edge_distance[v_idx, w_idx]
    local_adj_mean = 0.5 * (adj_u_lens + adj_v_lens) + 1e-6
    local_slack_ratio = tour_edge_lengths / local_adj_mean

    # Step 5: Geometric mean scale of the connected nodes
    edge_scale = np.sqrt(node_scale[u_idx] * node_scale[v_idx])
    relative_scale = tour_edge_lengths / edge_scale

    # Step 6: Sigmoid rank and local utilization penalties
    rank_penalty = 1.0 / (1.0 + np.exp(-3.5 * (local_slack_ratio - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (norm_density[u_idx] + norm_density[v_idx])
    local_penalty = relative_scale * usage_factor * density_factor

    # Step 7: Adaptive blend based on global search frequency
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Step 8: Coefficient of variation (CV)-adaptive percentile cutoff for bottleneck/bridge edges
    mean_len = np.mean(tour_edge_lengths)
    std_len = np.std(tour_edge_lengths)
    cv = std_len / (mean_len + 1e-6)
    cutoff_percentile = np.clip(85.0 - 25.0 * cv, 60.0, 90.0)
    threshold = np.percentile(tour_edge_lengths, cutoff_percentile)

    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= (1.2 + 0.4 * np.clip(cv, 0.0, 1.0))

    # Step 9: Symmetrically accumulate penalties into updated distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
