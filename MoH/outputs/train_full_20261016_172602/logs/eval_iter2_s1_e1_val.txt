# ===== Heuristic Code =====
#{Identifies high-utility tour edges using Z-score sigmoid gated cost-frequency metrics to penalize them, while strictly discounting only the k-nearest low-frequency neighbors of penalized endpoints with distance-normalized incentives to carve spatially viable escape routes.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    utilities = (costs * (1.0 + weights)) / (1.0 + used)
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities == np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Penalize the selected tour edges
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # K-Nearest-Neighbor Directed Inflow: discount only high-quality spatial neighbors
    k_neighbors = min(6, n - 2)
    base_discount_rate = 0.03

    for u in penalized_nodes:
        orig_row = edge_distance[u]
        # Candidate neighbors excluding self and current adjacent tour neighbors
        candidate_indices = np.argsort(orig_row)
        valid_candidates = []
        for cand in candidate_indices:
            if cand == u:
                continue
            # Prefer low-frequency edges
            if edge_n_used[u, cand] <= 1:
                valid_candidates.append(cand)
            if len(valid_candidates) == k_neighbors:
                break

        if valid_candidates:
            cands = np.array(valid_candidates, dtype=np.intp)
            cand_costs = orig_row[cands]
            min_cand_cost = cand_costs[0] + 1e-8
            # Distance-normalized incentive scaling: closer neighbors receive a stronger discount
            rel_closeness = min_cand_cost / (cand_costs + 1e-8)
            freq_damp = 1.0 / (1.0 + edge_n_used[u, cands])
            discount = updated_edge_distance[u, cands] * (base_discount_rate * rel_closeness * freq_damp)

            updated_edge_distance[u, cands] -= discount
            updated_edge_distance[cands, u] = updated_edge_distance[u, cands]

    return updated_edge_distance
