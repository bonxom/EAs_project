# {Vectorized adaptive edge penalization with non-monotonic memory decay that scales penalties dynamically while relaxing non-critical edges to prevent landscape saturation and escape stagnation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    mean_utility = np.mean(utilities)
    avg_edge = np.mean(tour_lengths)

    # Adaptive penalty scaling based on utility spread to avoid stagnation
    spread = max_utility - mean_utility
    penalty_scale = 0.25 + 0.15 * (spread / (max_utility + 1e-12))
    penalty = penalty_scale * avg_edge

    threshold = 0.95 * max_utility
    high_utility_mask = utilities >= threshold

    # Fractional continuous relaxation for low-utility tour edges to prevent saturation
    decay_factor = 0.98
    low_utility_mask = ~high_utility_mask

    # Apply continuous relaxation on low-utility tour edges
    decay_u = u_idx[low_utility_mask]
    decay_v = v_idx[low_utility_mask]
    decay_diff = (1.0 - decay_factor) * tour_lengths[low_utility_mask]
    updated_edge_distance[decay_u, decay_v] -= decay_diff
    updated_edge_distance[decay_v, decay_u] -= decay_diff

    # Penalize candidate bottleneck / high-utility edges
    sel_u = u_idx[high_utility_mask]
    sel_v = v_idx[high_utility_mask]
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
