# ===== Heuristic Code =====
#{This algorithm synthesizes Guided Local Search with node spatial centrality scaling, amplifying geometric detour and turn penalties in dense central regions where alternatives exist while moderating penalties in sparse peripheral zones and incentivizing promising shortcut chords.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Node centrality profiling: mean distance of each node relative to the global mean distance
    global_mean_dist = np.mean(edge_distance)
    node_mean_dist = np.mean(edge_distance, axis=1)
    # Higher centrality_ratio corresponds to dense central nodes; lower to peripheral nodes
    centrality_ratio = global_mean_dist / (node_mean_dist + 1e-8)

    # Vectorized consecutive triplets along the tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay historical edge utilization counts
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour ratio and turn angle estimation at apex node v
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn

    # Centrality at apex v scales geometric inefficiency: sharp turns in dense centers are heavily penalized
    apex_centrality = centrality_ratio[v_idx]
    geom_inefficiency = detour_ratio * (1.0 + 0.45 * acute_deficit * apex_centrality)

    # Distribute geometric factor across adjacent edges
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Edge-level spatial centrality (average centrality of incident vertices)
    edge_centrality = 0.5 * (centrality_ratio[u_idx] + centrality_ratio[v_idx])

    # Modulated utilities incorporating base distance, usage frequency, geometry, and spatial density
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.30 * local_geom_factor * edge_centrality)

    # Top-k edge penalization
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.40 * local_geom_factor[idx] * edge_centrality[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Incentive mechanism: discount shortcut chords spanning inefficient detours in dense/central clusters
    chord_attractiveness = detour_ratio * apex_centrality
    high_detour_indices = np.argsort(chord_attractiveness)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            chord_centrality = 0.5 * (centrality_ratio[u] + centrality_ratio[w])
            discount = 0.08 * global_mean_dist * (chord_centrality / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
