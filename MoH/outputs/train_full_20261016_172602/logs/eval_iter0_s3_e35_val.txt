# {Calibrate guided local search penalties dynamically by scaling unitless multi-factor edge scores with problem-scale-invariant tour dispersion metrics and mean edge lengths.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    # Compute local topological density using k-nearest neighbors
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-8)
    node_density = node_density / (np.mean(node_density) + 1e-8)

    # Edge indices along the incumbent tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Baseline scale metrics of the tour
    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths)
    # Dimensionally consistent base metric (accounting for variance and mean)
    base_metric = avg_tour_length + 0.5 * std_tour_length + 1e-8

    # Dimensionless normalized length and rank penalty
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-8)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Historical usage damping
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])

    # Topological context factor
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Balance global and local components
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    unitless_score = w_global * rank_penalty + w_local * local_penalty

    # Amplify penalties for the longest quartile of edges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    unitless_score[high_penalty_mask] *= 1.5

    # Calibrate dimensional penalty directly proportional to scale and problem size
    penalty_intensity = 0.35 / (1.0 + np.log1p(freq_mean))
    calibrated_penalty = base_metric * penalty_intensity * unitless_score

    # Apply symmetric updates to distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), calibrated_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), calibrated_penalty)

    return updated_edge_distance
