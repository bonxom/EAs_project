# ===== Heuristic Code =====
# {Compute vectorized tour edge utilities and apply feature-proportional graded penalties scaled by each selected edge's utility excess relative to the median tour utility, avoiding flat matrix distortion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    median_utility = np.median(utilities)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    threshold = 0.90 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_utilities = utilities[selected_mask]

    denom = max(max_utility - median_utility, 1e-8)
    excess_ratios = np.maximum(0.0, (sel_utilities - median_utility) / denom)

    base_penalty = 0.3 * avg_edge
    penalties = base_penalty * (1.0 + excess_ratios)

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
