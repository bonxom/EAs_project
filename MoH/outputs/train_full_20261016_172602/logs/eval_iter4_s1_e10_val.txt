# ===== Heuristic Code =====
#{Enhance Guided Local Search by incorporating 2-hop chain usage fatigue into edge utility evaluation and applying selective top-quantile penalization while guiding exploration toward unvisited k-nearest neighbors of penalized endpoints.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # 2-hop adjacent edge fatigue: weight edges by adjacent edges' usage to penalize stubborn path motifs
    prev_used = np.roll(used, 1)
    next_used = np.roll(used, -1)
    chain_fatigue = 1.0 + 0.5 * (prev_used + next_used) / (1.0 + used)

    base_utilities = costs / (1.0 + used)
    utilities = base_utilities * chain_fatigue

    # Selective top-quantile penalization (top 25% or at least edges above 75th percentile)
    threshold = np.percentile(utilities, 75)
    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities == np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]

    # Modulated penalty proportional to edge cost and inverse square-root usage
    penalties = 0.25 * p_costs / np.sqrt(1.0 + p_used)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Targeted bonus to compact, unvisited nearest neighbors of penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k_neighbors = min(5, n - 1)
    for u in penalized_nodes:
        orig_dists = edge_distance[u].copy()
        orig_dists[u] = np.inf
        nn_indices = np.argpartition(orig_dists, k_neighbors)[:k_neighbors]
        unvisited_nn = nn_indices[edge_n_used[u, nn_indices] == 0]
        if unvisited_nn.size > 0:
            bonus = 0.015 * updated_edge_distance[u, unvisited_nn]
            updated_edge_distance[u, unvisited_nn] -= bonus
            updated_edge_distance[unvisited_nn, u] = updated_edge_distance[u, unvisited_nn]

    return updated_edge_distance
