#{Replace heuristic utility multipliers with classical Guided Local Search formulation using a dimension-free dynamic penalty parameter lambda proportional to average edge length to achieve scale-invariant disruption.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n == 0:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    tour_length = np.sum(costs)
    avg_edge_cost = tour_length / n

    # Voudouris & Tsang utility: cost / (1 + count)
    utilities = costs / (1.0 + used)

    # Identify edges with maximum utility (or top tier above median to maintain adequate disruption)
    max_utility = np.max(utilities)
    threshold = max_utility * 0.999999
    penalize_mask = utilities >= threshold

    # If too few edges meet threshold, penalize edges above the median utility
    if np.sum(penalize_mask) == 0:
        penalize_mask = utilities >= np.median(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]

    # Dimensionless self-scaling penalty: lambda = alpha * TourLength / n
    # Penalty increment = lambda = alpha * avg_edge_cost
    alpha = 0.3
    penalty_value = alpha * avg_edge_cost

    # Apply symmetric penalty updates
    updated_edge_distance[p_u, p_v] += penalty_value
    updated_edge_distance[p_v, p_u] += penalty_value

    return updated_edge_distance
