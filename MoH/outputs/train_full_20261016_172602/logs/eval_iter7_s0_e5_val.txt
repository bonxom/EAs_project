#{Decouple structural instance metrics from accumulated penalties by estimating base edge distances, normalizing them using robust tour dispersion, and calculating penalized utilities to guide targeted updates.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_edge_dists = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Reconstruct structural base metric by removing prior penalty distortions
    mean_current_dist = np.mean(current_edge_dists)
    base_penalty_unit = 0.3 * mean_current_dist
    estimated_base_lengths = np.maximum(1e-6, current_edge_dists - (tour_counts * base_penalty_unit))

    # Robust dispersion normalization (mean absolute deviation from median)
    median_length = np.median(estimated_base_lengths)
    dispersion = np.mean(np.abs(estimated_base_lengths - median_length)) + 1e-8
    normalized_lengths = (estimated_base_lengths - median_length) / dispersion

    # Decoupled utility calculation using structural topology relative to usage frequency
    structural_weights = np.exp(np.clip(normalized_lengths, -4.0, 4.0))
    utilities = structural_weights / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    penalty = 0.3 * np.mean(estimated_base_lengths)
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
