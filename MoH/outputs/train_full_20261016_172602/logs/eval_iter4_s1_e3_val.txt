# ===== Heuristic Code =====
#{Differential ruin-and-recreate synthesizes mutual k-nearest neighbor topological filtering into the vectorized guided local search framework, discounting the penalty utility of mutual k-NN backbone edges to selectively target non-mutual, bridging connections for symmetric penalization and incident decay.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Identify tour edges
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # 2. Geometric Mutual k-Nearest Neighbor (k-NN) Evaluation
    # Determine adaptive k (e.g., k = min(10, n - 1))
    k = min(10, n - 1)
    if k > 0:
        # np.argpartition to find k nearest neighbors for each node (excluding self)
        # We find the threshold distance to the k-th nearest neighbor
        kth_dists = np.partition(edge_distance, k, axis=1)[:, k]
        # An edge (u, v) is in u's k-NN if cost <= kth_dists[u]
        is_u_knn = costs <= kth_dists[u_idx]
        is_v_knn = costs <= kth_dists[v_idx]
        mutual_knn = is_u_knn & is_v_knn
    else:
        mutual_knn = np.zeros(n, dtype=bool)

    # Mutual k-NN factor: mutual nearest neighbors are structural backbone, reduce their utility
    knn_discount = np.where(mutual_knn, 0.5, 1.0)

    # 3. Vectorized Utility Computation
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Incorporate mutual k-NN metric into utility evaluation
    utilities = ((costs * (1.0 + weights)) / (1.0 + used)) * knn_discount

    # 4. Median-based selective thresholding
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # 5. Symmetrical Penalty Application
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # 6. Targeted Node-Incident Discounting for Unused Alternative Edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
