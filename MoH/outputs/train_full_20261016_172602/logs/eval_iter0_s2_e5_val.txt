# ===== Heuristic Code =====
# {A hybrid guided local search strategy that applies high-intensity multi-factor modulated penalties to top utility tour edges while regularizing all previously visited edges with an entropy-gated background penalty field to avoid entrapment in alternate local basins.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)
    w_idx = np.roll(u_idx, -2)

    # Trace decay
    edge_n_used *= 0.95

    # Utility calculation on current tour edges
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    k = max(1, int(np.ceil(0.15 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Search stagnation metrics
    tour_usage = edge_n_used[u_idx, v_idx]
    avg_tour_usage = np.mean(tour_usage)
    max_used = np.max(edge_n_used) + 1e-12
    overlap_ratio = np.mean(tour_usage > 1.0)
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (avg_tour_usage / max_used)

    # Node entropy and rigidity
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    # Path-curvature segment weights
    segment_weights = np.zeros(num_nodes)
    for i in range(num_nodes):
        u, v, w = u_idx[i], v_idx[i], w_idx[i]
        segment_weights[v] = 0.5 * (edge_n_used[u, v] + edge_n_used[v, w])
    norm_segment_weights = segment_weights / (np.max(segment_weights) + 1e-12)

    # Graph-usage and spatial centrality
    node_centrality = np.sum(edge_n_used, axis=1)
    norm_centrality = node_centrality / (np.max(node_centrality) + 1e-12)
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial_centrality = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Adaptive base penalty multiplier
    adaptive_lambda = (0.15 + 0.20 * np.tanh(stagnation_factor)) * global_mean_dist

    # Background entropy-gated regularization on all active historical edges
    active_mask = edge_n_used > 0
    gate_matrix = 0.5 * (entropy_rigidity[:, None] + entropy_rigidity[None, :])
    background_penalty = 0.05 * adaptive_lambda * edge_n_used * (1.0 + gate_matrix)
    updated_edge_distance[active_mask] += background_penalty[active_mask]

    # Targeted high-intensity modulated penalty for selected top-k tour edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        c_mod = 0.25 * (norm_centrality[u] + norm_centrality[v])
        sc_mod = 0.25 * (norm_spatial_centrality[u] + norm_spatial_centrality[v])
        r_mod = 0.20 * (entropy_rigidity[u] + entropy_rigidity[v])
        s_mod = 0.15 * (norm_segment_weights[u] + norm_segment_weights[v])
        modulation = 1.0 + c_mod + sc_mod + r_mod + s_mod
        targeted_penalty = adaptive_lambda * edge_n_used[u, v] * modulation
        updated_edge_distance[u, v] += targeted_penalty
        updated_edge_distance[v, u] += targeted_penalty

    return updated_edge_distance
