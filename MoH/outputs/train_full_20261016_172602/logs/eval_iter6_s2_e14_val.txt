#{Guided Local Search edge penalization enhanced by multi-scale topological detour evaluation incorporating extended 4-point quad perimeter slack and spatial k-nearest neighbor detour margins with selective shortcut discounting.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Node sequences for 4-point window evaluation (u -> v -> w -> z)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    z_idx = np.roll(tour, -3)

    edge_n_used *= 0.96

    # Base edge lengths
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wz = edge_distance[w_idx, z_idx]
    d_uw = edge_distance[u_idx, w_idx]
    d_uz = edge_distance[u_idx, z_idx]
    d_vz = edge_distance[v_idx, z_idx]

    # Multi-scale detour evaluation:
    # 1. Extended 4-point quad slack: compares 3-edge path (u-v-w-z) to diagonal shortcuts (u-z or u-w + w-z or u-v + v-z)
    path_uvwz = d_uv + d_vw + d_wz
    quad_shortcut = np.minimum(d_uz, np.minimum(d_uw + d_wz, d_uv + d_vz))
    quad_detour_ratio = path_uvwz / (quad_shortcut + 1e-8)

    # 2. Local 3-point turning angle & detour
    tri_detour = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn

    # 3. Spatial nearest neighbor detour margin: checks if v has close non-tour neighbors bridging u or w
    # Evaluate slack against mean distance to capture bridge-edge bottlenecks
    global_mean_dist = np.mean(edge_distance)
    slack_margin = (path_uvwz - quad_shortcut) / (global_mean_dist + 1e-8)

    # Composite multi-scale geometric inefficiency
    geom_inefficiency = (0.5 * tri_detour + 0.5 * quad_detour_ratio) * (1.0 + 0.3 * acute_deficit) + 0.2 * np.maximum(0.0, slack_margin)

    # Smooth over adjacent edges
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base utility scaled by inverse frequency and geometric factor
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply penalization to high-utility edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Multi-scale shortcut discounting: incentivize promising 3-point and 4-point chords
    high_detour_indices = np.argsort(quad_detour_ratio)[-k:]
    for idx in high_detour_indices:
        # Check both 2-hop (u, w) and 3-hop (u, z) shortcuts
        for target_node in (w_idx[idx], z_idx[idx]):
            u = u_idx[idx]
            freq = edge_n_used[u, target_node]
            if freq < 1.0:
                discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
                updated_edge_distance[u, target_node] = max(1e-6, updated_edge_distance[u, target_node] - discount)
                updated_edge_distance[target_node, u] = updated_edge_distance[u, target_node]

    return updated_edge_distance
