# ===== Heuristic Code =====
#{This strategy penalizes high-utility tour edges and constructs an alternating 4-opt / cycle-exchange surrogate by solving a maximum-weight alternating matching on candidate replacement cross-edges, concentrating structured negative-cost incentives along coherent non-adjacent cycles.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # 1. Identify tour edges and compute guided local search utilities
    tour = np.asarray(local_opt_tour, dtype=np.int64)
    next_tour = np.roll(tour, -1)
    costs = edge_distance[tour, next_tour]
    used = edge_n_used[tour, next_tour]

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Select candidate tour edges to break (top-quantile utility)
    k_penalize = max(2, min(n // 4, int(np.ceil(n * 0.25))))
    penalized_indices = np.argsort(utilities)[-k_penalize:]

    max_util = np.max(utilities) + 1e-8
    for idx in penalized_indices:
        u, v = tour[idx], next_tour[idx]
        edge_use = edge_n_used[u, v]
        penalty = (costs[idx] / (1.0 + np.sqrt(edge_use + 1.0))) * (1.0 + utilities[idx] / max_util)
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 2. Formulate alternating cycle surrogate (identifying 4-opt alternate reconnecting moves)
    # Penalized segments represent edges E_remove = {(u_i, v_i)}
    # We search for pairs of penalized edges (e_i, e_j) that can form a valid 4-edge alternating cycle
    # with candidate replacement edges: (u_i, u_j) and (v_i, v_j) or (u_i, v_j) and (v_i, u_j)
    penalized_list = list(penalized_indices)
    penalized_set = set(penalized_list)

    candidate_cycles = []
    num_p = len(penalized_list)

    for i in range(num_p):
        idx_i = penalized_list[i]
        u1, v1 = tour[idx_i], next_tour[idx_i]
        c1 = costs[idx_i]

        for j in range(i + 1, num_p):
            idx_j = penalized_list[j]
            # Ensure edges are non-adjacent along the tour
            dist_idx = abs(idx_i - idx_j)
            if dist_idx <= 1 or dist_idx >= n - 1:
                continue

            u2, v2 = tour[idx_j], next_tour[idx_j]
            c2 = costs[idx_j]

            # Reconnection configuration A: (u1, u2) and (v1, v2)
            cost_gain_A = (c1 + c2) - (edge_distance[u1, u2] + edge_distance[v1, v2])
            use_penalty_A = 0.5 * (edge_n_used[u1, u2] + edge_n_used[v1, v2])
            score_A = cost_gain_A / (1.0 + np.sqrt(use_penalty_A))

            # Reconnection configuration B: (u1, v2) and (v1, u2)
            cost_gain_B = (c1 + c2) - (edge_distance[u1, v2] + edge_distance[v1, u2])
            use_penalty_B = 0.5 * (edge_n_used[u1, v2] + edge_n_used[v1, u2])
            score_B = cost_gain_B / (1.0 + np.sqrt(use_penalty_B))

            if score_A >= score_B:
                best_score = score_A
                best_edges = ((u1, u2), (v1, v2))
            else:
                best_score = score_B
                best_edges = ((u1, v2), (v1, u2))

            candidate_cycles.append((best_score, idx_i, idx_j, best_edges))

    # Sort alternating cycles by replacement desirability (maximum weight matching surrogate)
    candidate_cycles.sort(key=lambda x: x[0], reverse=True)

    matched_tour_edges = set()
    base_reduction_ratio = 0.08

    for score, idx_i, idx_j, repl_edges in candidate_cycles:
        if idx_i in matched_tour_edges or idx_j in matched_tour_edges:
            continue

        matched_tour_edges.add(idx_i)
        matched_tour_edges.add(idx_j)

        # Inject coordinated structural reductions along replacement cycle edges
        for a, b in repl_edges:
            edge_cost = edge_distance[a, b]
            decay = edge_cost * base_reduction_ratio * (1.0 / (1.0 + np.log1p(edge_n_used[a, b])))
            updated_edge_distance[a, b] = max(1e-4, updated_edge_distance[a, b] - decay)
            updated_edge_distance[b, a] = max(1e-4, updated_edge_distance[b, a] - decay)

    return updated_edge_distance
