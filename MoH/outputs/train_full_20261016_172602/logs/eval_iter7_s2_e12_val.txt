# ===== Heuristic Code =====
#{A topological cluster ruin-and-recreate strategy that locates the highest-density continuous sub-path of geometric-frequency penalized tour edges, heavily penalizes all intra-cluster connections to shatter local topology, and discounts bridging edges across the cluster boundaries to drive macro-reconnection.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Apply standard frequency decay
    edge_n_used *= 0.96

    # Extract consecutive tour triples
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Calculate geometric path inefficiency: detour ratio and turning angle deficit
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base edge GLS utilities
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Identify continuous sub-path cluster with maximal aggregated penalty utility
    cluster_len = max(4, int(np.clip(0.20 * num_nodes, 4, max(4, num_nodes - 2))))
    extended_utils = np.concatenate([utilities, utilities[:cluster_len - 1]])
    window_scores = np.convolve(extended_utils, np.ones(cluster_len), mode='valid')
    best_start = int(np.argmax(window_scores[:num_nodes]))

    # Continuous cluster indices in the cyclic tour
    cluster_indices = [(best_start + i) % num_nodes for i in range(cluster_len)]
    cluster_nodes = tour[cluster_indices]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.35 * global_mean_dist

    # 1. Heavily penalize the direct tour edges within the ruined cluster
    for idx in cluster_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.5
        edge_n_used[v, u] += 1.5
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        penalty = adaptive_lambda * freq_penalty * (1.0 + 0.5 * local_geom_factor[idx])
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 2. Shatter the intra-cluster topology by penalizing all pairwise internal connections
    intra_penalty = 0.5 * adaptive_lambda
    cluster_mesh_u, cluster_mesh_v = np.meshgrid(cluster_nodes, cluster_nodes, indexing='ij')
    mask_distinct = cluster_mesh_u != cluster_mesh_v
    updated_edge_distance[cluster_mesh_u[mask_distinct], cluster_mesh_v[mask_distinct]] += intra_penalty

    # 3. Create strong bridge links discounting external bypasses around the cluster
    entry_boundary = tour[(best_start - 1) % num_nodes]
    exit_boundary = tour[(best_start + cluster_len) % num_nodes]
    bypass_discount = 0.25 * global_mean_dist

    updated_edge_distance[entry_boundary, exit_boundary] = max(
        1e-6, updated_edge_distance[entry_boundary, exit_boundary] - bypass_discount
    )
    updated_edge_distance[exit_boundary, entry_boundary] = updated_edge_distance[entry_boundary, exit_boundary]

    # Reward long-distance external bridge shortcuts to alternative non-cluster nodes
    stride = max(2, (num_nodes - cluster_len) // 4)
    external_targets = tour[[(best_start + cluster_len + i * stride) % num_nodes for i in range(1, 4)]]
    for ext in external_targets:
        if ext != entry_boundary and ext != exit_boundary:
            disc = 0.15 * global_mean_dist / (1.0 + edge_n_used[entry_boundary, ext])
            updated_edge_distance[entry_boundary, ext] = max(1e-6, updated_edge_distance[entry_boundary, ext] - disc)
            updated_edge_distance[ext, entry_boundary] = updated_edge_distance[entry_boundary, ext]

    # Standard shortcut rewards on top geometric detour positions outside the ruined cluster
    k_shortcuts = max(1, int(np.ceil(0.08 * num_nodes)))
    high_detour_indices = np.argsort(detour_ratio)[-k_shortcuts:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
