# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Coordinated decay of edge frequency memory
    edge_n_used *= 0.95

    # Edge distances along the tour
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wx = edge_distance[w_idx, x_idx]

    # Chord shortcuts: 2-opt skip (u -> w) and 3-opt skip (u -> x)
    d_uw = edge_distance[u_idx, w_idx]
    d_ux = edge_distance[u_idx, x_idx]

    # 1-hop detour and turning angle deficit
    detour_2opt = (d_uv + d_vw) / (d_uw + 1e-8)
    denom_cos = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom_cos, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_ineff_2 = detour_2opt * (1.0 + 0.5 * acute_deficit)

    # 2-hop detour accounting for higher-order crossing structures
    detour_3opt = (d_uv + d_vw + d_wx) / (d_ux + 1e-8)
    geom_ineff_3 = detour_3opt

    # Combined local geometric stress on edges
    local_geom_factor = 0.4 * (geom_ineff_2 + np.roll(geom_ineff_2, 1)) + 0.2 * (geom_ineff_3 + np.roll(geom_ineff_3, 2))

    # GLS edge utility calculation
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Self-adaptive perturbation size bounded between 10% and 20%
    mean_freq = np.mean(base_freq)
    adaptive_ratio = np.clip(0.12 + 0.02 * np.tanh(mean_freq - 1.0), 0.08, 0.22)
    k = max(1, int(np.ceil(adaptive_ratio * num_nodes)))

    # Compute tour-scale metric for robust scaling
    tour_len = np.sum(d_uv)
    avg_tour_edge = tour_len / num_nodes
    adaptive_lambda = 0.35 * avg_tour_edge

    # Penalize inefficient edges in local optimum
    top_k_indices = np.argsort(utilities)[-k:]
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.4 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Metric-preserving proportional incentives on shortcuts (prevents negative distortion and cycling)
    # 2-hop shortcut incentives
    top_detour_2 = np.argsort(detour_2opt)[-k:]
    for idx in top_detour_2:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            incentive_ratio = 0.15 * (1.0 / (1.0 + freq))
            new_val = updated_edge_distance[u, w] * (1.0 - incentive_ratio)
            updated_edge_distance[u, w] = max(1e-6, new_val)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    # 3-hop shortcut incentives for long-range crossing relief
    top_detour_3 = np.argsort(detour_3opt)[-max(1, k // 2):]
    for idx in top_detour_3:
        u, x = u_idx[idx], x_idx[idx]
        freq = edge_n_used[u, x]
        if freq < 0.5:
            incentive_ratio = 0.10 * (1.0 / (1.0 + freq))
            new_val = updated_edge_distance[u, x] * (1.0 - incentive_ratio)
            updated_edge_distance[u, x] = max(1e-6, new_val)
            updated_edge_distance[x, u] = updated_edge_distance[u, x]

    return updated_edge_distance
