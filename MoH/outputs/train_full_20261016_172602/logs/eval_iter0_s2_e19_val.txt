# ===== Heuristic Code =====
# {A dual-tier adaptive guided local search penalty heuristic that selectively decays edge usage memory based on spatial centrality while penalizing stagnant tour edges scaled by stagnation factor, spatial proximity, and node centrality.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    # 1. Spatial centrality calculation: nodes closer to the barycenter have higher spatial centrality
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial_centrality = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Dual-tier decay: spatially central edges retain memory longer (slower decay), peripheral edges refresh faster
    # Decay factor per edge matrix: base decay modulated between ~0.90 (peripheral) and ~0.98 (central)
    spatial_edge_factor = 0.5 * (norm_spatial_centrality[:, None] + norm_spatial_centrality[None, :])
    decay_matrix = 0.90 + 0.08 * spatial_edge_factor
    edge_n_used *= decay_matrix

    # 2. Vectorized representation of the current local optimal tour
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]

    # Utility function balancing long edge lengths and low frequency of penalization
    utilities = tour_dist / (1.0 + tour_used)
    k = max(1, int(np.ceil(0.15 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Increment usage count for selected bottleneck edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # 3. Dynamic stagnation factor based on tour usage and repeated penalization
    tour_usage = edge_n_used[u_idx, v_idx]
    avg_tour_usage = np.mean(tour_usage)
    max_used = np.max(edge_n_used) + 1e-12
    overlap_ratio = np.mean(tour_usage > 1.0)
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (avg_tour_usage / max_used)

    # Base adaptive penalty scale
    adaptive_lambda = (0.15 + 0.20 * np.tanh(stagnation_factor)) * global_mean_dist

    # 4. Centrality index based on accumulated edge penalties
    node_centrality = np.sum(edge_n_used, axis=1)
    norm_centrality = node_centrality / (np.max(node_centrality) + 1e-12)

    # 5. Dual-tier penalty scaling applied to top-k edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        c_mod = 0.35 * (norm_centrality[u] + norm_centrality[v])
        sc_mod = 0.25 * (norm_spatial_centrality[u] + norm_spatial_centrality[v])
        modulation = 1.0 + c_mod + sc_mod

        penalty = adaptive_lambda * edge_n_used[u, v] * modulation
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
