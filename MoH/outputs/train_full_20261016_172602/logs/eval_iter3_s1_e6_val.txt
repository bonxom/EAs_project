# ===== Heuristic Code =====
#{Guided local search edge perturbation combining z-score utility penalties with targeted 2-opt reconnecting chord discounts calibrated by dual slack estimations on incident penalized vertices.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Targeted dual slack estimation for alternative 2-opt reconnecting shortcuts
    # Compute node potentials (dual relaxation approximation) using 1-tree / tour incident costs
    pred_idx = np.roll(u_idx, 1)
    costs_pred = edge_distance[pred_idx, u_idx].astype(np.float64)
    potentials = np.zeros(updated_edge_distance.shape[0], dtype=np.float64)
    potentials[u_idx] = 0.5 * (costs + costs_pred)

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    tour_pos = np.empty(updated_edge_distance.shape[0], dtype=np.intp)
    tour_pos[u_idx] = np.arange(n)

    for u in penalized_nodes:
        pos_u = tour_pos[u]
        # Evaluate candidate chords not immediately adjacent in the tour
        # Sample non-adjacent nodes with small reduced cost: d(u, w) - pi(u) - pi(w)
        d_u = edge_distance[u]
        reduced_costs = d_u - potentials[u] - potentials

        # Exclude self and immediate tour neighbors
        exclude_nodes = [u, u_idx[(pos_u - 1) % n], u_idx[(pos_u + 1) % n]]
        reduced_costs[exclude_nodes] = np.inf

        # Select best negative reduced cost chords (promising 2-opt re-connections)
        promising_mask = (reduced_costs < 0.0) & (edge_n_used[u] == 0)
        if np.any(promising_mask):
            candidates = np.where(promising_mask)[0]
            if len(candidates) > 5:
                top_indices = np.argpartition(reduced_costs[candidates], 5)[:5]
                candidates = candidates[top_indices]

            decay = 0.02 * (1.0 / (1.0 + np.exp(reduced_costs[candidates] / (std_cost + 1e-8))))
            updated_edge_distance[u, candidates] -= updated_edge_distance[u, candidates] * decay
            updated_edge_distance[candidates, u] = updated_edge_distance[u, candidates]

    return updated_edge_distance
