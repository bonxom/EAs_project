# ===== Heuristic Code =====
#{Synthesize static distance centrality with dynamic entropy rigidity by scaling penalties through geometric-mean local k-nearest neighbor distance shells while dampening penalties on bridge edges identified by peripheral centrality thresholds.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # 1. Guided Local Search: identify and penalize highest utility tour edges
    tour_distances = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_distances / (1.0 + tour_counts)
    max_util = np.max(utilities)
    max_mask = np.isclose(utilities, max_util)
    for i in range(num_nodes):
        if max_mask[i]:
            u, v = u_idx[i], v_idx[i]
            edge_n_used[u, v] += 1
            edge_n_used[v, u] += 1

    # 2. Node-level static spatial centrality & peripheral thresholding
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Peripheral nodes are those with low spatial centrality
    peripheral_score = 1.0 - norm_spatial
    # Peripheral bridge dampening factor: edges connecting peripheral nodes are preserved
    bridge_affinity = peripheral_score[:, None] * peripheral_score[None, :]
    bridge_dampening = 1.0 / (1.0 + 1.5 * bridge_affinity)

    # 3. Dynamic entropy rigidity
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    # 4. K-nearest neighbor shell relative distance normalization
    k = max(2, min(num_nodes - 1, int(np.sqrt(num_nodes))))
    sorted_dists = np.partition(edge_distance, kth=k, axis=1)[:, :k]
    knn_mean = np.mean(sorted_dists, axis=1, keepdims=True) + 1e-12
    # Relative edge length within node's k-NN shell
    rel_knn_dist = edge_distance / knn_mean
    # Geometric mean of asymmetric k-NN shells across edge (u, v)
    knn_geom_weight = np.sqrt(rel_knn_dist * rel_knn_dist.T)
    knn_weight_clamped = np.clip(knn_geom_weight, 0.5, 3.0)

    # 5. Stagnation-aware adaptive penalty base
    tour_usage_updated = edge_n_used[u_idx, v_idx]
    overlap_ratio = np.mean(tour_usage_updated > 1.0)
    mean_rigidity = np.mean(entropy_rigidity)
    stagnation_factor = np.clip(overlap_ratio + mean_rigidity, 0.0, 2.0)
    adaptive_lambda = (0.20 + 0.10 * stagnation_factor) * global_mean_dist

    # 6. Recombination: modulate penalty with static, entropy, knn shell, and bridge dampening
    rigidity_field = 0.5 * (entropy_rigidity[:, None] + entropy_rigidity[None, :])
    spatial_field = 0.5 * (norm_spatial[:, None] + norm_spatial[None, :])
    core_field = 1.0 + 0.40 * spatial_field + 0.40 * rigidity_field

    penalty_matrix = (adaptive_lambda * edge_n_used * core_field * 
                      knn_weight_clamped * bridge_dampening)

    updated_edge_distance += penalty_matrix
    return updated_edge_distance
