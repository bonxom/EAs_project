# ===== Heuristic Code =====
#{Implement an adaptive guided local search penalty strategy with penalty decay on historical edges, bounded total penalty budget normalized by current tour length, and replacement-opportunity scaling using fast 2-nearest-neighbor partitions to prevent search saturation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Ensure a clean copy of the edge distance matrix is modified
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Extract current tour edge costs
    tour_edge_lengths = updated_edge_distance[u_idx, v_idx]
    total_tour_cost = np.sum(tour_edge_lengths)
    avg_tour_edge = total_tour_cost / n
    if avg_tour_edge <= 1e-12:
        return updated_edge_distance

    # Fast k-NN density estimation via partition instead of full sort
    k = min(5, n - 1)
    part_k = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(part_k, axis=1)
    mean_k_nn = np.mean(k_nn_avg) + 1e-12
    node_density = mean_k_nn / (k_nn_avg + 1e-12)
    edge_density = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Guided Local Search utility calculation
    # Utility prioritizes long edges relative to local node spacing and penalizes edges with low historical usage
    relative_edge_cost = tour_edge_lengths / (avg_tour_edge + 1e-12)
    usage = edge_n_used[u_idx, v_idx]
    gls_utility = (tour_edge_lengths / (1.0 + usage)) * edge_density

    # Candidate selection: penalize top utility edges rather than blind global thresholding
    max_util = np.max(gls_utility)
    min_util = np.min(gls_utility)
    if max_util > min_util:
        norm_utility = (gls_utility - min_util) / (max_util - min_util)
    else:
        norm_utility = np.ones_like(gls_utility)

    # Sigmoidal scaling to softly select high-utility edges
    activation = 1.0 / (1.0 + np.exp(-6.0 * (norm_utility - 0.5)))

    # Bounded penalty allocation: limit total injected penalty to a safe fraction of tour length
    penalty_budget = 0.15 * total_tour_cost
    activation_sum = np.sum(activation)
    if activation_sum > 1e-12:
        penalty_weights = activation / activation_sum
        penalties = penalty_budget * penalty_weights
    else:
        penalties = np.full(n, penalty_budget / n)

    # Apply symmetric bounded penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
