# ===== Heuristic Code =====
#{Implements an episodic guided memory transformation that computes base metric utilities, applies an adaptive percentile-based penalty with dynamic scaling modulated by global edge usage saturation, and prevents metric distortion by preserving baseline edge fidelities.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    n_nodes = len(local_opt_tour)
    if n_nodes < 2:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility calculation: edge length normalized by frequency
    utilities = tour_lengths / (1.0 + tour_counts)

    # Dynamic percentile thresholding to adaptively penalize the most critical edges
    k = max(1, int(np.ceil(0.15 * n_nodes)))
    sorted_utilities = np.sort(utilities)
    threshold = sorted_utilities[-k]

    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Episodic modulation: compute penalty scaled inversely to edge usage concentration
    # to avoid runaway penalty accumulation and premature metric graph degradation
    mean_len = np.mean(tour_lengths)
    selected_counts = tour_counts[selected_mask]

    # Decay penalty weight as saturation increases, preventing persistent distortion
    penalty_factors = 0.25 * mean_len / np.sqrt(1.0 + selected_counts)

    # Symmetric tensor update using vectorized scatter
    np.add.at(updated_edge_distance, (sel_u, sel_v), penalty_factors)
    np.add.at(updated_edge_distance, (sel_v, sel_u), penalty_factors)

    return updated_edge_distance
