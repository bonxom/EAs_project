# ===== Heuristic Code =====
#{Guided Local Search with reciprocal dual-operator edge modification that penalizes geometrically inefficient tour edges while symmetrically discounting candidate nearest-neighbor bridging edges scaled by bottleneck reduction.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize inefficient edges and apply reciprocal bridge discounting
    nn_count = min(3, num_nodes - 1)

    for idx in top_k_indices:
        u = u_idx[idx]
        v = v_idx[idx]
        w = w_idx[idx]

        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

        # Dual-operator: identify bottleneck reduction around inefficient node v
        direct_cost = edge_distance[u, v] + edge_distance[v, w]
        shortcut_gain = direct_cost - edge_distance[u, w]

        # Discount candidate nearest-neighbor bridges for u and w proportional to bottleneck relief
        if shortcut_gain > 0:
            scale_factor = shortcut_gain / (direct_cost + 1e-8)
            discount_base = 0.12 * global_mean_dist * scale_factor

            # Find closest neighbors of u and w (excluding v)
            cand_u = np.argsort(edge_distance[u])[:nn_count + 2]
            cand_w = np.argsort(edge_distance[w])[:nn_count + 2]

            for cand, node in ((cand_u, u), (cand_w, w)):
                added = 0
                for nbr in cand:
                    if nbr != node and nbr != v:
                        disc = discount_base / (1.0 + 0.5 * edge_n_used[node, nbr])
                        new_val = max(1e-6, updated_edge_distance[node, nbr] - disc)
                        updated_edge_distance[node, nbr] = new_val
                        updated_edge_distance[nbr, node] = new_val
                        added += 1
                        if added >= nn_count:
                            break

    return updated_edge_distance
