# ===== Heuristic Code =====
#{Calculate edge utility util(e) = c(e) / (1 + p(e)) modulated by reciprocal nearest-neighbor dispersion and selectively apply targeted penalties to the top-quantile utility-maximizing tour edges to escape local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    penalties = edge_n_used[u_idx, v_idx]

    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_mean = np.mean(sorted_dists[:, 1:k+1], axis=1)
    scale_factor = 0.5 * (k_nn_mean[u_idx] + k_nn_mean[v_idx]) + 1e-6

    utilities = (tour_lengths / (1.0 + penalties)) / scale_factor

    max_util = np.max(utilities)
    threshold = max(max_util * 0.85, np.percentile(utilities, 80))
    selected_mask = utilities >= threshold

    base_penalty = 0.3 * np.mean(tour_lengths) / (1.0 + 0.1 * penalties)
    penalty_values = base_penalty * (utilities / (max_util + 1e-6))

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_pen = penalty_values[selected_mask]

    np.add.at(updated_edge_distance, (sel_u, sel_v), sel_pen)
    np.add.at(updated_edge_distance, (sel_v, sel_u), sel_pen)

    return updated_edge_distance
