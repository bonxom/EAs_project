# ===== Heuristic Code =====
# {Combines consecutive bottleneck utility augmentation and vectorized penalization with an adaptive rank-decayed reward on promising unvisited incident edges to accelerate escape from local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    base_utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Bottleneck context: augment utility using adjacent tour edges
    prev_util = np.roll(base_utilities, 1)
    next_util = np.roll(base_utilities, -1)
    bottleneck_utilities = base_utilities + 0.35 * np.maximum(prev_util, next_util)

    threshold = np.median(bottleneck_utilities)
    max_util = np.max(bottleneck_utilities) + 1e-8
    penalize_mask = bottleneck_utilities >= threshold

    # Vectorized penalty computation and symmetric assignment
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = bottleneck_utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Adaptive rank-scaled rewarding for unvisited incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k_nn = min(8, n - 1)

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        unused_indices = np.where(unused_mask)[0]

        if unused_indices.size > 0:
            sub_costs = edge_distance[u, unused_indices]
            k_eff = min(k_nn, unused_indices.size)
            best_order = np.argpartition(sub_costs, k_eff - 1)[:k_eff]
            best_order = best_order[np.argsort(sub_costs[best_order])]

            chosen_v = unused_indices[best_order]
            ranks = np.arange(k_eff, dtype=np.float64)
            # Rank-decayed reward factors favoring the nearest unvisited alternatives
            decay_factors = 0.04 / (1.0 + 0.5 * ranks)
            rewards = edge_distance[u, chosen_v] * decay_factors

            new_vals = np.maximum(1e-6, updated_edge_distance[u, chosen_v] - rewards)
            updated_edge_distance[u, chosen_v] = new_vals
            updated_edge_distance[chosen_v, u] = new_vals

    return updated_edge_distance
