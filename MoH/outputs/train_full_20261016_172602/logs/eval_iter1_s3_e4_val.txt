#{Replaces rigid convex weighting with a unified continuous topological score that couples hyperbolic bridging tension, local k-NN contour slack, and edge-specific frequency decay into a smoothly scaled additive penalty.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Measure local topological scales using k-NN neighborhood distances
    k = min(6, n - 1)
    sorted_dists = np.partition(edge_distance, k, axis=1)[:, 1:k + 1]
    local_scale = np.mean(sorted_dists, axis=1) + 1e-6

    # Extract tour directed transitions
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    lengths = edge_distance[u_idx, v_idx]
    mean_length = np.mean(lengths) + 1e-6
    relative_global = lengths / mean_length

    # Local contour slack: ratio of edge length to the joint k-NN ambient scale of endpoints
    contour_slack = lengths / (0.5 * (local_scale[u_idx] + local_scale[v_idx]))

    # Bounded non-linear bridging tension identifying long cluster-spanning bottlenecks
    bridging_tension = np.tanh(1.5 * (contour_slack - 1.0)) + 1.0

    # Edge-specific frequency attenuation
    usage = edge_n_used[u_idx, v_idx]
    usage_decay = 1.0 / (1.0 + 0.35 * usage)

    # Unified continuous topological score
    # Scales proportionally with edge length and dynamic bridging impact
    penalty = (mean_length * 0.15) * relative_global * bridging_tension * usage_decay

    # Symmetrically apply penalties along the tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
