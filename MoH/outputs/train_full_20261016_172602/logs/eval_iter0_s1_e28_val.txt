# ===== Heuristic Code =====
#{Multi-scale n-gram chain fragmentation combines 2-edge, 3-edge, and 4-edge persistence metrics to apply super-linear penalties to rigid sub-tours while actively guiding escape via discounted 2-chord and 3-chord bridge transitions and selective node-incident unvisited edge deflation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Clone edge distance matrix to ensure clean deterministic updates
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 6:
        return updated_edge_distance

    tour = np.array(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Step 2: Extract edge costs and usage frequencies along the local optimum tour
    e_costs = edge_distance[u_idx, v_idx].astype(np.float64)
    e_used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Step 3: Multi-scale chain co-occurrence n-grams (2-edge, 3-edge, 4-edge)
    e_used_p1 = np.roll(e_used, -1)
    e_used_m1 = np.roll(e_used, 1)
    e_used_p2 = np.roll(e_used, -2)

    chain_2 = np.minimum(e_used, e_used_p1)
    chain_3 = np.minimum(np.minimum(e_used_m1, e_used), e_used_p1)
    chain_4 = np.minimum(chain_2, np.minimum(np.roll(e_used, -2), np.roll(e_used, -3)))

    # Step 4: Compute normalized structural rigidity and utility scores
    mean_cost = np.mean(e_costs)
    std_cost = np.std(e_costs) + 1e-8
    rel_cost = (e_costs - mean_cost) / std_cost
    cost_sig = 1.0 / (1.0 + np.exp(-rel_cost))

    avg_chain2 = np.mean(chain_2) + 1e-8
    avg_chain3 = np.mean(chain_3) + 1e-8

    chain_persistence = (
        0.5 * (chain_2 / avg_chain2) +
        0.35 * (chain_3 / avg_chain3) +
        0.15 * (chain_4 / (np.mean(chain_4) + 1e-8))
    )
    rigidity_factor = 1.0 + np.power(chain_persistence, 1.6)

    # Utility: high-cost, high-rigidity edges are primary candidates for disruption
    utilities = (e_costs * (1.0 + 1.25 * cost_sig) * rigidity_factor) / (1.0 + np.sqrt(e_used + 1.0))

    threshold = np.quantile(utilities, 0.45)
    penalize_mask = utilities >= threshold

    # Step 5: Apply super-linear penalty to rigid segments
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = e_costs[penalize_mask]
    p_used = e_used[penalize_mask]
    p_rigidity = rigidity_factor[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * p_rigidity * 1.15
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Step 6: Dual-Memory Chord Fragmentation - Discount bypassing 2-chords (u->w) and 3-chords (u->x)
    chord2_mask = chain_2 >= np.median(chain_2)
    c2_u = u_idx[chord2_mask]
    c2_w = w_idx[chord2_mask]

    c2_used = edge_n_used[c2_u, c2_w]
    c2_rare = c2_used <= np.median(c2_used)
    sel_c2_u = c2_u[c2_rare]
    sel_c2_w = c2_w[c2_rare]

    discount2 = 0.08 * updated_edge_distance[sel_c2_u, sel_c2_w] / (1.0 + np.sqrt(c2_used[c2_rare] + 1.0))
    updated_edge_distance[sel_c2_u, sel_c2_w] -= discount2
    updated_edge_distance[sel_c2_w, sel_c2_u] -= discount2

    # 3-chord bypass for persistent 3-chains to enable broader reconnection
    chord3_mask = chain_3 >= np.quantile(chain_3, 0.70)
    c3_u = u_idx[chord3_mask]
    c3_x = x_idx[chord3_mask]

    c3_used = edge_n_used[c3_u, c3_x]
    c3_rare = c3_used <= np.median(c3_used)
    sel_c3_u = c3_u[c3_rare]
    sel_c3_x = c3_x[c3_rare]

    discount3 = 0.04 * updated_edge_distance[sel_c3_u, sel_c3_x] / (1.0 + np.sqrt(c3_used[c3_rare] + 1.0))
    updated_edge_distance[sel_c3_u, sel_c3_x] -= discount3
    updated_edge_distance[sel_c3_x, sel_c3_u] -= discount3

    # Step 7: Deflate completely unvisited edges incident to severely penalized nodes
    severe_mask = utilities >= np.quantile(utilities, 0.75)
    severely_penalized = np.unique(np.concatenate((u_idx[severe_mask], v_idx[severe_mask])))

    decay_ratio = 0.035
    for node in severely_penalized:
        unused_mask = (edge_n_used[node] == 0)
        unused_mask[node] = False
        if np.any(unused_mask):
            vals = updated_edge_distance[node, unused_mask]
            delta = vals * decay_ratio
            updated_edge_distance[node, unused_mask] -= delta
            updated_edge_distance[unused_mask, node] = updated_edge_distance[node, unused_mask]

    return updated_edge_distance
