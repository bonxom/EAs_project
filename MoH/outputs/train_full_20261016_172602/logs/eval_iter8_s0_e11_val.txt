# ===== Heuristic Code =====
#{An adaptive guided local search penalty strategy that evaluates edge frequency saturation and route overlap to dynamically scale penalty magnitude and utility quantile thresholds based on stagnation intensity.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy of the distance matrix to preserve original data
    updated_edge_distance = np.copy(edge_distance)

    # Vectorized extraction of tour directed edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Standard Guided Local Search utility calculation
    utilities = tour_lengths / (1.0 + tour_counts)

    # Measure tour-level stagnation and edge saturation
    max_count = np.max(tour_counts)
    mean_count = np.mean(tour_counts)
    median_count = np.median(tour_counts)

    # Quantify recurrence and stagnation: higher recurrence -> higher stagnation ratio
    # Scaled monotonically between 0.0 (fresh search) and 1.0 (severely cycling)
    stagnation_ratio = float(np.clip((mean_count + median_count * 0.5) / (max_count + 1.0), 0.0, 1.0))

    # Dynamically scale penalty intensity: relaxed (0.15) for exploration, aggressive (0.75) for deep stagnation
    penalty_factor = 0.15 + 0.60 * stagnation_ratio
    avg_edge = np.mean(tour_lengths)
    penalty = penalty_factor * avg_edge

    # Dynamically adjust threshold quantile: targeting fewer, critical edges when progressing,
    # and broader sets of penalised edges when stuck in recurring basins
    quantile_target = 0.95 - 0.20 * stagnation_ratio
    threshold = np.quantile(utilities, quantile_target)

    # Select edges meeting or exceeding the adaptive utility threshold
    selected_mask = utilities >= threshold

    # Robust fallback: ensure at least the edge with the maximum utility is penalized
    if not np.any(selected_mask):
        selected_mask = utilities >= np.max(utilities)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetric penalty update using vectorized advanced indexing
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
