# ===== Heuristic Code =====
#{Entropy-governed stagnation-adaptive scaling dynamically measures search state dispersion via the normalized entropy of edge usage counts, transitioning Guided Local Search penalties from focused high-regret edge increments to broader disruptive exploration when stagnation is detected.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    mean_len = np.mean(tour_edge_lengths) + 1e-8
    rel_lengths = tour_edge_lengths / mean_len

    tour_edge_counts = edge_n_used[u_idx, v_idx].astype(np.float64)
    total_usage = np.sum(tour_edge_counts)

    # Calculate Shannon entropy of the tour edge usage distribution
    if total_usage > 0:
        p = tour_edge_counts / total_usage
        nonzero_p = p[p > 0]
        entropy = -np.sum(nonzero_p * np.log(nonzero_p))
        max_entropy = np.log(n)
        normalized_entropy = entropy / (max_entropy + 1e-8)
    else:
        normalized_entropy = 0.0

    # Stagnation indicator: low entropy means a few edges dominate cycling; high entropy or high total usage signals stagnation
    usage_intensity = total_usage / (n + 1e-8)
    stagnation_factor = 1.0 / (1.0 + np.exp(-3.0 * (usage_intensity - 2.0)))
    dispersion_weight = 0.3 * normalized_entropy + 0.7 * stagnation_factor

    # Regret-based edge utility calculation: cost / (1 + count)
    edge_utilities = tour_edge_lengths / (1.0 + tour_edge_counts)
    max_util = np.max(edge_utilities) + 1e-8
    norm_utilities = edge_utilities / max_util

    # Dynamic scaling: early/active phase concentrates on top utilities; stagnant phase applies broader penalties
    sharpness = 8.0 * (1.0 - 0.7 * dispersion_weight)
    util_weights = np.exp(sharpness * (norm_utilities - np.median(norm_utilities)))
    util_weights = util_weights / (np.max(util_weights) + 1e-8)

    # Base penalty proportional to edge length and exploration stage
    base_penalty = 0.15 * mean_len * (1.0 + 1.5 * dispersion_weight)
    penalties = base_penalty * (util_weights + 0.25 * dispersion_weight * rel_lengths)

    # Symmetric update
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
