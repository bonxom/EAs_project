# ===== Heuristic Code =====
#{This algorithm dynamically modulates memory decay via tour length dispersion, normalizes penalty utilities using median absolute deviation to adaptively penalize the most anomalous edges, and applies locally bounded shortcut incentives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Dynamic decay factor calibrated by coefficient of variation of the tour edge lengths
    mean_tour_d = np.mean(d_uv)
    std_tour_d = np.std(d_uv)
    cv = std_tour_d / (mean_tour_d + 1e-8)
    decay_factor = float(np.clip(0.98 - 0.08 * cv, 0.90, 0.98))
    edge_n_used *= decay_factor

    # Geometric detour calculation and acute angle metric
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base utilities
    base_freq = edge_n_used[u_idx, v_idx]
    raw_utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Robust normalization via Median Absolute Deviation (MAD)
    med = np.median(raw_utilities)
    mad = np.median(np.abs(raw_utilities - med)) + 1e-8
    norm_utilities = (raw_utilities - med) / mad

    # Adaptive penalty budget selection: select edges significantly above the median or at least top 8%
    k_min = max(1, int(np.ceil(0.08 * num_nodes)))
    significant_mask = norm_utilities > 1.0
    k = max(k_min, int(np.sum(significant_mask)))
    k = min(k, max(1, int(0.20 * num_nodes)))

    top_k_indices = np.argsort(raw_utilities)[-k:]

    # Penalize candidate edges proportional to their local edge length rather than global mean
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.4 * local_geom_factor[idx]
        penalty = 0.30 * d_uv[idx] * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Shortcut discount bounded strictly by local edge metrics
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            # Bound shortcut discount relative to actual shortcut edge length to prevent negative/near-zero distortion
            max_discount = 0.20 * d_uw[idx]
            discount = min(max_discount, 0.12 * d_uw[idx] * (1.0 / (1.0 + freq)))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
