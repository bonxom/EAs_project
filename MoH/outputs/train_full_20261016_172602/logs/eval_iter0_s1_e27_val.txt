# ===== Heuristic Code =====
#{This algorithm penalizes long and frequently used tour edges using utility-guided penalties and computes a Minimum Spanning Tree over penalized incident nodes to selectively decay low-usage MST backbone edges as gradient pathways.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Initialize copy of distance matrix to preserve original
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Step 2: Extract current tour edges and calculate penalty utilities
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Step 3: Identify edges exceeding median utility for penalization
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Penalize high-utility edges to force local search escape
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Step 4: MST-Guided Backbone Decay on nodes incident to penalized edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k = len(penalized_nodes)

    if k >= 2:
        # Construct submatrix of original distances among penalized nodes
        sub_dist = edge_distance[np.ix_(penalized_nodes, penalized_nodes)].astype(np.float64)

        # Deterministic Prim's MST over penalized nodes
        in_mst = np.zeros(k, dtype=bool)
        min_cost_to_tree = np.full(k, np.inf, dtype=np.float64)
        parent = np.full(k, -1, dtype=np.intp)

        min_cost_to_tree[0] = 0.0
        for _ in range(k):
            # Select unvisited node with minimum distance to current tree
            unvisited_costs = np.where(in_mst, np.inf, min_cost_to_tree)
            curr = np.argmin(unvisited_costs)
            in_mst[curr] = True

            # Update connections to other unvisited nodes
            curr_dist = sub_dist[curr]
            better_mask = (~in_mst) & (curr_dist < min_cost_to_tree)
            min_cost_to_tree[better_mask] = curr_dist[better_mask]
            parent[better_mask] = curr

        # Identify MST backbone edges and decay them if they have low usage
        mst_u_sub = parent[1:]
        mst_v_sub = np.arange(1, k, dtype=np.intp)
        mst_u = penalized_nodes[mst_u_sub]
        mst_v = penalized_nodes[mst_v_sub]

        mst_used = edge_n_used[mst_u, mst_v]
        # Actively decay low-usage backbone edges (used <= 1)
        decay_mask = mst_used <= 1
        d_u = mst_u[decay_mask]
        d_v = mst_v[decay_mask]

        if len(d_u) > 0:
            decay_factor = 0.05
            reductions = updated_edge_distance[d_u, d_v] * decay_factor
            updated_edge_distance[d_u, d_v] -= reductions
            updated_edge_distance[d_v, d_u] -= reductions

    # Step 5: Mild Tabu-Biased decay on remaining completely unused adjacent edges
    decay_ratio = 0.015
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
