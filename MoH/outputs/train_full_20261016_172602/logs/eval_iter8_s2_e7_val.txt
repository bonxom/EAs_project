# ===== Heuristic Code =====
#{We replace isolated top-k edge selection with a deterministic string-ruin mechanism that identifies contiguous high-utility tour segments of lengths 3 to 7, applying structured chain penalties and bridge discounts to break 2-opt cycles and enforce coherent subpath reconstruction.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=int)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    edge_utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Deterministic coherent chain ruin (string selection)
    target_penalized_edges = max(1, int(np.ceil(0.14 * num_nodes)))
    penalized_count = 0
    penalized_mask = np.zeros(num_nodes, dtype=bool)

    # Seed candidates prioritized by individual utility
    seed_order = np.argsort(edge_utilities)[::-1]

    for seed in seed_order:
        if penalized_count >= target_penalized_edges:
            break
        if penalized_mask[seed]:
            continue

        # Deterministically select chain length between 3 and 7 based on node index and tour scale
        chain_len = 3 + ((seed * 7 + num_nodes) % 5)
        chain_len = min(chain_len, num_nodes - 1)

        # Form a contiguous block of edges starting from the seed
        chain_indices = [(seed + i) % num_nodes for i in range(chain_len)]

        # Check if the block has minimal overlap with existing penalized edges
        overlap = sum(penalized_mask[idx] for idx in chain_indices)
        if overlap > chain_len // 2:
            continue

        for idx in chain_indices:
            if not penalized_mask[idx]:
                penalized_mask[idx] = True
                penalized_count += 1
                u, v = u_idx[idx], v_idx[idx]
                edge_n_used[u, v] += 1.0
                edge_n_used[v, u] += 1.0
                freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
                geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
                penalty = adaptive_lambda * freq_penalty * geom_weight
                updated_edge_distance[u, v] += penalty
                updated_edge_distance[v, u] += penalty

        # Apply deterministic reconstruction incentive across the expelled chain ends
        start_node = u_idx[chain_indices[0]]
        end_node = v_idx[chain_indices[-1]]
        freq_bridge = edge_n_used[start_node, end_node]
        if freq_bridge < 2.0:
            discount = 0.15 * global_mean_dist * (1.0 / (1.0 + freq_bridge))
            new_cost = max(1e-6, updated_edge_distance[start_node, end_node] - discount)
            updated_edge_distance[start_node, end_node] = new_cost
            updated_edge_distance[end_node, start_node] = new_cost

    # Targeted shortcut promotion for high local detours
    k_detour = max(1, target_penalized_edges // 2)
    high_detour_indices = np.argsort(detour_ratio)[-k_detour:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            new_dist = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_dist
            updated_edge_distance[w, u] = new_dist

    return updated_edge_distance
