# ===== Heuristic Code =====
#{This algorithm replaces static top-k penalization with an entropy-gated selection operator that dynamically expands or contracts candidate edge sets based on tour utility dispersion and stagnation, penalizing critical edges and incentivizing shortcuts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Evaporate edge usage counters smoothly
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric inefficiency and detour analysis
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Guided Local Search (GLS) utility evaluation
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Entropy-rigidity / usage dispersion computation on normalized utilities
    util_shifted = utilities - np.min(utilities)
    util_sum = np.sum(util_shifted)
    if util_sum > 1e-8:
        probs = util_shifted / util_sum
        # Avoid log(0)
        nz_probs = probs[probs > 1e-12]
        entropy = -np.sum(nz_probs * np.log(nz_probs))
        max_entropy = np.log(len(probs)) if len(probs) > 1 else 1.0
        normalized_entropy = entropy / (max_entropy + 1e-8)
    else:
        normalized_entropy = 1.0

    # Dynamic thresholding based on dispersion: low entropy (frozen sub-tours) triggers larger candidate sets
    # High entropy (diffuse search) focuses strictly on the sharpest peaks
    alpha = np.clip(1.0 - normalized_entropy, 0.0, 1.0)
    threshold_ratio = 0.90 - 0.25 * alpha
    max_utility = np.max(utilities)
    penalize_mask = utilities >= (max_utility * threshold_ratio)

    # Ensure dynamic cardinality bounds between 1 edge and up to 25% of the tour
    k_min = 1
    k_max = max(1, int(np.ceil((0.05 + 0.20 * alpha) * num_nodes)))
    candidate_indices = np.where(penalize_mask)[0]

    if len(candidate_indices) > k_max:
        candidate_indices = candidate_indices[np.argsort(utilities[candidate_indices])[-k_max:]]
    elif len(candidate_indices) < k_min:
        candidate_indices = np.argsort(utilities)[-k_min:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.25 + 0.15 * alpha) * global_mean_dist

    # Apply symmetric penalization to candidate edges
    for idx in candidate_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recombine: incentivize high-detour shortcuts proportional to dispersion
    k_detour = max(1, int(np.ceil(0.10 * num_nodes)))
    high_detour_indices = np.argsort(detour_ratio)[-k_detour:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
