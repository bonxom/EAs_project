# ===== Heuristic Code =====
#{This strategy reconstructs GLS penalties by replacing log-dampened frequency with direct linear usage modulated by geometric turn deficit and topological endpoint centrality, while retaining selective shortcut discounting.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Trace decay
    edge_n_used *= 0.95

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour and turn deficit analysis
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # GLS utilities biased by geometric inefficiency
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Update frequencies of chosen edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Endpoint centrality modulations
    node_centrality = np.sum(edge_n_used, axis=1)
    max_c = np.max(node_centrality) + 1e-12
    norm_centrality = node_centrality / max_c

    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    norm_spatial_centrality = (global_mean_dist / (node_mean_dist + 1e-12))
    norm_spatial_centrality /= (np.max(norm_spatial_centrality) + 1e-12)

    adaptive_lambda = 0.20 * global_mean_dist

    # Apply linear usage penalty modulated by centrality and local geometry
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        c_mod = 0.25 * (norm_centrality[u] + norm_centrality[v])
        sc_mod = 0.25 * (norm_spatial_centrality[u] + norm_spatial_centrality[v])
        geom_weight = 1.0 + 0.4 * local_geom_factor[idx]
        modulation = (1.0 + c_mod + sc_mod) * geom_weight

        penalty = adaptive_lambda * edge_n_used[u, v] * modulation
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate: selective shortcut discounting on high detour segments
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
