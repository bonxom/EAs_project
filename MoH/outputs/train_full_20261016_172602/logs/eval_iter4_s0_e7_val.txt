#{Implement a continuous guided local search mechanism with an adaptive global evaporation discount that decays existing penalties while dynamically penalizing maximum-utility tour edges to prevent penalty saturation and facilitate neighborhood cycling.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create a copy of the edge distance matrix to avoid in-place side effects
    updated_edge_distance = np.copy(edge_distance)

    # Tour edge indices
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Compute base edge lengths and usage counts
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Continuous penalty evaporation: gently decay inflated non-zero edge distances across the tour edges
    # to release previously over-penalized optimal edges as search neighborhoods shift
    evaporation_factor = 0.995
    updated_edge_distance[u_idx, v_idx] *= evaporation_factor
    updated_edge_distance[v_idx, u_idx] = updated_edge_distance[u_idx, v_idx]

    # Evaluate Guided Local Search utilities: length normalized by historical frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Scale penalty relative to tour cost and usage density
    avg_edge = np.mean(tour_lengths)
    penalty = 0.25 * avg_edge

    # Select edges achieving near-maximum utility for guided diversification
    threshold = 0.98 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply symmetric additive penalty to prioritized candidate edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
