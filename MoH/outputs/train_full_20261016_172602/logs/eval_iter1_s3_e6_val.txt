# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length

    # Calculate detour cost: min_{w != u, v} (d(u, w) + d(w, v) - d(u, v))
    # Sample candidate intermediate nodes via the tour's nearest neighbor pool or global distance slice
    # Vectorized detour cost over candidate nodes
    d_u = edge_distance[u_idx]  # shape: (n, n_total)
    d_v = edge_distance[v_idx]  # shape: (n, n_total)
    detours = d_u + d_v - tour_edge_lengths[:, None]

    # Mask out self-loops (w == u or w == v)
    row_indices = np.arange(len(u_idx))
    detours[row_indices, u_idx] = np.inf
    detours[row_indices, v_idx] = np.inf

    # Detour cost indicates replacability: lower detour cost means easily bypassed in 2-opt/3-opt
    min_detour = np.min(detours, axis=1)
    normalized_detour = min_detour / avg_tour_length

    # Flexibility factor: high if detour cost is low (easily replaced), low for backbone edges
    detour_factor = 1.0 / (1.0 + np.exp(3.0 * (normalized_detour - np.median(normalized_detour))))

    # Rank penalty based on relative length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Usage damping
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])

    # Combine relative length, replacability (detour factor), and usage damping
    local_penalty = relative_length * usage_factor * (1.0 + 1.5 * detour_factor)

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Boost penalty for long, non-bottleneck edges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = (tour_edge_lengths >= threshold) & (normalized_detour < np.percentile(normalized_detour, 75))
    combined_penalty[high_penalty_mask] *= 1.5

    # Symmetric edge distance update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
