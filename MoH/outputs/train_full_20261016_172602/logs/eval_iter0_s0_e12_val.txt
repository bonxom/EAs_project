# {Implement a dynamic Guided Local Search penalty mechanism that penalizes only the edges achieving the maximum utility (or top-k with dynamic adaptive threshold based on usage saturation) and applies a scaled additive penalty to preserve high-quality subpaths while breaking local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    tour_length = sum(tour_lengths)
    avg_edge = tour_length / float(n) if n > 0 else 1.0

    # Calculate utility according to Guided Local Search formulation
    utilities = [edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in tour_edges]
    max_utility = max(utilities) if utilities else 0.0

    if max_utility <= 0.0:
        return updated_edge_distance

    # Dynamic utility threshold: selectively target the maximum utility edges
    # Stagnation indicator based on average usage of edges in the current tour
    mean_used = np.mean([edge_n_used[u, v] for u, v in tour_edges])
    # When stagnation increases, broaden threshold slightly, otherwise strictly focus on the worst edge(s)
    threshold_ratio = max(0.92, 0.99 - 0.02 * min(mean_used, 3.0))

    # Standard GLS penalty parameter lambda proportional to tour length
    # lambda = alpha * (Tour_Length / n)
    penalty_lambda = 0.30 * avg_edge

    for (u, v), util in zip(tour_edges, utilities):
        if util >= threshold_ratio * max_utility:
            penalty = penalty_lambda
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty

    return updated_edge_distance
