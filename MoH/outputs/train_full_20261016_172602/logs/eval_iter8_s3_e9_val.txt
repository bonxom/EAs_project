#{We implement a dual-space perturbation operator that simultaneously penalizes high-cost tour edges using GLS utility with frequency damping and injects proactive incentive bonuses into underutilized k-nearest neighbor candidate non-tour edges to guide structural reconnection.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Tour edge penalization via GLS-style utility
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    mean_len = np.mean(tour_lengths) + 1e-8
    tour_counts = edge_n_used[u_idx, v_idx]

    # Utility balances edge cost and penalization frequency
    tour_utility = tour_lengths / (1.0 + 0.5 * tour_counts)
    max_utility = np.max(tour_utility) + 1e-8
    rel_utility = tour_utility / max_utility

    # Non-linear penalty scaling focused on top-utility (worst) edges
    penalty_scale = 0.15 * mean_len
    tour_penalties = penalty_scale * (rel_utility ** 2)

    # Apply symmetric penalties to current tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), tour_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), tour_penalties)

    # 2. Proactive incentive bonuses (distance reductions) on promising non-tour edges
    # Identify k-nearest neighbors for structural reconnection
    k_cand = min(8, n - 1)
    # Get top k candidate neighbors for each node (excluding self)
    sorted_neighbors = np.argsort(edge_distance, axis=1)[:, 1:k_cand + 1]

    # Create mask of existing tour edges to ensure bonuses only target non-tour connections
    tour_mask = np.zeros((n, n), dtype=bool)
    tour_mask[u_idx, v_idx] = True
    tour_mask[v_idx, u_idx] = True

    rows = np.repeat(np.arange(n), k_cand)
    cols = sorted_neighbors.ravel()

    # Filter to non-tour candidates with row < col to maintain symmetry without double counting
    valid_mask = (~tour_mask[rows, cols]) & (rows < cols)
    cand_u = rows[valid_mask]
    cand_v = cols[valid_mask]

    if len(cand_u) > 0:
        cand_dists = edge_distance[cand_u, cand_v]
        cand_used = edge_n_used[cand_u, cand_v]

        # Proactively incentivize short, rarely explored bridging edges
        attractiveness = (mean_len / (cand_dists + 1e-6)) / (1.0 + 0.3 * cand_used)
        bonus_scale = 0.05 * mean_len
        # Normalized incentive bound
        bonus = bonus_scale * np.clip(attractiveness / (np.max(attractiveness) + 1e-8), 0.0, 1.0)

        np.add.at(updated_edge_distance, (cand_u, cand_v), -bonus)
        np.add.at(updated_edge_distance, (cand_v, cand_u), -bonus)

    return updated_edge_distance
