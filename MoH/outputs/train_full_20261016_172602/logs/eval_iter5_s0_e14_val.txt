# ===== Heuristic Code =====
#{This algorithm replaces the hard max-utility threshold with a deterministic percentile-rank soft weighting scheme that penalizes edges above the 75th utility percentile scaled continuously by their relative utility rank.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Rank-based soft thresholding: penalize edges in the top quartile of utility
    # with strength proportional to their relative utility above the 75th percentile
    p75 = np.percentile(utilities, 75.0)
    max_utility = np.max(utilities)

    if max_utility > p75:
        # Scale penalties smoothly between 0.5 * base_penalty and base_penalty for top edges
        relative_rank = (utilities - p75) / (max_utility - p75)
        selected_mask = utilities >= p75
        penalties = base_penalty * (0.5 + 0.5 * relative_rank[selected_mask])
    else:
        selected_mask = utilities >= (0.99 * max_utility)
        penalties = base_penalty

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
