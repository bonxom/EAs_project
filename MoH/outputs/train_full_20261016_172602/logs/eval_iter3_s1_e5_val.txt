# ===== Heuristic Code =====
#{Dynamically calibrate utility thresholding via variance-sensitive Pareto quantiles and adaptively reward unused candidate edges scaled inversely by local node clustering density to preserve metric structure while escaping local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Adaptive quantile threshold based on saturation variance across tour edges
    used_var = np.var(used)
    quantile_level = float(np.clip(50.0 + 25.0 / (1.0 + used_var), 50.0, 80.0))
    threshold = np.percentile(utilities, quantile_level)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities == np.max(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Proportional GLS penalties
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k_nn = min(15, n - 1)

    # Vectorized topological neighborhood density evaluation to modulate decay
    for u in penalized_nodes:
        row_dist = edge_distance[u]
        # Identify nearest metric neighbors to estimate local clustering density
        nearest = np.argpartition(row_dist, k_nn)[:k_nn + 1]
        nearest = nearest[nearest != u][:k_nn]

        sub = edge_distance[np.ix_(nearest, nearest)]
        diag_mask = ~np.eye(len(nearest), dtype=bool)
        local_density = np.mean(sub[diag_mask]) / (np.mean(row_dist[nearest]) + 1e-8)

        # Scale decay inversely with clustering density to avoid over-discounting dense clusters
        adaptive_decay = float(np.clip(0.015 * local_density, 0.002, 0.035))

        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False

        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * adaptive_decay
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
