#{This algorithm prevents utility drift by computing Guided Local Search edge utilities from the unpenalized baseline distances while adaptively scaling penalties on the surrogate distance matrix based on edge usage frequency.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_lengths = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Recover original invariant base edge costs by discounting accumulated penalties
    avg_current = np.mean(current_lengths)
    base_lengths = current_lengths / (1.0 + 0.3 * counts)
    avg_base = np.mean(base_lengths)

    # Compute Guided Local Search utility relative to base geometric length
    utilities = base_lengths / (1.0 + counts)
    max_utility = np.max(utilities)

    # Adaptive penalty proportional to base tour length and scale
    penalty_factor = 0.3 * avg_base
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically increment surrogate distance matrix
    updated_edge_distance[sel_u, sel_v] += penalty_factor
    updated_edge_distance[sel_v, sel_u] += penalty_factor

    return updated_edge_distance
