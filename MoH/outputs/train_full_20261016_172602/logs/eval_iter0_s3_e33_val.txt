#{Replaces isotropic k-NN density with an anisotropic surplus delay metric that penalizes incumbent edges based on their excess cost relative to low-cost shortcut alternatives derived from endpoint neighbor structures, scaled by frequency dampening and relative edge lengths.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Compute anisotropic shortcut baseline using the 2nd and 3rd nearest neighbors of endpoints
    # (avoiding the 0-th self-loop distance)
    k_cand = min(4, n)
    sorted_dists = np.partition(edge_distance, kth=range(1, k_cand), axis=1)[:, 1:k_cand]
    sorted_dists.sort(axis=1)

    if sorted_dists.shape[1] >= 2:
        alt_u = 0.5 * (sorted_dists[u_idx, 0] + sorted_dists[u_idx, 1])
        alt_v = 0.5 * (sorted_dists[v_idx, 0] + sorted_dists[v_idx, 1])
    else:
        alt_u = sorted_dists[u_idx, 0]
        alt_v = sorted_dists[v_idx, 0]

    # Baseline alternative cost for the edge endpoints
    baseline_alt = 0.5 * (alt_u + alt_v)

    # Surplus delay: excess distance spent over readily available local shortcuts
    surplus_delay = tour_edge_lengths - baseline_alt
    surplus_ratio = np.maximum(0.0, surplus_delay) / (tour_edge_lengths + 1e-6)

    # Tour-wide relative edge length normalization
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length

    # Rank penalty based on sigmoid transition of relative length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Frequency dampening factor
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])

    # Local anisotropic penalty combining surplus ratio and relative edge length
    local_penalty = relative_length * (1.0 + 2.0 * surplus_ratio) * usage_factor

    # Dynamic weighting balancing global tour scale and local surplus penalty
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Amplify top quartile inefficient edges
    threshold = np.percentile(tour_edge_lengths, 75)
    combined_penalty[tour_edge_lengths >= threshold] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
