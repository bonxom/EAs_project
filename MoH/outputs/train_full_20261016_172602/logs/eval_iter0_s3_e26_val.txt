# ===== Heuristic Code =====
#{Geometric angularity estimation via law of cosines computes acute turning penalties for consecutive tour edges, combining geometric sharpness, edge length relativity, and usage frequency to distort the distance matrix away from jagged local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Nodes in cyclical order: prev (p), curr (u), next (v), after_next (w)
    p_idx = np.roll(local_opt_tour, 1)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    w_idx = np.roll(local_opt_tour, -2)

    # Edge distances along the consecutive path triplets
    d_uv = edge_distance[u_idx, v_idx]
    d_pu = edge_distance[p_idx, u_idx]
    d_pv = edge_distance[p_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Law of cosines to estimate cosine of angle at u (p-u-v) and angle at v (u-v-w)
    cos_u = (d_pu**2 + d_uv**2 - d_pv**2) / (2.0 * np.maximum(d_pu * d_uv, 1e-8))
    cos_v = (d_uv**2 + d_vw**2 - d_uw**2) / (2.0 * np.maximum(d_uv * d_vw, 1e-8))
    cos_u = np.clip(cos_u, -1.0, 1.0)
    cos_v = np.clip(cos_v, -1.0, 1.0)

    # Sharp turning: acute angles at u and v correspond to cos > 0
    # Higher sharpness penalty for sharper/jagged turns (acute inner angles)
    sharpness_u = np.maximum(0.0, cos_u)
    sharpness_v = np.maximum(0.0, cos_v)
    angular_penalty = 1.0 + 1.5 * (sharpness_u + sharpness_v)

    # Relative edge length penalty
    avg_tour_length = np.mean(d_uv)
    relative_length = d_uv / (avg_tour_length + 1e-8)
    length_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Usage frequency damping to avoid over-penalizing infrequently visited edges
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.25 * usage)

    # Local density context using nearest neighbors
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn = min(6, n)
    k_nn_avg = np.mean(sorted_dists[:, 1:k_nn], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-8)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx]) / np.mean(node_density)

    # Combined penalty modulated by geometric sharpness and edge properties
    total_penalty = relative_length * angular_penalty * usage_factor * (0.5 + 0.5 * density_factor) * (0.6 * length_penalty + 0.4)

    # Extra penalty for the longest jagged edges in the tour
    threshold = np.percentile(d_uv, 75)
    total_penalty[d_uv >= threshold] *= 1.3

    # Symmetric update
    np.add.at(updated_edge_distance, (u_idx, v_idx), total_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), total_penalty)

    return updated_edge_distance
