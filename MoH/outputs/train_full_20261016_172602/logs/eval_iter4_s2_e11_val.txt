# ===== Heuristic Code =====
#{Dynamically balance GLS length penalties with topological tour-centrality and acute inflection entropy to disrupt structural bridge stagnation while guiding alternative multi-hop shortcut incentives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decayed usage tracking to allow cyclical search exploration
    edge_n_used *= 0.96

    # Edge distances along tour triplets
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour and turn deficit metrics
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Local inflection factor capturing neighborhood geometry
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Tour-relative edge centrality / displacement context:
    # Measure relative stretch compared to average chord distance to identify structural bridge vs local clutter
    mean_local_dist = 0.5 * (d_uv + np.roll(d_uv, 1))
    stretch_factor = d_uv / (mean_local_dist + 1e-8)

    # Usage frequency along current edges
    base_freq = edge_n_used[u_idx, v_idx]

    # Entropic dynamic utility criterion: penalize edges that combine length, geometric inflection, and high relative stretch
    # Dampening heavily penalized edges through frequency saturation
    normalized_stretch = np.clip(stretch_factor, 0.5, 2.5)
    utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.40 * local_geom_factor) * (0.8 + 0.2 * normalized_stretch)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply adaptive penalties to selected bottleneck edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.45 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Shortcut discounting: reward low-frequency 2-hop shortcuts at acute detour points
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
