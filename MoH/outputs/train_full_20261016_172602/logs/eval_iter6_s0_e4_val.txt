# ===== Heuristic Code =====
#{Smooth edge utilities across adjacent tour edges using a cyclic local neighborhood kernel to penalize contiguous structural bottlenecks while maintaining symmetric penalties proportional to edge lengths and inverse usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate base utility inversely scaled by edge usage
    base_utilities = tour_lengths / (1.0 + tour_counts)

    # Smooth utilities using adjacent edges in the tour to capture local topological bottlenecks
    prev_utilities = np.roll(base_utilities, 1)
    next_utilities = np.roll(base_utilities, -1)
    smoothed_utilities = base_utilities + 0.25 * (prev_utilities + next_utilities)

    max_utility = np.max(smoothed_utilities)
    avg_edge = np.mean(tour_lengths)

    # Determine edges to penalize based on smoothed utility threshold
    threshold = 0.99 * max_utility
    selected_mask = smoothed_utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Penalize selected edges proportionally to average tour length and relative edge length
    penalties = 0.3 * avg_edge * (tour_lengths[selected_mask] / (avg_edge + 1e-9))

    # Apply symmetric penalty without in-place mutation of external state
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
