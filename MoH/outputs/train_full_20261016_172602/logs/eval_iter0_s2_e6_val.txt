# ===== Heuristic Code =====
# {Enhances guided local search penalties on penalized tour edges through an anisotropic bottleneck detector based on k-nearest-neighbor harmonic radii ratios coupled with entropy rigidity and tour stagnation modulation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)
    w_idx = np.roll(u_idx, -2)

    # Evaporate prior edge usage
    edge_n_used *= 0.95

    # Compute utilities and penalize the top candidate edges in the tour
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    k_pen = max(1, int(np.ceil(0.15 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k_pen:]
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Stagnation factor based on tour edge re-use and overlap
    tour_usage = edge_n_used[u_idx, v_idx]
    avg_tour_usage = np.mean(tour_usage)
    max_used = np.max(edge_n_used) + 1e-12
    overlap_ratio = np.mean(tour_usage > 1.0)
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (avg_tour_usage / max_used)

    # Entropy rigidity calculation for node usage distribution
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    # Localized geometric bottleneck detector using k-nearest-neighbor radius
    knn = max(3, min(8, num_nodes - 1))
    # Partition to find k nearest distances per node excluding self (index 0 if dist is 0)
    sorted_dists = np.partition(edge_distance, knn, axis=1)[:, :knn + 1]
    # Exclude distance to self (0.0)
    knn_dists = np.where(sorted_dists > 1e-12, sorted_dists, np.nan)
    knn_radius = np.nanmean(knn_dists, axis=1)
    knn_radius = np.nan_to_num(knn_radius, nan=np.mean(edge_distance)) + 1e-12

    # Segment usage weights
    segment_weights = np.zeros(num_nodes)
    for i in range(num_nodes):
        u, v, w = u_idx[i], v_idx[i], w_idx[i]
        segment_weights[v] = 0.5 * (edge_n_used[u, v] + edge_n_used[v, w])
    norm_segment_weights = segment_weights / (np.max(segment_weights) + 1e-12)

    # Spatial centrality metrics
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial_centrality = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Overall adaptive penalty scaling factor
    adaptive_lambda = (0.15 + 0.20 * np.tanh(stagnation_factor)) * global_mean_dist

    # Apply anisotropic modulated penalties
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        d_uv = edge_distance[u, v]

        # Harmonic mean of incident nodes' k-NN radii to detect bridge edges
        r_u, r_v = knn_radius[u], knn_radius[v]
        harmonic_r = (2.0 * r_u * r_v) / (r_u + r_v + 1e-12)
        bottleneck_ratio = d_uv / harmonic_r
        bottleneck_mod = np.clip(bottleneck_ratio, 0.5, 3.0) - 1.0

        sc_mod = 0.20 * (norm_spatial_centrality[u] + norm_spatial_centrality[v])
        r_mod = 0.25 * (entropy_rigidity[u] + entropy_rigidity[v])
        s_mod = 0.15 * (norm_segment_weights[u] + norm_segment_weights[v])
        b_mod = 0.30 * bottleneck_mod

        modulation = 1.0 + sc_mod + r_mod + s_mod + b_mod
        penalty = adaptive_lambda * edge_n_used[u, v] * modulation

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
