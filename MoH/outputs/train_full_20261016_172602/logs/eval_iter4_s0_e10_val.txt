#{Replace the rigid threshold with a rank-based quantile selector that scales penalties dynamically using edge utility variance and tour length diversity to break cyclic trapping.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    # Adaptive top-k selection based on count diversity and tour variance
    n = len(local_opt_tour)
    unique_counts = np.unique(tour_counts).size
    k = max(1, min(int(np.ceil(unique_counts / 2.0)), max(1, n // 5)))

    top_indices = np.argsort(utilities)[-k:]

    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    base_penalty = 0.25 * avg_edge + 0.1 * std_edge

    # Rank-proportional fine-grained penalty to break sub-tour symmetry
    sel_u = u_idx[top_indices]
    sel_v = v_idx[top_indices]
    ranks = np.arange(1, k + 1)
    penalties = base_penalty * (1.0 + 0.2 * (ranks / float(k)))

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
