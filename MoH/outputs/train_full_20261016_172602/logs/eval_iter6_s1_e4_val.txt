# ===== Heuristic Code =====
#{This algorithm refines Guided Local Search by penalizing above-median utility tour edges via z-score sigmoid weighting and selectively discounting the nearest unused candidate edges per penalized node using distance-proportional incentives rather than a flat decay.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    mean_penalized_cost = np.mean(p_costs) if len(p_costs) > 0 else mean_cost

    for u in penalized_nodes:
        unused_indices = np.where((edge_n_used[u] == 0))[0]
        unused_indices = unused_indices[unused_indices != u]
        if len(unused_indices) > 0:
            c_vals = updated_edge_distance[u, unused_indices]
            # Prioritize top nearest unused neighbors (up to 5) to prevent distorting unpromising long edges
            if len(c_vals) > 5:
                top_k_idx = np.argpartition(c_vals, 5)[:5]
                cand_idx = unused_indices[top_k_idx]
                cand_costs = c_vals[top_k_idx]
            else:
                cand_idx = unused_indices
                cand_costs = c_vals

            decay_rates = np.clip(0.015 * (mean_penalized_cost / (cand_costs + 1e-8)), 0.005, 0.04)
            discount = cand_costs * decay_rates
            updated_edge_distance[u, cand_idx] -= discount
            updated_edge_distance[cand_idx, u] -= discount

    return updated_edge_distance
