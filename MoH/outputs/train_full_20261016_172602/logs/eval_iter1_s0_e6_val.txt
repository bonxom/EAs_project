#{Select candidate edges via a dynamic high-percentile threshold tempered by an inverse-frequency decay weighting to protect essential bridge connections from excessive penalization.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Vectorized GLS utility evaluation based on edge cost over usage count
    utilities = tour_lengths / (1.0 + tour_counts)

    # Percentile-based selection to target top utility candidates dynamically
    utility_cutoff = np.percentile(utilities, 85.0)
    selected_mask = utilities >= utility_cutoff

    # Base penalty scaled to instance-specific average edge length
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Frequency decay damping factor to prevent over-penalizing critical bridge edges
    decay_weights = 1.0 / np.sqrt(1.0 + tour_counts[selected_mask])
    adjusted_penalties = base_penalty * decay_weights

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply symmetric penalties using NumPy vectorized indexing
    updated_edge_distance[sel_u, sel_v] += adjusted_penalties
    updated_edge_distance[sel_v, sel_u] += adjusted_penalties

    return updated_edge_distance
