#{Adaptive dual-scale Guided Local Search penalty combining dynamic log(n)-proportional k-NN density estimation with z-score edge length normalization and search-frequency decay.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Dynamic neighborhood size proportional to log(n) for topology-aware density estimation
    k = max(3, min(n - 1, int(np.ceil(2.0 * np.log(n)))))
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Extract current tour edges and metrics
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Variance-aware z-score scaling for edge lengths
    mean_len = np.mean(tour_edge_lengths)
    std_len = np.std(tour_edge_lengths) + 1e-6
    z_scores = (tour_edge_lengths - mean_len) / std_len

    # Sigmoidal global rank penalty informed by z-scores
    rank_penalty = 1.0 / (1.0 + np.exp(-1.5 * z_scores))

    # Local penalty combining usage dampening and structural density of endpoints
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    relative_length = tour_edge_lengths / (mean_len + 1e-6)
    local_penalty = relative_length * usage_factor * density_factor

    # Frequency-modulated dual-scale balancing
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Symmetric edge distance update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
