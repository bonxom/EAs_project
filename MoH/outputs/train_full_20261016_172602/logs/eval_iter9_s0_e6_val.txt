# ===== Heuristic Code =====
#{Decouple utility evaluation from cumulative penalty distortion by recovering the unpenalized ground-truth distance matrix using edge usage counts and applying vectorized Guided Local Search penalties to the augmented matrix.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_aug_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate adaptive penalty scaling factor based on current tour characteristics
    avg_edge = np.mean(tour_aug_lengths)
    n_nodes = len(local_opt_tour)
    penalty_factor = 0.3 * avg_edge / (1.0 + 0.05 * np.log1p(n_nodes))

    # Recover original ground-truth costs by subtracting accumulated penalties
    # Ground truth: original_distance = augmented_distance - penalty_factor * count
    orig_tour_lengths = np.maximum(tour_aug_lengths - penalty_factor * tour_counts, 1e-8)

    # Guided Local Search utility evaluated strictly on original costs
    utilities = orig_tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply penalty symmetrically to the augmented matrix
    updated_edge_distance[sel_u, sel_v] += penalty_factor
    updated_edge_distance[sel_v, sel_u] += penalty_factor

    return updated_edge_distance
