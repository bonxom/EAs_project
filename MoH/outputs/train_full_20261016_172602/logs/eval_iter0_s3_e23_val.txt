# ===== Heuristic Code =====
#{A multi-scale harmonic density penalty strategy that combines sigmoid tour rank penalties with the harmonic mean of endpoint densities and an edge-bridging saturation factor to aggressively penalize long edges trapped in dense clusters while protecting valid inter-cluster bridges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Extract tour node indices
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Efficient k-nearest neighbors distance computation
    k = min(5, n - 1)
    k_nn_dists = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    node_scale = np.mean(k_nn_dists, axis=1) + 1e-6

    # Node densities and harmonic mean between endpoints
    node_density = 1.0 / node_scale
    norm_density = node_density / (np.mean(node_density) + 1e-6)
    d_u = norm_density[u_idx]
    d_v = norm_density[v_idx]
    harmonic_density = (2.0 * d_u * d_v) / (d_u + d_v + 1e-6)

    # Cluster scale and bridging ratio
    cluster_scale = np.sqrt(node_scale[u_idx] * node_scale[v_idx])
    bridging_ratio = tour_edge_lengths / cluster_scale

    # Bridge modulation factor: high for long intra-cluster edges, dampened for legitimate bridges
    bridge_factor = 1.0 + np.tanh(bridging_ratio - 1.0)

    # Sigmoid-ranked baseline from Parent 1
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Local penalty combining harmonic density, bridge factor, and usage frequency
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    local_penalty = relative_length * harmonic_density * bridge_factor * usage_factor

    # Adaptive global vs local balancing
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Threshold-based boost for upper-quartile edges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_mask = tour_edge_lengths >= threshold
    combined_penalty[high_mask] *= 1.5

    # Symmetric matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
