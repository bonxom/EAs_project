# ===== Heuristic Code =====
#{This strategy adaptively scales the edge usage decay rate and late-stage penalty weights using the global search progression indicated by mean tour edge frequency, preventing cyclic entrapment while amplifying penalties on disproportionately long edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Estimate local node density using k-nearest neighbor distances
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Extract current tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Search-phase adaptive dynamics based on average edge usage across the current tour
    freq_mean = np.mean(edge_counts)
    decay_rate = 0.25 + 0.15 * np.log1p(freq_mean)
    usage_factor = 1.0 / (1.0 + decay_rate * edge_counts)

    # Compute rank-based logistic penalty and density-informed local penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Dynamically balance exploration (global ranking) vs exploitation (local density/usage)
    w_global = 0.6 / (1.0 + 0.08 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Scale penalty by base tour length to maintain consistent dimensional perturbation
    combined_penalty = combined_penalty * (0.05 * avg_tour_length)

    # Amplify penalties on the longest quartile of edges in the current tour
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Symmetrically update distances
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
