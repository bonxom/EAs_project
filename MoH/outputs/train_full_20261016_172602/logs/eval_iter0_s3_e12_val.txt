# ===== Heuristic Code =====
#{Decouple penalty accumulation using a sub-linear square-root saturation factor combined with an edge stagnation filter to avoid over-distorting recurrent cyclic edges into degenerate minima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)
    p = np.roll(local_opt_tour, 1)
    w = np.roll(local_opt_tour, -2)

    tour_lens = edge_distance[u, v]
    usage = edge_n_used[u, v]

    k = min(6, n - 1)
    k_nn_dists = np.partition(edge_distance, k, axis=1)[:, 1:k]
    node_scale = np.mean(k_nn_dists, axis=1) + 1e-6
    cluster_scale = np.sqrt(node_scale[u] * node_scale[v])
    bridging_ratio = tour_lens / cluster_scale

    local_adj_mean = 0.5 * (edge_distance[p, u] + edge_distance[v, w]) + 1e-6
    slack_ratio = tour_lens / local_adj_mean

    mean_l = np.mean(tour_lens) + 1e-6
    std_l = np.std(tour_lens)
    cv = std_l / mean_l
    rel_global = tour_lens / mean_l

    # Sub-linear saturation factor replacing linear dampening
    usage_factor = 1.0 / np.sqrt(1.0 + usage)

    # Edge stagnation dampener: heavily penalize overused cyclic edges to avoid degenerate distortion
    mean_usage = np.mean(usage)
    stagnation_factor = np.where(usage > 2.0 * mean_usage + 1.0, 0.5, 1.0)

    bridge_weight = np.tanh(bridging_ratio - 1.0)
    topology_score = 0.5 * rel_global + 0.3 * slack_ratio + 0.2 * (bridge_weight + 1.0)

    cutoff_pct = np.clip(75.0 - 15.0 * cv, 50.0, 85.0)
    threshold = np.percentile(topology_score, cutoff_pct)

    penalties = topology_score * usage_factor * stagnation_factor
    boost_mask = topology_score >= threshold
    penalties[boost_mask] *= (1.25 + 0.25 * np.clip(cv, 0.0, 1.0))

    np.add.at(updated_edge_distance, (u, v), penalties)
    np.add.at(updated_edge_distance, (v, u), penalties)
    return updated_edge_distance
