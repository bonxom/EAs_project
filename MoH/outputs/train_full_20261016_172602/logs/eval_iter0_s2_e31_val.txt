# ===== Heuristic Code =====
#{A geometric detour and angle-deficit penalty strategy that evaluates consecutive tour triplets to identify sharp turnings and triangle-inequality slack, compounding penalties onto geometrically inefficient edges while discounting low-cost shortcut alternatives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    tour = np.asarray(local_opt_tour)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Standard decay to prevent long-term saturation
    edge_n_used *= 0.96

    # Geometric triplet analysis: (u -> v -> w) vs shortcut (u -> w)
    # detour_ratio quantifies triangle-inequality slack: (d(u,v) + d(v,w)) / d(u,w)
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Avoid division by zero with safe epsilon
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)

    # Identify acute angles / sharp back-and-forth turns using cosine approximation:
    # cos_v = (d_uv^2 + d_vw^2 - d_uw^2) / (2 * d_uv * d_vw)
    # Smaller / negative cos_v implies acute detour or sharp turnaround
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)

    # Acute deficit: high when turning angle is sharp (cos_turn < 0)
    acute_deficit = 1.0 - cos_turn

    # Geometric inefficiency score for each node transition
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Distribute node-level geometric inefficiency back to edges (u, v) and (v, w)
    # An edge's penalty utility incorporates standard distance, usage frequency, and geometric flaw
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Combined utility: combines GLS utility with incoming and outgoing geometric tension
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Select edges contributing the most to spatial bottlenecks and topological crossings
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Global reference distance
    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply targeted penalties
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Actively discount valid shortcut chords (u, w) where detour ratio is high
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        shortcut_cost = edge_distance[u, w]
        freq = edge_n_used[u, w]

        # Reward shortcut edges that haven't been heavily explored yet
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
