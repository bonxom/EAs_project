# ===== Heuristic Code =====
#{This algorithm detects search stagnation using edge frequency concentration and residual geometric variance to dynamically modulate memory decay from soft retention to an aggressive flash-reset, applying variance-calibrated subgradient penalties to inefficient edges while selectively rewarding unvisited geometric shortcuts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Analyze stagnation and saturation via tour edge frequency concentration
    tour_freqs = edge_n_used[u_idx, v_idx]
    max_freq = np.max(tour_freqs) if len(tour_freqs) > 0 else 0.0
    mean_freq = np.mean(tour_freqs) if len(tour_freqs) > 0 else 0.0
    stagnation_ratio = mean_freq / (1.0 + np.std(tour_freqs))

    # Stagnation-triggered adaptive memory reset: aggressive flash-reset when cycling/stagnant
    if max_freq > 8.0 or stagnation_ratio > 3.0:
        decay_rate = 0.70
    elif max_freq > 4.0 or stagnation_ratio > 1.8:
        decay_rate = 0.88
    else:
        decay_rate = 0.98
    edge_n_used *= decay_rate

    # Geometric triplet metrics: detour ratio and acute angular turn deficit
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    current_freqs = edge_n_used[u_idx, v_idx]

    # Utilities balanced by inverse frequency and geometric inefficiency
    utilities = (d_uv / (1.0 + current_freqs)) * (1.0 + 0.35 * local_geom_factor)

    # Dynamic sub-gradient penalty scaling calibrated against local search residual variance
    global_mean_dist = np.mean(edge_distance)
    util_std = np.std(utilities)
    util_mean = np.mean(utilities) + 1e-8
    residual_cv = np.clip(util_std / util_mean, 0.1, 2.0)
    adaptive_lambda = (0.25 + 0.15 * residual_cv) * global_mean_dist

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Reward promising shortcut topological skips that bypass high-detour vertices
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = (0.08 + 0.04 * (1.0 - decay_rate)) * global_mean_dist * (1.0 / (1.0 + freq))
            new_dist = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_dist
            updated_edge_distance[w, u] = new_dist

    return updated_edge_distance
