# ===== Heuristic Code =====
#{Guided Local Search is improved by penalizing only the highest-utility tour edges proportional to tour length scaled by classical GLS utility c_ij / (1 + p_ij), avoiding over-penalizing the entire tour and retaining structural neighborhood quality.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(local_opt_tour, -1)

    # Base cost and usage of edges in the current local optimum
    costs = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Classical GLS utility: utility = cost / (1 + count)
    utilities = costs / (1.0 + counts)

    max_util = np.max(utilities)
    if max_util <= 1e-12:
        return updated_edge_distance

    # Select only the top edges with maximum or near-maximum utility (top 15-20% or within 90% of max)
    threshold = 0.85 * max_util
    target_mask = utilities >= threshold

    # Standard GLS penalty factor: lambda = alpha * (tour_length / n)
    tour_len = np.sum(costs)
    avg_edge_len = tour_len / float(n)
    penalty_value = 0.3 * avg_edge_len

    sel_u = u_idx[target_mask]
    sel_v = v_idx[target_mask]

    np.add.at(updated_edge_distance, (sel_u, sel_v), penalty_value)
    np.add.at(updated_edge_distance, (sel_v, sel_u), penalty_value)

    return updated_edge_distance
