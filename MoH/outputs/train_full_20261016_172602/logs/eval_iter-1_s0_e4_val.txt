# ===== Heuristic Code =====
#{This heuristic introduces directional penalties along the tour traversal orientation scaled by edge cost and usage frequency, perturbing the search space while preserving baseline connectivity.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    # Calculate directed tour edges and directional utilities
    forward_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in forward_edges]
    mean_length = float(np.mean(tour_lengths)) if tour_lengths else 1.0

    # Evaluate directional utility based on length and directional usage count
    utilities = [edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in forward_edges]
    max_utility = max(utilities) if utilities else 0.0

    # Base penalty scale proportional to average edge length
    penalty_scale = 0.3 * mean_length

    # Apply asymmetric forward-oriented penalties to top utility edges, with mild reverse feedback
    for (u, v), util in zip(forward_edges, utilities):
        if util >= 0.8 * max_utility:
            penalty = penalty_scale / (1.0 + edge_n_used[u, v])
            # Primary asymmetric penalty along traversal direction
            updated_edge_distance[u, v] += penalty
            # Partial counter-penalty to encourage structural diversion without full metric distortion
            updated_edge_distance[v, u] += 0.3 * penalty

    return updated_edge_distance
