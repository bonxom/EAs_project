# ===== Heuristic Code =====
#{This strategy establishes a bimodal dual-potential field by heavily penalizing high-regret tour edges while applying multiplicative pseudo-distance deflation to unvisited candidate edges in low-usage sparse neighborhoods to drive cross-cluster exploration.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # 1. Evaluate tour edges and compute regret against 2-nearest alternative connections
    tour_dists = edge_distance[u_idx, v_idx]
    mean_tour_dist = np.mean(tour_dists) + 1e-8

    # Estimate k-nearest neighborhood distances for node scale
    k = min(8, n - 1)
    knn_indices = np.argpartition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    knn_dists = np.take_along_axis(edge_distance, knn_indices, axis=1)
    node_scale = np.mean(knn_dists, axis=1) + 1e-8

    # Edge cost relative to endpoints' local scales
    edge_scale = 0.5 * (node_scale[u_idx] + node_scale[v_idx])
    regret_ratio = tour_dists / edge_scale
    usage_factor = 1.0 / np.sqrt(1.0 + edge_n_used[u_idx, v_idx])

    # High-regret penalization: add scaled penalty to active tour edges
    penalty = (tour_dists * 0.4 + edge_scale * 0.3) * np.maximum(0.2, regret_ratio - 0.8) * usage_factor
    # Extra boost for upper quartile edges
    p75 = np.percentile(tour_dists, 75)
    penalty[tour_dists >= p75] *= 1.4

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    # 2. Multiplicative deflation (bonuses) on unvisited/low-usage sparse candidate edges
    # Identify promising non-tour edges among k-nearest neighbors
    source_nodes = np.repeat(np.arange(n), k)
    target_nodes = knn_indices.flatten()

    # Mask out edges currently in the tour
    tour_edge_mask = np.zeros((n, n), dtype=bool)
    tour_edge_mask[u_idx, v_idx] = True
    tour_edge_mask[v_idx, u_idx] = True

    cand_mask = ~tour_edge_mask[source_nodes, target_nodes]
    cand_u = source_nodes[cand_mask]
    cand_v = target_nodes[cand_mask]

    # Compute deflation factor proportional to under-exploration and edge compactness
    cand_usage = edge_n_used[cand_u, cand_v]
    exploration_potential = 1.0 / (1.0 + cand_usage)
    cand_lengths = edge_distance[cand_u, cand_v]
    relative_cand_length = cand_lengths / (0.5 * (node_scale[cand_u] + node_scale[cand_v]))

    # Apply discount: closer and less-used candidate edges receive stronger deflation
    discount = 0.20 * exploration_potential * np.exp(-np.clip(relative_cand_length, 0.0, 3.0))
    deflation_factors = 1.0 - discount

    # Symmetrically deflate candidate edges
    updated_edge_distance[cand_u, cand_v] *= deflation_factors
    updated_edge_distance[cand_v, cand_u] = updated_edge_distance[cand_u, cand_v]

    # Ensure non-negative distances and zero self-distances
    np.fill_diagonal(updated_edge_distance, 0.0)
    np.maximum(updated_edge_distance, 0.0, out=updated_edge_distance)

    return updated_edge_distance
