# ===== Heuristic Code =====
#{This algorithm dynamically modulates guided local search penalties by blending structural node-density and edge usage decay with smooth sigmoidal rank penalties, transitioning from early structure-guided exploration to late rank-dominated exploitation based on cumulative tour usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Measure local geometry via nearest neighbor density
    k_nn = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k_nn + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Extract tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Phase transition parameter derived from search progression (average usage frequency)
    mean_usage = np.mean(edge_counts)
    # Early phase: w_structural dominates; Late phase: w_rank dominates
    w_rank = 1.0 / (1.0 + np.exp(-0.25 * (mean_usage - 4.0)))
    w_structural = 1.0 - w_rank

    # Structural penalty: length scaled by incident node density and inverse usage
    usage_decay = 1.0 / (1.0 + 0.35 * edge_counts)
    incident_density = 0.5 * (node_density[u_idx] + node_density[v_idx])
    structural_penalty = relative_length * incident_density * usage_decay

    # Rank penalty: smooth sigmoidal penalty targeting longest edges
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Differential blend bridging exploration and exploitation
    combined_penalty = (w_structural * structural_penalty + w_rank * rank_penalty)

    # Proportional scaling by base edge distance scale to ensure effective GLS perturbation
    scale = avg_tour_length * 0.1
    effective_penalty = scale * combined_penalty

    # Boost penalty on top quartile bottleneck edges
    p75 = np.percentile(tour_edge_lengths, 75)
    effective_penalty[tour_edge_lengths >= p75] *= 1.35

    # Symmetric matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), effective_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), effective_penalty)

    return updated_edge_distance
