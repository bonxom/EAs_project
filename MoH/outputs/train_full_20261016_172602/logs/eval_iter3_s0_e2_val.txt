# ===== Heuristic Code =====
#{Vectorized adaptive guided local search that penalizes all tour edges exceeding an 85% maximum-utility threshold conditioned on being above the tour's average edge cost to promote effective multi-edge diversification.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Relaxed multi-edge selection threshold combined with above-average cost filtering
    selected_mask = (utilities >= 0.85 * max_utility) & (tour_lengths >= avg_edge)

    # Ensure at least the maximal utility edge is penalized if the combined condition is too strict
    if not np.any(selected_mask):
        selected_mask = (utilities >= 0.99 * max_utility)

    penalty = 0.3 * avg_edge
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
