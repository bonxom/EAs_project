# ===== Heuristic Code =====
#{Replaces full distance sorting with fast partial partitioning for k-NN density estimation, applies continuous smooth utility scoring normalized by median tour length, and dynamically scales perturbation magnitude based on tour edge frequency saturation to avoid stagnation without disruptive metric distortion.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Fast k-nearest neighbor distance approximation using partition instead of full sort
    k = min(5, n - 1)
    k_smallest = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    k_nn_mean = np.mean(k_smallest, axis=1) + 1e-8
    node_density = 1.0 / k_nn_mean
    node_density /= np.mean(node_density)

    # Extract consecutive tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    median_len = np.median(tour_edge_lengths) + 1e-8
    mean_len = np.mean(tour_edge_lengths) + 1e-8

    # Edge usage count for edges currently in the tour
    counts = edge_n_used[u_idx, v_idx]
    mean_usage = np.mean(counts)

    # Dynamic temperature factor that scales perturbation when cycling in familiar edges
    dispersion_factor = 1.0 + np.tanh(mean_usage / 4.0)

    # Relative edge length compared to median tour length
    rel_len = tour_edge_lengths / median_len

    # Continuous smooth ranking: penalize longer edges smoothly without abrupt percentile cliffs
    length_penalty = 1.0 / (1.0 + np.exp(-3.0 * (rel_len - 1.0)))

    # Topological density context (edges connecting dense nodes have higher penalty to encourage short jumps)
    density_context = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Guided local search penalty utility: length / (1 + count)
    gls_utility = (tour_edge_lengths / (1.0 + counts)) / mean_len

    # Smooth combined utility score
    combined_score = gls_utility * length_penalty * density_context

    # Scale additive penalty proportional to the median tour edge length and dynamic temperature
    penalty_magnitude = 0.3 * median_len * dispersion_factor * combined_score

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty_magnitude)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty_magnitude)

    return updated_edge_distance
