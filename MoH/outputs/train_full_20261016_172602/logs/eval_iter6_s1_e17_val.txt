#{Standard guided local search penalty with calibrated scaling based on tour length and maximum utility edge selection, preventing metric divergence while precisely forcing escaping moves.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Create a safe copy of the edge distance matrix
    updated_edge_distance = np.copy(edge_distance).astype(np.float64)
    n = len(local_opt_tour)

    # Step 2: Form tour edges
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Step 3: Extract current costs and frequency counts for tour edges
    costs = updated_edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Step 4: Classical GLS utility calculation: c_ij / (1 + count_ij)
    utilities = costs / (1.0 + counts)

    # Step 5: Identify edges with maximum utility (or very close to maximum)
    max_util = np.max(utilities)
    threshold = max_util - 1e-9
    penalize_mask = utilities >= threshold

    # Step 6: Scale penalty relative to total tour cost and size to balance exploration/exploitation
    tour_length = np.sum(costs)
    lambda_param = 0.3 * (tour_length / float(n))

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]

    # Apply symmetric additive penalty
    updated_edge_distance[p_u, p_v] += lambda_param
    updated_edge_distance[p_v, p_u] += lambda_param

    # Ensure zero self-distances
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
