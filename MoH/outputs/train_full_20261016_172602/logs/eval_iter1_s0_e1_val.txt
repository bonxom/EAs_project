# ===== Heuristic Code =====
#{A multi-tiered Guided Local Search penalty strategy that evaluates edge utilities along the tour and applies scaled, proportional penalties to top-percentile bottleneck edges to prevent stagnation while preserving search structure.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate Guided Local Search (GLS) edge utilities
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility <= 0:
        return updated_edge_distance

    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Multi-tiered thresholding for penalizing bottleneck edges
    # Tier 1: Highest utility edges (>= 0.95 * max_utility) receive full penalty
    tier1_mask = utilities >= 0.95 * max_utility
    # Tier 2: Sub-peak utility edges ([0.85, 0.95) * max_utility) receive partial penalty
    tier2_mask = (utilities >= 0.85 * max_utility) & (~tier1_mask)

    # Apply Tier 1 updates symmetrically
    u1, v1 = u_idx[tier1_mask], v_idx[tier1_mask]
    updated_edge_distance[u1, v1] += base_penalty
    updated_edge_distance[v1, u1] += base_penalty

    # Apply Tier 2 updates symmetrically
    u2, v2 = u_idx[tier2_mask], v_idx[tier2_mask]
    updated_edge_distance[u2, v2] += 0.5 * base_penalty
    updated_edge_distance[v2, u2] += 0.5 * base_penalty

    return updated_edge_distance
