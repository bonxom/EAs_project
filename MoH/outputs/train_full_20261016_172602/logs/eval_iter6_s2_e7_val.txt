# ===== Heuristic Code =====
# {A mathematical bound-guided GLS framework that dynamically penalizes geometrically inefficient tour edges using dual-slack surrogate costs and node degree variance, while applying safely bounded chord incentives.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # 1. Decay historical edge usage to maintain recency
    edge_n_used *= 0.95

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # 2. Geometric detour and acute angle inefficiency calculation
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # 3. Asymmetric edge contribution: penalize longer edges within detour triplets proportionally
    d_mean_tour = np.mean(d_uv)
    edge_length_bias = d_uv / (d_mean_tour + 1e-8)

    local_geom = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    base_freq = edge_n_used[u_idx, v_idx]

    # 4. Surrogate dual utility: GLS utility scaled by geometric and length-biased inefficiency
    utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom) * np.sqrt(edge_length_bias)

    # 5. Dynamic k selection based on tour length dispersion (adapts to cluster density)
    tour_std = np.std(d_uv)
    rel_std = tour_std / (d_mean_tour + 1e-8)
    k_fraction = np.clip(0.12 + 0.08 * rel_std, 0.10, 0.22)
    k = max(1, int(np.ceil(k_fraction * num_nodes)))

    top_k_indices = np.argsort(utilities)[-k:]

    # 6. Global instance scaling & adaptive Lagrangian multiplier
    global_mean_dist = np.mean(edge_distance)
    overlap_ratio = np.mean(base_freq > 1.0)
    adaptive_lambda = (0.25 + 0.15 * np.tanh(overlap_ratio * 2.0)) * global_mean_dist

    # Node centrality memory
    node_usage = np.sum(edge_n_used, axis=1)
    norm_node_usage = node_usage / (np.max(node_usage) + 1e-8)

    # 7. Apply selective penalties with asymmetric regularization
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        node_mod = 1.0 + 0.25 * (norm_node_usage[u] + norm_node_usage[v])
        geom_mod = 1.0 + 0.40 * local_geom[idx]

        penalty = adaptive_lambda * freq_penalty * node_mod * geom_mod
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 8. Bounded chord incentive (discounts 2-hop shortcuts safely relative to original distance)
    chord_k = max(1, int(np.ceil(0.5 * k)))
    high_detour_indices = np.argsort(detour_ratio)[-chord_k:]

    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        orig_chord_dist = edge_distance[u, w]
        freq = edge_n_used[u, w]
        if freq < 0.5:
            # Discount bound: never drop below 75% of the original chord distance
            max_discount = 0.25 * orig_chord_dist
            nominal_discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            discount = min(max_discount, nominal_discount)

            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
