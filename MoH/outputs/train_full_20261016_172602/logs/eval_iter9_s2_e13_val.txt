# ===== Heuristic Code =====
#{This strategy dynamically penalizes tour edges using an adaptive statistical inflection threshold on geometric-guided utilities coupled with a usage-driven stagnation detector to expand or contract the penalization set and reward high-potential shortcut candidates.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Natural memory decay
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour and corner sharpness calculation
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # GLS-style utility accounting for distance, recurrence, and geometry
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Stagnation-aware adaptive cardinality budget & inflection thresholding
    mean_freq = np.mean(base_freq)
    stagnation_factor = np.clip(mean_freq / 2.0, 0.0, 1.0)

    # High stagnation lowers threshold multiplier and expands budget
    alpha = 1.25 - 0.55 * stagnation_factor
    min_budget = max(1, int(np.ceil((0.04 + 0.08 * stagnation_factor) * num_nodes)))
    max_budget = max(min_budget, int(np.ceil((0.15 + 0.15 * stagnation_factor) * num_nodes)))

    u_mean = np.mean(utilities)
    u_std = np.std(utilities)
    threshold = u_mean + alpha * u_std

    # Candidate indices exceeding inflection threshold
    selected_mask = utilities >= threshold
    selected_indices = np.where(selected_mask)[0]

    # Enforce adaptive cardinality bounds
    if len(selected_indices) < min_budget:
        selected_indices = np.argsort(utilities)[-min_budget:]
    elif len(selected_indices) > max_budget:
        top_in_selected = np.argsort(utilities[selected_indices])[-max_budget:]
        selected_indices = selected_indices[top_in_selected]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.25 + 0.15 * stagnation_factor) * global_mean_dist

    # Apply GLS penalties to structural bottleneck edges
    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Reward promising shortcut edges bypassing extreme detours
    k_discount = len(selected_indices)
    high_detour_indices = np.argsort(detour_ratio)[-k_discount:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
