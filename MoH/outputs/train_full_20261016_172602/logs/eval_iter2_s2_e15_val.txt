#{Guided Local Search edge distance augmentation featuring stagnation-driven dynamic penalty evaporation and variance-scaled geometric penalties with bounded shortcut bonuses to escape cyclic basins.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Analyze tour edge statistics and dispersion to detect stagnation / basin entrapment
    tour_dist_mean = np.mean(d_uv)
    tour_dist_std = np.std(d_uv)
    cv = tour_dist_std / (tour_dist_mean + 1e-8)

    # Adaptive stagnation-driven evaporation:
    # High edge usage concentration or low coefficient of variation indicates trapped cyclic states,
    # triggering sharp annealed decay to evaporate over-penalized edges.
    max_used = np.max(edge_n_used)
    stagnation_factor = np.clip(max_used / (15.0 + 1e-8), 0.0, 1.0)
    decay_rate = 0.94 - 0.12 * stagnation_factor + 0.04 * np.clip(cv, 0.0, 1.0)
    edge_n_used *= decay_rate

    # Geometric detour calculation and directional slack
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.45 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # GLS Edge utility prioritizing long, geom-inefficient, and under-penalized edges
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.30 * local_geom_factor)

    k = max(1, int(np.ceil(0.12 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    # Bounded adaptive penalty scaled with stagnation and tour dispersion
    adaptive_lambda = (0.25 + 0.10 * (1.0 - stagnation_factor)) * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        # Sub-linear saturation to prevent unbounded explosion
        freq_penalty = np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.35 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Shortcut incentive bonuses for bypassing high-detour intermediate vertices
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            min_bound = 0.3 * edge_distance[u, w]
            new_val = max(min_bound, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
