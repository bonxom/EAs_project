# ===== Heuristic Code =====
#{Modulate guided local search edge penalties and chord incentives using node centrality ratios and incident edge congestion pressures to aggressively perturb interior bottleneck nodes while shielding sparse boundary vertices.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    # Precompute global and node-level distance centrality
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    # Node centrality: interior nodes have low node_mean_dist -> high centrality ratio
    node_centrality = global_mean_dist / (node_mean_dist + 1e-8)

    # Node-level incident churn/congestion from edge_n_used
    node_incident_churn = np.sum(edge_n_used, axis=1)
    mean_churn = np.mean(node_incident_churn) + 1e-8
    relative_node_churn = node_incident_churn / mean_churn

    # Combined node congestion factor
    node_pressure = 0.5 * (node_centrality + relative_node_churn)

    # Geometric metrics on the local tour
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Incident-node pressure for each tour edge (u, v)
    edge_node_pressure = 0.5 * (node_pressure[u_idx] + node_pressure[v_idx])

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Refined utility calculation with incident-node congestion pressure
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.30 * local_geom_factor) * (0.8 + 0.4 * edge_node_pressure)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.4 * local_geom_factor[idx]
        congestion_mod = 0.75 + 0.5 * edge_node_pressure[idx]

        penalty = adaptive_lambda * freq_penalty * geom_weight * congestion_mod
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate shortcuts prioritizing paths that bypass congested or high-detour intermediate vertices
    intermediate_pressure = node_pressure[v_idx]
    chord_attractiveness = detour_ratio * (0.8 + 0.4 * intermediate_pressure)
    high_detour_indices = np.argsort(chord_attractiveness)[-k:]

    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            chord_centrality = 0.5 * (node_centrality[u] + node_centrality[w])
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq)) * (0.8 + 0.4 * chord_centrality)
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
