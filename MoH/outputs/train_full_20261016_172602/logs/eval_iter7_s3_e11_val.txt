# ===== Heuristic Code =====
#{A dynamic Guided Local Search updating scheme that evaporates non-penalized edge weights using a decay factor while applying density-aware, usage-discounted penalties to long tour edges and enforcing a prohibitive tabu cost on the most frequently overused edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Compute node density using k-nearest neighbors
    k = min(6, n - 1)
    if k > 0:
        sorted_dists = np.sort(edge_distance, axis=1)
        k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
        node_density = 1.0 / (k_nn_avg + 1e-6)
        node_density = node_density / (np.mean(node_density) + 1e-6)
    else:
        node_density = np.ones(n)

    # 2. Extract current tour edges and properties
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    avg_tour_len = np.mean(tour_lengths) + 1e-6
    relative_lengths = tour_lengths / avg_tour_len

    # 3. Dynamic penalty evaporation: slightly decay edge distances to prevent monotonic runaway
    evaporation_factor = 0.995
    updated_edge_distance *= evaporation_factor

    # 4. Standard Guided Local Search utility calculation
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    usage = edge_n_used[u_idx, v_idx]
    usage_damping = 1.0 / (1.0 + 0.3 * usage)

    # Sigmoidal edge length sensitivity combined with density and usage damping
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_lengths - 1.0)))
    local_penalty = relative_lengths * usage_damping * density_factor

    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    penalty = (w_global * rank_penalty + w_local * local_penalty) * avg_tour_len * 0.15

    # Amplify penalties for the worst edges in the tour (top quartile)
    high_threshold = np.percentile(tour_lengths, 75)
    penalty[tour_lengths >= high_threshold] *= 1.5

    # 5. Adaptive tabu mechanism: prohibitively penalize overused tour edges to force structural diversification
    max_usage = np.max(usage)
    if max_usage >= 3:
        tabu_threshold = max(3.0, np.percentile(usage, 85))
        is_tabu = usage >= tabu_threshold
        # Assign a prohibitive additive penalty based on max tour length
        tabu_cost = 5.0 * np.max(tour_lengths)
        penalty[is_tabu] += tabu_cost

    # 6. Apply symmetric penalty updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    # Preserve zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
