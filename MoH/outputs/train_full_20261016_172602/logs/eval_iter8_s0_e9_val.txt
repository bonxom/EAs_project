#{A dual-objective augmented Lagrangian penalty strategy decomposes edge costs via dynamic matrix contraction decay combined with dual-scaled spatial bottlenecks and usage memory multipliers on top-utility tour segments.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy edge distance matrix to ensure input immutability
    updated_edge_distance = np.copy(edge_distance)

    # Step 2: Global contraction (Augmented Lagrangian decay) to dissipate stale penalties
    decay_lambda = 0.995
    updated_edge_distance = updated_edge_distance * decay_lambda

    # Step 3: Extract tour edge segments using vectorized shift indexing
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Step 4: Dual-objective utility computation: ratio of original metric cost to historical frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility <= 0:
        return updated_edge_distance

    # Step 5: Adaptive thresholding targeting top-utility bottleneck edges
    threshold = 0.90 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_lengths = tour_lengths[selected_mask]
    sel_counts = tour_counts[selected_mask]

    # Step 6: Augmented Lagrangian penalty decomposition: spatial scaling factor combined with memory regularization
    mean_tour_cost = np.mean(tour_lengths)
    spatial_factor = sel_lengths / (mean_tour_cost + 1e-12)
    memory_regularizer = 1.0 / np.sqrt(1.0 + sel_counts)
    augmented_penalty = 0.45 * mean_tour_cost * spatial_factor * memory_regularizer

    # Step 7: Symmetric update using vectorized advanced indexing
    np.add.at(updated_edge_distance, (sel_u, sel_v), augmented_penalty)
    np.add.at(updated_edge_distance, (sel_v, sel_u), augmented_penalty)

    # Ensure zero-diagonal invariant
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
