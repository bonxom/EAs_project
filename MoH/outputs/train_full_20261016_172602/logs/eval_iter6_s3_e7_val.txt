#{This algorithm scales Guided Local Search penalties by the mean tour edge length, uses partition-based k-NN density estimation for speed, and selectively targets top-quantile edge utility traps using an adaptive Lagrangian-inspired penalty structure.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Fast k-NN distance approximation via argpartition instead of full sort
    k = min(6, n - 1)
    knn_part = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(knn_part, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Scale invariance: compute scale factor from current tour
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Utility formulation: high cost per usage, weighted by local neighborhood density
    # Classical GLS utility: c_ij / (1 + p_ij), here enriched with spatial density
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    edge_utility = (relative_length / (1.0 + usage)) * density_factor

    # Rank penalty based on relative length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Selective penalization: focus primarily on high-utility bottlenecks
    top_threshold = np.percentile(edge_utility, 65)
    selectivity_mask = edge_utility >= top_threshold

    # Base penalty proportional to the actual distance scale of the tour
    lambda_scale = 0.15 * avg_tour_length
    penalty_magnitude = lambda_scale * (0.6 * rank_penalty + 0.4 * edge_utility)

    # Amplify the penalty on the worst bottleneck edges to force structural diversification
    penalty_magnitude[selectivity_mask] *= 1.8
    # Attenuate penalty on low-utility edges to preserve good tour sub-paths
    penalty_magnitude[~selectivity_mask] *= 0.2

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty_magnitude)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty_magnitude)

    return updated_edge_distance
