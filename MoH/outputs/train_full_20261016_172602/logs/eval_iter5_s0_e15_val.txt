# ===== Heuristic Code =====
#{Decouple utility evaluation from accumulated penalties by estimating original base distances via penalty subtraction against edge counts, computing true geometric utility ratios to apply scale-invariant symmetric perturbations.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_edge_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Establish dynamic penalty parameter from current tour scale
    avg_edge = np.mean(current_edge_lengths)
    lambda_factor = 0.3
    penalty_step = lambda_factor * avg_edge

    # Decouple: recover original geometric distances by subtracting accrued penalties
    # Base distance estimate: d_orig = max(d_current - count * penalty_step, epsilon)
    original_edge_lengths = np.maximum(current_edge_lengths - (tour_counts * penalty_step), 1e-6)

    # Dual-matrix decoupling: calculate utility strictly from unpenalized ground-truth geometry
    utilities = original_edge_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Select edges achieving near-maximal true utility to escape local traps
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply symmetric penalty updates to the routing distance matrix
    updated_edge_distance[sel_u, sel_v] += penalty_step
    updated_edge_distance[sel_v, sel_u] += penalty_step

    return updated_edge_distance
