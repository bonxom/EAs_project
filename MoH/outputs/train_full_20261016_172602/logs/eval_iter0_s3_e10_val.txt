# ===== Heuristic Code =====
#{Apply an exponential memory decay to historical edge penalties inferred from accumulated edge usage and base distances, followed by targeted utility-based GLS penalties for tour edges based on density-normalized relative length.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Decay historical penalties: penalize only edges with usage, decay excess distance towards base
    # In standard GLS, total distance = original_distance + lambda * penalties.
    # When persistent penalties decay, edge_distance is decayed slightly across all edges where edge_n_used > 0
    # to rehabilitate previously penalized good edges.
    decay = 0.985
    # Smooth decay applied to excess edge distance based on usage
    active_edges = edge_n_used > 0
    if np.any(active_edges):
        min_base = np.minimum(edge_distance[active_edges], updated_edge_distance[active_edges])
        updated_edge_distance[active_edges] = min_base + decay * (updated_edge_distance[active_edges] - min_base)

    # 2. Compute local density context using k-nearest neighbors
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # 3. Extract current tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # 4. Standard Guided Local Search utility calculation: c_ij / (1 + p_ij)
    # Scaled by local density and relative length rank
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Sigmoidal rank penalty blended with utility
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * (avg_tour_length / n)

    # 5. Amplify penalties on worst edges (upper quartile of tour)
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # 6. Apply penalty symmetrically to tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
