# ===== Heuristic Code =====
#{The algorithm identifies critical edges in the local optimal tour using utility-based penalization while applying a decay-and-reinforce mechanism driven by global usage frequencies to distinguish transient local traps from persistent structural bottlenecks.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy of the distance matrix to prevent in-place modification
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge indices representing the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Retrieve current edge distances and cumulative usage counts for tour edges
    tour_lengths = updated_edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Compute Guided Local Search utility: prioritize longer edges with fewer historical penalties
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Determine characteristic scale of the tour
    avg_edge = np.mean(tour_lengths)

    # Select candidate edges with utility near the maximum
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    # Dual-phase penalty scaling: reinforce edges that persistently recur, damp transient ones
    # Edges frequently used relative to the tour average usage receive stronger progressive penalization
    mean_count = np.mean(tour_counts) + 1e-6
    reinforcement_factor = 0.25 + 0.15 * np.tanh(tour_counts[selected_mask] / mean_count)
    penalties = reinforcement_factor * avg_edge

    # Apply symmetric updates to selected bottleneck edges
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
