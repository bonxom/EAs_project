# ===== Heuristic Code =====
# {Replaces triplet geometric metrics with a coordinate-free 4-node quadrangle tension and cross-detection metric based on Cayley-Menger / Ptolemy inequalities to penalize entwined tour edges and incentivize untangled diagonal shortcuts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Define 4-node consecutive segment indices: u -> v -> w -> x
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Edge usage decay to avoid permanent lockouts
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wx = edge_distance[w_idx, x_idx]
    d_ux = edge_distance[u_idx, x_idx]
    d_uw = edge_distance[u_idx, w_idx]
    d_vx = edge_distance[v_idx, x_idx]

    # 4-node segment tension and cross-detection metric using metric/Ptolemaic slack:
    # In Euclidean/metric geometry, crossed chords satisfy Ptolemy-like tension:
    # A planar twist/crossing or convoluted segment often has (d_uv + d_wx) > (d_uw + d_vx) or (d_uv + d_vw + d_wx) >> d_ux.
    current_path_cost = d_uv + d_vw + d_wx
    chord_diag_cost = d_uw + d_vx

    # Segment elongation ratio (stretch compared to direct span d_ux)
    elongation_ratio = current_path_cost / (d_ux + 1e-8)

    # Cross-twist indicator: how much current edges exceed diagonal shortcuts
    twist_slack = np.maximum(0.0, (d_uv + d_wx) - (d_uw + d_vx)) / (chord_diag_cost + 1e-8)

    # 4-node quadruplet geometric tension score
    quad_tension = elongation_ratio * (1.0 + 1.5 * twist_slack)

    # Symmetrize/smooth tension over incident edges (u, v)
    local_segment_tension = 0.34 * quad_tension + 0.33 * np.roll(quad_tension, 1) + 0.33 * np.roll(quad_tension, 2)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility ranking modulated by segment tension
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.40 * local_segment_tension)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        tension_weight = 1.0 + 0.5 * local_segment_tension[idx]
        penalty = adaptive_lambda * freq_penalty * tension_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Recreate shortcut incentive on the best uncrossed diagonal chords (u, w) or (u, x)
    high_tension_indices = np.argsort(quad_tension)[-k:]
    for idx in high_tension_indices:
        u, x = u_idx[idx], x_idx[idx]
        freq_ux = edge_n_used[u, x]
        if freq_ux < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq_ux))
            updated_edge_distance[u, x] = max(1e-6, updated_edge_distance[u, x] - discount)
            updated_edge_distance[x, u] = updated_edge_distance[u, x]

    return updated_edge_distance
