# ===== Heuristic Code =====
#{Dynamically modulates the Guided Local Search penalty magnitude using tour edge length variance and edge usage saturation to adaptively balance intensification on homogeneous tours with aggressive diversification out of deep local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Adaptive penalty scaling based on edge dispersion and search depth/stagnation
    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    cv = std_len / (mean_len + 1e-12)

    # Measure tour entrapment: proportion and magnitude of repeatedly penalized edges
    mean_count = np.mean(tour_counts)
    stagnation_factor = 1.0 + np.log1p(mean_count)

    # Modulate coefficient: higher when variance is low (subtle differences) or stagnation is high
    adaptive_coeff = (0.15 + 0.35 / (1.0 + cv)) * stagnation_factor
    penalty = adaptive_coeff * mean_len

    # Select candidate edges with utility near the maximum
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
