#{A dynamic edge penalization scheme that smoothly scales guided local search penalties using variance-adaptive continuous power-law amplification and inverse edge-usage damping without hard threshold discontinuities.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Efficient k-nearest neighbor density calculation
    k = min(6, n)
    sorted_dists = np.partition(edge_distance, k - 1, axis=1)[:, 1:k]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density /= np.mean(node_density)

    # Extract tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths)
    norm_std = std_tour_length / (avg_tour_length + 1e-6)

    # Relative normalized edge length
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Continuous power-law multiplier adapting to tour dispersion
    gamma = 1.0 + 0.5 * np.tanh(norm_std)
    power_scaling = np.power(np.maximum(relative_length, 0.0), gamma)

    # Soft rank penalty and frequency damping
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    local_penalty = relative_length * usage_factor * density_factor

    # Dynamic global vs local balancing
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * power_scaling

    # Symmetric penalty update along the tour
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
