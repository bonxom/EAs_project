# ===== Heuristic Code =====
#{This strategy penalizes high-utility tour edges using normalized utility-frequency weighting while creating candidate-directed gradient funnels by applying distance-scaled discounts exclusively to the k-nearest unvisited neighbors of penalized vertices.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k = min(5, max(1, n - 2))
    base_discount = 0.05

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        candidate_indices = np.where(unused_mask)[0]

        if candidate_indices.size > 0:
            candidate_costs = edge_distance[u, candidate_indices]

            if candidate_indices.size > k:
                k_best_pos = np.argpartition(candidate_costs, k)[:k]
                selected_indices = candidate_indices[k_best_pos]
                selected_costs = candidate_costs[k_best_pos]
            else:
                selected_indices = candidate_indices
                selected_costs = candidate_costs

            c_min = np.min(selected_costs)
            c_max = np.max(selected_costs)

            if c_max > c_min:
                norm_factors = 1.0 - (selected_costs - c_min) / (c_max - c_min)
            else:
                norm_factors = np.ones_like(selected_costs)

            discounts = base_discount * (0.5 + 0.5 * norm_factors)
            updated_edge_distance[u, selected_indices] -= updated_edge_distance[u, selected_indices] * discounts
            updated_edge_distance[selected_indices, u] = updated_edge_distance[u, selected_indices]

    return updated_edge_distance
