# ===== Heuristic Code =====
#{Dynamically scales the penalty by modulating a base coefficient with the tour's coefficient of variation and edge reuse frequency, applying aggressive perturbations during stagnant cycles and softer adjustments in novel regions.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)

    # Calculate dispersion coefficient to reflect the heterogeneity of edge costs
    cv = std_edge / (avg_edge + 1e-12)
    dispersion_factor = np.clip(1.0 + cv, 0.5, 2.5)

    # Measure search stagnation using the mean and maximum usage of the current tour edges
    mean_count = np.mean(tour_counts)
    max_count = np.max(tour_counts)
    stagnation_factor = np.log1p(mean_count) + 0.5 * np.log1p(max_count)

    # Adapt penalty: low penalty when exploring novel regions, high when cycling frequently
    base_factor = 0.15
    dynamic_weight = base_factor * dispersion_factor * (1.0 + stagnation_factor)
    penalty = dynamic_weight * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
