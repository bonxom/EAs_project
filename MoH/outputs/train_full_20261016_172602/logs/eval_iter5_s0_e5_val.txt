# ===== Heuristic Code =====
#{Dynamically scale edge penalties via tour length dispersion and distribute soft rank-proportional penalties across edges in the top utility quantile.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    k = max(1, int(np.ceil(0.20 * n_edges)))

    top_indices = np.argsort(utilities)[-k:]
    ranks = np.arange(1, k + 1, dtype=np.float64)
    rank_weights = ranks / np.sum(ranks)

    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    cv = std_len / (mean_len + 1e-12)
    dynamic_scale = mean_len * (0.15 + 0.35 * np.clip(cv, 0.0, 1.0))

    penalties = dynamic_scale * rank_weights * k
    sel_u = u_idx[top_indices]
    sel_v = v_idx[top_indices]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
