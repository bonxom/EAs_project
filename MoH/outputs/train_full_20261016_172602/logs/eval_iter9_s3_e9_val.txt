# ===== Heuristic Code =====
#{A dynamic topological alpha-nearness penalty mechanism that measures structural criticality using 1-tree edge slack relative to the minimum spanning tree neighborhood, penalizing topologically non-critical and overused edges on the tour.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # 1-tree / dynamic topological context approximation (alpha-nearness proxy)
    # For each node, find the top 2 shortest incident edges to approximate local 1-tree bounds
    # Mask out the self-loops with infinity
    dist_no_self = np.copy(edge_distance)
    np.fill_diagonal(dist_no_self, np.inf)

    sorted_dists = np.partition(dist_no_self, kth=2, axis=1)
    min_edge_1 = sorted_dists[:, 0]
    min_edge_2 = sorted_dists[:, 1]

    # Alpha-nearness slack: difference between actual edge cost and the sum of baseline local incident costs
    # alpha_uv reflects how far edge (u, v) is from being an essential minimum 1-tree structural edge
    alpha_slack = tour_edge_lengths - 0.5 * (min_edge_1[u_idx] + min_edge_1[v_idx])
    alpha_slack = np.maximum(alpha_slack, 0.0)

    # Directional dispersion / local baseline scale
    local_baseline = 0.5 * (min_edge_2[u_idx] + min_edge_2[v_idx]) + 1e-6
    relative_slack = alpha_slack / local_baseline

    # Sigmoidal penalty scaling based on topological non-criticality (higher slack -> higher penalty)
    criticality_penalty = 1.0 / (1.0 + np.exp(-3.0 * (relative_slack - np.mean(relative_slack))))

    # Memory penalty component incorporating usage frequency to avoid cycling
    usage_factor = 1.0 / (1.0 + 0.35 * edge_n_used[u_idx, v_idx])

    # Global tour scale normalization
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    normalized_length = tour_edge_lengths / avg_tour_length

    # Adaptive weights balancing topological slack and frequency-guided search diversification
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_topo = 0.65 / (1.0 + 0.08 * freq_mean)
    w_freq = 1.0 - w_topo

    combined_penalty = (w_topo * criticality_penalty + w_freq * usage_factor) * normalized_length

    # Accentuate penalties on excessively sub-optimal edges exceeding the 75th percentile of slack
    slack_threshold = np.percentile(alpha_slack, 75)
    combined_penalty[alpha_slack >= slack_threshold] *= 1.4

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
