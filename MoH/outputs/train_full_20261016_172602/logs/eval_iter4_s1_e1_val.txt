# ===== Heuristic Code =====
#{This algorithm penalizes the top 50% utility-scored tour edges using z-score weighting and frequency damping, then selectively discounts distances only towards the k-nearest low-frequency candidate neighbors of penalized endpoints to incentivize high-quality alternative connections without metric distortion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # Confine reward discounting strictly to top k-nearest neighbors with low frequency
    k = min(8, n - 1)
    decay_ratio = 0.02

    for u in penalized_nodes:
        # Retrieve original metric distances from node u
        orig_dists = edge_distance[u]
        # Get nearest candidates excluding self
        candidate_indices = np.argpartition(orig_dists, k + 1)[:k + 2]
        candidate_indices = candidate_indices[candidate_indices != u][:k]

        # Filter strictly by low usage frequency (edge_n_used <= 1)
        valid_mask = edge_n_used[u, candidate_indices] <= 1
        valid_candidates = candidate_indices[valid_mask]

        if valid_candidates.size > 0:
            discount = updated_edge_distance[u, valid_candidates] * decay_ratio
            updated_edge_distance[u, valid_candidates] -= discount
            updated_edge_distance[valid_candidates, u] -= discount

    return updated_edge_distance
