# ===== Heuristic Code =====
#{Vectorized Guided Local Search penalty updating edges above the average tour cost with continuous power-scaled utility weights to smoothly diversify away from local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    avg_tour_cost = np.mean(tour_lengths)
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility > 0:
        norm_utility = utilities / max_utility
        # Smooth scaling active only on above-average edges, combining continuous utility weighting
        excess_ratio = np.maximum(0.0, (tour_lengths - avg_tour_cost) / (avg_tour_cost + 1e-12))
        weights = (norm_utility ** 2) * np.tanh(excess_ratio)
        penalties = 0.3 * avg_tour_cost * weights

        # In case all edges are below average (uniform tour), fallback to the highest utility edge
        if np.max(penalties) == 0.0:
            top_mask = utilities >= (0.99 * max_utility)
            penalties = np.where(top_mask, 0.3 * avg_tour_cost, 0.0)

        updated_edge_distance[u_idx, v_idx] += penalties
        updated_edge_distance[v_idx, u_idx] += penalties

    return updated_edge_distance
