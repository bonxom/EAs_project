# ===== Heuristic Code =====
#{A dual-operator guided local search perturbator that applies frequency-dampened utilities to penalize high-cost tour edges while selectively discounting the top-k nearest low-frequency candidate edges via proximity-weighted incentives to direct search toward promising shortcuts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # Target discounts strictly on the top-k cheapest low-frequency incident edges
    k = min(5, n - 1)
    base_discount = 0.04

    for u in penalized_nodes:
        row_costs = edge_distance[u].astype(np.float64)
        row_used = edge_n_used[u].astype(np.float64)

        # Exclude self-loop and heavily used edges
        median_row_used = np.median(row_used)
        candidate_mask = (row_used <= median_row_used)
        candidate_mask[u] = False

        cand_indices = np.where(candidate_mask)[0]
        if len(cand_indices) == 0:
            continue

        cand_costs = row_costs[cand_indices]
        if len(cand_indices) > k:
            top_k_rel_idx = np.argpartition(cand_costs, k)[:k]
            selected_neighbors = cand_indices[top_k_rel_idx]
        else:
            selected_neighbors = cand_indices

        selected_costs = row_costs[selected_neighbors]
        min_c = np.min(selected_costs)
        max_c = np.max(selected_costs)
        range_c = max_c - min_c + 1e-8

        # Proximity weight: closer edges receive higher incentive relative to distant ones
        proximity_weights = 1.0 - (selected_costs - min_c) / range_c
        freq_damping = 1.0 / (1.0 + row_used[selected_neighbors])
        discounts = updated_edge_distance[u, selected_neighbors] * base_discount * proximity_weights * freq_damping

        updated_edge_distance[u, selected_neighbors] -= discounts
        updated_edge_distance[selected_neighbors, u] = updated_edge_distance[u, selected_neighbors]

    return updated_edge_distance
