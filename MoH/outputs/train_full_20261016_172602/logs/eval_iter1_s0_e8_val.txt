#{Scale the Guided Local Search edge penalties continuously based on each selected edge's relative normalized utility relative to the maximum utility to differentiate penalty magnitudes.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy of the distance matrix to preserve original values
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge indices representing the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Vectorized computation of edge utilities based on edge length and usage count
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    if max_utility <= 0:
        return updated_edge_distance

    # Filter edges close to the maximum utility
    threshold = 0.85 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_utilities = utilities[selected_mask]

    # Calculate baseline penalty scaled to the average edge length of the tour
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Continuous penalty scaling proportional to normalized individual edge utility
    continuous_penalties = base_penalty * (sel_utilities / max_utility)

    # Symmetrically apply continuous penalties using NumPy advanced indexing
    updated_edge_distance[sel_u, sel_v] += continuous_penalties
    updated_edge_distance[sel_v, sel_u] += continuous_penalties

    return updated_edge_distance
