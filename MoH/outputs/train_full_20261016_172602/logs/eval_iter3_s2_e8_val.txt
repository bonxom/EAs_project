# ===== Heuristic Code =====
#{Synthesize vectorized triangle-slack geometric inefficiency with density-based inter-cluster bridge identification to dynamically scale edge penalties and encourage structural macroscopic tour diversification.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=int)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric triplet metrics
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Topological density / cluster awareness:
    # Compute k-nearest neighbor average distance for each node as local density proxy
    k_nn = min(5, num_nodes - 1)
    sorted_dists = np.partition(edge_distance, k_nn, axis=1)[:, 1:k_nn+1]
    local_density = np.mean(sorted_dists, axis=1)  # Higher means sparse/bridge node, lower means dense cluster

    # Bridge score: edges connecting disparate density zones or spanning far relative to local neighborhood
    bridge_factor = d_uv / (np.sqrt(local_density[u_idx] * local_density[v_idx]) + 1e-8)
    bridge_weight = np.clip(bridge_factor, 0.5, 3.0)

    base_freq = edge_n_used[u_idx, v_idx]

    # Synthesize utility: scale by bridge score to prioritize breaking inefficient inter-cluster connections
    utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.30 * local_geom_factor) * (1.0 + 0.25 * bridge_weight)

    # Adaptive perturbation size based on current tour edge usage saturation
    mean_used = np.mean(base_freq)
    adaptive_ratio = np.clip(0.12 + 0.05 * np.tanh(mean_used / 3.0), 0.10, 0.22)
    k = max(2, int(np.ceil(adaptive_ratio * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.28 + 0.10 * np.tanh(mean_used / 4.0)) * global_mean_dist

    # Apply penalties
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_w = 1.0 + 0.4 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_w
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Guided shortcut discounting on high detour segments with low prior usage
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.12 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
