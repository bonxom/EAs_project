# ===== Heuristic Code =====
#{This metaheuristic identifies high-utility consecutive edge segments in the local optimum using vectorized tour metrics and penalizes their component edges with a magnitude scaled by the current tour's average edge length.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    edge_utilities = tour_lengths / (1.0 + tour_counts)
    segment_utilities = edge_utilities + np.roll(edge_utilities, -1)

    max_utility = np.max(segment_utilities)
    if max_utility <= 0:
        return updated_edge_distance

    avg_tour_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_tour_edge
    threshold = 0.99 * max_utility

    selected_segments = np.where(segment_utilities >= threshold)[0]

    penalize_mask = np.zeros(n, dtype=bool)
    penalize_mask[selected_segments] = True
    penalize_mask[(selected_segments + 1) % n] = True

    sel_u = u_idx[penalize_mask]
    sel_v = v_idx[penalize_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
