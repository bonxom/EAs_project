# ===== Heuristic Code =====
#{A vectorized guided local search strategy that applies sigmoid z-score and square-root frequency penalized perturbations to long, overused tour edges while reciprocally discounting candidate shortcut edges to top-k nearest neighbors via vector rank-scaled rewards.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    if penalized_nodes.size == 0:
        return updated_edge_distance

    k = min(8, n - 2)
    if k <= 0:
        return updated_edge_distance

    sub_dists = edge_distance[penalized_nodes].copy()
    sub_dists[np.arange(len(penalized_nodes)), penalized_nodes] = np.inf

    k_candidates = np.argpartition(sub_dists, kth=k, axis=1)[:, :k]
    row_indices = np.arange(len(penalized_nodes))[:, None]
    cand_dists = sub_dists[row_indices, k_candidates]
    sorted_order = np.argsort(cand_dists, axis=1)
    k_sorted_neighbors = np.take_along_axis(k_candidates, sorted_order, axis=1)

    u_expanded = np.repeat(penalized_nodes[:, None], k, axis=1)
    cand_used = edge_n_used[u_expanded, k_sorted_neighbors]
    valid_mask = cand_used <= 1

    rank_factors = 1.0 / (1.0 + np.arange(k, dtype=np.float64)[None, :] * 0.3)
    base_decay = 0.035

    valid_u = u_expanded[valid_mask]
    valid_v = k_sorted_neighbors[valid_mask]
    valid_decay = (updated_edge_distance[valid_u, valid_v] * 
                   base_decay * 
                   np.repeat(rank_factors, len(penalized_nodes), axis=0)[valid_mask])

    updated_edge_distance[valid_u, valid_v] -= valid_decay
    updated_edge_distance[valid_v, valid_u] = updated_edge_distance[valid_u, valid_v]

    return updated_edge_distance
