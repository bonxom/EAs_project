#{Replace the hard 75th percentile cliff with a smooth, continuous Gompertz utility function based on excess edge length over local topological neighborhood bounds, computed efficiently via np.partition, scaled dynamically to the distance metric.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Efficient k-nearest neighbor distance approximation via partition instead of full sort
    k = min(5, n - 1)
    k_part = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(k_part, axis=1)

    # Endpoints
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    eps = 1e-6
    relative_length = tour_edge_lengths / (avg_tour_length + eps)

    # Topological baseline: average k-NN scale of edge endpoints
    topo_baseline = 0.5 * (k_nn_avg[u_idx] + k_nn_avg[v_idx]) + eps
    excess_ratio = tour_edge_lengths / topo_baseline

    # Smooth Gompertz utility function for continuous edge-length penalization
    # S-shaped transition avoiding discontinuous cliffs, saturating smoothly for long edges
    gompertz_penalty = np.exp(-np.exp(-2.0 * (excess_ratio - 1.2)))

    # Inverse frequency damping to avoid cyclic lock
    usage_factor = 1.0 / (1.0 + 0.35 * edge_n_used[u_idx, v_idx])

    # Density weighting: higher density regions allow easier re-routing
    node_density = 1.0 / topo_baseline
    node_density = node_density / (np.mean(node_density) + eps)

    # Dynamic step size calibrated to current tour edge metric scale
    scale_factor = 0.15 * avg_tour_length

    # Unified continuous penalty
    penalty = scale_factor * relative_length * gompertz_penalty * usage_factor * node_density

    # Symmetrically apply penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
