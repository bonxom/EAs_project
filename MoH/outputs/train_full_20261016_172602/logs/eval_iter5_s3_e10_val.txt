# ===== Heuristic Code =====
#{Replace O(n^2 log n) full matrix sorting with a fast O(n) localized incident minimum surrogate to compute edge sparsity ratios and dynamically scale Guided Local Search penalties based on tour length deviations and visitation frequencies.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    mean_tour_len = np.mean(tour_lengths) + 1e-8

    # Fast localized sparsity baseline: compute minimum non-zero incident edge weight per node without full sorting
    # Masking self-loops (diagonal) with infinity
    masked_dists = np.where(np.eye(n, dtype=bool), np.inf, edge_distance)
    min_incident = np.min(masked_dists, axis=1)

    # Minimum spanning tree / Delaunay sparsity surrogate:
    # Ratio of current edge cost to the baseline incident minimum of its endpoints
    baseline_incident = 0.5 * (min_incident[u_idx] + min_incident[v_idx]) + 1e-8
    incident_sparsity_ratio = tour_lengths / baseline_incident

    # Normalized relative length against average tour edge length
    rel_cost = tour_lengths / mean_tour_len

    # Guided Local Search utility: scale penalty directly with cost and sparsity, dampened by edge usage frequency
    freq = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.5 * freq)

    # Composite penalty balancing relative tour excess and geometric deviation from minimum incident edges
    cost_penalty = 1.0 / (1.0 + np.exp(-3.0 * (rel_cost - 1.0)))
    sparsity_penalty = np.tanh(incident_sparsity_ratio - 1.0)
    sparsity_penalty = np.maximum(0.0, sparsity_penalty)

    penalty = (0.5 * cost_penalty + 0.5 * sparsity_penalty) * usage_factor * mean_tour_len * 0.15

    # Boost penalties on the longest tour segments to enforce structural ruin
    threshold = np.percentile(tour_lengths, 75)
    penalty[tour_lengths >= threshold] *= 1.4

    # Apply symmetric updates to the distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
