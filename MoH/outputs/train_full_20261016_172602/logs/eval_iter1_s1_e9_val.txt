# ===== Heuristic Code =====
#{Adaptive Pareto/quantile filtering with spatial clustering replaces the flat median threshold by dynamically setting the penalization quantile using stagnation metrics and clustering contiguous high-utility subpaths along the tour to induce structured topological escapes while discounting promising unused incident edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Dynamic quantile threshold based on usage saturation/stagnation
    mean_used = np.mean(used)
    max_used = np.max(used)
    stagnation_ratio = mean_used / (max_used + 1.0)
    # Stagnant search increases ruin intensity (lowers quantile threshold to ruin more edges)
    target_quantile = np.clip(0.80 - 0.40 * stagnation_ratio, 0.40, 0.85)
    cutoff = np.quantile(utilities, target_quantile)

    # Identify candidate edges
    candidate_mask = utilities >= cutoff
    if not np.any(candidate_mask):
        candidate_mask = utilities >= np.median(utilities)

    # Spatial clustering: expand isolated candidates into contiguous subpaths along the cyclic tour
    # If edge i is selected, adjacent edges on the tour with above-average utility get clustered
    mean_util = np.mean(utilities)
    spatial_mask = np.copy(candidate_mask)
    left_neighbors = np.roll(candidate_mask, 1)
    right_neighbors = np.roll(candidate_mask, -1)

    # Expand to create contiguous ruined blocks (subpaths) where utility is not strictly low
    spatial_mask = spatial_mask | ((left_neighbors | right_neighbors) & (utilities > mean_util))

    penalize_indices = np.where(spatial_mask)[0]
    p_u = u_idx[penalize_indices]
    p_v = v_idx[penalize_indices]
    p_costs = costs[penalize_indices]
    p_used = used[penalize_indices]
    p_utils = utilities[penalize_indices]

    max_util = np.max(utilities) + 1e-8
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Apply symmetric edge penalization
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Discount incident unexplored/promising edges from perturbed subpath vertices
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.025
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
