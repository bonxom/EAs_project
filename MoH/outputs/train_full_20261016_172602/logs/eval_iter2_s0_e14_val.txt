# ===== Heuristic Code =====
#{A robust percentile and z-score based edge selection scheme that penalizes edges exceeding an adaptive statistical threshold and ranks a guaranteed subset of high-utility, high-length edges to escape local optima efficiently.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to prevent in-place mutation of the input
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge indices across the tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge penalties using utility formulation
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    # Target top quartile or at least a deterministic minimal fraction to ensure diversity
    k = max(1, int(np.ceil(0.15 * n_edges)))

    # Distribution-based edge selection using utility z-scores and high-percentile thresholding
    mean_util = np.mean(utilities)
    std_util = np.std(utilities)
    z_scores = (utilities - mean_util) / (std_util + 1e-12)

    # Pareto/Percentile selection: top-k utilities combined with statistical outliers (z-score > 1.0)
    top_k_threshold = np.partition(utilities, n_edges - k)[n_edges - k]
    selected_mask = (utilities >= top_k_threshold) | (z_scores > 1.0)

    # Scale penalty proportionally to both average tour edge length and individual edge utility weight
    avg_edge = np.mean(tour_lengths)
    norm_utilities = utilities / (np.max(utilities) + 1e-12)
    penalties = 0.3 * avg_edge * (1.0 + norm_utilities[selected_mask])

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Vectorized symmetric penalty application
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
