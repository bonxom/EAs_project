# ===== Heuristic Code =====
#{Augment Guided Local Search edge penalties by replacing static 2-hop triplets with multi-hop chordal evaluations across 2-hop and 3-hop segments normalized by local k-nearest neighbor spatial density to strongly penalize proximity-violating edges while proactively discounting unvisited shortcut chords.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # 1. Edge utilization frequency decay
    edge_n_used *= 0.96

    # 2. Extract multi-hop sequence indices (u -> v -> w -> z)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    z_idx = np.roll(tour, -3)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wz = edge_distance[w_idx, z_idx]

    # Chordal shortcut distances
    d_uw = edge_distance[u_idx, w_idx]
    d_uz = edge_distance[u_idx, z_idx]

    # 3. Local spatial density estimation around each vertex (mean distance to 4 nearest neighbors)
    # Using partition for efficient local proximity metric
    k_nn = min(5, num_nodes - 1)
    # Exclude self-distance (which is 0) by taking k_nn + 1 smallest elements
    part_dists = np.partition(edge_distance, k_nn, axis=1)[:, 1:k_nn + 1]
    local_density = np.mean(part_dists, axis=1)  # shape (num_nodes,)
    eps = 1e-8

    # Local reference densities for the path segments
    density_v = local_density[v_idx] + eps
    density_w = local_density[w_idx] + eps

    # Multi-hop chordal slack normalized against local density
    # 2-hop detour slack: path (u-v-w) vs chord (u-w)
    path_2hop = d_uv + d_vw
    slack_2hop = np.maximum(0.0, path_2hop - d_uw) / density_v

    # 3-hop detour slack: path (u-v-w-z) vs chord (u-z)
    path_3hop = d_uv + d_vw + d_wz
    slack_3hop = np.maximum(0.0, path_3hop - d_uz) / (0.5 * (density_v + density_w))

    # Integrated multi-hop inefficiency measure
    chordal_inefficiency = 0.65 * slack_2hop + 0.35 * slack_3hop

    # Distribute chordal inefficiency across incident edges
    # Edge (u, v) is penalized if it forms part of inefficient 2-hop or 3-hop configurations
    geom_edge_factor = (
        chordal_inefficiency +
        np.roll(chordal_inefficiency, 1) +
        0.5 * np.roll(chordal_inefficiency, 2)
    ) / 2.5

    # 4. Compute GLS utilities incorporating chordal geometric violations
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.40 * geom_edge_factor)

    # 5. Penalize worst edges
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * geom_edge_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 6. Proactive bonus discounting on promising multi-hop chords
    # Prioritize 2-hop chords where detour slack is highest and usage is low
    chord_rank = np.argsort(slack_2hop)[-k:]
    for idx in chord_rank:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    # Proactive bonus for high-benefit 3-hop chords
    chord3_rank = np.argsort(slack_3hop)[-max(1, k // 2):]
    for idx in chord3_rank:
        u, z = u_idx[idx], z_idx[idx]
        freq = edge_n_used[u, z]
        if freq < 0.5:
            discount = 0.06 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, z] = max(1e-6, updated_edge_distance[u, z] - discount)
            updated_edge_distance[z, u] = updated_edge_distance[u, z]

    return updated_edge_distance
