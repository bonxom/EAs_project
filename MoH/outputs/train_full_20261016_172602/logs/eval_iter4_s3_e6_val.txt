#{Refactor edge penalization using continuous Voudouris-style GLS utility with structural bottleneck detection, replacing step thresholds and sigmoids with a scale-invariant power-law amplification modulated by historical edge frequency and neighborhood density.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Extract edges from the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_usage = edge_n_used[u_idx, v_idx]

    # Estimate local neighborhood scale using k-NN distances to detect bottleneck edges
    k = min(6, n)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k], axis=1)
    local_scale = 0.5 * (k_nn_avg[u_idx] + k_nn_avg[v_idx]) + 1e-8

    # Bottleneck ratio measures how stretched an edge is relative to local endpoints' neighborhoods
    bottleneck_ratio = tour_lengths / local_scale

    # Adaptive GLS-style utility with dynamic usage parameter
    # lambda balances exploration vs intensification dynamically based on tour length scale
    avg_tour_edge = np.mean(tour_lengths) + 1e-8
    lambda_param = 0.3 * (1.0 + np.mean(tour_usage) / (np.max(tour_usage) + 1.0))

    # Augmented utility combining cost, structural bottleneck factor, and usage frequency
    structural_cost = tour_lengths * np.sqrt(bottleneck_ratio)
    raw_utility = structural_cost / (1.0 + lambda_param * tour_usage)

    # Smooth power-law scaling relative to max tour utility (Pareto-style soft thresholding)
    max_utility = np.max(raw_utility)
    if max_utility > 1e-8:
        normalized_utility = raw_utility / max_utility
    else:
        normalized_utility = np.zeros_like(raw_utility)

    gamma = 2.5
    amplified_utility = np.power(normalized_utility, gamma)

    # Penalty scaled to tour magnitude to preserve scale invariance
    penalty = amplified_utility * (0.25 * avg_tour_edge)

    # Symmetric in-place addition
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
