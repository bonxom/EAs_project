# ===== Heuristic Code =====
#{A bi-level guided perturbation heuristic that penalizes tour edges using normalized cost-to-usage utilities while directly donating a conserved fraction of the accumulated penalty to mutual nearest-neighbor shortcut edges incident to the penalized endpoints.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=int)
    u_idx = tour
    v_idx = np.roll(tour, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Strict normalized cost-to-usage utility evaluation
    mean_c = np.mean(costs)
    std_c = np.std(costs) + 1e-8
    norm_cost = (costs - mean_c) / std_c
    cost_factor = 1.0 / (1.0 + np.exp(-norm_cost))
    utilities = (costs * (1.0 + 2.0 * cost_factor)) / (1.0 + used)

    # Top-utility edge selection for guided penalty
    threshold = np.percentile(utilities, 65.0)
    penalized_mask = utilities >= threshold
    if not np.any(penalized_mask):
        penalized_mask = utilities >= np.median(utilities)

    max_util = np.max(utilities) + 1e-8
    penalized_edges = []
    node_donations = {}

    for i in range(n):
        if penalized_mask[i]:
            u, v = int(u_idx[i]), int(v_idx[i])
            c_val = costs[i]
            u_cnt = used[i]

            # Penalize the selected inefficient tour edge
            penalty = (c_val / (1.0 + np.sqrt(u_cnt + 1.0))) * (0.8 + 0.4 * (utilities[i] / max_util))
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty
            penalized_edges.append((u, v, penalty))

            # Donate a portion of the penalty to incident nodes for shortcut encouragement
            donation = 0.25 * penalty
            node_donations[u] = node_donations.get(u, 0.0) + donation
            node_donations[v] = node_donations.get(v, 0.0) + donation

    tour_edge_set = set()
    for i in range(n):
        a, b = int(u_idx[i]), int(v_idx[i])
        tour_edge_set.add((min(a, b), max(a, b)))

    k_neighbors = min(10, max(3, n // 10))
    candidate_neighbors = {}
    penalized_nodes = list(node_donations.keys())

    for u in penalized_nodes:
        dists = edge_distance[u].copy()
        dists[u] = np.inf
        sorted_indices = np.argsort(dists)
        valid = []
        for v in sorted_indices:
            v = int(v)
            edge_key = (min(u, v), max(u, v))
            if edge_key not in tour_edge_set and edge_n_used[u, v] <= 2:
                valid.append(v)
                if len(valid) >= k_neighbors:
                    break
        candidate_neighbors[u] = valid

    # Bi-level reduction: incentivize mutual-nearest-neighbor shortcuts using donated penalty pools
    processed_shortcuts = set()
    for u in penalized_nodes:
        cand_u = candidate_neighbors[u]
        for rank_u, v in enumerate(cand_u):
            pair = (min(u, v), max(u, v))
            if pair in processed_shortcuts:
                continue
            if v in candidate_neighbors and u in candidate_neighbors[v]:
                rank_v = candidate_neighbors[v].index(u)
                rank_discount = 1.0 / (1.0 + 0.25 * (rank_u + rank_v))
                available_donation = 0.5 * (node_donations[u] + node_donations[v])
                reduction = min(
                    updated_edge_distance[u, v] * 0.20,
                    available_donation * rank_discount / (1.0 + edge_n_used[u, v])
                )
                updated_edge_distance[u, v] = max(1e-4, updated_edge_distance[u, v] - reduction)
                updated_edge_distance[v, u] = updated_edge_distance[u, v]
                processed_shortcuts.add(pair)

    return updated_edge_distance
