# ===== Heuristic Code =====
#{This algorithm penalizes long, acute, detour-heavy tour edges via memory-decayed GLS utilities and simultaneously applies correlated 2-opt discount pairs to both candidate shortcut and reconnecting edges weighted by k-nearest-neighbor slack to steer search into structurally valid improvements.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Identify cyclic indices
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Evaporate edge usage memory
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric metrics
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Penalize inefficient edges with GLS
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Correlated ruin-and-relink 2-opt shortcut discounting
    # Evaluate 2-opt swap candidates between edge (i, i+1) and edge (j, j+1)
    # Target promising shortcuts from high-utility penalized edges
    discount_budget = max(1, int(np.ceil(0.08 * num_nodes)))

    # Precompute k-NN threshold proxy (mean distance of 5th nearest neighbor across sample)
    knn_k = min(5, max(1, num_nodes - 1))
    knn_slack_threshold = np.partition(edge_distance, knn_k, axis=1)[:, knn_k]

    # Rank moves using potential delta savings: (d(i, i+1) + d(j, j+1)) - (d(i, j) + d(i+1, j+1))
    best_pairs = []
    for idx_i in top_k_indices:
        i_node = u_idx[idx_i]
        i_next = v_idx[idx_i]
        d_orig1 = edge_distance[i_node, i_next]

        # Scan through candidate disjoint edges (j, j+1)
        for offset in range(2, min(num_nodes - 1, 15)):
            idx_j = (idx_i + offset) % num_nodes
            j_node = u_idx[idx_j]
            j_next = v_idx[idx_j]

            d_orig2 = edge_distance[j_node, j_next]
            d_new1 = edge_distance[i_node, j_node]
            d_new2 = edge_distance[i_next, j_next]

            # Structural gain
            gain = (d_orig1 + d_orig2) - (d_new1 + d_new2)

            # KNN alignment slack
            slack1 = max(0.0, knn_slack_threshold[i_node] - d_new1)
            slack2 = max(0.0, knn_slack_threshold[i_next] - d_new2)

            pair_score = gain + 0.5 * (slack1 + slack2)
            best_pairs.append((pair_score, i_node, j_node, i_next, j_next))

    best_pairs.sort(key=lambda x: x[0], reverse=True)

    applied = 0
    for score, u1, v1, u2, v2 in best_pairs:
        if applied >= discount_budget:
            break
        freq1 = edge_n_used[u1, v1]
        freq2 = edge_n_used[u2, v2]

        if freq1 < 2.0 and freq2 < 2.0:
            discount_base = 0.08 * global_mean_dist
            disc1 = discount_base / (1.0 + freq1)
            disc2 = discount_base / (1.0 + freq2)

            updated_edge_distance[u1, v1] = max(1e-6, updated_edge_distance[u1, v1] - disc1)
            updated_edge_distance[v1, u1] = updated_edge_distance[u1, v1]
            updated_edge_distance[u2, v2] = max(1e-6, updated_edge_distance[u2, v2] - disc2)
            updated_edge_distance[v2, u2] = updated_edge_distance[u2, v2]

            applied += 1

    return updated_edge_distance
