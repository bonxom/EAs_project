# ===== Heuristic Code =====
# {This algorithm replaces the static 75th percentile cutoff on raw lengths with a dynamic dispersion-adaptive threshold scaled by the coefficient of variation and applied to a composite topological tension metric to adaptively boost penalties on bottleneck edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Compute local k-NN density
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn = min(6, n)
    k_nn_avg = np.mean(sorted_dists[:, 1:k_nn], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Invariant core penalty terms
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Recreated Ruin Target: Dynamic dispersion-adaptive threshold on composite topological tension
    k_nn_pair_scale = 0.5 * (k_nn_avg[u_idx] + k_nn_avg[v_idx]) + 1e-6
    topological_tension = (tour_edge_lengths / k_nn_pair_scale) * np.sqrt(density_factor)

    mean_tension = np.mean(topological_tension)
    std_tension = np.std(topological_tension)
    cv = std_tension / (mean_tension + 1e-6)

    # Dynamically shift percentile cutoff between 60th and 90th percentile based on dispersion
    adaptive_percentile = np.clip(85.0 - 25.0 * cv, 60.0, 90.0)
    tension_threshold = np.percentile(topological_tension, adaptive_percentile)

    # Apply progressive adaptive boost on edges exceeding topological tension
    excess_ratio = np.maximum(0.0, (topological_tension - tension_threshold) / (std_tension + 1e-6))
    boost_factor = 1.0 + 0.5 * np.tanh(excess_ratio)
    combined_penalty = combined_penalty * boost_factor

    # Symmetrically add penalties to the distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
