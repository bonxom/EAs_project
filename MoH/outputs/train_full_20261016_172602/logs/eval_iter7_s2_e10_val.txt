# ===== Heuristic Code =====
#{A multi-hop 4-node quartet geometric analysis evaluates criss-cross chain detours and cross-connection shortcuts (u-w and v-x) alongside consecutive edge utilities to penalize intertwined tour segments while selectively discounting promising shortcut alternatives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Define consecutive 4-node quartet chain: u -> v -> w -> x
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    x_idx = np.roll(tour, -3)

    # Frequency memory decay
    edge_n_used *= 0.96

    # Path distances along the 3 consecutive links
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_wx = edge_distance[w_idx, x_idx]

    # Cross connections and end-to-end shortcut
    d_uw = edge_distance[u_idx, w_idx]
    d_vx = edge_distance[v_idx, x_idx]
    d_ux = edge_distance[u_idx, x_idx]

    # Multi-hop chain path length vs alternate uncrossing connections
    chain_len = d_uv + d_vw + d_wx
    # Compare against 2-opt and 3-opt style bypasses: (u->w + v->x) or direct end-to-end + interior
    cross_alt = d_uw + d_vx
    criss_cross_detour = chain_len / (cross_alt + 1e-8)

    # Turning angle deficits for u-v-w and v-w-x
    denom_uvw = 2.0 * d_uv * d_vw + 1e-8
    cos_turn_uvw = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom_uvw, -1.0, 1.0)
    acute_deficit_uvw = 1.0 - cos_turn_uvw

    denom_vwx = 2.0 * d_vw * d_wx + 1e-8
    cos_turn_vwx = np.clip((d_vw**2 + d_wx**2 - d_vx**2) / denom_vwx, -1.0, 1.0)
    acute_deficit_vwx = 1.0 - cos_turn_vwx

    # Combined quartet inefficiency score
    quartet_inefficiency = criss_cross_detour * (1.0 + 0.25 * (acute_deficit_uvw + acute_deficit_vwx))

    # Distribute quartet geometric penalties back onto edges (each edge v-w participates in overlapping quartets)
    # Edge v-w is central in quartet (u, v, w, x), trailing in (prev_u, u, v, w), leading in (v, w, x, next_x)
    geom_factor_vw = (
        0.50 * quartet_inefficiency +
        0.25 * np.roll(quartet_inefficiency, 1) +
        0.25 * np.roll(quartet_inefficiency, -1)
    )
    # Shift geom_factor so indices align with edge (u_idx, v_idx)
    geom_factor_uv = np.roll(geom_factor_vw, 1)

    # Guided Local Search base utility on tour edges
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * geom_factor_uv)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize inefficient / high-utility edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * geom_factor_uv[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Reward promising uncrossing shortcuts (u-w and v-x) identified from tangled quartets
    high_detour_quartets = np.argsort(criss_cross_detour)[-k:]
    for idx in high_detour_quartets:
        u, w = u_idx[idx], w_idx[idx]
        v, x = v_idx[idx], x_idx[idx]

        # Shortcut u-w
        if edge_n_used[u, w] < 1.0:
            discount_uw = 0.08 * global_mean_dist / (1.0 + edge_n_used[u, w])
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount_uw)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

        # Shortcut v-x
        if edge_n_used[v, x] < 1.0:
            discount_vx = 0.08 * global_mean_dist / (1.0 + edge_n_used[v, x])
            updated_edge_distance[v, x] = max(1e-6, updated_edge_distance[v, x] - discount_vx)
            updated_edge_distance[x, v] = updated_edge_distance[v, x]

    return updated_edge_distance
