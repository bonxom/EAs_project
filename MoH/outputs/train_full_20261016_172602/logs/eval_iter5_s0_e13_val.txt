# {Dynamically scale Guided Local Search edge penalties within [0.1, 0.6] proportional to stagnation measured via the coefficient of variation in edge reuse frequencies and utilities, breaking persistent local optima without introducing non-determinism.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Adaptive Ruin-and-Recreate penalty scaling based on stagnation / edge reuse intensity
    mean_count = np.mean(tour_counts)
    std_count = np.std(tour_counts)
    std_util = np.std(utilities)
    mean_util = np.mean(utilities)

    # Higher reuse and lower relative utility variation indicate deepening stagnation in local optima
    cv_count = std_count / (mean_count + 1e-6)
    cv_util = std_util / (mean_util + 1e-6)
    stagnation_metric = (mean_count / (mean_count + 5.0)) * (1.0 / (1.0 + cv_util)) + 0.5 * (1.0 / (1.0 + cv_count))

    # Map stagnation metric smoothly into [0.1, 0.6]
    alpha = 0.1 + 0.5 / (1.0 + np.exp(-3.0 * (stagnation_metric - 0.75)))

    avg_edge = np.mean(tour_lengths)
    penalty = alpha * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
