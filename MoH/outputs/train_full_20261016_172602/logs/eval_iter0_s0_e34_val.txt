# {Replace uniform near-maximal thresholding with a deterministic percentile-rank and dynamic utility-spread selection that scales penalties proportionally to break cyclic trap loops while penalizing diverse high-utility edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    avg_edge = np.mean(tour_lengths)
    n_edges = len(local_opt_tour)

    # Deterministic top-k percentile selection (penalize top 10-25% based on tour size, min 1)
    k = max(1, int(np.ceil(0.15 * n_edges)))
    top_indices = np.argsort(utilities)[-k:]

    # Rank-proportional penalty scaling to diversify impact and prevent cyclic 1-2 edge oscillation
    ranks = np.linspace(0.5, 1.0, k)
    base_penalty = 0.3 * avg_edge
    penalties = base_penalty * ranks

    sel_u = u_idx[top_indices]
    sel_v = v_idx[top_indices]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
