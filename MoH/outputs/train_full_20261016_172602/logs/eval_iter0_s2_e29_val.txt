# ===== Heuristic Code =====
#{A geometric detour and angle-deficit penalty strategy that detects consecutive path inefficiencies and crossing hazards by evaluating local detour ratios and triangle deficiencies, selectively penalizing geometrically awkward edges while discounting low-cost shortcut alternatives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)
    w_idx = np.roll(u_idx, -2)

    # Slow decay on usage frequency to maintain long-term memory
    edge_n_used *= 0.96

    # Base tour edge lengths and usage
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]
    freq_uv = edge_n_used[u_idx, v_idx]

    # Evaluate local geometric detour ratio: (d(u,v) + d(v,w)) / d(u,w)
    # A high ratio signifies an acute angle, path zigzag, or near-collinear overshoot
    detour_ratio = (d_uv + d_vw) / np.maximum(d_uw, 1e-6)

    # Angle-deficit indicator: absolute detour excess relative to direct shortcut
    detour_excess = (d_uv + d_vw) - d_uw

    # Compute GLS utility modulated by geometric awkwardness
    # Edges belonging to severe detours are given higher utility for penalization
    geom_weight = 1.0 + np.tanh(detour_ratio - 1.0) + (detour_excess / (np.mean(d_uv) + 1e-6))
    utilities = (d_uv * geom_weight) / (1.0 + freq_uv)

    # Select top bottleneck edges that also exhibit geometric inefficiency
    k = max(1, int(np.ceil(0.12 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Register frequency updates
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    global_mean_dist = np.mean(edge_distance)
    mean_freq = np.mean(edge_n_used[u_idx, v_idx])
    stagnation_factor = 1.0 + np.tanh(mean_freq)

    # Adaptive penalty scaling
    base_lambda = 0.22 * global_mean_dist * stagnation_factor

    # Apply selective geometric penalties
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        geom_factor = min(2.5, 1.0 + 0.5 * detour_ratio[idx])
        penalty = base_lambda * (1.0 + np.log1p(edge_n_used[u, v])) * geom_factor
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Inversely discount shortcut edges across acute vertices to encourage 2-opt untangling
    for idx in top_k_indices:
        u = u_idx[idx]
        w = w_idx[idx]
        if edge_n_used[u, w] < 0.35:
            shortcut_discount = 0.08 * global_mean_dist / (1.0 + edge_n_used[u, w])
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - shortcut_discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
