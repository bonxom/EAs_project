# ===== Heuristic Code =====
# {Compute Guided Local Search utility values scaled by current tour metrics and instance variance to penalize the most critical bottleneck edges with scale-invariant additive penalties.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    costs = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Compute GLS utilities for current tour edges
    utilities = costs / (1.0 + counts)

    # Scale-invariant penalty parameter based on instance edge distance statistics
    mean_cost = np.mean(costs)
    std_cost = np.std(edge_distance)
    if std_cost < 1e-6:
        std_cost = mean_cost if mean_cost > 1e-6 else 1.0

    # Base penalty proportional to mean tour edge length and normalized by standard deviation
    lambda_param = 0.3 * mean_cost * (mean_cost / std_cost)

    max_util = np.max(utilities)
    # Target edges with utility within top threshold or near maximum utility
    util_mask = utilities >= (0.85 * max_util)

    # Calculate calibrated penalty for selected edges
    penalties = np.zeros(n, dtype=np.float64)
    penalties[util_mask] = lambda_param / (1.0 + counts[util_mask])

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
