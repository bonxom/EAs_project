# ===== Heuristic Code =====
#{EntropyGatedSelectiveFeedback: Recombines top-k edge selection with a memory decay factor and applies targeted congestion modulation composed of spatial centrality, entropy rigidity, and node interaction dispersion only to candidate penalty edges scaled by a non-linear tanh stagnation factor.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # Memory forgetting factor to allow de-penalization of previously penalized edges
    edge_n_used *= 0.95

    # Compute utilities along current tour edges
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    # Select top-k candidate edges with the highest utility
    k = max(1, int(np.ceil(0.15 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Increment usage counts symmetrically for top-k edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Stagnation metrics based on tour overlap and usage
    tour_usage = edge_n_used[u_idx, v_idx]
    avg_tour_usage = np.mean(tour_usage)
    max_used = np.max(edge_n_used) + 1e-12
    overlap_ratio = np.mean(tour_usage > 1.0)
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (avg_tour_usage / max_used)

    # Spatial centrality
    node_mean_dist = np.mean(edge_distance, axis=1)
    global_mean_dist = np.mean(node_mean_dist)
    spatial_centrality = global_mean_dist / (node_mean_dist + 1e-12)
    norm_spatial = spatial_centrality / (np.max(spatial_centrality) + 1e-12)

    # Entropy rigidity calculation
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)

    # Usage centrality
    node_usage_centrality = np.sum(edge_n_used, axis=1)
    norm_usage = node_usage_centrality / (np.max(node_usage_centrality) + 1e-12)

    # Non-linear adaptive penalty base
    adaptive_lambda = (0.15 + 0.20 * np.tanh(stagnation_factor)) * global_mean_dist

    # Apply targeted congestion modulation strictly to the top-k edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        spatial_term = norm_spatial[u] + norm_spatial[v]
        rigidity_term = entropy_rigidity[u] + entropy_rigidity[v]
        dispersion_term = norm_usage[u] * norm_usage[v]

        congestion_multiplier = 1.0 + 0.35 * spatial_term + 0.35 * rigidity_term + 0.30 * dispersion_term
        penalty = adaptive_lambda * edge_n_used[u, v] * congestion_multiplier

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
