# ===== Heuristic Code =====
#{A vectorized Guided Local Search penalty mechanism that penalizes maximal utility edges while dynamically cooling and resetting over-penalized backbone edges using usage frequency percentiles to prevent saturation and escape local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate guided local search utility
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Identify high-utility edges to penalize
    threshold = 0.99 * max_utility
    penalize_mask = utilities >= threshold

    # Detect frequency saturation and backbone stagnation across the tour
    max_count = np.max(tour_counts)
    if max_count > 0:
        # Edges heavily over-penalized relative to average tour usage receive a cooling discount
        mean_count = np.mean(tour_counts)
        cooling_mask = (tour_counts > (mean_count + 1.5 * np.std(tour_counts))) & (~penalize_mask)
        if np.any(cooling_mask):
            cool_u = u_idx[cooling_mask]
            cool_v = v_idx[cooling_mask]
            cool_amount = 0.15 * avg_edge * (tour_counts[cooling_mask] / max_count)
            updated_edge_distance[cool_u, cool_v] = np.maximum(0.0, updated_edge_distance[cool_u, cool_v] - cool_amount)
            updated_edge_distance[cool_v, cool_u] = updated_edge_distance[cool_u, cool_v]

    # Apply scaled penalty to selected edges
    penalty = 0.3 * avg_edge
    sel_u = u_idx[penalize_mask]
    sel_v = v_idx[penalize_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
