#{Replace static k-NN node density with geometric corner sharpness derived from the cosine rule to penalize inefficient acute zigzag angles between consecutive tour edges while modulating penalties by relative length and historical usage.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Identify predecessor, current, successor, and next-successor nodes in tour
    p_idx = np.roll(local_opt_tour, 1)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    w_idx = np.roll(local_opt_tour, -2)

    # Tour edge lengths
    d_uv = edge_distance[u_idx, v_idx]
    d_pu = edge_distance[p_idx, u_idx]
    d_pv = edge_distance[p_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Estimate incident angle sharpness via cosine theorem on triangles (p, u, v) and (u, v, w)
    # Acute angles (zigzag turnarounds) yield cos(theta) close to -1, maximizing (1 - cos(theta))
    cos_theta_u = (d_pu**2 + d_uv**2 - d_pv**2) / (2.0 * np.maximum(d_pu * d_uv, 1e-6))
    cos_theta_v = (d_uv**2 + d_vw**2 - d_uw**2) / (2.0 * np.maximum(d_vw * d_uv, 1e-6))
    cos_theta_u = np.clip(cos_theta_u, -1.0, 1.0)
    cos_theta_v = np.clip(cos_theta_v, -1.0, 1.0)

    sharpness_u = 1.0 - cos_theta_u
    sharpness_v = 1.0 - cos_theta_v
    corner_sharpness = 0.5 * (sharpness_u + sharpness_v)
    corner_sharpness = corner_sharpness / (np.mean(corner_sharpness) + 1e-6)

    # Relative edge length penalty
    avg_tour_length = np.mean(d_uv)
    relative_length = d_uv / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Damping based on edge usage history
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])

    # Geometric bottleneck penalty targeting sharp turns and long edges
    local_penalty = relative_length * usage_factor * corner_sharpness

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Emphasize highest-cost edges
    threshold = np.percentile(d_uv, 75)
    high_penalty_mask = d_uv >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Symmetrically apply penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
