# ===== Heuristic Code =====
# {This strategy penalizes high-utility tour edges based on standardized length and usage, then applies rank-decayed discounts exclusively to top k-nearest neighbor incident edges of penalized vertices to guide search toward promising alternative connections.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k = min(15, n - 1)
    base_bonus_factor = 0.05

    for u in penalized_nodes:
        row_dist = edge_distance[u]
        # Partition to find top k nearest neighbors excluding self
        candidate_indices = np.argpartition(row_dist, k)[:k + 1]
        candidate_indices = candidate_indices[candidate_indices != u][:k]

        # Sort candidates strictly by original edge metric distance to establish rank
        sorted_order = np.argsort(row_dist[candidate_indices])
        sorted_neighbors = candidate_indices[sorted_order]

        # Determine candidate edge usage
        nbr_used = edge_n_used[u, sorted_neighbors]
        ranks = np.arange(1, len(sorted_neighbors) + 1, dtype=np.float64)

        # Rank-proportional discount favoring unvisited or minimally used nearest neighbors
        discount_factors = (base_bonus_factor / ranks) / (1.0 + np.sqrt(nbr_used))
        discounts = updated_edge_distance[u, sorted_neighbors] * discount_factors

        updated_edge_distance[u, sorted_neighbors] -= discounts
        updated_edge_distance[sorted_neighbors, u] = updated_edge_distance[u, sorted_neighbors]

    return updated_edge_distance
