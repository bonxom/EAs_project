# ===== Heuristic Code =====
#{An entropy-and-Gini-driven weighting scheme dynamically transitions between targeting stubborn cyclic trap edges when usage inequality is high and penalizing raw edge lengths when usage across the tour is uniform.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Precompute k-NN spatial density
    k = min(6, n)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Tour edge attributes
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx].astype(np.float64)

    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Core components: Rank/length penalty and usage-damped local density penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_counts)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Calculate Gini coefficient of tour edge usage frequencies to measure usage inequality
    counts_sorted = np.sort(edge_counts)
    total_usage = np.sum(counts_sorted)
    if total_usage > 0:
        index = np.arange(1, n + 1)
        gini = (2.0 * np.sum(index * counts_sorted) - (n + 1) * total_usage) / (n * total_usage)
        gini = np.clip(gini, 0.0, 1.0)
    else:
        gini = 0.0

    # Usage variance and dynamic weighting between trap edge targeting vs raw length dominance
    # High Gini/variance indicates stubborn cyclic traps; low indicates uniform exploration
    w_trap = gini
    w_length = 1.0 - w_trap

    # Trap edge prioritization factor: edges with usage higher than mean receive extra focus when Gini is high
    mean_count = total_usage / n
    trap_factor = 1.0 + (edge_counts / (mean_count + 1.0)) * w_trap

    combined_penalty = (w_length * rank_penalty + (1.0 + w_trap) * local_penalty) * trap_factor

    # Boost top percentile penalties to break stubborn long edges
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
