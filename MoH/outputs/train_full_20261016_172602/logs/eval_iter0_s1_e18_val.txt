# ===== Heuristic Code =====
# {Combines bottleneck-aware guided penalization of high-utility tour edges with severed-cut segment identification to selectively discount reconnecting edges between opposing path endpoints and their nearest low-usage neighbors.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=np.intp)
    u_idx = tour
    v_idx = np.roll(tour, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    base_utilities = (costs * (1.0 + weights)) / (1.0 + used)

    prev_util = np.roll(base_utilities, 1)
    next_util = np.roll(base_utilities, -1)
    bottleneck_utilities = base_utilities + 0.35 * np.maximum(prev_util, next_util)

    threshold = np.median(bottleneck_utilities)
    max_util = np.max(bottleneck_utilities) + 1e-8
    penalized_mask = bottleneck_utilities >= threshold

    p_u = u_idx[penalized_mask]
    p_v = v_idx[penalized_mask]
    p_costs = costs[penalized_mask]
    p_used = used[penalized_mask]
    p_utils = bottleneck_utilities[penalized_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_indices = np.where(penalized_mask)[0]
    cut_u = tour[penalized_indices]
    cut_v = tour[(penalized_indices + 1) % n]
    endpoints = np.unique(np.concatenate([cut_u, cut_v]))

    # Cross-connecting reward between severed cut endpoints
    if len(cut_u) > 1:
        for u in cut_u:
            for v in cut_v:
                if u != v:
                    u_cnt = edge_n_used[u, v]
                    reconnect_factor = 0.04 / (1.0 + np.sqrt(u_cnt))
                    reward = reconnect_factor * edge_distance[u, v]
                    val = max(1e-6, updated_edge_distance[u, v] - reward)
                    updated_edge_distance[u, v] = val
                    updated_edge_distance[v, u] = val

    # Targeted dampening for nearest low-usage neighbors of broken endpoints
    k_nn = min(5, n - 1)
    for u in endpoints:
        orig_row = edge_distance[u]
        nn_candidates = np.argpartition(orig_row, k_nn)[:k_nn + 1]
        for v in nn_candidates:
            if v != u:
                u_cnt = edge_n_used[u, v]
                decay_ratio = 0.03 / (1.0 + u_cnt)
                val = max(1e-6, updated_edge_distance[u, v] - decay_ratio * edge_distance[u, v])
                updated_edge_distance[u, v] = val
                updated_edge_distance[v, u] = val

    return updated_edge_distance
