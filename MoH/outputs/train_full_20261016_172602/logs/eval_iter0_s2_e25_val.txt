# ===== Heuristic Code =====
#{Key idea: Construct an affinity/conductance graph from edge distances and frequency, compute the Fiedler vector via the normalized Laplacian to identify structural bottleneck bridge edges across graph cuts, and selectively penalize high-utility edges connecting distinct clusters to diversify the local search.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy of the edge distance matrix
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    u_idx = np.asarray(local_opt_tour, dtype=int)
    v_idx = np.roll(u_idx, -1)

    # Decay usage frequencies
    edge_n_used *= 0.95

    # Global mean distance scale
    mean_dist = np.mean(edge_distance) + 1e-12

    # Guided local search utility computation
    tour_dist = edge_distance[u_idx, v_idx]
    tour_used = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_used)

    # Approximate graph conductance / affinity matrix for spectral cut analysis
    # High conductance where distance is small and edge has been frequently explored
    affinity = np.exp(-edge_distance / mean_dist) * (1.0 + np.log1p(edge_n_used))
    np.fill_diagonal(affinity, 0.0)

    # Normalized Laplacian and Fiedler vector approximation (power iteration / deflation)
    degrees = np.sum(affinity, axis=1) + 1e-12
    inv_sqrt_d = 1.0 / np.sqrt(degrees)
    # Normalized adjacency S = D^(-1/2) * W * D^(-1/2)
    norm_adj = affinity * np.outer(inv_sqrt_d, inv_sqrt_d)

    # Dominant eigenvector of norm_adj is d^(1/2) / ||d^(1/2)||
    v0 = np.sqrt(degrees)
    v0 /= np.linalg.norm(v0) + 1e-12

    # Power iteration with deflation to extract the second eigenvector (Fiedler vector proxy)
    # Deterministic initialization based on coordinate variance proxy
    x = np.sin(np.arange(num_nodes) * (2.0 * np.pi / num_nodes))
    x -= np.dot(x, v0) * v0
    x_norm = np.linalg.norm(x)
    if x_norm > 1e-12:
        x /= x_norm
    else:
        x = np.linspace(-1.0, 1.0, num_nodes)
        x -= np.dot(x, v0) * v0
        x /= np.linalg.norm(x) + 1e-12

    for _ in range(12):
        x = norm_adj.dot(x)
        x -= np.dot(x, v0) * v0
        norm_val = np.linalg.norm(x)
        if norm_val > 1e-12:
            x /= norm_val
        else:
            break

    # Spectral resistance / cut metric: difference across the partition
    spectral_diff = np.abs(x[u_idx] - x[v_idx])
    max_spec = np.max(spectral_diff) + 1e-12
    norm_spectral = spectral_diff / max_spec

    # Modulate utility with spectral cut bridge potential
    # Edges crossing structural cuts with high usage get prioritized for disruption
    combined_scores = utilities * (1.0 + 0.8 * norm_spectral)

    k = max(1, int(np.ceil(0.18 * num_nodes)))
    top_k_indices = np.argsort(combined_scores)[-k:]

    # Increment usage for penalized edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

    # Adaptive penalty scaling based on stagnation
    tour_usage = edge_n_used[u_idx, v_idx]
    overlap_ratio = np.mean(tour_usage > 1.0)
    stagnation_factor = 1.0 + 1.5 * overlap_ratio + (np.mean(tour_usage) / (np.max(edge_n_used) + 1e-12))
    adaptive_lambda = (0.18 + 0.22 * np.tanh(stagnation_factor)) * mean_dist

    # Apply penalties incorporating spectral bridge weights
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        bridge_boost = 1.0 + 0.75 * norm_spectral[idx]
        penalty = adaptive_lambda * (1.0 + 0.25 * edge_n_used[u, v]) * bridge_boost
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
