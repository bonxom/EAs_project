# ===== Heuristic Code =====
#{An adaptive reinforcement feedback scheme dynamically modulates the penalty magnitude based on stagnation metrics derived from edge usage statistics and tour edge length variance to escape deep local basins while preserving search efficiency.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility calculation
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Adaptive reinforcement feedback:
    # 1. Edge usage saturation captures how deeply the current tour edges have been entrenched/visited.
    mean_count = np.mean(tour_counts)
    max_count = np.max(tour_counts)
    stagnation_ratio = mean_count / (1.0 + max_count)

    # 2. Coefficient of variation (CV) of tour edge lengths measures search state homogeneity.
    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    cv_edge = std_edge / (avg_edge + 1e-12)

    # Dynamically ramp up penalty when stuck in repetitive configurations and adapt to tour dispersion
    dynamic_scale = 0.15 + 0.35 * (1.0 / (1.0 + np.exp(-4.0 * (stagnation_ratio - 0.5)))) + 0.20 * np.tanh(cv_edge)
    penalty = dynamic_scale * avg_edge

    # Select candidate edges with highest utility (guided perturbation)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically penalize selected edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
