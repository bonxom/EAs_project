# ===== Heuristic Code =====
#{An adaptive guided local search scheme that penalizes maximal utility edges using a normalized tour-length scale modulated by edge usage frequency and relative dispersion to dynamically balance intensification and diversification.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility calculation balancing cost against prior penalty frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility <= 0:
        return updated_edge_distance

    # Adaptive penalty scaling derived from the tour cost statistics and problem dimension
    n = len(local_opt_tour)
    mean_cost = np.mean(tour_lengths)
    std_cost = np.std(tour_lengths)

    # Adaptive parameter lambda: scaled by tour mean and normalized dispersion
    dispersion_factor = 1.0 + (std_cost / (mean_cost + 1e-12))
    base_penalty = (mean_cost / np.sqrt(n)) * dispersion_factor * 0.3

    # Identify edges within top utility threshold
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Frequency-attenuated penalty to avoid runaway inflation on frequently penalized edges
    edge_penalties = base_penalty / np.sqrt(1.0 + tour_counts[selected_mask])

    # Apply symmetric updates
    np.add.at(updated_edge_distance, (sel_u, sel_v), edge_penalties)
    np.add.at(updated_edge_distance, (sel_v, sel_u), edge_penalties)

    return updated_edge_distance
