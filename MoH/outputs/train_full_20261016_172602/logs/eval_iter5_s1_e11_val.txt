# ===== Heuristic Code =====
#{Penalize high-utility edges in the local optimum and selectively discount the top-k nearest non-tour incident neighbors inversely proportional to their relative proximity rank to guide search diversification into promising candidate subgraphs.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Calculate costs and historical usage frequencies for tour edges
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Cost-frequency utility calculation
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Penalize above-median utility edges
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Build tour adjacency lookup for quick exclusion
    prev_idx = np.roll(u_idx, 1)
    tour_neighbors = np.zeros((n, 2), dtype=np.intp)
    tour_neighbors[u_idx, 0] = prev_idx
    tour_neighbors[u_idx, 1] = v_idx

    # Candidate-subgraph-directed incentive: discount top-k nearest incident non-tour neighbors
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k = min(5, n - 3)
    if k > 0:
        base_decay = 0.05
        decay_weights = base_decay / (np.arange(1, k + 1, dtype=np.float64))

        for u in penalized_nodes:
            row_costs = edge_distance[u].astype(np.float64)
            sorted_indices = np.argsort(row_costs)

            # Select nearest non-tour, non-self candidate neighbors
            candidates = []
            excluded = {u, tour_neighbors[u, 0], tour_neighbors[u, 1]}
            for cand in sorted_indices:
                if cand not in excluded:
                    candidates.append(cand)
                    if len(candidates) == k:
                        break

            if candidates:
                cand_arr = np.array(candidates, dtype=np.intp)
                actual_k = len(cand_arr)
                discounts = updated_edge_distance[u, cand_arr] * decay_weights[:actual_k]
                updated_edge_distance[u, cand_arr] -= discounts
                updated_edge_distance[cand_arr, u] -= discounts

    return updated_edge_distance
