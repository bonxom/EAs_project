# ===== Heuristic Code =====
# {Hybridizes sigmoid z-score cost weighting with endpoint entropy-derived motif rigidity to penalize rigid high-utility edges while selectively decaying unused incident edges of penalized nodes.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Sigmoid z-score cost weights
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Endpoint entropy and motif rigidity
    total_node_usage = np.sum(edge_n_used, axis=1, dtype=np.float64)
    u_total = total_node_usage[u_idx] + 1e-9
    v_total = total_node_usage[v_idx] + 1e-9
    p_u = edge_n_used[u_idx] / u_total[:, None]
    p_v = edge_n_used[v_idx] / v_total[:, None]
    log_p_u = np.log(p_u + 1e-12)
    log_p_v = np.log(p_v + 1e-12)
    entropy_u = -np.sum(p_u * log_p_u, axis=1)
    entropy_v = -np.sum(p_v * log_p_v, axis=1)
    max_entropy = np.log(max(n - 1.0, 2.0))
    motif_rigidity = 1.0 - (0.5 * (entropy_u + entropy_v) / (max_entropy + 1e-9))
    motif_rigidity = np.clip(motif_rigidity, 0.0, 1.0)

    # Augmented utility combining cost weighting, usage, and motif rigidity
    utilities = (costs * (1.0 + weights) / (1.0 + used)) * (1.0 + 1.5 * motif_rigidity)

    # Dynamic thresholding based on upper-percentile selection
    threshold = np.percentile(utilities, 60.0)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_rigidity = motif_rigidity[penalize_mask]
    max_util = np.max(utilities) + 1e-8
    p_utils = utilities[penalize_mask]

    # Entropy-modulated penalty scaling
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * (0.5 + 0.5 * p_rigidity)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Incident unused edge decay for penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
