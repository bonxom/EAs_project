# ===== Heuristic Code =====
#{This strategy classifies tour edges into intra-cluster outliers, bridge edges, and standard segments using fast k-NN partitioning, applying maturity-modulated asymmetric penalties that aggressively eliminate inefficient intra-cluster links while protecting essential topological bridges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # 1. Compute k-NN density using fast np.partition
    k = min(6, n - 1)
    k_smallest = np.partition(edge_distance, kth=k, axis=1)[:, 1:k + 1]
    k_nn_avg = np.mean(k_smallest, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # 2. Tour edge properties and decomposition
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    edge_usages = edge_n_used[u_idx, v_idx]

    avg_tour_length = np.mean(tour_lengths)
    relative_lengths = tour_lengths / (avg_tour_length + 1e-6)
    edge_density = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # 3. Dynamic search maturity
    mean_usage = np.mean(edge_usages)
    tau = 1.0 / (1.0 + 0.1 * mean_usage)
    usage_factor = 1.0 / (1.0 + 0.3 * edge_usages)

    # 4. Standard logistic rank-based penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_lengths - 1.0)))

    # 5. Topological classification: intra-cluster outliers vs. bridges
    # Intra-cluster outliers: long edges within dense regions
    # Bridges: long edges connecting sparse regions (essential macro-structure)
    is_long = relative_lengths > 1.0
    is_dense = edge_density > 1.0

    intra_outlier_mask = is_long & is_dense
    bridge_mask = is_long & (~is_dense)

    # Baseline combined penalty
    combined_penalty = relative_lengths * rank_penalty * usage_factor

    # Asymmetric modulation
    # Severely penalize intra-cluster shortcuts, scaled by search maturity tau
    combined_penalty[intra_outlier_mask] *= (1.5 + 0.5 * tau) * edge_density[intra_outlier_mask]

    # Attenuate penalties on bridge edges to preserve macro-topology
    combined_penalty[bridge_mask] *= (0.6 / (1.0 + 0.2 * (2.0 - edge_density[bridge_mask])))

    # Symmetric distance matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
