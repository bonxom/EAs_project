# ===== Heuristic Code =====
#{Adaptively balance exploratory utility penalties against disruptive structural mutations using the Shannon entropy of historical edge penalties to measure basin stagnation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Identify edges on the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_usage = edge_n_used[u_idx, v_idx]

    # Metric embedding: k-nearest neighbor density
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Standard Guided Local Search (GLS) exploration utility
    avg_tour_len = np.mean(tour_lengths) + 1e-6
    rel_lengths = tour_lengths / avg_tour_len
    gls_utility = rel_lengths / (1.0 + 0.3 * tour_usage)

    # Disruptive structural penalty factor for escaping narrow basins
    rank_disrupt = 1.0 / (1.0 + np.exp(-4.0 * (rel_lengths - 1.0)))
    disruptive_penalty = rank_disrupt * density_factor

    # Compute information-theoretic entropy of penalty distribution over tour edges
    total_usage = np.sum(tour_usage)
    if total_usage > 0:
        p = tour_usage / total_usage
        p_pos = p[p > 0]
        entropy = -np.sum(p_pos * np.log(p_pos))
        max_entropy = np.log(n)
        normalized_entropy = entropy / (max_entropy + 1e-6)
    else:
        normalized_entropy = 1.0

    # High entropy implies well-dispersed exploration; low entropy indicates basin stagnation
    w_disrupt = 1.0 - normalized_entropy
    w_gls = normalized_entropy

    # Dynamic penalty scaling based on tour edge scale
    penalty_magnitude = 0.25 * avg_tour_len
    combined_penalty = penalty_magnitude * (w_gls * gls_utility + w_disrupt * disruptive_penalty)

    # Extra boost for extreme long edges
    threshold = np.percentile(tour_lengths, 75)
    combined_penalty[tour_lengths >= threshold] *= 1.5

    # Apply symmetric additive updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
