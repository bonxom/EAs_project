# ===== Heuristic Code =====
#{A hybrid metaheuristic penalty strategy that combines fast partition-based k-NN density normalization with Parent 1's rank-density weighting and a continuous progressive threshold boost for high-cost edges to escape local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Computational efficiency: np.partition instead of full sort for k-nearest neighbors
    k = min(5, n - 1)
    k_nn_dists = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(k_nn_dists, axis=1)

    # Exact density normalization from Parent 1
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Sigmoidal global rank penalty and density-modulated local penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Adaptive global vs local weight trade-off
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Continuous progressive boost between 1.25 and 1.50 for top quartile edges
    p75 = np.percentile(tour_edge_lengths, 75)
    max_len = np.max(tour_edge_lengths)
    spread = max_len - p75 + 1e-6
    high_mask = tour_edge_lengths >= p75

    final_penalty = combined_penalty.copy()
    progressive_boost = 1.25 + 0.25 * ((tour_edge_lengths[high_mask] - p75) / spread)
    final_penalty[high_mask] *= progressive_boost

    # Symmetrically apply penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), final_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), final_penalty)

    return updated_edge_distance
