# {Replace static k-NN density with dynamic detour replaceability that penalizes edges based on their relative detour gap to alternative nodes, weighted by relative tour length and historical usage dampening.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Calculate best alternative 2-hop detour cost for each tour edge (u, v): min_{w != u, v} (d(u,w) + d(w,v) - d(u,v))
    # Vectorized computation: evaluate triangular detour gaps across all candidate nodes w
    d_u_all = edge_distance[u_idx, :]  # Shape: (n, n)
    d_v_all = edge_distance[v_idx, :]  # Shape: (n, n)
    detour_costs = d_u_all + d_v_all - tour_edge_lengths[:, None]

    # Mask out w == u and w == v by setting them to infinity
    row_indices = np.arange(n)[:, None]
    mask = (np.arange(n) == u_idx[:, None]) | (np.arange(n) == v_idx[:, None])
    detour_costs[mask] = np.inf

    min_detour_gap = np.min(detour_costs, axis=1)

    # Edges with smaller detour gaps are more easily replaced/reconnected, so amplify their penalization
    # Normalize detour gap relative to average tour edge length
    rel_detour_gap = min_detour_gap / (avg_tour_length + 1e-6)
    detour_factor = 1.0 / (1.0 + np.exp(3.0 * (rel_detour_gap - 0.5)))

    # Standard GLS historical usage dampening and rank penalty
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    local_penalty = relative_length * usage_factor * detour_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
