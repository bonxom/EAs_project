# ===== Heuristic Code =====
#{The algorithm performs multi-scale topological ruin by segmenting tour edges into structural clusters via cost-scale percentiles, applying continuous rank-proportional penalties that aggressively disrupt high-utility long-distance bridges while preserving tight local clusters, paired with incident unexplored edge discounts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Topological clustering based on edge length scales
    # Differentiating macro-level bottleneck connectors from intra-cluster short connections
    q25, q75 = np.percentile(costs, [25.0, 75.0])
    iqr = q75 - q25 + 1e-8
    structural_bridge_factor = np.clip((costs - q25) / iqr, 0.0, 2.5)

    # Continuous rank-proportional modulation over edge utilities
    order = np.argsort(utilities)
    ranks = np.empty_like(order, dtype=np.float64)
    ranks[order] = np.linspace(0.0, 1.0, n)

    # Continuous penalty activation: spares the dense cluster core (low ranks)
    # and progressively concentrates strong perturbations on bottleneck bridges (high ranks)
    activation = np.maximum(0.0, (ranks - 0.4) / 0.6) ** 1.5
    scale_multiplier = 0.5 + structural_bridge_factor

    penalties = activation * scale_multiplier * (costs / (1.0 + np.sqrt(used + 1.0)))

    penalize_mask = penalties > 0.0
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_penalties = penalties[penalize_mask]

    updated_edge_distance[p_u, p_v] += p_penalties
    updated_edge_distance[p_v, p_u] += p_penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.025

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
