#{Decouple gating temperature from distance scale by standardizing utilities using utility IQR/standard deviation, ensuring scale-invariant and robust guided local search penalties.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]

    costs = np.array([edge_distance[u, v] for u, v in tour_edges], dtype=np.float64)
    median_cost = np.median(costs)
    mean_cost = np.mean(costs)

    utilities = np.zeros(n, dtype=np.float64)
    for i in range(n):
        u, v = tour_edges[i]
        used = edge_n_used[u, v]
        c = costs[i]
        excess_ratio = max(0.0, (c - median_cost) / (median_cost + 1e-8))
        utilities[i] = (c / (1.0 + used)) * (1.0 + excess_ratio)

    q75_util = np.percentile(utilities, 75)
    q25_util = np.percentile(utilities, 25)
    iqr_util = q75_util - q25_util
    std_util = np.std(utilities)
    scale_util = iqr_util if iqr_util > 1e-8 else (std_util if std_util > 1e-8 else 1.0)

    penalized_nodes = set()
    for idx in range(n):
        u, v = tour_edges[idx]
        used = edge_n_used[u, v]
        c = costs[idx]
        z_gate = (utilities[idx] - q75_util) / scale_util
        gate_weight = 1.0 / (1.0 + np.exp(-3.0 * z_gate))

        if gate_weight > 0.05:
            base_penalty = (c / (1.0 + np.sqrt(used + 1.0))) * (1.0 + c / (mean_cost + 1e-8))
            penalty = base_penalty * gate_weight
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty
            penalized_nodes.add(u)
            penalized_nodes.add(v)

    mean_tour_cost = mean_cost + 1e-8
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            local_density = np.mean(edge_distance[u, unused_mask]) / mean_tour_cost
            adaptive_decay = 0.02 / (1.0 + max(0.1, local_density))
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * adaptive_decay
            for v_idx in np.where(unused_mask)[0]:
                updated_edge_distance[v_idx, u] = updated_edge_distance[u, v_idx]

    return updated_edge_distance
