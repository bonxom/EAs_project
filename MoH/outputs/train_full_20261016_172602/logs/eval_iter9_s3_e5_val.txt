#{Adapts edge penalties dynamically using z-score standardization of tour edge lengths and edge usage frequencies, modulated by an endogenous stagnation metric to perturb entrenched long edges proportionally.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_usage = edge_n_used[u_idx, v_idx]

    # Dynamic standardization: Z-scores of tour lengths
    len_mean = np.mean(tour_lengths)
    len_std = np.std(tour_lengths)
    z_len = (tour_lengths - len_mean) / (len_std + 1e-8)

    # Dynamic standardization: Z-scores of edge usage (frequency penalty discount)
    use_mean = np.mean(tour_usage)
    use_std = np.std(tour_usage)
    z_use = (tour_usage - use_mean) / (use_std + 1e-8) if use_std > 1e-8 else np.zeros_like(tour_usage, dtype=float)

    # Endogenous stagnation estimate based on average usage relative to spread
    stagnation_ratio = use_mean / (use_mean + 1.0)

    # Adaptive utility computation: balance long edges and penalize heavily repeated edges
    # Lower penalty discount for highly worn edges to encourage structural diversification
    adaptive_utility = np.maximum(0.0, z_len) / (1.0 + np.exp(z_use))

    # Adaptive threshold determined by stagnation level
    # Higher stagnation targets more edges by dynamically shifting the cutoff
    adaptive_cutoff = np.quantile(adaptive_utility, np.clip(1.0 - stagnation_ratio, 0.5, 0.9))
    boost_mask = adaptive_utility >= adaptive_cutoff

    # Scale penalty relative to average tour edge length to keep distance metric well-conditioned
    base_penalty = len_mean * (adaptive_utility / (1.0 + np.mean(adaptive_utility) + 1e-8))
    base_penalty[boost_mask] *= (1.0 + stagnation_ratio)

    # Symmetrically apply penalties to the identified tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), base_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), base_penalty)

    return updated_edge_distance
