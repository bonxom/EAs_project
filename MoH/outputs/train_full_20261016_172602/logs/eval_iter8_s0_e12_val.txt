# ===== Heuristic Code =====
#{This algorithm creates an explicit copy of the distance matrix, calculates vectorized Guided Local Search utility scores along the local tour, and selects the deterministic top-K highest-utility edges via np.argpartition to apply a bounded symmetric penalty, preventing over-penalization under clustered utilities.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy to avoid mutating the original input matrix
    updated_edge_distance = edge_distance.copy()

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Compute Guided Local Search edge utility: utility = cost / (1 + count)
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    # Select top-k critical edges to penalize (at least 1, bounded to at most 5% or 3 edges)
    k = max(1, min(3, int(np.ceil(0.05 * n_edges))))

    # Efficient top-k partition to avoid full sort overhead and avoid over-penalization
    top_k_indices = np.argpartition(utilities, -k)[-k:]

    # Deterministic adaptive penalty scaled by mean tour edge length
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    sel_u = u_idx[top_k_indices]
    sel_v = v_idx[top_k_indices]

    # Apply symmetric penalty using advanced indexing
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
