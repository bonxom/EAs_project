# ===== Heuristic Code =====
#{A dual-frequency oscillatory memory strategy that decouples edge usage into short-term recency and long-term frequency components, penalizing high-recency bottleneck edges while actively discounting under-sampled promising connections to drive directed exploration.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    # Decouple usage: integer floor tracks long-term frequency, fractional part tracks fast recency
    # Decay recency component quickly while preserving cumulative frequency
    recency = edge_n_used - np.floor(edge_n_used)
    recency *= 0.70

    # Apply standard slow decay to the base usage matrix
    edge_n_used *= 0.98

    # Guided Local Search utility calculation balancing edge distance and long-term frequency
    tour_dist = edge_distance[u_idx, v_idx]
    tour_freq = edge_n_used[u_idx, v_idx]
    utilities = tour_dist / (1.0 + tour_freq)

    # Adaptive selection of penalized edges based on utility ranking
    k = max(1, int(np.ceil(0.12 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        # Boost short-term recency layer
        recency[u, v] = min(1.0, recency[u, v] + 0.5)
        recency[v, u] = recency[u, v]

    # Re-embed updated recency into edge_n_used
    edge_n_used += 0.05 * recency

    # Global scale reference
    global_mean_dist = np.mean(edge_distance)

    # Calculate stagnation based on recency spikes on current tour
    tour_recency = recency[u_idx, v_idx]
    short_term_trap = np.mean(tour_recency)
    stagnation_factor = 1.0 + 2.0 * short_term_trap

    adaptive_lambda = (0.20 + 0.15 * np.tanh(stagnation_factor)) * global_mean_dist

    # Apply penalties to trapped edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        recency_penalty_mod = 1.0 + 1.5 * recency[u, v]
        penalty = adaptive_lambda * (1.0 + np.log1p(edge_n_used[u, v])) * recency_penalty_mod
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Inversely reward or shield globally under-sampled, promising low-cost edges adjacent to penalized nodes
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        for node in (u, v):
            row_dist = edge_distance[node]
            row_used = edge_n_used[node]
            # Identify under-sampled candidate neighbors
            candidate_mask = (row_used < 0.25) & (row_dist > 0)
            if np.any(candidate_mask):
                cand_indices = np.where(candidate_mask)[0]
                best_cand = cand_indices[np.argmin(row_dist[cand_indices])]
                discount = 0.05 * global_mean_dist * (1.0 / (1.0 + row_used[best_cand]))
                updated_edge_distance[node, best_cand] = max(1e-6, updated_edge_distance[node, best_cand] - discount)
                updated_edge_distance[best_cand, node] = updated_edge_distance[node, best_cand]

    return updated_edge_distance
