# ===== Heuristic Code =====
#{This strategy introduces a penalty evaporation decay operator across the edge distance matrix prior to adding Guided Local Search penalties scaled by relative tour length, inverted edge usage, and upper-quartile outliers.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy edge distance matrix to preserve inputs
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Step 2: Edge penalty dissipation / cooling
    # Apply uniform decay factor on accumulated penalties (modulating toward initial base distances estimated via min values or decay)
    # Using decay factor gamma = 0.98 on historical adjustments relative to symmetry
    decay_factor = 0.98
    updated_edge_distance = updated_edge_distance * decay_factor

    # Step 3: Compute localized k-nearest neighbor density
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Step 4: Extract current local optimal tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)

    # Step 5: Calculate relative edge length and GLS-guided penalty terms
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Step 6: Heavily penalize structural outliers (upper quartile)
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Scale penalty with respect to average tour edge length to keep magnitude consistent after decay
    penalty_scale = avg_tour_length * 0.05
    scaled_penalty = combined_penalty * penalty_scale

    # Step 7: Apply symmetric additive penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), scaled_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), scaled_penalty)

    return updated_edge_distance
