# ===== Heuristic Code =====
#{An adaptive Guided Local Search penalty mechanism dynamically scales the percentile threshold and amplification factor based on the coefficient of variation of tour edge lengths and stagnation frequency to intensely break bottleneck edges during stagnation while evenly dispersing perturbations during diversification.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Precompute k-nearest-neighbor based local node density
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Extract current tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths)
    cov_tour_length = std_tour_length / (avg_tour_length + 1e-6)

    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    tour_edge_usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.3 * tour_edge_usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Balance global rank-based and local density-usage penalties
    freq_mean = np.mean(tour_edge_usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Dynamic percentile threshold and amplification driven by stagnation and variance
    # High frequency / high variance indicates persistent bottleneck edges needing sharp, focused penalization
    stagnation_ratio = freq_mean / (1.0 + freq_mean)
    percentile_val = np.clip(60.0 + 30.0 * (1.0 / (1.0 + np.exp(-3.0 * (cov_tour_length - 0.5)))) * stagnation_ratio, 60.0, 95.0)
    multiplier = 1.2 + 1.2 * stagnation_ratio + 0.6 * np.tanh(cov_tour_length)

    threshold = np.percentile(tour_edge_lengths, percentile_val)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= multiplier

    # Scale penalty magnitude relative to current mean edge length
    scaled_penalty = combined_penalty * (0.1 * avg_tour_length)

    # Symmetric edge distance update
    np.add.at(updated_edge_distance, (u_idx, v_idx), scaled_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), scaled_penalty)

    return updated_edge_distance
