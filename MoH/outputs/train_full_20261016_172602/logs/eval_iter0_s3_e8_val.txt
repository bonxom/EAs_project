# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Extract consecutive edge endpoints from the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    # Calculate local density around endpoints based on k-nearest neighbors
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Compute Guided Local Search utility ratio adjusted by density
    avg_tour_length = np.mean(tour_edge_lengths)
    utility = (tour_edge_lengths / (1.0 + edge_counts)) * density_factor

    # Identify bottleneck candidates: strictly upper quartile by length and utility
    length_threshold = np.percentile(tour_edge_lengths, 75)
    utility_threshold = np.percentile(utility, 75)
    selective_mask = (tour_edge_lengths >= length_threshold) & (utility >= utility_threshold)

    # Ensure at least the top utility edge is penalized if the joint mask is too strict
    if not np.any(selective_mask):
        selective_mask = utility >= utility_threshold

    # Scale penalty relative to average tour edge length to make meaningful perturbations
    base_penalty = 0.15 * avg_tour_length
    penalties = np.zeros_like(tour_edge_lengths)
    penalties[selective_mask] = base_penalty * (utility[selective_mask] / (np.max(utility) + 1e-6))

    # Apply symmetric updates to the distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
