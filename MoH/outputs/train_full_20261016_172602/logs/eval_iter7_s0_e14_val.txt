# ===== Heuristic Code =====
# {Applies a smooth continuous softmax-like utility-weighted penalty across tour edges to penalize high-utility edges proportionately while avoiding hard threshold cutoffs and premature convergence.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    # Soft continuous penalty assignment via temperature-scaled normalized utility
    u_min = np.min(utilities)
    u_max = np.max(utilities)
    denom = u_max - u_min if u_max > u_min else 1.0

    # Scaled continuous weights favoring higher utility edges smoothly
    norm_utilities = (utilities - u_min) / denom
    weights = np.exp(4.0 * (norm_utilities - 1.0))

    avg_edge = np.mean(tour_lengths)
    penalties = 0.3 * avg_edge * weights

    # Apply symmetric penalties without loops
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
