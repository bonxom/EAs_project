# ===== Heuristic Code =====
#{Replaces static k-NN density with a minimum spanning tree detour-centrality metric to strongly penalize inefficient bottleneck bridges between clusters while preserving inverse frequency dampening.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Fast Prim's MST to compute minimum incident tree cost and MST connectivity baseline
    selected = np.zeros(n, dtype=bool)
    min_cost = np.full(n, np.inf)
    parent = np.full(n, -1, dtype=int)

    selected[0] = True
    min_cost[1:] = edge_distance[0, 1:]
    parent[1:] = 0

    mst_incident_min = np.full(n, np.inf)

    for _ in range(n - 1):
        unselected_idx = np.where(~selected)[0]
        v = unselected_idx[np.argmin(min_cost[unselected_idx])]
        u = parent[v]
        selected[v] = True

        weight = edge_distance[u, v]
        if weight < mst_incident_min[u]:
            mst_incident_min[u] = weight
        if weight < mst_incident_min[v]:
            mst_incident_min[v] = weight

        unselected_mask = ~selected
        closer_mask = edge_distance[v] < min_cost
        update_mask = unselected_mask & closer_mask
        min_cost[update_mask] = edge_distance[v, update_mask]
        parent[update_mask] = v

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Assess bottleneck detour centrality: ratio of edge length to the incident MST spine
    mst_baseline = np.maximum(mst_incident_min[u_idx], mst_incident_min[v_idx])
    bottleneck_ratio = tour_edge_lengths / (mst_baseline + 1e-6)
    bottleneck_factor = np.log1p(np.maximum(0.0, bottleneck_ratio - 1.0)) + 1.0

    # Symmetric additive penalty with relative length and inverse frequency dampener
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    local_penalty = relative_length * usage_factor * bottleneck_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * avg_tour_length * 0.05

    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
