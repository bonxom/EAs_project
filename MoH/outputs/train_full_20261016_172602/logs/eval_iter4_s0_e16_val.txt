# ===== Heuristic Code =====
#{Applies an evaporative memory mechanism by decaying all previously accumulated edge perturbations toward baseline costs while symmetrically penalizing the highest-utility edges of the current local optimum tour.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an independent copy of edge_distance
    updated_edge_distance = np.copy(edge_distance)

    # Identify directed tour edges symmetrically
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Vectorized computation of edge utilities on the current tour
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    # Evaporative decay factor: dissipate existing penalty accumulation across all edges
    # Dissipates historical penalties proportional to edge visit frequency
    decay_rate = 0.95
    updated_edge_distance = updated_edge_distance * decay_rate + (1.0 - decay_rate) * (edge_distance / (1.0 + 0.05 * edge_n_used))

    # Determine the maximal-utility threshold for penalization
    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    # Compute adaptive penalty scale based on the local tour's average edge cost
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Apply symmetric additive penalization to selected maximal-utility edges
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    # Ensure zero diagonal
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
