# ===== Heuristic Code =====
#{Replaces global cost z-scores with node-centric k-nearest-neighbor relative regret ranks to construct a localized utility metric, penalizing structurally redundant high-regret tour edges while discounting unused incident low-regret candidate alternatives.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Node-centric relative regret evaluation using k-nearest neighbors
    k = min(10, max(2, n - 1))

    # Find k+1 smallest distances per row (includes self-distance 0 at index 0)
    part_u = np.partition(edge_distance[u_idx], k, axis=1)[:, :k+1]
    part_v = np.partition(edge_distance[v_idx], k, axis=1)[:, :k+1]

    # Minimum non-zero neighbor distance (ideal closest alternative)
    min_dist_u = part_u[:, 1]
    min_dist_v = part_v[:, 1]

    # Average distance within the localized k-nearest neighborhood
    knn_mean_u = np.mean(part_u[:, 1:k+1], axis=1)
    knn_mean_v = np.mean(part_v[:, 1:k+1], axis=1)

    # Symmetrized baseline and scale reflecting local node density
    local_baseline = 0.5 * (min_dist_u + min_dist_v)
    local_scale = 0.5 * (knn_mean_u + knn_mean_v) - local_baseline + 1e-8

    # Normalized localized regret: structurally necessary long edges have low relative regret
    local_regret = np.maximum(0.0, (costs - local_baseline) / local_scale)
    regret_weights = 1.0 / (1.0 + np.exp(-local_regret))

    # Frequency-attenuated utility based on localized regret
    utilities = (costs * (1.0 + regret_weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Penalize inferior tour edges with high local regret
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Symmetrical reward-discount for promising unused candidate edges incident to penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
