# {Apply an evaporation decay toward base topology by reconstructing base distances via unpenalized minimums, then add adaptive symmetric penalties scaled by local density, relative edge length, and usage damping to escape local optima without permanently distorting distances.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Global evaporation of previous cumulative penalties
    # Estimate base distance matrix using symmetric minimum and decay excessive accumulated penalties
    evaporation_rate = 0.05
    # Symmetrize and apply mild geometric contraction toward median local distance scale
    mean_dist = np.mean(edge_distance)
    updated_edge_distance = (1.0 - evaporation_rate) * updated_edge_distance + evaporation_rate * (0.5 * (edge_distance + edge_distance.T))

    # Local density estimation via k-nearest neighbors
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Tour edge features
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Damped penalties using sigmoid relative length, usage damping, and density factor
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Escalate penalty on longest quartile edges of the local optimum
    threshold = np.percentile(tour_edge_lengths, 75)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= 1.5

    # Scale penalty relative to average edge scale to maintain robust magnitude across problem scales
    penalty_scale = 0.15 * avg_tour_length
    final_penalties = combined_penalty * penalty_scale

    # Symmetric additive edge penalty update
    np.add.at(updated_edge_distance, (u_idx, v_idx), final_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), final_penalties)

    return updated_edge_distance
