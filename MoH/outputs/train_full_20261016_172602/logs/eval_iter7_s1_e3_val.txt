# ===== Heuristic Code =====
#{An adaptive Guided Local Search distance updater combining global evaporation, sigmoid z-score edge utilities, temperature-annealed softmax rank penalties on candidate edges, and usage-scaled incident edge discounting.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    max_used = np.max(edge_n_used)
    if max_used > 0:
        evap_factor = 0.996
        updated_edge_distance = updated_edge_distance * evap_factor + edge_distance * (1.0 - evap_factor)

    base_costs = edge_distance[u_idx, v_idx].astype(np.float64)
    mod_costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(mod_costs)
    std_cost = np.std(mod_costs) + 1e-8
    z_scores = (mod_costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (base_costs * (1.0 + weights)) / (1.0 + used)

    med_util = np.median(utilities)
    penalize_mask = utilities >= med_util

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = base_costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    k = len(p_utils)
    if k > 0:
        # Annealed softmax rank weighting over penalized candidates
        order = np.argsort(np.argsort(p_utils))
        ranks = order.astype(np.float64) / max(k - 1, 1)
        temp = 0.65
        softmax_weights = np.exp(ranks / temp)
        softmax_weights /= np.mean(softmax_weights)

        fatigue = np.minimum(p_used / (np.mean(p_used) + 1.0), 3.0)
        boost = 1.0 + 0.25 * (np.exp(fatigue) - 1.0)

        penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * softmax_weights * boost
        updated_edge_distance[p_u, p_v] += penalties
        updated_edge_distance[p_v, p_u] += penalties

        penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
        decay_ratio = 0.02 + 0.015 * min(max_used / 20.0, 1.0)

        for u in penalized_nodes:
            unused_mask = (edge_n_used[u] == 0)
            unused_mask[u] = False
            if np.any(unused_mask):
                current_vals = updated_edge_distance[u, unused_mask]
                updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
                updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
