# ===== Heuristic Code =====
#{Dynamically adjust the utility penalty quantile and incident edge discount rate using normalized Shannon entropy and the variance of edge usage to aggressively disrupt stagnation while softly guiding active diversification.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Measure stagnation and diversity loss across the tour edges using usage entropy and variance
    total_used = np.sum(used)
    if total_used > 0:
        probs = used / total_used
        nonzero_p = probs[probs > 0]
        entropy = -np.sum(nonzero_p * np.log(nonzero_p))
        max_entropy = np.log(float(n))
        norm_entropy = entropy / (max_entropy + 1e-9)
    else:
        norm_entropy = 1.0

    usage_var = np.var(used)
    stagnation_factor = (1.0 - norm_entropy) * (1.0 - np.exp(-usage_var / (np.mean(used) + 1.0 + 1e-8)))
    stagnation_factor = np.clip(stagnation_factor, 0.0, 1.0)

    # Dynamic quantile: high entropy (diversifying) penalizes top ~15-20%, while stagnation penalizes up to ~65%
    cutoff_percentile = 80.0 - 45.0 * stagnation_factor

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.percentile(utilities, cutoff_percentile)
    penalize_mask = utilities >= threshold
    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.median(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    max_util = np.max(utilities) + 1e-8
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util) * (1.0 + 0.5 * stagnation_factor)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Stagnation-adaptive discount rate for unvisited alternatives
    decay_ratio = 0.015 + 0.035 * stagnation_factor
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            discounts = updated_edge_distance[u, unused_mask] * decay_ratio
            updated_edge_distance[u, unused_mask] -= discounts
            updated_edge_distance[unused_mask, u] -= discounts

    return updated_edge_distance
