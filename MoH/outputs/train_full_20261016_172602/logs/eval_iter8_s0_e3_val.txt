# ===== Heuristic Code =====
# {A modular guided local search penalty strategy blending edge and adjacent segment utilities with a dual-tier penalization scheme targeting both the peak segment and high-utility individual edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = np.asarray(local_opt_tour, dtype=int)
    v_idx = np.roll(u_idx, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Stage 1: Vectorized base utility calculation blending single-edge and 2-edge segment utilities
    base_utilities = tour_lengths / (1.0 + tour_counts)
    next_utilities = np.roll(base_utilities, -1)
    segment_utilities = 0.5 * base_utilities + 0.5 * next_utilities

    avg_edge = np.mean(tour_lengths) if tour_lengths.size > 0 else 1.0
    base_penalty = 0.3 * avg_edge

    # Stage 2: Dual-tier thresholding
    # Tier 1: Standard proportional penalty for edges exceeding threshold relative to max base utility
    max_base_util = np.max(base_utilities)
    if max_base_util > 0:
        tier1_mask = base_utilities >= (0.95 * max_base_util)
        t1_u = u_idx[tier1_mask]
        t1_v = v_idx[tier1_mask]
        updated_edge_distance[t1_u, t1_v] += base_penalty
        updated_edge_distance[t1_v, t1_u] += base_penalty

    # Tier 2: Major penalty applied to edges forming the absolute top segment
    top_seg_idx = np.argmax(segment_utilities)
    next_seg_idx = (top_seg_idx + 1) % len(u_idx)
    major_penalty = 0.5 * avg_edge

    e1_u, e1_v = u_idx[top_seg_idx], v_idx[top_seg_idx]
    e2_u, e2_v = u_idx[next_seg_idx], v_idx[next_seg_idx]

    updated_edge_distance[e1_u, e1_v] += major_penalty
    updated_edge_distance[e1_v, e1_u] += major_penalty
    updated_edge_distance[e2_u, e2_v] += major_penalty
    updated_edge_distance[e2_v, e2_u] += major_penalty

    return updated_edge_distance
