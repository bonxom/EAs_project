# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)
    p = np.roll(local_opt_tour, 1)
    w = np.roll(local_opt_tour, -2)

    tour_lens = edge_distance[u, v]
    usage = edge_n_used[u, v]

    # Efficient rank computation: how many nodes in row u are closer to u than v is?
    # edge_distance[u, :] < tour_lens[:, None] gives strict rank
    rank_u_v = np.sum(edge_distance[u, :] < tour_lens[:, None], axis=1)
    rank_v_u = np.sum(edge_distance[v, :] < tour_lens[:, None], axis=1)

    # Mutual rank metrics
    mean_rank = 0.5 * (rank_u_v + rank_v_u)
    max_rank = np.maximum(rank_u_v, rank_v_u)
    normalized_rank = (mean_rank + 0.5 * max_rank) / np.maximum(1.0, float(n))

    # Local context comparison
    local_adj_mean = 0.5 * (edge_distance[p, u] + edge_distance[v, w]) + 1e-6
    slack_ratio = tour_lens / local_adj_mean

    # Global context comparison
    mean_l = np.mean(tour_lens) + 1e-6
    std_l = np.std(tour_lens)
    cv = std_l / mean_l
    rel_global = tour_lens / mean_l

    # Usage discounting
    usage_factor = 1.0 / (1.0 + 0.35 * usage)

    # Composite penalty score combining geometric mutual rank, global relative length, and local slack
    topology_score = 0.40 * rel_global + 0.25 * slack_ratio + 0.35 * (normalized_rank * 5.0)

    cutoff_pct = np.clip(75.0 - 15.0 * cv, 50.0, 85.0)
    threshold = np.percentile(topology_score, cutoff_pct)

    penalties = topology_score * usage_factor
    boost_mask = topology_score >= threshold
    penalties[boost_mask] *= (1.25 + 0.25 * np.clip(cv, 0.0, 1.0))

    np.add.at(updated_edge_distance, (u, v), penalties)
    np.add.at(updated_edge_distance, (v, u), penalties)

    return updated_edge_distance
