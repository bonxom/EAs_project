# ===== Heuristic Code =====
#{Recombine GLS utility penalization by selecting tour edges that both exceed the average tour edge length and meet an adaptive utility threshold, penalizing them proportional to their individual lengths scaled by utilization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Adaptive filter: must be high-utility and exceed the tour's average edge length
    # If no edge strictly exceeds the average, fall back to the top utility edges
    cost_mask = tour_lengths >= avg_edge
    utility_mask = utilities >= (0.95 * max_utility)
    selected_mask = cost_mask & utility_mask

    if not np.any(selected_mask):
        selected_mask = utilities >= (0.99 * max_utility)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Scale penalty by base edge cost to avoid disproportionate distortion across heterogeneous edges
    penalties = 0.3 * tour_lengths[selected_mask] / (1.0 + 0.1 * tour_counts[selected_mask])

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
