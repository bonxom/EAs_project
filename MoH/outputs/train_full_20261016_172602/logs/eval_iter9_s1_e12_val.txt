# ===== Heuristic Code =====
#{Dual-force guided local search that evaluates path-context utilities using joint contiguous 2-edge and 3-edge tour segment costs to penalize clustered high-cost sequences while symmetrically discounting unvisited incident shortcut edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Direct edge costs and usage frequencies along the tour
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Path context evaluation: adjacent edge costs along the tour
    prev_costs = np.roll(costs, 1)
    next_costs = np.roll(costs, -1)

    # 2-edge segment context (coupling each edge with its preceding and succeeding neighbor)
    local_path_costs = costs + 0.5 * (prev_costs + next_costs)

    # Contextual z-score weighting reflecting both individual and neighborhood cost intensity
    mean_path = np.mean(local_path_costs)
    std_path = np.std(local_path_costs) + 1e-8
    z_scores = (local_path_costs - mean_path) / std_path
    context_weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Path-aware utility combining contextual segment cost, frequency, and local variance
    utilities = (local_path_costs * (1.0 + context_weights)) / (1.0 + used)

    # Identify interconnected high-cost sequences
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold
    max_util = np.max(utilities) + 1e-8

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Scaled penalties proportional to joint context intensity
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Symmetrically apply penalties to the active unfavorable sequence edges
    np.add.at(updated_edge_distance, (p_u, p_v), penalties)
    np.add.at(updated_edge_distance, (p_v, p_u), penalties)

    # Symmetrically discount promising unvisited incident edges to induce topological escape
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            discounts = current_vals * decay_ratio
            updated_edge_distance[u, unused_mask] -= discounts
            updated_edge_distance[unused_mask, u] -= discounts

    return updated_edge_distance
