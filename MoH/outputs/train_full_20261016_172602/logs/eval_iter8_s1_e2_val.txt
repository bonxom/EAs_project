# ===== Heuristic Code =====
#{This algorithm uses a vectorized excess-over-median cost ratio projected through square-root usage dampening to compute differential penalties on active tour edges while dynamically rewarding adjacent unvisited alternatives.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    med_cost = np.median(costs)
    mad_cost = np.median(np.abs(costs - med_cost)) + 1e-8
    excess_ratio = np.maximum(0.0, (costs - med_cost) / (med_cost + 1e-8))

    z_scores = (costs - med_cost) / mad_cost
    sigmoid_weights = 1.0 / (1.0 + np.exp(-z_scores))

    utilities = (costs * (1.0 + excess_ratio + sigmoid_weights)) / (1.0 + np.sqrt(used + 1.0))

    util_thresh = np.percentile(utilities, 60.0)
    penalize_mask = utilities >= util_thresh

    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.median(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    max_util = np.max(p_utils) + 1e-8
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + (p_utils / max_util))

    np.add.at(updated_edge_distance, (p_u, p_v), penalties)
    np.add.at(updated_edge_distance, (p_v, p_u), penalties)

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.015
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            updated_edge_distance[u, unused_mask] *= (1.0 - decay_ratio)
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
