# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes == 0:
        return updated_edge_distance

    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    tour_distances = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided Local Search utility calculation
    utilities = tour_distances / (1.0 + tour_counts)
    max_util = np.max(utilities)
    max_mask = np.isclose(utilities, max_util, rtol=1e-5, atol=1e-8)

    # Increment penalization frequency for selected edges
    penalized_u = u_idx[max_mask]
    penalized_v = v_idx[max_mask]
    for u, v in zip(penalized_u, penalized_v):
        edge_n_used[u, v] += 1
        edge_n_used[v, u] += 1

    # Scale sensitivity: compute tour-based reference scale in O(N)
    mean_tour_dist = np.mean(tour_distances)

    # Stagnation factor based on edge reuse along current local optimum
    overlap_ratio = np.mean(tour_counts > 0)
    stagnation_factor = np.clip(overlap_ratio * 1.5, 0.0, 1.5)

    # Sub-linear base penalty coefficient
    base_lambda = (0.15 + 0.10 * stagnation_factor) * mean_tour_dist

    # Identify non-zero penalized edges sparsely to avoid dense O(N^2) allocations
    penalized_rows, penalized_cols = np.nonzero(edge_n_used)
    if len(penalized_rows) > 0:
        counts = edge_n_used[penalized_rows, penalized_cols]
        # Sub-linear scaling using sqrt(counts) prevents catastrophic distortion of the metric space
        penalties = base_lambda * np.sqrt(counts)
        updated_edge_distance[penalized_rows, penalized_cols] += penalties

    return updated_edge_distance
