#{Accelerate topological density estimation via np.partition and apply continuous smooth edge length scaling calibrated to tour distance dispersion to avoid metric distortion.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Fast k-NN computation using np.partition instead of full sorting
    k = min(6, n - 1)
    partitioned = np.partition(edge_distance, k, axis=1)
    k_nn_avg = np.mean(partitioned[:, 1:k+1], axis=1)

    # Node density estimation
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Tour edge extraction
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths) + 1e-6

    # Continuous standardized edge prominence (replaces hard threshold clipping)
    z_scores = (tour_edge_lengths - avg_tour_length) / std_tour_length
    length_boost = 1.0 / (1.0 + np.exp(-2.5 * z_scores))

    # Frequency damping and density factors
    usage_factor = 1.0 / (1.0 + 0.35 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    # Dynamic scale calibrated to typical edge distance scale
    base_scale = 0.15 * avg_tour_length
    penalty = base_scale * length_boost * usage_factor * density_factor

    # Symmetrically update edge distances
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
