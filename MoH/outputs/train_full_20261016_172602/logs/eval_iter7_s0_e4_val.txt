# ===== Heuristic Code =====
#{A variance-adaptive Guided Local Search perturbation that selects tour edges above a dynamic dispersion threshold based on utility standard deviation and scales penalties by both edge utility dispersion and tour-normalized cost.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate utility according to Guided Local Search principles
    utilities = tour_lengths / (1.0 + tour_counts)

    max_util = np.max(utilities)
    mean_util = np.mean(utilities)
    std_util = np.std(utilities)

    # Dynamic dispersion threshold: tighter selection when variance is high, broader when landscape is homogeneous
    k_dispersion = 0.5 if std_util > 1e-8 else 0.0
    threshold = max(max_util - k_dispersion * std_util, 0.95 * max_util)

    selected_mask = utilities >= threshold

    if np.any(selected_mask):
        sel_u = u_idx[selected_mask]
        sel_v = v_idx[selected_mask]
        sel_lengths = tour_lengths[selected_mask]
        sel_counts = tour_counts[selected_mask]
        sel_utilities = utilities[selected_mask]

        # Base penalty modulated by mean edge cost and relative utility dispersion
        avg_edge = np.mean(tour_lengths)
        variance_ratio = 1.0 + (sel_utilities - mean_util) / (std_util + 1e-8)
        variance_ratio = np.clip(variance_ratio, 0.5, 2.0)

        # Penalty magnitude dynamically scaled by individual edge variance and usage count
        penalties = 0.3 * avg_edge * variance_ratio / (1.0 + 0.1 * sel_counts)

        # Symmetric penalty assignment
        updated_edge_distance[sel_u, sel_v] += penalties
        updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
