# ===== Heuristic Code =====
#{Penalize inefficient tour edges via triangle slack and turning deficits while actively discounting high-potential cross-tour nearest-neighbor bridge edges to trigger macro-structural 3-opt and 4-opt reconnects.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # 1. Decay edge usage frequency
    edge_n_used *= 0.96

    # 2. Vectorized geometry analysis along the local optimal tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Evaluate edge removal utility
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply guided penalties to high-utility edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 3. Dynamic candidate-pool injection: cross-tour nearest-neighbor bridge shortcuts
    pos = np.empty(num_nodes, dtype=np.int64)
    pos[tour] = np.arange(num_nodes)

    # Evaluate the top penalized vertices as anchors for structural reconnection
    penalized_nodes = np.unique(np.concatenate([u_idx[top_k_indices], v_idx[top_k_indices]]))
    min_separation = max(3, num_nodes // 8)

    # Identify nearest geometric neighbors that lie on distant tour segments
    num_nn = min(8, num_nodes - 1)
    for p in penalized_nodes:
        row_dist = edge_distance[p]
        nn_candidates = np.argpartition(row_dist, num_nn)[:num_nn + 1]
        p_pos = pos[p]

        for q in nn_candidates:
            if p == q:
                continue
            q_pos = pos[q]
            circ_dist = abs(p_pos - q_pos)
            circ_dist = min(circ_dist, num_nodes - circ_dist)

            # Target cross-segment bridge edges
            if circ_dist >= min_separation:
                freq = edge_n_used[p, q]
                if freq < 2.0:
                    bridge_quality = global_mean_dist / (edge_distance[p, q] + 1e-8)
                    centrality_boost = 1.0 + min(2.0, bridge_quality * 0.1)
                    discount = 0.12 * global_mean_dist * centrality_boost / (1.0 + freq)

                    new_cost = max(1e-6, updated_edge_distance[p, q] - discount)
                    updated_edge_distance[p, q] = new_cost
                    updated_edge_distance[q, p] = new_cost

    return updated_edge_distance
