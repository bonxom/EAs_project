#{Modulate Guided Local Search penalties by scaling the base step size with the ratio of total tour cost to edge length standard deviation and weighting edge penalties by each edge's marginal excess over the median tour edge length.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an explicit copy to prevent in-place modification of input
    updated_edge_distance = np.copy(edge_distance)

    # Vectorized extraction of tour directed edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Standard GLS utility metric
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Select edges within 99% of maximum utility
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    if not np.any(selected_mask):
        return updated_edge_distance

    # Instance- and state-dependent scaling calculations
    n = len(tour_lengths)
    total_length = np.sum(tour_lengths)
    mean_length = total_length / n
    std_length = np.std(tour_lengths)
    median_length = np.median(tour_lengths)

    # Scale base penalty factor dynamically using tour dispersion
    dispersion_factor = mean_length / (std_length + 1e-8)
    base_penalty = 0.3 * mean_length / (1.0 + 0.1 * np.log1p(dispersion_factor))

    # Calculate edge-specific penalty weights based on excess over median
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_lengths = tour_lengths[selected_mask]

    excess_weights = 1.0 + np.maximum(0.0, (sel_lengths - median_length) / (std_length + 1e-8))
    penalties = base_penalty * excess_weights

    # Apply symmetric updates to distance matrix
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
