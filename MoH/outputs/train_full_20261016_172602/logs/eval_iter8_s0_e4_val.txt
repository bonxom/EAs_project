# ===== Heuristic Code =====
#{Guided Local Search update penalizing bottleneck edges identified via individual and 2-edge rolling segment utilities with sub-linear frequency dampening to disrupt clustered and isolated suboptimal paths.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utilities with standard GLS formula
    utilities = tour_lengths / (1.0 + tour_counts)
    max_util = np.max(utilities)

    # Calculate higher-order 2-edge segment utilities
    segment_utilities = utilities + np.roll(utilities, -1)
    max_segment_util = np.max(segment_utilities)

    # Adaptive threshold combining individual and segment bottlenecks
    threshold = max(0.95 * max_util, max_segment_util / 2.0)
    selected_mask = utilities >= threshold

    # In case no edge meets the threshold due to floating precision, select the maximum utility edge
    if not np.any(selected_mask):
        selected_mask[np.argmax(utilities)] = True

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_counts = tour_counts[selected_mask]

    # Penalty proportional to average tour edge, dampened sub-linearly by frequency to prevent cycles
    avg_tour_edge = np.mean(tour_lengths)
    penalty = (0.3 * avg_tour_edge) / np.sqrt(1.0 + sel_counts)

    # Symmetric vectorized update
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
