#{We replace linear density averaging with a geometric-mean neighborhood scale and a non-linear tanh bridging activation to aggressively penalize bottleneck edges spanning separated clusters while preserving frequency-dampened rank penalties.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Calculate k-NN characteristic scale for each node
    k = min(6, n - 1) if n > 1 else 1
    sorted_dists = np.sort(edge_distance, axis=1)
    node_scale = np.mean(sorted_dists[:, 1:k + 1], axis=1) + 1e-6

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Global rank penalty based on tour-relative edge length
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Geometric mean cluster scale and non-linear bridging activation
    cluster_scale = np.sqrt(node_scale[u_idx] * node_scale[v_idx])
    bridging_ratio = tour_edge_lengths / cluster_scale
    bridge_penalty = np.maximum(0.0, np.tanh(bridging_ratio - 1.0))

    # Historical usage dampening
    usage_factor = 1.0 / (1.0 + 0.3 * usage)

    # Dynamic weighting balancing global relative length and cluster bridging
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_bridge = 1.0 - w_global

    combined_penalty = (w_global * rank_penalty + w_bridge * bridge_penalty * relative_length) * usage_factor

    # Top quartile amplification
    threshold = np.percentile(tour_edge_lengths, 75)
    combined_penalty[tour_edge_lengths >= threshold] *= 1.5

    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
