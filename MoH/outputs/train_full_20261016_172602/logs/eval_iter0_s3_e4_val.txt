# ===== Heuristic Code =====
#{Exploit metric TSP geometric invariants and 2-opt/3-opt replacement slack potentials by penalizing tour edges exhibiting high bottleneck length relative to local shortcut gains and triangle inequality defect while modulating by historical usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u = local_opt_tour
    v = np.roll(local_opt_tour, -1)
    p = np.roll(local_opt_tour, 1)
    w = np.roll(local_opt_tour, -2)

    # Edge costs along the tour
    d_uv = edge_distance[u, v]
    d_pu = edge_distance[p, u]
    d_vw = edge_distance[v, w]
    d_pw = edge_distance[p, w]

    # 2-opt shortcut potential: gain of bypassing u-v via direct connection p-w
    # Larger potential means edge (u, v) creates a bottleneck that is amenable to uncrossing/shortcut
    shortcut_slack = (d_pu + d_vw) - d_pw
    rel_slack = np.maximum(0.0, d_uv - shortcut_slack)

    # Metric triangle inequality slack / geometric defect around edge (u, v)
    triangle_bound = d_pu + edge_distance[p, v]
    defect = np.maximum(1e-6, triangle_bound - d_uv)

    # Local k-nearest neighbor reference to gauge edge dilation
    k = min(8, n - 1)
    k_nn_dists = np.partition(edge_distance, k, axis=1)[:, 1:k]
    node_scale = np.mean(k_nn_dists, axis=1) + 1e-6
    geom_scale = 0.5 * (node_scale[u] + node_scale[v])
    dilation = d_uv / geom_scale

    # Combined geometric and 2-opt uncrossing potential
    uncrossing_score = (d_uv / (defect + 1e-6)) * (1.0 + rel_slack / (geom_scale + 1e-6))
    score = dilation * np.sqrt(np.maximum(1e-6, uncrossing_score))

    # Historical frequency modulation (guided local search style)
    usage = edge_n_used[u, v]
    usage_factor = 1.0 / (1.0 + 0.4 * usage)
    base_penalty = score * usage_factor

    # Adaptive amplification for the most critical edges
    mean_val = np.mean(base_penalty)
    std_val = np.std(base_penalty)
    threshold = mean_val + 0.5 * std_val
    boost = np.where(base_penalty >= threshold, 1.5, 1.0)
    penalties = base_penalty * boost

    # Scale penalties relative to average edge distance to preserve optimization dynamics
    avg_tour_dist = np.mean(d_uv) + 1e-6
    norm_penalties = (penalties / (np.max(penalties) + 1e-6)) * (0.35 * avg_tour_dist)

    np.add.at(updated_edge_distance, (u, v), norm_penalties)
    np.add.at(updated_edge_distance, (v, u), norm_penalties)

    return updated_edge_distance
