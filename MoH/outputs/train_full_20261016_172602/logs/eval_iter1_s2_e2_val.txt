# ===== Heuristic Code =====
#{This algorithm enhances Guided Local Search by eliminating risky metric deflation and instead applying geometrically-weighted, chord-viability modulated penalties to high-detour edges while decaying historical usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Identify 1-hop and 2-hop sequence indices along the tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay edge usage memory to avoid permanent entrapment
    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Calculate geometric detour ratios and angular deficits
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Symmetric geometric factor accounting for incoming and outgoing tour edges
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Base GLS utility: prioritize long, under-penalized, and geometrically sharp edges
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Chord viability factor: measures how much shorter the 2-hop shortcut is relative to the path
    chord_viability = np.clip((d_uv + d_vw - d_uw) / (d_uv + d_vw + 1e-8), 0.0, 1.0)

    # Determine top-k edges with highest utility to penalize
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply scaled penalization strictly preserving metric validity (no negative distance updates)
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        chord_modulator = 1.0 + 0.5 * chord_viability[idx]

        penalty = adaptive_lambda * freq_penalty * geom_weight * chord_modulator
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    return updated_edge_distance
