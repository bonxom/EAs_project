# ===== Heuristic Code =====
#{An edge-entropy and oscillation-aware adaptive cooling memory decay replaces uniform evaporation, enabling volatile edges to retain penalties longer while cold edges rapidly clear distortion to maintain structural landscape integrity.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Edge-entropy / oscillation-based adaptive cooling memory decay:
    # High-usage edges have higher memory retention (decay closer to 0.98-0.99),
    # while low-usage/cold edges decay much faster (down to 0.88) to clear topological distortion.
    max_freq = np.max(edge_n_used)
    if max_freq > 1e-6:
        normalized_freq = edge_n_used / max_freq
        decay_matrix = 0.88 + 0.11 * (normalized_freq / (1.0 + normalized_freq))
    else:
        decay_matrix = np.full_like(edge_n_used, 0.95)

    edge_n_used *= decay_matrix

    # Local tour geometry evaluation
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # GLS edge utility ranking
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # Adaptive penalty scaling driven by tour-level dispersion and edge bottleneck severity
    tour_mean_dist = np.mean(d_uv)
    tour_std_dist = np.std(d_uv)
    adaptive_lambda_base = 0.25 * tour_mean_dist + 0.15 * tour_std_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        local_scale = max(d_uv[idx] / (tour_mean_dist + 1e-8), 0.5)

        penalty = adaptive_lambda_base * freq_penalty * geom_weight * local_scale
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Strategic shortcut attraction to encourage viable alternative connections
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * tour_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
