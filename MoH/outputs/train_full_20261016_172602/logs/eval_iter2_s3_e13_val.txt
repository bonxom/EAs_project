# ===== Heuristic Code =====
#{Scale additive penalties proportional to the average tour edge length and allocate them to top-utility edges using Voudouris & Tsang's Guided Local Search utility formula combined with local density normalization.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create a copy to prevent in-place modification of input matrix
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Extract consecutive edges in the local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)

    # Compute local neighborhood scale via k-NN distance to capture spatial clustering
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)

    # Calculate density factor: edges spanning sparse or relatively stretched regions
    local_scale = 0.5 * (k_nn_avg[u_idx] + k_nn_avg[v_idx]) + 1e-6
    relative_edge_cost = tour_edge_lengths / local_scale

    # Guided Local Search (GLS) edge utility: c_ij / (1 + p_ij), scaled by local density
    usages = edge_n_used[u_idx, v_idx]
    utilities = relative_edge_cost / (1.0 + usages)

    # Target only high-utility edges exceeding an adaptive top-quantile threshold
    threshold = np.percentile(utilities, 65.0)
    eligible_mask = utilities >= threshold

    # Calibration parameter: scale additive penalty directly by instance average edge scale (avg_tour_length / n)
    base_penalty_scale = avg_tour_length / max(n, 1)

    # Additive penalty magnitude proportional to edge utility and instance scale
    penalties = np.zeros(n, dtype=edge_distance.dtype)
    penalties[eligible_mask] = (
        base_penalty_scale * (utilities[eligible_mask] / (np.mean(utilities) + 1e-6))
    )

    # Apply symmetric additive penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
