# ===== Heuristic Code =====
#{This algorithm combines Parent A's triplet-based geometric detour and turn deficit utility scoring with top-k selection, modulated by Parent B's dual-endpoint spatial centrality and Parent C's stagnation factor to penalize persistent central bottleneck edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour, dtype=int)

    # Cyclic shift indices for triplets (u -> v -> w)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Gradual decay of historical edge usage counts
    edge_n_used *= 0.96

    # Triplet distances
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # 1. Candidate Identification via Geometric Utility
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Symmetric local geometric factor across incident edges
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (d_uv / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # 2. Candidate Selection (Top ~14% edges)
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    # 3. Magnitude Modulation components
    # Global scale
    global_mean_dist = float(np.mean(edge_distance))
    lambda_base = 0.30 * global_mean_dist

    # Stagnation factor (Parent C) based on maximum and mean tour edge usage
    max_tour_freq = float(np.max(base_freq))
    mean_tour_freq = float(np.mean(base_freq))
    stagnation_factor = 1.0 + 0.25 * np.log1p(max_tour_freq) + 0.15 * np.log1p(mean_tour_freq)

    # Spatial centrality (Parent B): nodes closer on average to all others are more central
    mean_node_dist = np.mean(edge_distance, axis=1)
    min_mnd = np.min(mean_node_dist)
    max_mnd = np.max(mean_node_dist)
    mnd_range = max_mnd - min_mnd if (max_mnd - min_mnd) > 1e-8 else 1.0
    # Invert so lower average distance yields higher centrality in [1.0, 1.5]
    node_centrality = 1.0 + 0.5 * (1.0 - (mean_node_dist - min_mnd) / mnd_range)

    # Apply modulated penalties
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        spatial_weight = 0.5 * (node_centrality[u] + node_centrality[v])

        penalty = lambda_base * stagnation_factor * spatial_weight * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Incentive mechanism: reward under-used shortcut edges bypasses high-detour vertices
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
