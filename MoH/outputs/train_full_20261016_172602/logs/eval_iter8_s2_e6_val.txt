# ===== Heuristic Code =====
#{A Guided Local Search strategy penalizing geometrically inefficient tour edges while selectively incentivizing non-tour structural bridge edges among k-nearest neighbors between distant tour segments.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create a copy of the edge distance matrix to preserve original metrics
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour, dtype=int)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay historical edge utilization counts
    edge_n_used *= 0.95

    # Compute metric geometry and detour characteristics along the tour
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Penalization utility combining length, frequency, and local geometry
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Determine penalization budget
    k_penalize = max(1, int(np.ceil(0.15 * num_nodes)))
    top_penalize_indices = np.argsort(utilities)[-k_penalize:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply penalties to selected tour edges
    for idx in top_penalize_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Map tour positions to identify topological separations along the cycle
    tour_pos = np.empty(num_nodes, dtype=int)
    tour_pos[tour] = np.arange(num_nodes)

    # Structural bridge incentive: find non-tour edges among k-nearest neighbors
    # connecting distant tour segments (topological shortcut bridges)
    k_neighbors = min(8, num_nodes - 1)
    bridge_candidates = []

    # Identify nodes associated with penalized inefficient segments
    penalized_nodes = np.unique(np.concatenate((u_idx[top_penalize_indices], v_idx[top_penalize_indices])))

    for u in penalized_nodes:
        # Candidate neighbors sorted by true distance
        sorted_neighbors = np.argsort(edge_distance[u])
        count = 0
        pos_u = tour_pos[u]
        for v in sorted_neighbors:
            if u == v:
                continue
            pos_v = tour_pos[v]
            # Graph-cycle distance along current tour
            cycle_dist = abs(pos_u - pos_v)
            cycle_dist = min(cycle_dist, num_nodes - cycle_dist)

            # Target topological shortcuts: non-adjacent in tour and topologically distant
            if cycle_dist >= 3:
                # Structural bridge score: low utilization and short physical distance
                d_val = edge_distance[u, v]
                freq = edge_n_used[u, v]
                bridge_score = (global_mean_dist / (d_val + 1e-8)) / (1.0 + freq)
                bridge_candidates.append((bridge_score, u, v, d_val, freq))
                count += 1
                if count >= k_neighbors:
                    break

    # Apply temporary bridge incentives to top candidates without metric degeneration
    if bridge_candidates:
        bridge_candidates.sort(key=lambda item: item[0], reverse=True)
        num_incentives = min(len(bridge_candidates), max(1, int(np.ceil(0.10 * num_nodes))))
        for i in range(num_incentives):
            _, u, v, d_val, freq = bridge_candidates[i]
            # Bounded discount relative to edge's baseline distance and frequency
            max_allowed_discount = 0.25 * d_val
            discount = min(max_allowed_discount, 0.08 * global_mean_dist / (1.0 + freq))
            new_cost = max(1e-6, updated_edge_distance[u, v] - discount)
            updated_edge_distance[u, v] = new_cost
            updated_edge_distance[v, u] = new_cost

    return updated_edge_distance
