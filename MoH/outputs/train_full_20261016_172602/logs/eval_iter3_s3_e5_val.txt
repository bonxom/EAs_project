# ===== Heuristic Code =====
#{A guided local search matrix perturbation heuristic that smooths penalization using dynamic variance-normalized z-scores and continuous sigmoid modulation instead of hard thresholding, scaled by k-NN density and edge usage damping.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Fast approximate k-NN density using partial sorting instead of full sort
    k = min(5, n - 1)
    k_smallest = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(k_smallest, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    density_mean = np.mean(node_density)
    node_density = node_density / (density_mean + 1e-6)

    # Tour edge characteristics
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths) + 1e-6

    # Variance-normalized z-score for smooth transition rather than abrupt quantile jumps
    z_scores = (tour_edge_lengths - avg_tour_length) / std_tour_length
    continuous_scale = 1.0 + 1.0 / (1.0 + np.exp(-2.0 * z_scores))

    # Base penalties: relative length, usage damping, and density scaling
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global

    # Combined continuous penalty scaled smoothly by variance-aware modulation
    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * continuous_scale

    # Symmetric matrix augmentation
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
