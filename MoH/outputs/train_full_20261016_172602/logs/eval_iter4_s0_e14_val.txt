#{Derives baseline unpenalized edge lengths and an edge-incident centrality factor to compute structurally grounded edge utilities, selectively applying additive penalties to maximal utility edges without compounding existing distortions.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    avg_edge = np.mean(current_lengths)
    penalty_unit = 0.3 * avg_edge

    # Reconstruct original baseline distances (D_0) by subtracting accumulated penalties
    baseline_lengths = np.maximum(current_lengths - tour_counts * penalty_unit, 0.0)

    # Calculate node-incident baseline distance sums to capture vertex centrality/dispersion
    incident_sums = np.sum(np.maximum(edge_distance - edge_n_used * penalty_unit, 0.0), axis=1)
    centrality_factor = incident_sums[u_idx] + incident_sums[v_idx]
    norm_centrality = centrality_factor / (np.mean(incident_sums) * 2.0 + 1e-12)

    # Utility mapped using baseline geometry modulated by structural centrality and usage frequency
    utilities = (baseline_lengths * norm_centrality) / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty_unit
    updated_edge_distance[sel_v, sel_u] += penalty_unit

    return updated_edge_distance
