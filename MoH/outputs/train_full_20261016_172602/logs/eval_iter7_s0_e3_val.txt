# ===== Heuristic Code =====
#{A bimodal two-tier edge penalization strategy combining sharp disruption of near-maximal utility edges with gentle landscape smoothing on remaining above-average edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]
    utilities = costs / (1.0 + counts)

    max_utility = np.max(utilities)
    avg_tour_cost = np.mean(costs)

    # Primary Tier: high utility edges causing trap in local optimum
    primary_mask = utilities >= (0.98 * max_utility)
    primary_penalty = 0.35 * avg_tour_cost

    p_u = u_idx[primary_mask]
    p_v = v_idx[primary_mask]
    updated_edge_distance[p_u, p_v] += primary_penalty
    updated_edge_distance[p_v, p_u] += primary_penalty

    # Secondary Tier: above-average edges outside the primary set for landscape smoothing
    secondary_mask = (~primary_mask) & (costs > avg_tour_cost)
    if np.any(secondary_mask):
        s_u = u_idx[secondary_mask]
        s_v = v_idx[secondary_mask]
        secondary_penalties = 0.05 * (costs[secondary_mask] / (1.0 + counts[secondary_mask]))
        updated_edge_distance[s_u, s_v] += secondary_penalties
        updated_edge_distance[s_v, s_u] += secondary_penalties

    return updated_edge_distance
