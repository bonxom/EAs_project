# ===== Heuristic Code =====
#{Re-encodes edge utility using quantile ranking fused with local k-nearest-neighbor relative dispersion to eliminate spatial density bias across heterogeneous clusters, followed by adaptive Guided Local Search penalization and unused incident edge incentive discounts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # K-nearest neighbor relative dispersion metric to handle cluster vs boundary density
    k = min(max(3, int(np.sqrt(n))), n - 1)
    sorted_u = np.partition(edge_distance[u_idx], k, axis=1)[:, 1:k+1]
    sorted_v = np.partition(edge_distance[v_idx], k, axis=1)[:, 1:k+1]
    local_scale = 0.5 * (np.mean(sorted_u, axis=1) + np.mean(sorted_v, axis=1)) + 1e-8
    relative_costs = costs / local_scale

    # Rank-based quantile normalization of tour costs and relative costs
    ranks_cost = np.argsort(np.argsort(costs)).astype(np.float64) / (n - 1 if n > 1 else 1.0)
    ranks_rel = np.argsort(np.argsort(relative_costs)).astype(np.float64) / (n - 1 if n > 1 else 1.0)
    structural_weight = 0.5 * (ranks_cost + ranks_rel)

    # Re-encoded utility incorporating structural quantile bounds and relative dispersion
    utilities = (costs * (1.0 + structural_weight)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
