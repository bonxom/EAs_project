#{This algorithm refines Parent A's dual-scale adaptive penalty framework by replacing the abrupt 75th-percentile step-function threshold with a smooth, continuous sigmoid quantile gate that softly amplifies penalties on exceptionally long edges without introducing search landscape discontinuities.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Topological node density estimation via k-nearest neighbors
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # 2. Extract edge properties along the tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # 3. Dual-scale global vs local penalty calculations
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # 4. Smooth continuous extremity modulation (replaces discontinuous 75th-percentile step boost)
    p75 = np.percentile(tour_edge_lengths, 75)
    std_length = np.std(tour_edge_lengths) + 1e-6
    # Sigmoidal amplification centered around the upper quartile
    soft_quantile_gate = 1.0 + 0.5 / (1.0 + np.exp(-3.0 * (tour_edge_lengths - p75) / std_length))
    combined_penalty = combined_penalty * soft_quantile_gate

    # 5. Symmetric update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
