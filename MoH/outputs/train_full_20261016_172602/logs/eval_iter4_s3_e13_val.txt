# ===== Heuristic Code =====
#{This algorithm replaces static threshold multiplier clipping with a continuous Pareto-quantile power scaling and adaptive Gumbel-type tail amplifier driven by tour variance to penalize outlier edges smoothly.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # 1. Local topology / density estimation via k-nearest neighbors
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Tour edge indices
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # 2. Length metrics and continuous rank calculation
    avg_tour_length = np.mean(tour_edge_lengths)
    std_tour_length = np.std(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)

    # Continuous quantile / rank representation in [0, 1]
    rank_order = np.argsort(np.argsort(tour_edge_lengths))
    normalized_rank = rank_order / max(1, n - 1)

    # 3. Base multi-factor penalties
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    base_penalty = w_global * rank_penalty + w_local * local_penalty

    # 4. Continuous Pareto and Gumbel-like tail amplification
    # Standardized edge lengths for extreme value scaling
    z_scores = (tour_edge_lengths - avg_tour_length) / std_tour_length
    tail_activation = np.exp(np.clip(z_scores - 0.5, -5.0, 3.0))
    pareto_boost = 1.0 + 0.8 * (normalized_rank ** 2.5) * (tail_activation / (1.0 + tail_activation))

    combined_penalty = base_penalty * pareto_boost

    # 5. Symmetric distance matrix update
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
