# ===== Heuristic Code =====
#{Adaptive guided local search penalty dynamically scaled by the coefficient of variation and edge dispersion of the tour, adjusted by edge usage saturation to escape local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Vectorized GLS utility evaluation based on edge cost over usage count
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Compute adaptive dispersion metrics across the tour edges
    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    range_len = np.max(tour_lengths) - np.min(tour_lengths)

    # Measure tour-level edge usage saturation to gauge search stagnation
    mean_count = np.mean(tour_counts)
    stagnation_factor = mean_count / (1.0 + mean_count)

    # Dynamically scale penalty according to edge variability and stagnation state
    dispersion_scale = (std_len + 0.1 * range_len) / (mean_len + 1e-12)
    base_factor = np.clip(0.15 + 0.35 * dispersion_scale + 0.25 * stagnation_factor, 0.15, 0.75)
    penalty = base_factor * (mean_len + std_len)

    # Identify maximum-utility edges to penalize symmetrically
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
