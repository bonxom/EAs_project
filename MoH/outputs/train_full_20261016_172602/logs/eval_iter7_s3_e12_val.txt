# ===== Heuristic Code =====
#{This strategy partitions nodes into spatial spectral clusters via diffusion coordinates on a mutual k-NN graph, detects bottleneck inter-cluster bridge edges within the tour, and applies heavy anisotropic non-linear Guided Local Search penalties to trigger macro-level topological restructuring.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # 1. Core Guided Local Search edge statistics along the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_len = np.mean(tour_edge_lengths) + 1e-8
    relative_length = tour_edge_lengths / avg_tour_len

    # 2. Spectral spatial clustering ruin operator
    # Construct mutual k-nearest neighbor adjacency
    k = min(max(3, int(np.sqrt(n))), 15)
    knn_indices = np.argsort(edge_distance, axis=1)[:, 1:k+1]

    A = np.zeros((n, n), dtype=np.float64)
    rows = np.repeat(np.arange(n), k)
    cols = knn_indices.ravel()
    A[rows, cols] = 1.0
    # Symmetrize to form mutual/shared adjacency graph
    W = np.maximum(A, A.T)

    # Normalized Laplacian eigenvectors for spectral partitioning
    degrees = np.sum(W, axis=1)
    d_inv_sqrt = 1.0 / np.sqrt(np.maximum(degrees, 1e-5))
    L_norm = np.eye(n) - (W * d_inv_sqrt[:, None]) * d_inv_sqrt[None, :]

    # Compute the Fiedler vector (2nd smallest eigenvector)
    eigvals, eigvecs = np.linalg.eigh(L_norm)
    fiedler = eigvecs[:, 1] if n > 2 else np.zeros(n)

    # Assign nodes to spatial clusters based on median split along Fiedler coordinate
    med = np.median(fiedler)
    cluster_labels = (fiedler >= med).astype(np.int32)

    # Identify bridge edges connecting distinct spatial macro-clusters
    is_bridge = cluster_labels[u_idx] != cluster_labels[v_idx]

    # 3. Dynamic anisotropic Guided Local Search penalization
    # Local density estimation via k-NN scale
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_scale = np.mean(sorted_dists[:, 1:min(6, n)], axis=1) + 1e-6
    node_density = 1.0 / k_nn_scale
    node_density = node_density / np.mean(node_density)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])

    usage_counts = edge_n_used[u_idx, v_idx]
    frequency_discount = 1.0 / (1.0 + 0.25 * usage_counts)

    # Base GLS penalty combining relative length, usage, and local density
    base_penalty = avg_tour_len * (relative_length ** 1.5) * frequency_discount * density_factor * 0.4

    # Anisotropic non-linear penalty for inter-cluster boundary bridges
    # Bridges trapped in sub-optimal local configurations are heavily penalized
    cluster_penalty_mult = np.where(is_bridge, 2.5 + 1.5 * np.tanh(relative_length), 1.0)

    # Extra boost for disproportionately long edges above 75th percentile
    upper_threshold = np.percentile(tour_edge_lengths, 75)
    outlier_mult = np.where(tour_edge_lengths >= upper_threshold, 1.4, 1.0)

    combined_penalty = base_penalty * cluster_penalty_mult * outlier_mult

    # 4. Symmetrically apply penalties to the distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
