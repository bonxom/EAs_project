# {Replace static top-k selection with a deterministic multi-criteria Pareto-rank and dispersion roulette scheme that dynamically modulates penalization cardinality and spatial clustering based on utility margin and stagnation depth.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Core Guided Local Search edge utility calculation
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Search stagnation depth estimated by frequency concentration
    mean_freq = np.mean(base_freq)
    max_freq = np.max(base_freq)
    stagnation_ratio = max_freq / (mean_freq + 1e-6)

    # Dynamic cardinality bounded adaptively between 6% and 20%
    target_k = int(np.clip(np.round(num_nodes * (0.08 + 0.04 * np.tanh(stagnation_ratio - 1.5))), 1, max(1, int(0.20 * num_nodes))))

    # Multi-criteria scoring: utility normalized against maximum, detour inefficiency, and penalization dispersion
    u_norm = utilities / (np.max(utilities) + 1e-8)
    d_norm = detour_ratio / (np.max(detour_ratio) + 1e-8)
    combined_score = 0.65 * u_norm + 0.35 * d_norm

    # Deterministic dispersion sampling: greedy selection with structural suppression of immediate tour adjacencies
    selected_indices = []
    penalized_mask = np.zeros(num_nodes, dtype=bool)
    current_scores = combined_score.copy()

    for _ in range(target_k):
        best_candidate = np.argmax(current_scores)
        if current_scores[best_candidate] <= -1e8:
            break
        selected_indices.append(best_candidate)
        penalized_mask[best_candidate] = True

        # Suppress adjacent edges along the tour cycle to prevent structural shattering
        prev_idx = (best_candidate - 1) % num_nodes
        next_idx = (best_candidate + 1) % num_nodes
        current_scores[best_candidate] = -1e9
        current_scores[prev_idx] *= 0.45
        current_scores[next_idx] *= 0.45

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = (0.25 + 0.10 * np.tanh(stagnation_ratio - 1.0)) * global_mean_dist

    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # High-detour shortcut incentivization
    k_detour = max(1, target_k // 2)
    high_detour_indices = np.argsort(detour_ratio)[-k_detour:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
