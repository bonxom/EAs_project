# ===== Heuristic Code =====
#{This algorithm penalizes high-utility tour edges and symmetrically discounts coherent triangular shortcut subpaths around penalized segments to structurally guide local search out of local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    prev_idx = np.roll(u_idx, 1)
    next_idx = np.roll(v_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    p_prev = prev_idx[penalize_mask]
    p_next = next_idx[penalize_mask]

    discount_factor = 0.025

    # Bypass subpath 1: connect predecessor across edge (prev -> v)
    d1 = updated_edge_distance[p_prev, p_v] * discount_factor / (1.0 + edge_n_used[p_prev, p_v])
    updated_edge_distance[p_prev, p_v] -= d1
    updated_edge_distance[p_v, p_prev] -= d1

    # Bypass subpath 2: connect edge origin to successor (u -> next)
    d2 = updated_edge_distance[p_u, p_next] * discount_factor / (1.0 + edge_n_used[p_u, p_next])
    updated_edge_distance[p_u, p_next] -= d2
    updated_edge_distance[p_next, p_u] -= d2

    # Direct 2-hop bridge across penalized edge (prev -> next)
    d3 = updated_edge_distance[p_prev, p_next] * (discount_factor * 0.5) / (1.0 + edge_n_used[p_prev, p_next])
    updated_edge_distance[p_prev, p_next] -= d3
    updated_edge_distance[p_next, p_prev] -= d3

    np.maximum(updated_edge_distance, 1e-6, out=updated_edge_distance)
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
