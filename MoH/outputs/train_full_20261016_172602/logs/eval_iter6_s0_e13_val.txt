# ===== Heuristic Code =====
# {Adaptive variance- and skewness-driven edge penalties dynamically scale perturbation strength based on tour dispersion, heavily penalizing extreme outlier edges while softly penalizing moderately long edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Calculate distributional metrics of tour edge lengths
    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    eps = 1e-8

    # Measure sample skewness to detect asymmetric distributions with long outlier edges
    standardized_diff = (tour_lengths - mean_len) / (std_len + eps)
    skewness = np.mean(standardized_diff ** 3)
    # Bound skewness effect smoothly to keep scaling robust across diverse topologies
    skew_factor = 1.0 + np.tanh(np.maximum(0.0, skewness))

    # Adaptive base scaling factor driven by coefficient of variation and skewness
    cv = std_len / (mean_len + eps)
    base_factor = 0.2 + 0.3 * np.tanh(cv * skew_factor)

    # Identify high-utility edges eligible for penalty
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_lengths = tour_lengths[selected_mask]

    # Modulate edge-specific penalty based on its z-score deviation relative to the mean
    z_scores = np.maximum(0.0, (sel_lengths - mean_len) / (std_len + eps))
    outlier_weights = 1.0 + np.log1p(z_scores)

    # Compute vectorized adaptive penalties
    penalties = base_factor * mean_len * outlier_weights

    # Symmetric in-place updates via NumPy advanced indexing
    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
