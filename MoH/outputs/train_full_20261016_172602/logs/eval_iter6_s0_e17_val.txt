# ===== Heuristic Code =====
#{An adaptive guided local search penalty operator penalizes all edges achieving the maximum utility scaled by tour length and instance size, with an incremental penalty on high-frequency edges to escape stagnation.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    lengths = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Standard Guided Local Search utility calculation
    utilities = lengths / (1.0 + counts)
    max_utility = np.max(utilities)

    if max_utility <= 0:
        return updated_edge_distance

    # Select edges with utility at or near the maximum
    threshold = max_utility * 0.999999
    selected_mask = utilities >= threshold

    # Standard GLS penalty parameter: alpha * TourLength / n
    tour_length = np.sum(lengths)
    alpha = 0.3
    base_penalty = alpha * (tour_length / n)

    # Apply base penalty to selected maximum utility edges
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += base_penalty
    updated_edge_distance[sel_v, sel_u] += base_penalty

    # Secondary stagnation escape: penalize the single most over-used edge if counts are high
    max_count = np.max(counts)
    if max_count >= 2:
        top_freq_idx = np.argmax(counts)
        freq_u = u_idx[top_freq_idx]
        freq_v = v_idx[top_freq_idx]
        extra_penalty = 0.5 * base_penalty
        updated_edge_distance[freq_u, freq_v] += extra_penalty
        updated_edge_distance[freq_v, freq_u] += extra_penalty

    return updated_edge_distance
