# ===== Heuristic Code =====
# {This heuristic replaces single-edge utilities with a 2-hop contextual utility combining each edge's cost with its incident tour predecessors and successors normalized by visit frequencies, selectively penalizing high-cost local detour structures.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Base edge lengths and counts along the local tour
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # 2-hop neighborhood context: adjacent tour edge lengths
    prev_lengths = np.roll(tour_lengths, 1)
    next_lengths = np.roll(tour_lengths, -1)

    # Contextual detour cost: weighted combination of edge length and incident edges
    context_cost = tour_lengths + 0.5 * (prev_lengths + next_lengths)

    # Contextual count accounting for visit frequencies of the edge and adjacent segments
    prev_counts = np.roll(tour_counts, 1)
    next_counts = np.roll(tour_counts, -1)
    effective_counts = 1.0 + tour_counts + 0.25 * (prev_counts + next_counts)

    # Contextual utilities evaluating bottleneck structures
    utilities = context_cost / effective_counts

    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
