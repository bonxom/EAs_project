#{Refine the Guided Local Search edge penalty update by properly penalizing high-utility tour edges scaled by average tour length while applying genuine multiplicative decay to accumulated penalties over previously penalized edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]
    decay_factor = 0.95

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    avg_tour_edge = np.mean(tour_lengths) if len(tour_lengths) > 0 else 1.0

    utilities = [edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in tour_edges]
    max_utility = max(utilities) if utilities else 0.0

    # Genuine decay: edge_n_used represents accumulated penalty counts; apply mild decay to penalized edges
    mask = edge_n_used > 0
    updated_edge_distance[mask] = edge_distance[mask] * decay_factor

    penalty_scale = 0.25 * avg_tour_edge
    for (u, v), util in zip(tour_edges, utilities):
        if util >= 0.85 * max_utility:
            add_penalty = penalty_scale / (1.0 + edge_n_used[u, v])
            updated_edge_distance[u, v] += add_penalty
            updated_edge_distance[v, u] += add_penalty

    return updated_edge_distance
