# ===== Heuristic Code =====
#{This algorithm combines incident node entropy-based rigidity with cost-frequency utility to selectively penalize only the top-quartile bottleneck tour edges while discounting unused incident edges to guide search toward unexplored basins.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Compute incident node usage distribution entropy to measure structural rigidity
    total_node_usage = np.sum(edge_n_used, axis=1, dtype=np.float64)
    u_total = total_node_usage[u_idx] + 1e-9
    v_total = total_node_usage[v_idx] + 1e-9

    p_u = edge_n_used[u_idx] / u_total[:, None]
    p_v = edge_n_used[v_idx] / v_total[:, None]

    # Shannon entropy of edge distribution for endpoints
    entropy_u = -np.sum(p_u * np.log(p_u + 1e-12), axis=1)
    entropy_v = -np.sum(p_v * np.log(p_v + 1e-12), axis=1)
    max_entropy = np.log(float(n - 1) + 1e-9)

    # Lower entropy implies rigid, concentrated choices; higher rigidity triggers stronger escape focus
    motif_rigidity = 1.0 - (0.5 * (entropy_u + entropy_v) / (max_entropy + 1e-9))
    motif_rigidity = np.clip(motif_rigidity, 0.0, 1.0)

    # Normalize cost with z-score weight
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Multi-factor utility combining cost, weight, usage, and structural rigidity
    utilities = (costs * (1.0 + weights) / (1.0 + used)) * (1.0 + 1.5 * motif_rigidity)

    # Selective 75th percentile cutoff to preserve good sub-tours and only penalize bottleneck edges
    cutoff = np.percentile(utilities, 75.0)
    penalize_mask = utilities >= cutoff

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_rigidity = motif_rigidity[penalize_mask]

    # Rigidity-scaled penalty calculation
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (0.5 + 0.5 * p_rigidity)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Active exploration incentive: slightly discount unused alternative edges incident to penalized nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
