# ===== Heuristic Code =====
#{Combines standardized z-score excess-cost utilities with continuous sublinear penalty scaling, paired with reciprocal k-nearest-neighbor shortcut discounts dynamically scaled by local tour density.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # 1. Non-linear utility combining standardized z-scores with excess-cost ratios over median
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    med_cost = np.median(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    excess_ratio = np.maximum(0.0, (costs - med_cost) / med_cost)

    # Sigmoidal non-linear weight on relative cost significance
    sig_weight = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + sig_weight) * (1.0 + excess_ratio)) / (1.0 + used)

    # 2. Continuous soft-threshold penalties scaled sublinearly (eliminating hard cutoffs)
    util_median = np.median(utilities)
    util_iqr = np.percentile(utilities, 75) - np.percentile(utilities, 25) + 1e-8
    # Smooth activation: softplus-like continuous scaling above base threshold
    soft_activation = 1.0 / (1.0 + np.exp(-(utilities - util_median) / util_iqr))

    # Sublinear penalty scaling based on edge usage and continuous utility strength
    penalties = (costs / (1.0 + np.sqrt(used + 1.0))) * soft_activation

    # Apply penalties symmetrically
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    # 3. Dynamic reciprocal k-NN shortcutting scaled by local tour density
    # Local density estimation: average incident edge cost on tour for each node
    prev_idx = np.roll(u_idx, 1)
    incident_costs = 0.5 * (edge_distance[prev_idx, u_idx] + edge_distance[u_idx, v_idx])
    density_map = np.zeros(updated_edge_distance.shape[0], dtype=np.float64)
    density_map[u_idx] = incident_costs
    avg_tour_density = np.mean(incident_costs) + 1e-8

    # Focus shortcutting around nodes affected by significant penalties
    active_mask = soft_activation >= 0.5
    candidate_nodes = np.unique(np.concatenate((u_idx[active_mask], v_idx[active_mask])))

    k = min(5, updated_edge_distance.shape[0] - 1)
    if k > 0 and len(candidate_nodes) > 0:
        # Precompute k-NN for candidate nodes based on base edge distances (excluding self)
        node_sub = candidate_nodes
        dist_sub = np.copy(edge_distance[node_sub, :])
        dist_sub[:, node_sub] = np.inf  # avoid self or identical node masking issues
        np.fill_diagonal(dist_sub[:, node_sub], np.inf)

        # Find top-k nearest neighbors for candidates
        knn_indices = np.argpartition(dist_sub, k, axis=1)[:, :k]

        # Check reciprocal k-NN conditions efficiently
        for i_idx, u in enumerate(node_sub):
            u_knn = knn_indices[i_idx]
            # Local density scaling: sparse regions (high incident cost) get larger discounts
            rel_sparsity = np.clip(density_map[u] / avg_tour_density, 0.5, 2.0)
            base_decay = 0.02 * rel_sparsity

            for v in u_knn:
                if u == v or edge_n_used[u, v] > 0:
                    continue
                # Reciprocal check: is u also within top-k of v?
                row_v = edge_distance[v, :].copy()
                row_v[v] = np.inf
                v_knn = np.argpartition(row_v, k)[:k]
                if u in v_knn:
                    discount = updated_edge_distance[u, v] * base_decay
                    updated_edge_distance[u, v] -= discount
                    updated_edge_distance[v, u] -= discount

    return updated_edge_distance
