# ===== Heuristic Code =====
#{Spatially-biased memory evaporation scales edge frequency retention according to incident node centrality and congestion, shielding central hubs from cycling while rapidly clearing peripheral edges, combined with geometric Guided Local Search penalties and high-detour shortcut promotions.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # 1. Spatially-biased non-uniform frequency evaporation
    # Compute node centrality: mean proximity to all other nodes (lower average distance = higher centrality)
    avg_node_dist = np.mean(edge_distance, axis=1)
    mean_dist_all = np.mean(avg_node_dist)
    # Proximity centrality score in [0, 1]: hubs closer to center have higher values
    closeness = 1.0 / (1.0 + (avg_node_dist / (mean_dist_all + 1e-8)))

    # Compute node congestion from incident search frequency
    incident_freq = np.sum(edge_n_used, axis=1) / 2.0
    max_incident = np.max(incident_freq) + 1e-8
    congestion = incident_freq / max_incident

    # Node retention factor: hubs and congested nodes retain memory longer (up to ~0.98), peripheral nodes evaporate faster (~0.90)
    node_retention = 0.90 + 0.08 * (0.6 * closeness + 0.4 * congestion)

    # Edge retention matrix via outer mean
    retention_matrix = 0.5 * (node_retention[:, None] + node_retention[None, :])
    edge_n_used *= retention_matrix

    # 2. Geometric inefficiency & GLS utility computation along the local optimal tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Utility incorporating edge distance, usage frequency, and geometric shape
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # 3. Penalize top-k edges with high utility
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]
    adaptive_lambda = 0.30 * mean_dist_all

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 4. Shortcut promotion for high detour transitions
    high_detour_indices = np.argsort(detour_ratio)[-k:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * mean_dist_all * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
