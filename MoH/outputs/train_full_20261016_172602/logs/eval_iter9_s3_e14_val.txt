#{Replace static k-NN density with Gabriel-graph-inspired mutual proximity and topological bottleneck estimation to strongly penalize inter-cluster bridge edges while shielding tightly-coupled structural clusters.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-8
    relative_length = tour_edge_lengths / avg_tour_length

    # Topology-aware Gabriel / Relative Neighborhood mutual proximity:
    # Estimate the empty-sphere / empty-lune violation count in the local neighborhood
    # A true Gabriel edge (u, v) satisfies d(u, w)^2 + d(v, w)^2 >= d(u, v)^2 for all w.
    # An edge with many intermediate nodes violating this condition is a bridge crossing clusters.
    k_cand = min(12, n)
    # Get indices of nearest neighbors for each node to evaluate mutual structural enclosure efficiently
    nn_indices = np.argpartition(edge_distance, k_cand, axis=1)[:, :k_cand]

    d_uv = tour_edge_lengths
    d_uv_sq = d_uv ** 2

    # Compute cluster separation factor for tour edges:
    # Count how many shared/local neighbor nodes violate the Gabriel property
    # Sample candidate intermediate nodes from the union of top neighbors of u and v
    gabriel_violations = np.zeros(n, dtype=np.float64)
    for i in range(n):
        u = u_idx[i]
        v = v_idx[i]
        cand_nodes = np.unique(np.concatenate((nn_indices[u], nn_indices[v])))
        cand_nodes = cand_nodes[(cand_nodes != u) & (cand_nodes != v)]
        if len(cand_nodes) > 0:
            d_uw_sq = edge_distance[u, cand_nodes] ** 2
            d_vw_sq = edge_distance[v, cand_nodes] ** 2
            # Violation when cand node falls within the diametral ball of (u, v)
            violations = np.sum((d_uw_sq + d_vw_sq) < d_uv_sq[i])
            gabriel_violations[i] = violations / len(cand_nodes)

    # Bridge edges cross clusters: high Gabriel violation ratio and longer than local scales
    bridge_factor = 1.0 + 2.0 * gabriel_violations

    # Mutual proximity density: harmonic mean of local Gabriel/RNG neighbor distances
    # Nodes in dense clusters have small Gabriel margins
    sorted_dists = np.sort(edge_distance, axis=1)
    k_sub = min(5, n - 1)
    local_scale = np.mean(sorted_dists[:, 1:k_sub + 1], axis=1)
    mutual_scale = np.sqrt(local_scale[u_idx] * local_scale[v_idx]) + 1e-8
    dilation_ratio = d_uv / mutual_scale

    # GLS rank and frequency attenuation
    rank_penalty = 1.0 / (1.0 + np.exp(-3.5 * (relative_length - 1.0)))
    usage_factor = 1.0 / (1.0 + 0.35 * edge_n_used[u_idx, v_idx])

    # Combined topological penalty: prioritize dilating bridge edges while protecting tight bottlenecks
    topo_penalty = rank_penalty * usage_factor * bridge_factor * np.log1p(dilation_ratio)

    # Scale penalty proportionally to average tour length to maintain consistent impact
    penalty_magnitude = topo_penalty * (0.15 * avg_tour_length)

    # Extra boost for long, high-violation edges in upper quartile
    threshold = np.percentile(tour_edge_lengths, 75)
    boost_mask = tour_edge_lengths >= threshold
    penalty_magnitude[boost_mask] *= 1.4

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty_magnitude)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty_magnitude)

    return updated_edge_distance
