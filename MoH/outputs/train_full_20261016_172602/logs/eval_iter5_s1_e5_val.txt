# ===== Heuristic Code =====
#{Dynamically adjusts the GLS edge penalization threshold via utility skewness and applies rank-proportional cost discounts to incident unused edges to escape local optima efficiently.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Skewness-based adaptive percentile threshold
    mean_util = np.mean(utilities)
    std_util = np.std(utilities) + 1e-8
    skewness = np.mean(((utilities - mean_util) / std_util) ** 3)
    target_percentile = np.clip(50.0 + skewness * 10.0, 50.0, 85.0)
    threshold = np.percentile(utilities, target_percentile)

    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    # Fallback if threshold filters out all edges
    if not np.any(penalize_mask):
        penalize_mask = utilities >= np.median(utilities)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # Rank-proportional discount for incident unused edges
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        k = np.count_nonzero(unused_mask)
        if k > 0:
            indices = np.where(unused_mask)[0]
            candidate_costs = edge_distance[u, indices]
            # Rank edges ascending by base cost (shortest edges get highest discount)
            rank_order = np.argsort(candidate_costs)
            ranks = np.empty_like(rank_order)
            ranks[rank_order] = np.arange(k)
            # Normalized rank weight: 1.0 for shortest, down to 0.0 for longest
            rank_weights = 1.0 - (ranks / max(1.0, float(k - 1)))
            discount_rates = 0.005 + 0.035 * rank_weights

            discounts = updated_edge_distance[u, indices] * discount_rates
            updated_edge_distance[u, indices] -= discounts
            updated_edge_distance[indices, u] = updated_edge_distance[u, indices]

    return updated_edge_distance
