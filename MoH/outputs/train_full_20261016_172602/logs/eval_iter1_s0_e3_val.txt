# ===== Heuristic Code =====
#{This algorithm enhances Guided Local Search by incorporating an exponential memory decay on historically penalized edges before symmetrically applying new penalties to maximum-utility edges of the local optimum, preventing metric distortion and premature convergence.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to avoid modifying the input in place
    updated_edge_distance = np.copy(edge_distance)

    # Identify edges belonging to the current local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Calculate edge lengths and utilization counts
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Apply an exponential memory decay on previously penalized edges to preserve metric structure
    decay_factor = 0.98
    active_mask = edge_n_used > 0
    updated_edge_distance[active_mask] = (
        decay_factor * updated_edge_distance[active_mask] + (1.0 - decay_factor) * edge_distance[active_mask]
    )

    # Vectorized computation of Guided Local Search utilities
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Calculate deterministic penalty proportional to average tour edge length
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Select edges achieving near-maximum utility to penalize
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically increment distances for the targeted bottleneck edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
