# ===== Heuristic Code =====
# {Guided local search penalizes high-utility tour edges scaled by cost and frequency while applying a directional regret-based discount strictly targeting the top-k nearest unvisited neighbors of penalized nodes to focus exploration without distorting the global metric space.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    k_neighbors = min(5, max(1, n - 2))
    base_discount = 0.05

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        candidates = np.where(unused_mask)[0]

        if len(candidates) == 0:
            continue

        cand_costs = edge_distance[u, candidates]
        num_k = min(k_neighbors, len(candidates))
        best_k_idx = np.argpartition(cand_costs, num_k - 1)[:num_k]
        top_candidates = candidates[best_k_idx]
        sorted_order = np.argsort(cand_costs[best_k_idx])
        top_candidates = top_candidates[sorted_order]
        top_costs = edge_distance[u, top_candidates]

        c_min = top_costs[0]
        c_ref = top_costs[-1] if num_k > 1 else (c_min * 1.5 + 1e-8)
        regret_denom = (c_ref - c_min) + 1e-8

        regret_weights = (c_ref - top_costs) / regret_denom
        discounts = base_discount * (1.0 + regret_weights) * 0.5 * top_costs

        for v, disc in zip(top_candidates, discounts):
            new_val = max(1e-6, updated_edge_distance[u, v] - disc)
            updated_edge_distance[u, v] = new_val
            updated_edge_distance[v, u] = new_val

    return updated_edge_distance
