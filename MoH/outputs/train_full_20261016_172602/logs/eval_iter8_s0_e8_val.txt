# ===== Heuristic Code =====
#{Cooperative multi-trajectory guided penalization with path-relinking backbone preservation, dynamically penalizing non-backbone high-utility edges via restricted candidate-set frequency differentials.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Clone distance matrix to guarantee pure side-effect-free execution
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    if n <= 3:
        return updated_edge_distance

    # Step 2: Extract current tour directed and symmetric edges
    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Step 3: Compute candidate edge utilities based on cost and usage frequency
    utilities = tour_lengths / (1.0 + tour_counts)

    # Step 4: Identify pseudo-backbone edges via frequency differential and low relative utility
    mean_count = np.mean(tour_counts)
    std_count = np.std(tour_counts)
    backbone_threshold = mean_count + 0.5 * std_count

    # Non-backbone edges are prone to ruin-and-relink perturbation
    non_backbone_mask = tour_counts <= backbone_threshold

    # Restrict candidate set to high-utility non-backbone segments
    if np.any(non_backbone_mask):
        candidate_utilities = utilities[non_backbone_mask]
        max_u = np.max(candidate_utilities)
        selected_mask = non_backbone_mask & (utilities >= 0.85 * max_u)
    else:
        max_u = np.max(utilities)
        selected_mask = utilities >= 0.90 * max_u

    # Step 5: Compute adaptive dynamic penalty modulated by graph dispersion
    avg_tour_edge = np.mean(tour_lengths)
    dynamic_scale = 0.35 + 0.15 * (np.max(tour_lengths) / (avg_tour_edge + 1e-12))
    dynamic_scale = min(max(dynamic_scale, 0.2), 0.8)
    base_penalty = dynamic_scale * avg_tour_edge

    # Step 6: Vectorized modulated penalization on both symmetric matrix entries
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Modulate individual edge penalty by its relative utility ranking
    rel_weights = utilities[selected_mask] / (max_u + 1e-12)
    penalties = base_penalty * (0.8 + 0.4 * rel_weights)

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
