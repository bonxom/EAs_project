# ===== Heuristic Code =====
# {This algorithm combines cost z-scores, edge usage counts, and incident node entropy rigidity into a unified utility metric, applying continuous sigmoid soft-gating across all tour edges to calculate smooth additive penalties and proportional reciprocal decay on untraversed incident edges without hard thresholding.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # 1. Edge cost z-score activation
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost

    # 2. Incident node usage entropy (motif rigidity)
    total_node_usage = np.sum(edge_n_used, axis=1, dtype=np.float64)
    u_total = total_node_usage[u_idx] + 1e-9
    v_total = total_node_usage[v_idx] + 1e-9

    p_u = edge_n_used[u_idx] / u_total[:, None]
    p_v = edge_n_used[v_idx] / v_total[:, None]

    log_p_u = np.log(p_u + 1e-12)
    log_p_v = np.log(p_v + 1e-12)

    entropy_u = -np.sum(p_u * log_p_u, axis=1)
    entropy_v = -np.sum(p_v * log_p_v, axis=1)
    max_entropy = np.log(n - 1.0 + 1e-9)
    motif_rigidity = 1.0 - (0.5 * (entropy_u + entropy_v) / (max_entropy + 1e-9))
    motif_rigidity = np.clip(motif_rigidity, 0.0, 1.0)

    # 3. Unified utility scoring
    base_utility = (costs / (1.0 + np.sqrt(used + 1.0))) * (1.0 + 1.5 * motif_rigidity)
    util_mean = np.mean(base_utility)
    util_std = np.std(base_utility) + 1e-8
    norm_util = (base_utility - util_mean) / util_std

    # 4. Continuous soft-threshold sigmoid gating (eliminates hard cutoffs)
    temperature = 0.8
    activation = 1.0 / (1.0 + np.exp(-norm_util / temperature))

    # 5. Continuous additive edge penalties
    penalties = costs * activation * (0.2 + 0.4 * motif_rigidity + 0.2 / (1.0 + used))
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    # 6. Smooth localized reciprocal edge decay on unused incident edges
    node_decay_intensity = np.zeros(n, dtype=np.float64)
    np.add.at(node_decay_intensity, u_idx, activation * 0.015)
    np.add.at(node_decay_intensity, v_idx, activation * 0.015)

    active_nodes = np.where(node_decay_intensity > 0.005)[0]
    for u in active_nodes:
        decay_factor = min(node_decay_intensity[u], 0.05)
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            decay = updated_edge_distance[u, unused_mask] * decay_factor
            updated_edge_distance[u, unused_mask] -= decay
            updated_edge_distance[unused_mask, u] -= decay

    return updated_edge_distance
