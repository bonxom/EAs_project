# ===== Heuristic Code =====
# {This heuristic integrates vectorized edge selection and average tour length scaling with a continuous relative-utility power-law penalty, penalizing candidate edges smoothly in proportion to their relative utility above an activation threshold to avoid rigid cutoffs while focusing penalties on near-maximal utility edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utilities: longer edges with lower usage yield higher utility for penalization
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility > 0:
        avg_edge = np.mean(tour_lengths)
        base_penalty = 0.3 * avg_edge

        # Continuous relative-utility scaling for edges meeting a flexible activation baseline
        activation_threshold = 0.85
        relative_utility = utilities / max_utility
        active_mask = relative_utility >= activation_threshold

        if np.any(active_mask):
            sel_u = u_idx[active_mask]
            sel_v = v_idx[active_mask]
            sel_rel_util = relative_utility[active_mask]
            sel_counts = tour_counts[active_mask]

            # Smooth power-law weighting focused on near-maximal edges, attenuated by historical frequency
            weights = ((sel_rel_util - activation_threshold) / (1.0 - activation_threshold)) ** 2
            penalties = (base_penalty / (1.0 + sel_counts)) * weights

            # Apply symmetric penalty updates
            np.add.at(updated_edge_distance, (sel_u, sel_v), penalties)
            np.add.at(updated_edge_distance, (sel_v, sel_u), penalties)

    return updated_edge_distance
