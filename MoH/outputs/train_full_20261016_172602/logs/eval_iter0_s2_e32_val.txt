# ===== Heuristic Code =====
#{Dynamic distribution-aware thresholding identifies high-stress tour segments via robust z-score filtering, dynamically penalizing contiguous clusters of geometric inefficiency while selectively rewarding viable shortcuts.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Dynamic distribution-aware thresholding using robust statistics
    util_median = np.median(utilities)
    util_mad = np.median(np.abs(utilities - util_median)) + 1e-8
    util_zscores = (utilities - util_median) / (1.4826 * util_mad)

    # Adaptive threshold scaling with tour heterogeneity
    heterogeneity = np.std(utilities) / (np.mean(utilities) + 1e-8)
    z_cutoff = max(0.5, 1.2 - 0.4 * np.tanh(heterogeneity))

    # Candidate selection and contiguous stress expansion
    high_stress_mask = util_zscores >= z_cutoff
    if not np.any(high_stress_mask):
        high_stress_mask = utilities >= np.percentile(utilities, 85.0)

    # Expand contiguous high-stress subpaths to reconstruct entire poorly routed clusters
    clustered_stress_mask = high_stress_mask | (np.roll(high_stress_mask, 1) & (util_zscores >= 0.2)) | (np.roll(high_stress_mask, -1) & (util_zscores >= 0.2))

    # Bounded selection size (between 5% and 25% of the tour)
    selected_indices = np.where(clustered_stress_mask)[0]
    min_k = max(1, int(0.05 * num_nodes))
    max_k = max(min_k, int(0.25 * num_nodes))
    if len(selected_indices) < min_k:
        selected_indices = np.argsort(utilities)[-min_k:]
    elif len(selected_indices) > max_k:
        top_subset = np.argsort(utilities[selected_indices])[-max_k:]
        selected_indices = selected_indices[top_subset]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in selected_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Dynamically select shortcut candidates matching cluster distribution
    k_shortcuts = len(selected_indices)
    high_detour_indices = np.argsort(detour_ratio)[-k_shortcuts:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
