# ===== Heuristic Code =====
#{We replace the static 75th-percentile multiplier with an Extreme Value Theory tail surge that dynamically scales penalties above the 75th percentile using standardized interquartile tail dispersion to aggressively penalize extreme outlier edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # K-NN local density estimation
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Extract tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Global logistic rank penalty
    avg_tour_length = np.mean(tour_edge_lengths)
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-6)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Local penalty component with density and usage frequency decay
    usage_factor = 1.0 / (1.0 + 0.3 * edge_n_used[u_idx, v_idx])
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Adaptive global-local balance based on search history
    freq_mean = np.mean(edge_n_used[u_idx, v_idx])
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # EVT-based generalized upper tail scaling replacing static multiplier
    p25 = np.percentile(tour_edge_lengths, 25)
    p75 = np.percentile(tour_edge_lengths, 75)
    iqr = np.maximum(p75 - p25, 1e-6)

    # Standardized excess over the 75th percentile threshold
    tail_excess = np.maximum(0.0, (tour_edge_lengths - p75) / iqr)
    # Exponential super-linear surge for extreme outliers with base scaling
    tail_multiplier = np.where(tour_edge_lengths >= p75, 1.25 + 0.75 * np.expm1(np.minimum(tail_excess, 3.0)), 1.0)
    combined_penalty *= tail_multiplier

    # Apply symmetric penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
