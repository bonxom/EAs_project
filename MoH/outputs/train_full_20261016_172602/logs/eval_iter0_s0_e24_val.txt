# ===== Heuristic Code =====
#{Guided Local Search with Incident Node Penalty Dilution: Identify maximum-utility bottleneck edges in the tour, penalize them directly, and dilute a fraction of the penalty across incident edges connected to their endpoint nodes to prevent cheap adjacent detour cycling.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to avoid modifying original array
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge endpoints along the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Compute edge utilities using standard guided local search formulation
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    # Determine bottleneck edges exceeding utility threshold
    max_utility = np.max(utilities)
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Base penalty derived from the average edge length of the tour
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Dilution factor: direct penalty on the bottleneck edge and distributed penalty on incident edges
    direct_weight = 0.8
    incident_weight = 0.1
    direct_penalty = base_penalty * direct_weight
    incident_penalty = base_penalty * incident_weight

    # Apply direct penalty to selected bottleneck edges
    updated_edge_distance[sel_u, sel_v] += direct_penalty
    updated_edge_distance[sel_v, sel_u] += direct_penalty

    # Distribute penalty to incident edges of the endpoint nodes to discourage adjacent detour loops
    unique_incident_nodes = np.unique(np.concatenate([sel_u, sel_v]))
    updated_edge_distance[unique_incident_nodes, :] += incident_penalty
    updated_edge_distance[:, unique_incident_nodes] += incident_penalty

    # Reset diagonal distances to 0.0
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
