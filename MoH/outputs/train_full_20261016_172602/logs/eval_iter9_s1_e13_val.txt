# ===== Heuristic Code =====
# {Replace static median thresholding with an adaptive quantile-annealed Pareto selection where edge stagnation and cost leverage dynamically contract the penalty set to surgical bottleneck edges while scaling incentive discounts on incident unvisited paths.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Compute cost and frequency statistics
    c_min, c_max = np.min(costs), np.max(costs)
    cost_span = (c_max - c_min) if (c_max - c_min) > 1e-8 else 1.0
    norm_cost = (costs - c_min) / cost_span

    u_max = np.max(used)
    norm_used = used / (u_max if u_max > 0 else 1.0)

    # Dynamic quantile annealing: deeper search stagnation (high max usage) sharpens the cutoff
    # High stagnation contracts the selection to the top high-leverage bottlenecks (70% - 90% quantile)
    stagnation_ratio = float(np.tanh(u_max / 10.0))
    quantile_target = 0.50 + 0.40 * stagnation_ratio

    # Dual-factor utility: high cost and high persistence under local moves
    utilities = (costs * (1.0 + norm_cost)) / (1.0 + np.log1p(used))

    # Pareto-dominance score: combination of normalized edge cost and recurrence
    pareto_leverage = 0.6 * norm_cost + 0.4 * norm_used
    joint_score = utilities * (1.0 + pareto_leverage)

    cutoff = np.quantile(joint_score, quantile_target)
    penalize_mask = joint_score >= cutoff

    # Fallback to at least the single maximum edge if quantile selects empty set
    if not np.any(penalize_mask):
        penalize_mask[np.argmax(joint_score)] = True

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_scores = joint_score[penalize_mask]
    max_score = np.max(p_scores) + 1e-8

    # Apply adaptive penalties to bottleneck edges
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_scores / max_score)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Selective incentive discounting for incident unvisited edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02 * (1.0 + stagnation_ratio)

    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
