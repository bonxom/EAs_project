# ===== Heuristic Code =====
#{A surgical GLS update that combines geometric detour analysis with thresholded elitist utility penalization (penalizing only edges within 85% of maximum utility, capped at 4% of the tour) and targeted shortcut discounting to perturb local optima without destroying tour backbone structure.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    if num_nodes < 4:
        return updated_edge_distance

    tour = np.asarray(local_opt_tour)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Decay historical edge usage trace symmetrically
    edge_n_used *= 0.96

    # Edge distances along consecutive triplets (u -> v -> w)
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    # Geometric detour calculation and triangle turn-angle deficit
    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / denom, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))

    # Classical GLS utility augmented by local geometric distortion
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Elitist thresholding: target only edges within 85% of maximum utility, bounded between 1 and 4% of nodes
    max_util = np.max(utilities)
    threshold = 0.85 * max_util
    candidate_indices = np.where(utilities >= threshold)[0]

    max_k = max(1, int(np.ceil(0.04 * num_nodes)))
    if len(candidate_indices) > max_k:
        candidate_indices = candidate_indices[np.argsort(utilities[candidate_indices])[-max_k:]]

    # Global scaling factor for additive penalties
    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply surgical penalties to selected high-utility bottleneck edges
    for idx in candidate_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Incentivize natural 2-opt shortcut chords with extreme detours
    k_shortcuts = max(1, int(np.ceil(0.03 * num_nodes)))
    high_detour_indices = np.argsort(detour_ratio)[-k_shortcuts:]
    for idx in high_detour_indices:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            new_val = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[u, w] = new_val
            updated_edge_distance[w, u] = new_val

    return updated_edge_distance
