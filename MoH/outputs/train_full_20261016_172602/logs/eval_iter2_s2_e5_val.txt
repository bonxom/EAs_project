# ===== Heuristic Code =====
#{Evaluates multi-scale k-hop chains (k in {2, 3, 4}) to identify topologically contorted subpaths against direct chord spans, applying correlated block-level penalties to entangled segments and reciprocal shortcut incentives to provoke macro-reconfigurations.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour, dtype=np.intp)

    # Smooth decay of historical usage
    edge_n_used *= 0.95
    global_mean_dist = float(np.mean(edge_distance))

    # Edge vectors along the tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    e_dist = edge_distance[u_idx, v_idx]
    e_freq = edge_n_used[u_idx, v_idx]

    # Base GLS utility for individual edges
    base_edge_utility = e_dist / (1.0 + e_freq)

    # Multi-scale subpath analysis for chain lengths L in {2, 3, 4}
    # Accumulate contortion strain and identify candidate chords for shortcuts
    edge_contortion = np.zeros(num_nodes, dtype=np.float64)
    best_shortcut_gain = np.zeros((num_nodes, 5), dtype=np.float64)

    for L in (2, 3, 4):
        if L >= num_nodes:
            continue
        # Direct chord between start of chain and end of chain
        end_idx = np.roll(tour, -L)
        chord_dist = edge_distance[u_idx, end_idx]

        # Cumulative path distance along the L hops
        cum_path_dist = np.zeros(num_nodes, dtype=np.float64)
        for step in range(L):
            step_u = np.roll(tour, -step)
            step_v = np.roll(tour, -(step + 1))
            cum_path_dist += edge_distance[step_u, step_v]

        chord_ratio = cum_path_dist / (chord_dist + 1e-8)
        chain_strain = np.maximum(0.0, chord_ratio - 1.0) / float(L)

        # Distribute chain contortion across the constituent edges
        for step in range(L):
            edge_contortion += np.roll(chain_strain, step)

        best_shortcut_gain[:, L] = chord_ratio

    # Integrated block utility incorporating multi-scale topological inefficiency
    block_utility = base_edge_utility * (1.0 + 0.40 * edge_contortion)

    # Select top contorted subpath seeds for block penalization
    num_seeds = max(1, int(np.ceil(0.12 * num_nodes)))
    seed_indices = np.argsort(block_utility)[-num_seeds:]

    adaptive_lambda = 0.32 * global_mean_dist

    # Correlated block penalization: penalize seed edges and correlated contiguous neighbors
    penalized_mask = np.zeros(num_nodes, dtype=bool)
    for seed in seed_indices:
        # Penalize a coherent 2-edge block centered on the seed
        for offset in (0, 1):
            idx = (seed + offset) % num_nodes
            u, v = u_idx[idx], v_idx[idx]
            edge_n_used[u, v] += 1.0
            edge_n_used[v, u] += 1.0

            freq_factor = 1.0 + np.log1p(edge_n_used[u, v])
            strain_weight = 1.0 + 0.30 * edge_contortion[idx]
            penalty = adaptive_lambda * freq_factor * strain_weight

            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty
            penalized_mask[idx] = True

    # Multi-scale shortcut incentive: discount prominent chords bridging highly contorted blocks
    num_shortcuts = max(1, int(np.ceil(0.10 * num_nodes)))
    max_chord_ratios = np.max(best_shortcut_gain[:, 2:5], axis=1)
    best_hops = np.argmax(best_shortcut_gain[:, 2:5], axis=1) + 2
    shortcut_candidates = np.argsort(max_chord_ratios)[-num_shortcuts:]

    for idx in shortcut_candidates:
        if max_chord_ratios[idx] > 1.25:
            L = best_hops[idx]
            u = u_idx[idx]
            w = np.roll(tour, -L)[idx]
            freq = edge_n_used[u, w]
            if freq < 1.0:
                discount = 0.12 * global_mean_dist * (1.0 / (1.0 + freq))
                new_dist = max(1e-6, updated_edge_distance[u, w] - discount)
                updated_edge_distance[u, w] = new_dist
                updated_edge_distance[w, u] = new_dist

    return updated_edge_distance
