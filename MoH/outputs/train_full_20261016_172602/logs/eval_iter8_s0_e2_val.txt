# ===== Heuristic Code =====
#{Vectorized evaluation of single-edge and two-edge segment utilities with count-dampened penalty scaling to disrupt persistent subpaths without over-penalizing recurrent transitions.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # 1-edge utilities
    edge_utilities = tour_lengths / (1.0 + tour_counts)

    # 2-edge connected segment utilities
    seg_lengths = tour_lengths + np.roll(tour_lengths, -1)
    seg_counts = tour_counts + np.roll(tour_counts, -1)
    seg_utilities = seg_lengths / (2.0 + seg_counts)

    max_edge_u = np.max(edge_utilities)
    max_seg_u = np.max(seg_utilities)

    # Identify high utility edges and segments
    edge_mask = edge_utilities >= (0.95 * max_edge_u)
    seg_mask = seg_utilities >= (0.95 * max_seg_u)

    # Edges in top 2-edge segments
    seg_edge_mask = seg_mask | np.roll(seg_mask, 1)

    # Combined target mask
    target_mask = edge_mask | seg_edge_mask

    if not np.any(target_mask):
        target_mask = edge_utilities == max_edge_u

    avg_edge = np.mean(tour_lengths)

    # Dampened penalty inversely proportional to usage count
    penalties = (0.35 * avg_edge) / (1.0 + 0.5 * tour_counts[target_mask])

    sel_u = u_idx[target_mask]
    sel_v = v_idx[target_mask]

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
