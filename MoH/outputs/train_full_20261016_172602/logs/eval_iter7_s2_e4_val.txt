# ===== Heuristic Code =====
#{A bilateral curvature-informed basin tunneling heuristic that penalizes high-torsion, high-entropy tour edges using 4-node discrete geometric curvature while carving proactive distance contractions on promising unvisited shortcuts to escape local optima.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Copy distance matrix and decay historical edge frequencies
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)
    edge_n_used *= 0.955

    # Step 2: Establish 4-node consecutive sliding windows (t -> u -> v -> w)
    t_idx = np.roll(tour, 1)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Step 3: Compute discrete metrics and bilateral geometric torsion across windows
    d_tu = edge_distance[t_idx, u_idx]
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_tv = edge_distance[t_idx, v_idx]
    d_uw = edge_distance[u_idx, w_idx]
    d_tw = edge_distance[t_idx, w_idx]

    # Bilateral detour inefficiencies around edge (u, v)
    detour_left = (d_tu + d_uv) / (d_tv + 1e-8)
    detour_right = (d_uv + d_vw) / (d_uw + 1e-8)
    bilateral_detour = 0.5 * (detour_left + detour_right)

    # Curvature angle cosines at u and v
    cos_u = np.clip((d_tu**2 + d_uv**2 - d_tv**2) / (2.0 * d_tu * d_uv + 1e-8), -1.0, 1.0)
    cos_v = np.clip((d_uv**2 + d_vw**2 - d_uw**2) / (2.0 * d_uv * d_vw + 1e-8), -1.0, 1.0)
    torsion_factor = (2.0 - cos_u - cos_v) * bilateral_detour

    # Step 4: Compute node entropy rigidity to identify entrenched topological basins
    total_node_usage = np.sum(edge_n_used, axis=1, keepdims=True) + 1e-12
    usage_probs = edge_n_used / total_node_usage
    node_entropy = -np.sum(usage_probs * np.log(usage_probs + 1e-12), axis=1)
    max_entropy = np.log(num_nodes) + 1e-12
    entropy_rigidity = 1.0 - (node_entropy / max_entropy)
    edge_rigidity = 0.5 * (entropy_rigidity[u_idx] + entropy_rigidity[v_idx])

    # Step 5: Guided local search utilities modulated by bilateral torsion and entropy
    edge_freq = edge_n_used[u_idx, v_idx]
    utilities = (d_uv / (1.0 + edge_freq)) * (1.0 + 0.30 * torsion_factor) * (1.0 + 0.25 * edge_rigidity)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.28 * global_mean_dist

    # Step 6: Apply selective non-linear penalties to high-torsion edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        penalty_mod = 1.0 + 0.40 * torsion_factor[idx] + 0.30 * edge_rigidity[idx]
        penalty = adaptive_lambda * freq_penalty * penalty_mod
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Step 7: Proactive Basin Tunneling - inject distance contractions on promising unvisited shortcuts
    # Evaluate 2-hop (u, w) and 3-hop (t, w) chordal shortcuts across structurally inefficient windows
    shortcut_score_2hop = (d_tu + d_uv + d_vw) - d_uw
    shortcut_indices_2hop = np.argsort(shortcut_score_2hop)[-k:]

    shortcut_score_3hop = (d_tu + d_uv + d_vw) - d_tw
    shortcut_indices_3hop = np.argsort(shortcut_score_3hop)[-max(1, k // 2):]

    for idx in shortcut_indices_2hop:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            contraction = 0.12 * global_mean_dist * (1.0 / (1.0 + freq))
            contracted_val = max(1e-6, updated_edge_distance[u, w] - contraction)
            updated_edge_distance[u, w] = contracted_val
            updated_edge_distance[w, u] = contracted_val

    for idx in shortcut_indices_3hop:
        t, w = t_idx[idx], w_idx[idx]
        freq = edge_n_used[t, w]
        if freq < 0.5:
            contraction = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            contracted_val = max(1e-6, updated_edge_distance[t, w] - contraction)
            updated_edge_distance[t, w] = contracted_val
            updated_edge_distance[w, t] = contracted_val

    return updated_edge_distance
