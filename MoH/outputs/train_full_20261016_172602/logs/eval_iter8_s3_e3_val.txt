#{A hybrid guided local search penalty strategy that combines usage-adaptive global/local balancing with bridge-edge preservation for sparse inter-cluster links and search-progression-decayed top-quartile amplification.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # Calculate k-nearest neighbor density for structural context
    k = min(6, n - 1)
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k + 1], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Tour edge indices and base metrics
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    relative_length = tour_edge_lengths / avg_tour_length

    # Global logistic rank penalty and local density/usage-scaled penalty
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))
    edge_counts = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.3 * edge_counts)
    avg_density = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * avg_density

    # Usage-dependent dynamic weighting between global rank and local context
    freq_mean = np.mean(edge_counts)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Progression decay factor based on global iteration usage depth
    progression_decay = 1.0 / (1.0 + 0.05 * freq_mean)

    # Selective bridge protection: shield low-density inter-cluster edges from heavy penalization
    # Non-bridge top-quartile edges receive progression-decayed amplification
    q75 = np.percentile(tour_edge_lengths, 75)
    top_quartile_mask = tour_edge_lengths >= q75
    is_bridge = avg_density <= 1.0

    # Shield bridges by dampening their penalty, amplify non-bridge top-quartile edges
    combined_penalty[is_bridge] *= 0.5
    amplify_mask = top_quartile_mask & (~is_bridge)
    combined_penalty[amplify_mask] *= (1.5 * progression_decay)

    # Symmetrically apply penalties
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
