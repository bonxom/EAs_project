# ===== Heuristic Code =====
#{Implements a vectorized dual-condition GLS penalty that selectively penalizes edges exceeding the average tour cost while meeting a near-maximal utility threshold, scaling penalties dynamically by edge cost and usage.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_tour_cost = np.mean(tour_lengths)

    # Dual-condition filter: edge must be above average length and close to peak utility
    selected_mask = (tour_lengths >= avg_tour_cost) & (utilities >= 0.90 * max_utility)

    # Fallback to pure utility peak if no edge satisfies both conditions simultaneously
    if not np.any(selected_mask):
        selected_mask = utilities >= 0.99 * max_utility

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Dynamic penalty proportional to average edge length scaled by utilization
    penalties = 0.3 * avg_tour_cost / (1.0 + tour_counts[selected_mask])

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
