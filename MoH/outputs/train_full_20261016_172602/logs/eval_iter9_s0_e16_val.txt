# ===== Heuristic Code =====
# {Apply an evaporative decay factor to previously accumulated penalties when frequency-based stagnation occurs, dynamically restoring over-penalized edges while penalizing the highest-utility edges of the current local optimum.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Identify edges belonging to the current local optimum tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Calculate edge utility based on modified distance and historical utilization
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Detect stagnation: high maximum usage among current tour edges triggers decay
    max_usage = np.max(tour_counts)
    mean_usage = np.mean(tour_counts)

    if max_usage > 2.0 * (mean_usage + 1.0):
        # Evaporative decay step on penalized edges to prevent irreversible distortion
        penalized_mask = edge_n_used > 0
        decay_factor = 0.95
        updated_edge_distance[penalized_mask] = (
            updated_edge_distance[penalized_mask] * decay_factor + 
            edge_distance[penalized_mask] * (1.0 - decay_factor)
        )

    # Calculate adaptive penalty scaling with tour characteristics
    penalty = 0.35 * avg_edge / (1.0 + 0.05 * np.log1p(mean_usage))
    threshold = 0.98 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Apply symmetric penalty to targeted bottleneck edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
