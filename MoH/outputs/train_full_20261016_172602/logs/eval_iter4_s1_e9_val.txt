# ===== Heuristic Code =====
#{Adaptive Guided Local Search: penalizes the top 20% highest-utility tour edges to preserve promising subtours, combined with a targeted k-nearest-neighbor discount on the 4 shortest untried incident edges of penalized endpoints to guide search toward compact unvisited connections.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Adaptive Pareto-topological filter: select only top ~20% highest utility edges (at least 1 edge)
    top_k = max(1, int(np.ceil(0.20 * n)))
    sorted_order = np.argsort(utilities)
    top_indices = sorted_order[-top_k:]

    p_u = u_idx[top_indices]
    p_v = v_idx[top_indices]
    p_costs = costs[top_indices]
    p_used = used[top_indices]
    p_utils = utilities[top_indices]
    max_util = np.max(utilities) + 1e-8

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))

    # Localized k-nearest-neighbor attractiveness bonus for up to 4 untried incident edges
    knn = 4
    discount_ratio = 0.03
    for u in penalized_nodes:
        untried_mask = (edge_n_used[u] == 0)
        untried_mask[u] = False
        cand_indices = np.where(untried_mask)[0]
        if len(cand_indices) > 0:
            cand_dists = edge_distance[u, cand_indices]
            num_to_discount = min(knn, len(cand_indices))
            best_local_order = np.argpartition(cand_dists, num_to_discount - 1)[:num_to_discount]
            chosen_neighbors = cand_indices[best_local_order]

            discounts = updated_edge_distance[u, chosen_neighbors] * discount_ratio
            updated_edge_distance[u, chosen_neighbors] -= discounts
            updated_edge_distance[chosen_neighbors, u] -= discounts

    return updated_edge_distance
