# ===== Heuristic Code =====
#{Multi-scale geodesic curvature profiling evaluates both 2-hop and 3-hop polygonal expansion ratios alongside chord-crossing geometric entanglement proxies to penalize spatial zigzagging while systematically discounting underused multi-scale shortcuts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Node indices for consecutive windows across the tour: t -> u -> v -> w
    t_idx = np.roll(tour, 1)
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    # Smooth historical usage decay
    edge_n_used *= 0.96

    # Direct tour edge lengths
    d_tu = edge_distance[t_idx, u_idx]
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]

    # Shortcut chords for 2-hop (u->w) and 3-hop (t->w)
    d_uw = edge_distance[u_idx, w_idx]
    d_tw = edge_distance[t_idx, w_idx]
    d_tv = edge_distance[t_idx, v_idx]

    # Multi-scale perimeter-to-chord expansion ratios
    # 2-hop detour: (d_uv + d_vw) / d_uw
    detour_2hop = (d_uv + d_vw) / (d_uw + 1e-8)
    # 3-hop detour: (d_tu + d_uv + d_vw) / d_tw
    detour_3hop = (d_tu + d_uv + d_vw) / (d_tw + 1e-8)

    # Chord-crossing proxy: check if interior chords (t-v and u-w) indicate crossing / spatial zigzag
    # In a convex quad, d_tv + d_uw > d_tu + d_vw; an inverted ratio indicates severe entanglement
    entanglement = (d_tu + d_vw) / (d_tv + d_uw + 1e-8)

    # Compound multi-scale geodesic inefficiency profile
    multi_scale_inefficiency = detour_2hop + 0.5 * detour_3hop + 0.3 * entanglement

    # Local geometric factor integrating left and right context around edge (u, v)
    local_geom_factor = 0.5 * (multi_scale_inefficiency + np.roll(multi_scale_inefficiency, 1))

    # Base cost and historical frequencies
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]

    # Penalization utility combining edge distance, frequency, and multi-scale curvature
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply adaptive penalties to worst geometric edges
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Target 2-hop shortcut reductions where detour is largest and chord is underused
    high_detour_2hop = np.argsort(detour_2hop)[-k:]
    for idx in high_detour_2hop:
        u, w = u_idx[idx], w_idx[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.10 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    # Target top 3-hop shortcut reductions to bypass zigzagging segments
    k_3hop = max(1, int(np.ceil(0.07 * num_nodes)))
    high_detour_3hop = np.argsort(detour_3hop)[-k_3hop:]
    for idx in high_detour_3hop:
        t, w = t_idx[idx], w_idx[idx]
        freq = edge_n_used[t, w]
        if freq < 0.5:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[t, w] = max(1e-6, updated_edge_distance[t, w] - discount)
            updated_edge_distance[w, t] = updated_edge_distance[t, w]

    return updated_edge_distance
