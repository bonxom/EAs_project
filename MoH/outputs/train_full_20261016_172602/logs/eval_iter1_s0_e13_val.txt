# ===== Heuristic Code =====
# {Applies a continuous, rank-proportional soft-max style penalty distribution to high-utility edges alongside cooperative incident-edge ruin-and-recreate scaling to dynamically reshape the search landscape without uniform step thresholds.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)

    min_util = np.min(utilities)
    max_util = np.max(utilities)
    range_util = max_util - min_util

    # Scale utilities smoothly into [0, 1] relative ranking
    if range_util > 1e-12:
        norm_util = (utilities - min_util) / range_util
    else:
        norm_util = np.ones_like(utilities)

    # Power-law continuous scaling focusing penalties on the upper spectrum of utilities
    continuous_weights = np.power(norm_util, 3.0)

    # Base penalty proportional to average edge length modulated by the continuous utility profile
    avg_edge = np.mean(tour_lengths)
    edge_penalties = 0.5 * avg_edge * continuous_weights

    # Apply symmetric continuous penalty to the tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), edge_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), edge_penalties)

    return updated_edge_distance
