# ===== Heuristic Code =====
# {Vectorized high-utility tour bottleneck penalization combined with dynamic decay of past penalties on inactive edges to balance targeted perturbation and metric recovery.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Vectorized extraction of current tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Calculate tour edge statistics and utility metrics
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities) if len(utilities) > 0 else 0.0
    avg_tour_edge = np.mean(tour_lengths) if len(tour_lengths) > 0 else 1.0

    # Decay previously penalized edges outside the current tour to restore metric fidelity
    decay_factor = 0.95
    tour_mask = np.zeros_like(edge_distance, dtype=bool)
    tour_mask[u_idx, v_idx] = True
    tour_mask[v_idx, u_idx] = True

    inactive_penalized = (edge_n_used > 0) & (~tour_mask)
    updated_edge_distance[inactive_penalized] *= decay_factor

    # Selective perturbation of high-utility bottleneck edges in the local optimum
    threshold = 0.90 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_counts = tour_counts[selected_mask]

    penalties = (0.3 * avg_tour_edge) / (1.0 + 0.5 * sel_counts)

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
