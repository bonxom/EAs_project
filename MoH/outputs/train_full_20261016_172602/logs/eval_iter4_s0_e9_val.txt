#{Dynamically scale edge penalties using tour length dispersion (coefficient of variation) and edge utility dominance while breaking ties deterministically with relative length weighting.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    if max_utility <= 0.0:
        return updated_edge_distance

    mean_len = np.mean(tour_lengths)
    std_len = np.std(tour_lengths)
    cv = std_len / (mean_len + 1e-12)

    # Adaptive penalty scaling: increase pressure when edge lengths are uniform (low cv)
    # and modulate with edge usage stagnation
    base_factor = 0.25 + 0.5 * (1.0 / (1.0 + cv))
    mean_count = np.mean(tour_counts)
    stagnation_factor = 1.0 + np.log1p(mean_count)

    # Select candidate edges with close-to-maximum utility
    threshold = 0.98 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]
    sel_util = utilities[selected_mask]
    sel_len = tour_lengths[selected_mask]

    # Deterministic fine-grained differentiation based on edge length share
    differential_weights = sel_util / max_utility + 0.1 * (sel_len / (mean_len + 1e-12))
    penalties = base_factor * stagnation_factor * mean_len * differential_weights

    updated_edge_distance[sel_u, sel_v] += penalties
    updated_edge_distance[sel_v, sel_u] += penalties

    return updated_edge_distance
