# ===== Heuristic Code =====
#{Vectorize edge penalty evaluation by eliminating the full N-by-N sort and calculating an efficient, tour-localized normalized edge penalty scaled by edge usage frequency and relative length.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Vectorized extraction of tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    edge_counts = edge_n_used[u_idx, v_idx]

    # Efficient statistics derived purely from the O(N) tour edges
    mean_len = np.mean(tour_edge_lengths)
    std_len = np.std(tour_edge_lengths) + 1e-8

    # Standardized relative edge length (identifies disproportionately long edges)
    z_lengths = (tour_edge_lengths - mean_len) / std_len
    length_penalty = 1.0 / (1.0 + np.exp(-2.5 * z_lengths))

    # Frequency attenuation: penalize edges based on length scaled down by excessive reuse
    freq_attenuation = 1.0 / (1.0 + 0.25 * edge_counts)

    # Base penalty proportional to the edge length and its relative extremity
    penalty = (0.25 * mean_len) * length_penalty * freq_attenuation

    # Boost penalty for the top quartile of longest edges in the tour
    q75 = np.percentile(tour_edge_lengths, 75)
    high_mask = tour_edge_lengths >= q75
    penalty[high_mask] *= 1.4

    # Symmetric in-place addition to distance matrix
    np.add.at(updated_edge_distance, (u_idx, v_idx), penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalty)

    return updated_edge_distance
