# ===== Heuristic Code =====
# {Applies classical Guided Local Search edge utility evaluation directly to tour edges, penalizing the maximal utility subset with an adaptive scale while completely eliminating redundant full-matrix distance sorting.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create a copy of the edge distance matrix to avoid direct mutation
    updated_edge_distance = np.copy(edge_distance)

    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    # Extract consecutive edges in the local optimum tour
    u_idx = np.asarray(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    # Retrieve current edge distances and historical penalty/usage counters
    tour_dists = edge_distance[u_idx, v_idx]
    penalties = edge_n_used[u_idx, v_idx]

    # Compute classical GLS utility: util(u, v) = c(u, v) / (1 + p(u, v))
    utilities = tour_dists / (1.0 + penalties)

    # Identify maximum utility and select edges achieving or near the peak utility
    max_util = np.max(utilities)
    if max_util <= 0.0:
        return updated_edge_distance

    # Select edges within the top utility tier (e.g., matching or within 95% of max utility)
    target_mask = utilities >= (0.95 * max_util)

    # Calculate an adaptive penalty increment proportional to mean tour edge cost
    mean_tour_cost = np.mean(tour_dists)
    penalty_magnitude = 0.3 * mean_tour_cost

    # Apply scaled penalty selectively to the identified subset
    applied_penalties = penalty_magnitude * (utilities[target_mask] / max_util)

    target_u = u_idx[target_mask]
    target_v = v_idx[target_mask]

    # Symmetrically apply penalties to distance matrix
    np.add.at(updated_edge_distance, (target_u, target_v), applied_penalties)
    np.add.at(updated_edge_distance, (target_v, target_u), applied_penalties)

    return updated_edge_distance
