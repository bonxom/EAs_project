# ===== Heuristic Code =====
#{A guided local search strategy that applies tour-scale-calibrated saturating penalties to high-utility edges based on global tour cost while granting proportional discounts to under-explored incident alternative edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_tour_cost = np.mean(costs)
    std_tour_cost = np.std(costs) + 1e-8

    # Guided Local Search utility calculation: cost / (1 + used)
    utilities = costs / (1.0 + used)
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    # Diminishing memory saturation term: tanh prevents unbounded distance growth
    # Global tour-cost calibrated perturbation scale based on mean and deviation
    base_scale = 0.3 * mean_tour_cost + 0.1 * std_tour_cost
    saturation_factor = np.tanh(p_costs / (mean_tour_cost + 1e-8))
    memory_damping = 1.0 / np.sqrt(1.0 + p_used)

    penalties = base_scale * saturation_factor * memory_damping * (1.0 + p_utils / (np.max(utilities) + 1e-8))

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Targeted discount for unvisited or minimally explored incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_rate = 0.03
    for u in penalized_nodes:
        node_used = edge_n_used[u]
        incident_candidates = (node_used == 0)
        incident_candidates[u] = False
        if np.any(incident_candidates):
            current_vals = updated_edge_distance[u, incident_candidates]
            reduction = current_vals * discount_rate
            updated_edge_distance[u, incident_candidates] -= reduction
            updated_edge_distance[incident_candidates, u] -= reduction

    return updated_edge_distance
