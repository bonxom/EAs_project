# ===== Heuristic Code =====
#{Combines Parent 1's sigmoid rank-penalty and density-weighted local penalty with an adaptive cutoff percentile gated by tour length coefficient of variation to dynamically boost penalization on long dispersion-sensitive edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Compute node density using efficient partition for k-nearest neighbors
    k = min(5, n - 1)
    sorted_dists = np.partition(edge_distance, kth=k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / (np.mean(node_density) + 1e-6)

    # Extract tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Global rank penalty (sigmoid activation around normalized length)
    avg_tour_length = np.mean(tour_edge_lengths) + 1e-6
    std_tour_length = np.std(tour_edge_lengths)
    relative_length = tour_edge_lengths / avg_tour_length
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Local penalty modulated by edge usage and local node density
    usage_factor = 1.0 / (1.0 + 0.3 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Frequency-adjusted weighting between global rank penalty and local density penalty
    freq_mean = np.mean(usage)
    w_global = 0.6 / (1.0 + 0.1 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = w_global * rank_penalty + w_local * local_penalty

    # Dynamic percentile gating using coefficient of variation (CV)
    cv = std_tour_length / avg_tour_length
    cutoff_pct = np.clip(75.0 - 15.0 * cv, 55.0, 80.0)
    threshold = np.percentile(tour_edge_lengths, cutoff_pct)

    # Apply adaptive boost to edges exceeding the dynamic threshold
    boost_factor = 1.5 + 0.2 * np.clip(cv - 0.5, 0.0, 0.5)
    high_penalty_mask = tour_edge_lengths >= threshold
    combined_penalty[high_penalty_mask] *= boost_factor

    # Symmetrically update distances
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
