# ===== Heuristic Code =====
#{Penalizes inefficient edges using geometric detour and memory decay while attracting productive 2-opt cross-chords via a strictly bounded multiplicative discount scaled by utilization.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Attractor potential field: identify promising non-adjacent 2-opt cross-chords
    # Evaluate prospective 2-opt swaps between top penalized segments and nearby tour chords
    for p_idx in top_k_indices:
        u1, v1 = u_idx[p_idx], v_idx[p_idx]
        cur_len1 = edge_distance[u1, v1]

        # Check candidate chords with fixed relative offsets to maintain determinism and efficiency
        for offset in (2, 3, 4, num_nodes // 4, num_nodes // 2):
            q_idx = (p_idx + offset) % num_nodes
            if q_idx == p_idx or q_idx == (p_idx - 1) % num_nodes or q_idx == (p_idx + 1) % num_nodes:
                continue
            u2, v2 = u_idx[q_idx], v_idx[q_idx]
            cur_len2 = edge_distance[u2, v2]

            # Virtual 2-opt uncrossing test: (u1, v1) + (u2, v2) -> (u1, u2) + (v1, v2)
            cross_d1 = edge_distance[u1, u2]
            cross_d2 = edge_distance[v1, v2]
            delta = (cross_d1 + cross_d2) - (cur_len1 + cur_len2)

            # If the candidate reconnecting move shows structural promise or mild deficit
            if delta < 0.20 * (cur_len1 + cur_len2):
                # Multiplicative attractor discount strictly bounded in [0.85, 1.0)
                freq_1 = edge_n_used[u1, u2]
                freq_2 = edge_n_used[v1, v2]

                scale1 = 1.0 - 0.15 / (1.0 + 0.5 * freq_1)
                scale2 = 1.0 - 0.15 / (1.0 + 0.5 * freq_2)

                updated_edge_distance[u1, u2] = min(updated_edge_distance[u1, u2], edge_distance[u1, u2] * scale1)
                updated_edge_distance[u2, u1] = updated_edge_distance[u1, u2]
                updated_edge_distance[v1, v2] = min(updated_edge_distance[v1, v2], edge_distance[v1, v2] * scale2)
                updated_edge_distance[v2, v1] = updated_edge_distance[v1, v2]

    return updated_edge_distance
