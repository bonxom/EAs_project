# ===== Heuristic Code =====
#{Dual-action guided local search penalizes tour edges using sigmoid-gated cost-to-usage utilities boosted exponentially by rigid vectorized triangular detour deficits while discounting unvisited incident shortcuts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))

    # Vectorized Triangular Slack (Detour Deficit) across all potential bypass nodes w
    # For each edge (u_i, v_i), detour via w is c(u_i, w) + c(w, v_i) - c(u_i, v_i)
    # Shape: (n, num_nodes)
    u_all = edge_distance[u_idx, :]
    v_all = edge_distance[v_idx, :]
    two_opt_detours = u_all + v_all - costs[:, np.newaxis]

    # Exclude w = u and w = v by masking
    node_indices = np.arange(edge_distance.shape[0])
    mask_u = (node_indices == u_idx[:, np.newaxis])
    mask_v = (node_indices == v_idx[:, np.newaxis])
    two_opt_detours[mask_u | mask_v] = np.inf

    # Find minimum detour slack for each tour edge
    min_detour_slack = np.min(two_opt_detours, axis=1)
    # Normalize detour deficit: smaller slack implies easier local escape, larger slack implies topological rigidity
    # Low alternate bypass flexibility gets boosted to rupture locked configurations
    avg_edge_cost = np.mean(edge_distance) + 1e-8
    rel_slack = min_detour_slack / avg_edge_cost
    rigidity_boost = np.exp(-np.clip(rel_slack, 0.0, 3.0))

    utilities = ((costs * (1.0 + weights)) / (1.0 + used)) * (1.0 + rigidity_boost)
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Landscape shaping: discount promising unvisited incident edges
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
