#{Two-tier adaptive memory with soft baseline rejuvenation: applies global evaporation and stagnation-triggered baseline restoration to prevent metric divergence while maintaining asymmetric frequency-damped penalties on high-utility tour edges and rewarding unexplored incident connections.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Step 1: Create a safe working copy of the edge distance matrix
    updated_edge_distance = np.copy(edge_distance).astype(np.float64)
    n = edge_distance.shape[0]

    # Step 2: Global memory evaporation and stagnation reset trigger
    # Stagnation threshold: heavily traversed edges are progressively rejuvenated
    stagnation_threshold = 20.0
    stagnated_mask = edge_n_used >= stagnation_threshold
    if np.any(stagnated_mask):
        rejuvenation_factor = np.clip((edge_n_used[stagnated_mask] - stagnation_threshold) / stagnation_threshold, 0.0, 0.8)
        # Softly pull over-frequented edges back to the median edge distance baseline
        baseline_cost = np.median(edge_distance)
        updated_edge_distance[stagnated_mask] = (
            (1.0 - rejuvenation_factor) * updated_edge_distance[stagnated_mask] 
            + rejuvenation_factor * baseline_cost
        )

    # Step 3: Extract tour edges and compute utility metrics
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    costs = updated_edge_distance[u_idx, v_idx]
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    # Logistic weighting emphasizing disproportionately long edges
    weights = 1.0 / (1.0 + np.exp(-np.clip(z_scores, -10.0, 10.0)))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    # Step 4: Apply asymmetric frequency-damped penalties to high-utility edges
    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Step 5: Reward/discount unexplored incident edges to redirect trajectories
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_ratio = 0.03
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            discounts = updated_edge_distance[u, unused_mask] * discount_ratio
            updated_edge_distance[u, unused_mask] -= discounts
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    # Step 6: Maintain non-negativity and zero self-distances
    np.fill_diagonal(updated_edge_distance, 0.0)
    np.maximum(updated_edge_distance, 1e-6, out=updated_edge_distance)
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
