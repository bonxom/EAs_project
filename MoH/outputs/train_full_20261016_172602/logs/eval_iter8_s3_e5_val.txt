# ===== Heuristic Code =====
#{Scale-adaptive Guided Local Search penalizes candidate edges proportionally to edge cost, usage frequency, and baseline tour length while dynamically tuning the scaling parameter based on the ratio of active penalties.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    total_tour_length = np.sum(tour_edge_lengths)
    mean_edge_length = total_tour_length / n

    usage = edge_n_used[u_idx, v_idx]
    gls_utility = tour_edge_lengths / (1.0 + usage)

    max_util = np.max(gls_utility)
    penalize_mask = gls_utility >= (0.90 * max_util)

    num_penalized = np.count_nonzero(penalize_mask)
    ratio = num_penalized / n

    # Dynamic scaling lambda: higher when few edges are penalized, moderated when many edges qualify
    lambda_param = 0.30 / (1.0 + 2.0 * ratio)

    # Scale-invariant Guided Local Search penalty term
    penalties = lambda_param * gls_utility * mean_edge_length * penalize_mask.astype(np.float64)

    np.add.at(updated_edge_distance, (u_idx, v_idx), penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), penalties)

    return updated_edge_distance
