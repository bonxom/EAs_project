#{Adaptive Guided Local Search that infers original baseline edge lengths from the penalty history, evaluates utility ratios with variance-normalized thresholding, and applies scale-proportional symmetric penalties to escape recurring local minima without distortion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    current_lengths = edge_distance[u_idx, v_idx]
    counts = edge_n_used[u_idx, v_idx]

    # Infer/approximate base distances to prevent penalty compounding from distorting utility calculation
    n_nodes = len(local_opt_tour)
    mean_len = np.mean(current_lengths)
    std_len = np.std(current_lengths)

    # Base distance heuristic: discounting accumulated penalties scaled by usage count
    base_lengths = np.maximum(current_lengths / (1.0 + 0.15 * counts), 1e-6)

    # Classical GLS utility calculation using estimated base costs
    utilities = base_lengths / (1.0 + counts)

    max_utility = np.max(utilities)
    mean_utility = np.mean(utilities)

    # Adaptive thresholding: tighter when utilities are clustered, broader when peak is sharp
    spread = max_utility - mean_utility
    threshold = max(max_utility - 0.1 * spread, 0.95 * max_utility)
    selected_mask = utilities >= threshold

    # Fail-safe diversification: ensure at least one prominent edge is penalized
    if not np.any(selected_mask):
        selected_mask = utilities == max_utility

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Scale-proportional penalty incorporating both average tour scale and individual edge length
    penalty_magnitude = 0.25 * mean_len + 0.05 * std_len + 0.05 * base_lengths[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty_magnitude
    updated_edge_distance[sel_v, sel_u] += penalty_magnitude

    return updated_edge_distance
