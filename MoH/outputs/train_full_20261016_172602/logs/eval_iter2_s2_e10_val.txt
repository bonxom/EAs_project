# ===== Heuristic Code =====
#{This algorithm replaces local 3-point window evaluation with a multi-scale string detour metric spanning 4-node and 5-node subpaths against bounding chords, penalizing clustered inefficient sub-sequences while discounting prospective long-range shortcuts.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # Historical memory decay
    edge_n_used *= 0.96

    # Precompute tour node indices for multi-scale chords (1 to 4 hops ahead)
    i0 = tour
    i1 = np.roll(tour, -1)
    i2 = np.roll(tour, -2)
    i3 = np.roll(tour, -3)
    i4 = np.roll(tour, -4)

    # Distance of basic edges along the tour
    d01 = edge_distance[i0, i1]
    d12 = edge_distance[i1, i2]
    d23 = edge_distance[i2, i3]
    d34 = edge_distance[i3, i4]

    # Chord distances across multi-node spans
    d02 = edge_distance[i0, i2]  # 2-hop chord (3 nodes: i0-i1-i2)
    d03 = edge_distance[i0, i3]  # 3-hop chord (4 nodes: i0-i1-i2-i3)
    d04 = edge_distance[i0, i4]  # 4-hop chord (5 nodes: i0-i1-i2-i3-i4)

    # Multi-scale detour ratios comparing actual subpath length to bounding chord
    path_len_3 = d01 + d12
    detour_3 = path_len_3 / (d02 + 1e-8)

    path_len_4 = path_len_3 + d23
    detour_4 = path_len_4 / (d03 + 1e-8)

    path_len_5 = path_len_4 + d34
    detour_5 = path_len_5 / (d04 + 1e-8)

    # Multi-scale geometric inefficiency aggregated over sub-sequences
    # Evaluates how much an edge participates in sub-optimal 4-node and 5-node string detours
    seq_inefficiency = 0.3 * detour_3 + 0.4 * detour_4 + 0.3 * detour_5

    # Smooth inefficiency over the edges within the string
    edge_seq_factor = (
        seq_inefficiency +
        np.roll(seq_inefficiency, 1) +
        np.roll(seq_inefficiency, 2)
    ) / 3.0

    base_dist = d01
    base_freq = edge_n_used[i0, i1]

    # Guided local search utility scoring for edges
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.40 * edge_seq_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Apply penalties to edges belonging to the most inefficient sub-sequences
    for idx in top_k_indices:
        u, v = i0[idx], i1[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * edge_seq_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Prospective shortcut discounting across high-detour multi-node spans (3-hop and 4-hop chords)
    high_detour_4_indices = np.argsort(detour_4)[-k:]
    for idx in high_detour_4_indices:
        u, w = i0[idx], i3[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.12 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    high_detour_5_indices = np.argsort(detour_5)[-k:]
    for idx in high_detour_5_indices:
        u, w = i0[idx], i4[idx]
        freq = edge_n_used[u, w]
        if freq < 1.0:
            discount = 0.08 * global_mean_dist * (1.0 / (1.0 + freq))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

    return updated_edge_distance
