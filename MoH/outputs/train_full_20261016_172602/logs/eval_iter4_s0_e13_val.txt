# ===== Heuristic Code =====
#{An adaptive Guided Local Search penalty mechanism dynamically scales the additive edge penalty using the utility spread and the ratio of maximum to mean utility to avoid over-penalization in homogeneous tours.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Vectorized calculation of edge utilities
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    min_utility = np.min(utilities)
    mean_utility = np.mean(utilities)
    avg_edge = np.mean(tour_lengths)

    # Dynamically scale penalty based on utility spread and relative prominence of the bottleneck edge
    utility_spread = max_utility - min_utility
    rel_contrast = (max_utility / (mean_utility + 1e-12)) - 1.0
    modulation = np.clip(rel_contrast, 0.05, 1.5)

    # Compute adaptive penalty combining local spread with modulated tour scale
    penalty = 0.5 * utility_spread + 0.15 * modulation * avg_edge

    # Identify edges with maximal utility
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetric in-place additive penalization
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
