# ===== Heuristic Code =====
#{This strategy preserves Z-score and inverse-usage utility filtering while replacing monolithic additive penalties with a Pareto-blended penalty scale combining incident-edge bottleneck ratios with logarithmic usage saturation to selectively penalize redundant edges over essential bridges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]
    max_util = np.max(utilities) + 1e-8

    # Pareto-blended penalty reconstruction:
    # 1. Bottleneck ratio: compare edge cost against the minimum incident non-zero edge for its endpoints
    # Compute min incident cost for each node excluding self-loops
    masked_dist = np.where(edge_distance > 0, edge_distance, np.inf)
    min_incident = np.min(masked_dist, axis=1)
    min_incident_uv = np.minimum(min_incident[p_u], min_incident[p_v])
    min_incident_uv = np.where(np.isinf(min_incident_uv) | (min_incident_uv <= 0), 1e-6, min_incident_uv)

    # Bridge indicator: high bottleneck ratio implies the edge is much longer than local alternatives
    bottleneck_ratio = p_costs / min_incident_uv
    normalized_bottleneck = bottleneck_ratio / (1.0 + bottleneck_ratio)

    # 2. Logarithmic edge saturation factor to temper over-penalization of heavily used critical bridges
    saturation_damping = 1.0 / (1.0 + np.log1p(p_used))

    # 3. Pareto blending: balance relative excess cost with utility rank and saturation
    excess_factor = np.clip(normalized_bottleneck, 0.1, 1.0)
    penalties = p_costs * excess_factor * saturation_damping * (0.5 + 0.5 * (p_utils / max_util))

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * decay_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
