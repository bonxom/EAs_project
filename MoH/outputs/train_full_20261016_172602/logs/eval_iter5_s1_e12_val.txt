# ===== Heuristic Code =====
#{A leaky-integrator edge penalty strategy that applies soft exponential evaporation to historical edge costs before superimposing normalized cost-frequency utilities onto tour edges and discounting under-explored incident alternatives.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n == 0:
        return updated_edge_distance

    # 1. Soft evaporation (leaky integrator) to prevent permanent landscape distortion
    # Evaporate artificially inflated costs back toward base levels proportional to exploration frequency
    evaporation_factor = 0.995
    updated_edge_distance *= evaporation_factor

    # 2. Extract current local optimum tour edges
    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)
    costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # 3. Evaluate cost-frequency utility with bounded standardization
    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + np.log1p(used))

    # 4. Identify high-utility edges to penalize for diversification
    threshold = np.median(utilities)
    penalize_mask = utilities >= threshold

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    max_util = np.max(utilities) + 1e-8
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    # Apply symmetric penalties
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # 5. Discount under-explored alternative incident edges to guide search re-creation
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_ratio = 0.015

    for u in penalized_nodes:
        row_used = edge_n_used[u]
        min_used = np.min(row_used)
        promising_mask = (row_used <= min_used + 1.0)
        promising_mask[u] = False

        if np.any(promising_mask):
            discounts = updated_edge_distance[u, promising_mask] * discount_ratio
            updated_edge_distance[u, promising_mask] -= discounts
            updated_edge_distance[promising_mask, u] -= discounts

    # Guarantee non-negative distances and zero diagonal
    np.maximum(updated_edge_distance, 0.0, out=updated_edge_distance)
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
