# ===== Heuristic Code =====
# {Refine Guided Local Search by incorporating an adaptive penalty evaporation factor that decays previously accumulated penalties to prevent permanent landscape distortion while selectively penalizing maximal utility edges of the current local optimum with an adaptive threshold.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = edge_distance.shape[0]

    # Evaporate existing penalties slightly where edges have accumulated penalties to allow re-exploration
    # We identify penalized edges by checking where edge_n_used > 0
    penalized_mask = edge_n_used > 0
    decay_factor = 0.985
    # Apply global soft decay across previously penalized edges relative to base distances
    # Since only edge_distance is passed, we scale edges with usage > 0 toward lower values
    # but ensure it never falls below an estimated intrinsic distance floor
    # For stability, we directly apply a light dampening factor on penalized edges
    if np.any(penalized_mask):
        # Scale back the penalty proportion of distance for edges that are penalized but not in current tour
        updated_edge_distance[penalized_mask] = (
            updated_edge_distance[penalized_mask] * decay_factor
        )

    tour_edges = [(local_opt_tour[i], local_opt_tour[(i + 1) % n]) for i in range(n)]
    tour_lengths = [edge_distance[u, v] for u, v in tour_edges]
    tour_length = sum(tour_lengths)
    avg_edge = tour_length / float(n) if n > 0 else 1.0

    # Calculate utility according to Guided Local Search formulation
    utilities = [edge_distance[u, v] / (1.0 + edge_n_used[u, v]) for u, v in tour_edges]
    max_utility = max(utilities) if utilities else 0.0

    if max_utility <= 0.0:
        return updated_edge_distance

    # Dynamic utility threshold based on current edge usage saturation
    mean_used = float(np.mean([edge_n_used[u, v] for u, v in tour_edges]))
    threshold_ratio = max(0.92, 0.99 - 0.02 * min(mean_used, 3.0))

    # GLS additive penalty parameter lambda proportional to tour length
    penalty_lambda = 0.30 * avg_edge

    for (u, v), util in zip(tour_edges, utilities):
        if util >= threshold_ratio * max_utility:
            penalty = penalty_lambda
            updated_edge_distance[u, v] += penalty
            updated_edge_distance[v, u] += penalty

    return updated_edge_distance
