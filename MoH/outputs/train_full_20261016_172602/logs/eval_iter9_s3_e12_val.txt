# ===== Heuristic Code =====
#{Evaporate historic penalties across the distance matrix using frequency memory and inject a targeted asymmetric stress spike into the tour's highest-cost contiguous sub-path to drive directed structural diversification.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    # 1. Multiplicative evaporative factor (1 - alpha) based on usage memory
    # Edges that have accumulated excessive penalties gradually relax back toward base distances
    # Scale alpha adaptively by mean tour edge length
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = edge_distance[u_idx, v_idx]
    avg_tour_len = np.mean(tour_lengths) + 1e-8

    # Mild full-matrix penalty relaxation on frequently visited edges to avoid lock-in
    alpha = 0.05
    active_mask = edge_n_used > 0
    # Apply soft evaporation relative to minimum known edge cost to preserve non-negativity
    min_dist = np.min(edge_distance)
    excess_dist = updated_edge_distance - min_dist
    updated_edge_distance[active_mask] -= alpha * np.maximum(0.0, excess_dist[active_mask])

    # 2. Local density estimation for normalization
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn = min(6, n - 1)
    k_nn_avg = np.mean(sorted_dists[:, 1:k_nn + 1], axis=1) + 1e-6
    node_density = 1.0 / k_nn_avg
    node_density = node_density / np.mean(node_density)

    # 3. Base tour edge penalties proportional to relative length, usage, and density
    rel_len = tour_lengths / avg_tour_len
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    usage_factor = 1.0 / (1.0 + 0.25 * edge_n_used[u_idx, v_idx])

    base_penalties = 0.5 * avg_tour_len * (rel_len ** 1.5) * density_factor * usage_factor

    # 4. Asymmetric stress spike penalty on the single highest-stress sub-path
    # Window size covers ~10-20% of tour (at least 2, at most 8 edges)
    w = max(2, min(8, n // 5))
    extended_tour_lengths = np.concatenate([tour_lengths, tour_lengths[:w - 1]])
    # Compute moving sum of stress scores
    stress_scores = (tour_lengths / avg_tour_len) * (node_density[u_idx] + node_density[v_idx])
    extended_stress = np.concatenate([stress_scores, stress_scores[:w - 1]])
    window_stress = np.convolve(extended_stress, np.ones(w), mode='valid')

    # Identify the highest-stress sub-path index deterministically
    best_start = int(np.argmax(window_stress))
    spike_indices = (best_start + np.arange(w)) % n

    # Apply asymmetric ramped spike penalty to completely ruin this contiguous critical segment
    spike_weights = np.linspace(1.5, 3.0, w)
    base_penalties[spike_indices] += avg_tour_len * spike_weights

    # 5. Apply symmetric additive updates
    np.add.at(updated_edge_distance, (u_idx, v_idx), base_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), base_penalties)

    return updated_edge_distance
