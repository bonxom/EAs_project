#{A dynamic Guided Local Search penalty operator evaluating V-shaped marginal cost utility modulated by power-law contrast enhancement and local density to aggressively penalize inefficient long edges while preserving essential local cluster connections.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    # Compute local neighborhood density context
    sorted_dists = np.sort(edge_distance, axis=1)
    k_nn_avg = np.mean(sorted_dists[:, 1:min(6, n)], axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-6)
    node_density = node_density / np.mean(node_density)

    # Extract consecutive edges of the current local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]

    # Compute classic GLS utility: cost / (1 + count)
    edge_counts = edge_n_used[u_idx, v_idx]
    gls_utility = tour_edge_lengths / (1.0 + edge_counts)

    # Contrast enhancement: relative utility scaled through a power-law transformation
    mean_utility = np.mean(gls_utility) + 1e-6
    rel_utility = gls_utility / mean_utility
    contrast_utility = np.power(rel_utility, 2.5)

    # Combine with endpoint density to penalize long leaps across dense clusters
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    mean_tour_length = np.mean(tour_edge_lengths) + 1e-6

    # Step-size calibration proportional to mean tour length
    penalty_step = 0.3 * mean_tour_length
    combined_penalty = penalty_step * contrast_utility * density_factor / (1.0 + 0.1 * edge_counts)

    # Emphasize the top quartile of inefficient edges
    threshold = np.percentile(tour_edge_lengths, 75)
    combined_penalty[tour_edge_lengths >= threshold] *= 1.5

    # Apply penalties symmetrically
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
