# ===== Heuristic Code =====
#{A Guided Local Search penalty mechanism that combines relative edge length, usage frequency attenuation, and local density weighting with an adaptive Pareto-decay evaporation towards the baseline distance matrix to prevent metric space over-distortion.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 2:
        return updated_edge_distance

    # Evaporate accumulated distance inflation to preserve original metric structure
    evaporation_rate = 0.08
    # Using edge_n_used as proxy for previous distortion: damp previously penalized edges slightly
    penalized_mask = edge_n_used > 0
    if np.any(penalized_mask):
        updated_edge_distance[penalized_mask] = (
            (1.0 - evaporation_rate) * updated_edge_distance[penalized_mask]
            + evaporation_rate * (updated_edge_distance[penalized_mask] / (1.0 + 0.05 * edge_n_used[penalized_mask]))
        )

    # Calculate k-nearest neighbor density for nodes
    k = min(6, n - 1)
    sorted_dists = np.partition(edge_distance, k, axis=1)[:, 1:k+1]
    k_nn_avg = np.mean(sorted_dists, axis=1)
    node_density = 1.0 / (k_nn_avg + 1e-8)
    mean_density = np.mean(node_density)
    node_density = node_density / (mean_density + 1e-8)

    # Extract consecutive edges in local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_edge_lengths = edge_distance[u_idx, v_idx]
    avg_tour_length = np.mean(tour_edge_lengths)

    # Relative edge length penalty scaling
    relative_length = tour_edge_lengths / (avg_tour_length + 1e-8)
    rank_penalty = 1.0 / (1.0 + np.exp(-4.0 * (relative_length - 1.0)))

    # Usage attenuation and local clustering density factor
    usage = edge_n_used[u_idx, v_idx]
    usage_factor = 1.0 / (1.0 + 0.35 * usage)
    density_factor = 0.5 * (node_density[u_idx] + node_density[v_idx])
    local_penalty = relative_length * usage_factor * density_factor

    # Dynamic balance between global rank and local context
    freq_mean = np.mean(usage)
    w_global = 0.65 / (1.0 + 0.08 * freq_mean)
    w_local = 1.0 - w_global
    combined_penalty = (w_global * rank_penalty + w_local * local_penalty) * (avg_tour_length * 0.15)

    # Amplify penalties for the longest edges in the tour
    p75 = np.percentile(tour_edge_lengths, 75)
    combined_penalty[tour_edge_lengths >= p75] *= 1.4

    # Apply symmetric penalties to current tour edges
    np.add.at(updated_edge_distance, (u_idx, v_idx), combined_penalty)
    np.add.at(updated_edge_distance, (v_idx, u_idx), combined_penalty)

    return updated_edge_distance
