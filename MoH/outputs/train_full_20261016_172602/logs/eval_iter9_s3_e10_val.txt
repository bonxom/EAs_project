# ===== Heuristic Code =====
#{Replaces hand-crafted weight mixing with an information-theoretic utility-to-cost penalty ratio derived from edge length, usage frequency, and 1-tree node dual potentials to selectively penalize high-slack edges along the tour.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 3:
        return updated_edge_distance

    # Approximate Held-Karp / 1-tree dual node potentials using 2-nearest neighbor distances
    sorted_dists = np.sort(edge_distance, axis=1)
    # The average of the two closest edges serves as a strong 1-tree dual estimate pi_i
    dual_pi = 0.5 * (sorted_dists[:, 1] + sorted_dists[:, 2])

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    c_e = edge_distance[u_idx, v_idx]
    usage = edge_n_used[u_idx, v_idx]

    # Reduced cost / dual slack: c_bar(u, v) = c(u, v) - pi(u) - pi(v)
    dual_potential_sum = dual_pi[u_idx] + dual_pi[v_idx]
    slack = np.maximum(0.0, c_e - dual_potential_sum)

    # Information-theoretic utility-to-cost ratio:
    # High slack indicates an edge that is suboptimal compared to 1-tree relaxation,
    # weighted inversely by frequency to avoid stagnation.
    mean_edge_cost = np.mean(c_e) + 1e-6
    relative_slack = slack / mean_edge_cost
    utility = (c_e / mean_edge_cost) + relative_slack

    # Penalty scales with utility divided by usage frequency memory
    penalty = utility / (1.0 + 0.35 * usage)

    # Scale penalty relative to problem magnitude and apply symmetrically
    scale_factor = 0.15 * mean_edge_cost
    final_penalties = penalty * scale_factor

    np.add.at(updated_edge_distance, (u_idx, v_idx), final_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), final_penalties)

    return updated_edge_distance
