# ===== Heuristic Code =====
#{Applies geometric decay to historical edge penalties while penalizing maximum-utility tour edges scaled by stagnation-aware tour length and localized node usage density to prevent over-penalization and escape local optima.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)

    # Apply geometric memory decay to historical usage to avoid permanent over-penalization
    decay_factor = 0.92
    edge_n_used *= decay_factor

    u_idx = np.asarray(local_opt_tour)
    v_idx = np.roll(u_idx, -1)

    tour_distances = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Guided local search utility: d_ij / (1 + count)
    utilities = tour_distances / (1.0 + tour_counts)
    max_util = np.max(utilities)
    max_mask = np.isclose(utilities, max_util, rtol=1e-5, atol=1e-8)

    # Increment usage for maximum utility edges
    for i in range(num_nodes):
        if max_mask[i]:
            u, v = u_idx[i], v_idx[i]
            edge_n_used[u, v] += 1.0
            edge_n_used[v, u] += 1.0

    # Compute characteristic distance metrics
    tour_length = np.sum(tour_distances)
    avg_tour_edge = tour_length / num_nodes

    # Usage distribution and stagnation estimation
    high_use_edges = edge_n_used[u_idx, v_idx] > 1.5
    stagnation_ratio = np.mean(high_use_edges)

    # Local node-level usage intensity
    node_intensity = np.sum(edge_n_used, axis=1)
    max_intensity = np.max(node_intensity) + 1e-12
    norm_intensity = node_intensity / max_intensity

    # Structural density coupling
    density_matrix = 1.0 + 0.25 * (norm_intensity[:, None] + norm_intensity[None, :])

    # Adaptive penalty scaling parameter lambda
    lambda_param = (0.30 + 0.15 * stagnation_ratio) * avg_tour_edge

    # Additive penalty matrix scaled by decayed usage and density
    penalty_matrix = lambda_param * edge_n_used * density_matrix
    updated_edge_distance += penalty_matrix

    return updated_edge_distance
