# ===== Heuristic Code =====
#{We implement a dynamic Guided Local Search with exponential penalty evaporation and soft-reset wash based on cumulative edge frequency to restore geometric fidelity and prevent metric over-flattening while penalizing high-utility tour edges and discounting unexplored incident alternatives.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n <= 1:
        return updated_edge_distance

    # Evaporate historical penalty drift to restore true metric fidelity
    # Using edge_n_used to estimate modified edges and apply memory decay
    modified_mask = edge_n_used > 0
    if np.any(modified_mask):
        # Global exponential memory evaporation factor
        decay_factor = 0.985
        # Soft-reset ("penalization wash") scaling if search has heavily recycled edges
        max_used = np.max(edge_n_used)
        if max_used > 15:
            # Scale down excessive penalty inflation when trapped in a dense basin
            wash_ratio = np.clip(1.0 - (max_used - 15) * 0.015, 0.75, 1.0)
            decay_factor *= wash_ratio

        # Regress current matrix slightly towards geometric balance
        updated_edge_distance[modified_mask] *= decay_factor

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = updated_edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    mean_cost = np.mean(costs)
    std_cost = np.std(costs) + 1e-8
    z_scores = (costs - mean_cost) / std_cost
    weights = 1.0 / (1.0 + np.exp(-z_scores))
    utilities = (costs * (1.0 + weights)) / (1.0 + used)

    threshold = np.median(utilities)
    max_util = np.max(utilities) + 1e-8

    penalize_mask = utilities >= threshold
    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utilities[penalize_mask]

    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)
    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    discount_ratio = 0.02
    for u in penalized_nodes:
        unused_mask = (edge_n_used[u] == 0)
        unused_mask[u] = False
        if np.any(unused_mask):
            current_vals = updated_edge_distance[u, unused_mask]
            updated_edge_distance[u, unused_mask] -= current_vals * discount_ratio
            updated_edge_distance[unused_mask, u] = updated_edge_distance[u, unused_mask]

    return updated_edge_distance
