# ===== Heuristic Code =====
#{Dynamic Top-K Frequency-Adaptive Perturbation: Dynamically select the top-k highest-utility bottleneck edges based on tour length variance, penalizing them with a visit-attenuated logarithmic scaling and applying scaled incident node diffusion to escape deep local minima without distorting essential structural edges.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Copy edge distance matrix to avoid modifying original array directly
    updated_edge_distance = np.copy(edge_distance)

    # Extract consecutive edge endpoints along the local optimal tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Extract tour edge distances and visit frequencies
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Standard Guided Local Search edge utility calculation
    utilities = tour_lengths / (1.0 + tour_counts)

    # Calculate tour statistics
    n = len(local_opt_tour)
    avg_edge = np.mean(tour_lengths)
    std_edge = np.std(tour_lengths)
    coeff_var = std_edge / (avg_edge + 1e-12)

    # Dynamically select top-k edges based on coefficient of variation of tour edge lengths
    # Higher variance indicates uneven bottlenecks; lower variance calls for broader perturbations
    k = int(np.clip(np.round(1.0 + coeff_var * 4.0), 1, max(1, n // 5)))

    # Identify indices of the top-k utility edges
    top_k_indices = np.argpartition(utilities, -k)[-k:]
    sel_u = u_idx[top_k_indices]
    sel_v = v_idx[top_k_indices]
    sel_counts = tour_counts[top_k_indices]
    sel_lengths = tour_lengths[top_k_indices]

    # Attenuate penalty logarithmically with cumulative visit counts to prevent over-penalizing structural edges
    # Base penalty scales with edge length and inversely with logarithmic search memory
    attenuation = 1.0 / (1.0 + np.log1p(sel_counts))
    direct_penalties = 0.35 * sel_lengths * attenuation

    # Symmetric direct penalty assignment to top-k bottleneck edges
    for i in range(k):
        pen = direct_penalties[i]
        u, v = sel_u[i], sel_v[i]
        updated_edge_distance[u, v] += pen
        updated_edge_distance[v, u] += pen

    # Mild incident node dilution to discourage tight cyclic detours around penalized edges
    incident_penalty = 0.08 * avg_edge / (1.0 + np.log1p(np.mean(tour_counts)))
    unique_incident_nodes = np.unique(np.concatenate([sel_u, sel_v]))
    updated_edge_distance[unique_incident_nodes, :] += incident_penalty
    updated_edge_distance[:, unique_incident_nodes] += incident_penalty

    # Ensure self-loop distance remains zero
    np.fill_diagonal(updated_edge_distance, 0.0)

    return updated_edge_distance
