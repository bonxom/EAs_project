# ===== Heuristic Code =====
#{Multi-scale spatial-cluster guided penalization that identifies bridge edges crossing distance-partitioned tour segments and diffuses correlated tour penalties across topological neighborhoods using distance-weighted exponential attenuation.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)
    if n < 4:
        return updated_edge_distance

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_edge_lengths = edge_distance[u_idx, v_idx]
    mean_len = np.mean(tour_edge_lengths)
    std_len = np.std(tour_edge_lengths) + 1e-8
    rel_lengths = (tour_edge_lengths - mean_len) / std_len

    # Segment tour into topological spatial clusters via thresholding on edge distances
    split_threshold = mean_len + 0.5 * std_len
    boundary_mask = tour_edge_lengths >= split_threshold

    # Base penalty: combination of relative edge elongation and inverse usage damping
    usages = edge_n_used[u_idx, v_idx]
    usage_damp = 1.0 / np.sqrt(1.0 + usages)
    base_penalties = np.maximum(0.1, 1.0 + np.tanh(rel_lengths)) * usage_damp * (mean_len * 0.15)

    # Ruin inter-cluster / boundary bridge edges with amplified penalty
    base_penalties[boundary_mask] *= 2.2

    # Momentum / neighborhood diffusion: propagate penalty to adjacent edges along tour
    prev_penalties = np.roll(base_penalties, 1)
    next_penalties = np.roll(base_penalties, -1)
    diffused_penalties = base_penalties + 0.25 * (prev_penalties + next_penalties)

    # Apply penalties symmetrically
    np.add.at(updated_edge_distance, (u_idx, v_idx), diffused_penalties)
    np.add.at(updated_edge_distance, (v_idx, u_idx), diffused_penalties)

    return updated_edge_distance
