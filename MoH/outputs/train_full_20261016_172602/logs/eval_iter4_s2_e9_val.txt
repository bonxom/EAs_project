# ===== Heuristic Code =====
import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    # 1. Decay historical edge usage counts
    edge_n_used *= 0.96

    # Node sequence indexing along the tour
    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)
    z_idx = np.roll(tour, -3)

    # Geometric evaluations of consecutive triplets (u -> v -> w)
    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    # Frequency and geometry-guided utility for tour edges
    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    # Identify top-k penalized edges
    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = np.mean(edge_distance)
    adaptive_lambda = 0.30 * global_mean_dist

    # Penalize inefficient / frequently used edges
    penalized_nodes = set()
    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        penalized_nodes.add(u)
        penalized_nodes.add(v)

        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0

        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight

        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # 2. Ruin & Recreate replacement: Multi-hop Leapfrog and KNN Bridge Promotion
    # (a) 3-hop leapfrog discount (u -> z skipping v and w if path u-v-w-z is excessively long)
    d_wz = edge_distance[w_idx, z_idx]
    d_uz = edge_distance[u_idx, z_idx]
    leapfrog_detour = (d_uv + d_vw + d_wz) / (d_uz + 1e-8)
    top_leapfrogs = np.argsort(leapfrog_detour)[-k:]

    discount_base = 0.08 * global_mean_dist
    for idx in top_leapfrogs:
        u, z = u_idx[idx], z_idx[idx]
        freq = edge_n_used[u, z]
        if freq < 1.0:
            discount = discount_base / (1.0 + freq)
            updated_edge_distance[u, z] = max(1e-6, updated_edge_distance[u, z] - discount)
            updated_edge_distance[z, u] = updated_edge_distance[u, z]

    # (b) Topological spatial bridge promotion: Connect penalized nodes to their closest unselected neighbors
    # For each penalized node, find nearest neighbors that are not immediate tour neighbors
    for u in penalized_nodes:
        # Find nearest 5 neighbors in original metric
        nearest_neighbors = np.argsort(edge_distance[u])[:6]
        for neighbor in nearest_neighbors:
            if neighbor == u:
                continue
            freq = edge_n_used[u, neighbor]
            if freq < 0.5:
                # Modest discount to encourage exploration of topological bridges
                bridge_discount = 0.05 * global_mean_dist / (1.0 + freq)
                updated_edge_distance[u, neighbor] = max(1e-6, updated_edge_distance[u, neighbor] - bridge_discount)
                updated_edge_distance[neighbor, u] = updated_edge_distance[u, neighbor]

    return updated_edge_distance
