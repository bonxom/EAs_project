# ===== Heuristic Code =====
#{Guided local search penalizes high-utility geometrically inefficient edges while selectively rewarding unvisited 2-opt and node-relocation reconnection chords that strictly improve turning angles and detour geometry at sharp tour inflections.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    num_nodes = len(local_opt_tour)
    tour = np.asarray(local_opt_tour)

    u_idx = tour
    v_idx = np.roll(tour, -1)
    w_idx = np.roll(tour, -2)

    edge_n_used *= 0.96

    d_uv = edge_distance[u_idx, v_idx]
    d_vw = edge_distance[v_idx, w_idx]
    d_uw = edge_distance[u_idx, w_idx]

    detour_ratio = (d_uv + d_vw) / (d_uw + 1e-8)
    denom = 2.0 * d_uv * d_vw + 1e-8
    cos_turn = (d_uv**2 + d_vw**2 - d_uw**2) / denom
    cos_turn = np.clip(cos_turn, -1.0, 1.0)
    acute_deficit = 1.0 - cos_turn
    geom_inefficiency = detour_ratio * (1.0 + 0.5 * acute_deficit)

    base_dist = d_uv
    base_freq = edge_n_used[u_idx, v_idx]
    local_geom_factor = 0.5 * (geom_inefficiency + np.roll(geom_inefficiency, 1))
    utilities = (base_dist / (1.0 + base_freq)) * (1.0 + 0.35 * local_geom_factor)

    k = max(1, int(np.ceil(0.14 * num_nodes)))
    top_k_indices = np.argsort(utilities)[-k:]

    global_mean_dist = float(np.mean(edge_distance))
    adaptive_lambda = 0.30 * global_mean_dist

    for idx in top_k_indices:
        u, v = u_idx[idx], v_idx[idx]
        edge_n_used[u, v] += 1.0
        edge_n_used[v, u] += 1.0
        freq_penalty = 1.0 + np.log1p(edge_n_used[u, v])
        geom_weight = 1.0 + 0.5 * local_geom_factor[idx]
        penalty = adaptive_lambda * freq_penalty * geom_weight
        updated_edge_distance[u, v] += penalty
        updated_edge_distance[v, u] += penalty

    # Targeted guided reconnection chord rewards for acute inflection resolution
    acute_candidates = np.argsort(acute_deficit)[-k:]
    max_discount_cap = 0.15 * global_mean_dist

    for i in acute_candidates:
        u = u_idx[i]
        v = v_idx[i]
        w = w_idx[i]

        # Candidate 1: Direct 2.5-opt shortcut bypassing acute vertex v
        cur_cost_1 = d_uv[i] + d_vw[i]
        cand_cost_1 = edge_distance[u, w]
        if cand_cost_1 < cur_cost_1 and edge_n_used[u, w] < 1.0:
            net_gain_1 = cur_cost_1 - cand_cost_1
            discount_1 = min(max_discount_cap, 0.20 * net_gain_1 / (1.0 + edge_n_used[u, w]))
            updated_edge_distance[u, w] = max(1e-6, updated_edge_distance[u, w] - discount_1)
            updated_edge_distance[w, u] = updated_edge_distance[u, w]

        # Candidate 2: Evaluated 2-opt reconnection chord with opposite tour segment
        opp_idx = (i + num_nodes // 2) % num_nodes
        x = tour[opp_idx]
        y = tour[(opp_idx + 1) % num_nodes]

        cur_seg_cost = edge_distance[u, v] + edge_distance[x, y]
        cand_seg_cost = edge_distance[u, x] + edge_distance[v, y]
        if cand_seg_cost < cur_seg_cost and edge_n_used[u, x] < 1.0:
            net_gain_2 = cur_seg_cost - cand_seg_cost
            discount_2 = min(max_discount_cap, 0.15 * net_gain_2 / (1.0 + edge_n_used[u, x]))
            updated_edge_distance[u, x] = max(1e-6, updated_edge_distance[u, x] - discount_2)
            updated_edge_distance[x, u] = updated_edge_distance[u, x]

    return updated_edge_distance
