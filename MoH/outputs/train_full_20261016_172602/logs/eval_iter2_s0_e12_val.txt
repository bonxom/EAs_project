# ===== Heuristic Code =====
#{Applies an exponential decay factor to previous distance modifications to allow historically penalized edges to recover viability over time before augmenting the symmetric penalties on the highest GLS utility edges of the current local optimum.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    # Create an independent copy of edge_distance
    updated_edge_distance = np.copy(edge_distance)

    # Ruin & Recreate: apply a continuous decay/evaporation factor to existing penalty-modified edges
    decay_factor = 0.98
    modified_mask = edge_n_used > 0
    updated_edge_distance[modified_mask] *= decay_factor

    # Vectorized extraction of tour edges and Guided Local Search (GLS) utility evaluation
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = updated_edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    # Proportional penalty calculation based on the tour's average edge length
    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    # Top-utility edge selection and symmetric penalty assignment without Python loop overhead
    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold
    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
