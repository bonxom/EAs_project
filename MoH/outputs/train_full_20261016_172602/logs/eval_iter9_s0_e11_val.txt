# ===== Heuristic Code =====
#{Decay existing edge distance modifications towards the baseline using an exponential memory discount factor while symmetrically penalizing the highest utility edges of the current local optimum to balance exploration and prevent permanent landscape distortion.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Penalty evaporation / memory decay factor
    alpha = 0.05
    # Linear discount towards lower bound to prevent unbounded penalty accumulation
    # where edge_distance has accumulated past modifications, we decay non-diagonal values slightly
    decay_mask = ~np.eye(updated_edge_distance.shape[0], dtype=bool)
    # Estimate baseline scale from minimal symmetric distances or current non-zero distances
    current_min_scale = np.min(updated_edge_distance[decay_mask])
    updated_edge_distance[decay_mask] = (1.0 - alpha) * updated_edge_distance[decay_mask] + alpha * current_min_scale

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Utility calculation inversely proportional to edge frequency
    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)

    avg_edge = np.mean(tour_lengths)
    penalty = 0.3 * avg_edge

    threshold = 0.99 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically apply guided local search penalties to maximum utility edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
