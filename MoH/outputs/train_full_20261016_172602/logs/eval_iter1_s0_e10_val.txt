# ===== Heuristic Code =====
#{Neighborhood cluster-ruin identifies high-utility bottleneck edges in the local optimal tour and propagates exponentially decayed fractional penalties across their contiguous adjacent sub-path edges using vectorized roll operations, shattering trapping topological clusters without loop overhead.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Extract sequential tour edges
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    # Vectorized edge utility calculation
    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)
    base_penalty = 0.3 * avg_edge

    # Mark core edges meeting utility threshold
    core_mask = utilities >= (0.99 * max_utility)

    # Convolve/propagate ruin to adjacent contiguous edges (offsets: 0, +/-1, +/-2)
    # Decay factors for distance 0, 1, 2 along the tour sub-path
    offsets = [0, 1, -1, 2, -2]
    weights = [1.0, 0.5, 0.5, 0.25, 0.25]

    penalty_weights = np.zeros(len(local_opt_tour), dtype=np.float64)
    for offset, weight in zip(offsets, weights):
        rolled_core = np.roll(core_mask, offset)
        penalty_weights = np.maximum(penalty_weights, rolled_core.astype(np.float64) * weight)

    penalized_mask = penalty_weights > 0.0
    sel_u = u_idx[penalized_mask]
    sel_v = v_idx[penalized_mask]
    applied_penalties = base_penalty * penalty_weights[penalized_mask]

    # Symmetrically apply clustered penalties to shatter local entrapment
    updated_edge_distance[sel_u, sel_v] += applied_penalties
    updated_edge_distance[sel_v, sel_u] += applied_penalties

    return updated_edge_distance
