# ===== Heuristic Code =====
#{Adaptive Pareto-quantile penalization identifies non-dominated high-cost and low-frequency edges dynamically calibrated by tour cost variance, coupling them with targeted reinforcement of promising alternative incident edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    n = len(local_opt_tour)

    u_idx = np.array(local_opt_tour, dtype=np.intp)
    v_idx = np.roll(u_idx, -1)

    costs = edge_distance[u_idx, v_idx].astype(np.float64)
    used = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Measure coefficient of variation to adaptively size the perturbation set
    mean_cost = np.mean(costs)
    std_cost = np.std(costs)
    cv = std_cost / (mean_cost + 1e-8)

    # Adaptive quantile: dynamically target between top 15% and 40% based on dispersion
    quantile_level = float(np.clip(1.0 - (0.15 + 0.25 * np.tanh(cv * 2.0)), 0.60, 0.85))

    # Pareto dominance check: an edge is dominated if another edge has >= cost and <= used (with at least one strict)
    # Since n is typically 50-100, pairwise comparison is ultra-fast
    cost_diff = costs[:, None] - costs[None, :]
    used_diff = used[:, None] - used[None, :]
    strictly_better = (cost_diff < 0) & (used_diff >= 0) | (cost_diff <= 0) & (used_diff > 0)
    dominated = np.any(strictly_better, axis=1)

    # Utility formulation incorporating both cost prominence and low visitation
    cost_ranks = np.argsort(np.argsort(costs)).astype(np.float64) / max(n - 1, 1)
    freq_discount = 1.0 / (1.0 + np.sqrt(used))
    utility = cost_ranks * freq_discount

    # Combine Pareto non-dominance with dynamic high-utility quantile
    cutoff = np.quantile(utility, quantile_level)
    penalize_mask = (utility >= cutoff) | (~dominated & (costs > mean_cost))

    if not np.any(penalize_mask):
        penalize_mask = utility >= np.median(utility)

    p_u = u_idx[penalize_mask]
    p_v = v_idx[penalize_mask]
    p_costs = costs[penalize_mask]
    p_used = used[penalize_mask]
    p_utils = utility[penalize_mask]

    max_util = np.max(utility) + 1e-8
    penalties = (p_costs / (1.0 + np.sqrt(p_used + 1.0))) * (1.0 + p_utils / max_util)

    updated_edge_distance[p_u, p_v] += penalties
    updated_edge_distance[p_v, p_u] += penalties

    # Dual-action: discount unvisited / minimally used alternative edges connected to perturbed nodes
    penalized_nodes = np.unique(np.concatenate((p_u, p_v)))
    decay_ratio = 0.025
    for u in penalized_nodes:
        min_used = np.min(edge_n_used[u])
        promising_mask = (edge_n_used[u] <= min_used)
        promising_mask[u] = False
        if np.any(promising_mask):
            current_vals = updated_edge_distance[u, promising_mask]
            updated_edge_distance[u, promising_mask] -= current_vals * decay_ratio
            updated_edge_distance[promising_mask, u] = updated_edge_distance[u, promising_mask]

    return updated_edge_distance
