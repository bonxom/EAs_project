# ===== Heuristic Code =====
#{Information-theoretic frequency diversification computes local tour edge usage relative to global visitation entropy, penalizing dominant backbone edges proportionally to their redundancy and inversely to manifold diversity.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx].astype(np.float64)

    # Calculate global visitation statistics to gauge manifold concentration
    total_usage = np.sum(edge_n_used) / 2.0

    if total_usage > 0:
        # Probability distribution over all visited edges
        p_matrix = edge_n_used / (2.0 * total_usage)
        p_flat = p_matrix[p_matrix > 0]
        # Global entropy of edge exploration
        global_entropy = -np.sum(p_flat * np.log(p_flat + 1e-12))
        max_entropy = np.log(len(local_opt_tour) * (len(local_opt_tour) - 1) / 2.0 + 1e-12)
        # Normalized diversity metric in [0, 1]
        diversity = np.clip(global_entropy / (max_entropy + 1e-12), 0.05, 1.0)
    else:
        diversity = 1.0

    # Local edge frequency distribution within the current tour
    tour_p = (tour_counts + 1.0) / np.sum(tour_counts + 1.0)
    surprisal = -np.log(tour_p + 1e-12)
    redundancy = 1.0 / (surprisal + 1e-12)

    # Penalize over-represented backbone edges inversely proportional to global tour diversity
    manifold_diversification_pressure = (1.0 - diversity + 0.1) / diversity
    edge_utility = (tour_lengths * redundancy) * (1.0 + manifold_diversification_pressure * (tour_counts / (np.mean(tour_counts) + 1.0)))

    max_util = np.max(edge_utility)
    mean_length = np.mean(tour_lengths)

    # Select edges in the top percentile of utility to apply structured topological perturbation
    threshold = 0.85 * max_util
    selected_mask = edge_utility >= threshold

    penalty_magnitude = 0.35 * mean_length * (1.0 + manifold_diversification_pressure)

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty_magnitude
    updated_edge_distance[sel_v, sel_u] += penalty_magnitude

    return updated_edge_distance
