# ===== Heuristic Code =====
#{This heuristic replaces monolithic utility thresholding with a dynamic top-k Pareto selection that filters edges exceeding the median tour length and balances cost-to-frequency utility with adaptive penalty scaling.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    # Vectorized computation of structured utility balancing cost and historical frequency
    utilities = tour_lengths / (1.0 + tour_counts)

    n_edges = len(local_opt_tour)
    median_length = np.median(tour_lengths)

    # Filter candidates to ensure only sufficiently long edges are penalized, protecting short connector edges
    length_candidate_mask = tour_lengths >= median_length

    # Dynamically select top-k edges with highest utility among length-qualified candidates
    k = max(1, int(np.ceil(0.05 * n_edges)))

    if np.any(length_candidate_mask):
        candidate_indices = np.where(length_candidate_mask)[0]
        candidate_utilities = utilities[candidate_indices]

        actual_k = min(k, len(candidate_indices))
        # Deterministic top-k selection via partition
        top_k_rel_indices = np.argpartition(candidate_utilities, -actual_k)[-actual_k:]
        selected_indices = candidate_indices[top_k_rel_indices]
    else:
        actual_k = min(k, n_edges)
        selected_indices = np.argpartition(utilities, -actual_k)[-actual_k:]

    sel_u = u_idx[selected_indices]
    sel_v = v_idx[selected_indices]

    # Adaptive dynamic penalty scaled by relative utility and edge count dispersion
    avg_edge = np.mean(tour_lengths)
    selected_counts = tour_counts[selected_indices]
    penalty_factors = 0.25 * avg_edge * (1.0 + 1.0 / (1.0 + selected_counts))

    # Apply symmetric updates to the edge distance matrix
    updated_edge_distance[sel_u, sel_v] += penalty_factors
    updated_edge_distance[sel_v, sel_u] += penalty_factors

    return updated_edge_distance
