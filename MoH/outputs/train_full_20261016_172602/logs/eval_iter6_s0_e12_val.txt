# ===== Heuristic Code =====
#{Apply long-term memory evaporation to penalized edges via geometric decay while penalizing high-utility bottleneck edges in the local optimum to prevent landscape saturation and cyclic traps.}#

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)

    # Ruin step: apply episodic evaporation decay to edges previously penalized to avoid monotonic noise flattening
    decay_factor = 0.96
    penalized_mask = edge_n_used > 0
    if np.any(penalized_mask):
        # Evaporate excess penalty by pulling values back towards base scale proportionally
        excess_scale = edge_n_used / (1.0 + edge_n_used)
        updated_edge_distance[penalized_mask] -= (1.0 - decay_factor) * excess_scale[penalized_mask] * updated_edge_distance[penalized_mask]

    # Vectorized computation of edge utilities on the active tour
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)
    tour_lengths = updated_edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]

    utilities = tour_lengths / (1.0 + tour_counts)
    max_utility = np.max(utilities)
    avg_edge = np.mean(tour_lengths)

    # Dynamic penalty based on average edge cost and search saturation
    penalty = 0.25 * avg_edge
    threshold = 0.98 * max_utility
    selected_mask = utilities >= threshold

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    # Symmetrically update distances for identified bottleneck edges
    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
