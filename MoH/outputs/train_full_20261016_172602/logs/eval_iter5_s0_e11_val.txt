# ===== Heuristic Code =====
# {An adaptive ruin-and-recreate mechanism dynamically adjusts a percentile selection threshold and penalty scaling based on tour edge stagnation frequency to penalize the most critical bottleneck edges.}

import numpy as np

def update_edge_distance(edge_distance, local_opt_tour, edge_n_used):
    updated_edge_distance = np.copy(edge_distance)
    u_idx = local_opt_tour
    v_idx = np.roll(local_opt_tour, -1)

    tour_lengths = edge_distance[u_idx, v_idx]
    tour_counts = edge_n_used[u_idx, v_idx]
    utilities = tour_lengths / (1.0 + tour_counts)

    # Compute stagnation level based on mean usage of the current tour edges
    mean_count = np.mean(tour_counts)
    stagnation_ratio = mean_count / (1.0 + np.max(edge_n_used))

    # Dynamically widen the quantile threshold as stagnation intensifies
    # Base percentile is 90th percentile; lowers toward 70th percentile under stagnation
    quantile_cutoff = np.clip(90.0 - 20.0 * stagnation_ratio, 70.0, 95.0)
    threshold = np.percentile(utilities, quantile_cutoff)

    selected_mask = utilities >= threshold

    # Ensure at least the top-1 edge is selected if quantile misses due to ties/numerics
    if not np.any(selected_mask):
        selected_mask[np.argmax(utilities)] = True

    avg_edge = np.mean(tour_lengths)
    # Dynamically scale penalty with stagnation to escape deeper local basins
    penalty_factor = 0.3 * (1.0 + stagnation_ratio)
    penalty = penalty_factor * avg_edge

    sel_u = u_idx[selected_mask]
    sel_v = v_idx[selected_mask]

    updated_edge_distance[sel_u, sel_v] += penalty
    updated_edge_distance[sel_v, sel_u] += penalty

    return updated_edge_distance
